def _lp_radii(c):
    """Optimal radii for fixed centres: maximise sum(r) s.t. r_i + r_j <= d_ij and r_i <= wall distance."""
    n = len(c)
    iu = np.triu_indices(n, 1)
    d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1))
    m = len(d)
    A = np.zeros((m, n))
    A[np.arange(m), iu[0]] = 1
    A[np.arange(m), iu[1]] = 1
    wall = np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1]))
    res = linprog(-np.ones(n), A_ub=A, b_ub=d, bounds=[(0, max(w, 0)) for w in wall], method="highs")
    return res.x if res.success else None
