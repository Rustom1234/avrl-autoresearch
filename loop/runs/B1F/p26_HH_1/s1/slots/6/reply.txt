def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    tries, wins = np.zeros(3), np.zeros(3)
    disc_bounds = [(0.12, 0.2), (0.2, 0.35)]
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        if best is None:   # first start: a random restart
            c, r = _solve_once(n, rng)
            kind = -1
        else:              # adaptive kick, type chosen by (wins + 1) / (tries + 3)
            w = (wins + 1) / (tries + 3)
            kind = rng.choice(3, p=w / w.sum())
            c, r = best[0].copy(), best[1].copy()
            if kind < 2:   # disc kick: re-randomise the centres in a disc
                p, R = rng.uniform(0, 1, 2), rng.uniform(*disc_bounds[kind])
                m = np.hypot(*(c - p).T) < R
                k = m.sum()
                ang, rad = rng.uniform(0, 2 * np.pi, k), R * np.sqrt(rng.uniform(0, 1, k))
                c[m] = np.clip(p + np.c_[rad * np.cos(ang), rad * np.sin(ang)], 0, 1)
                r[m] = 0
            else:          # swap two circles of different radii, shrink both, then jitter all centres
                pairs = np.argwhere(np.abs(r[:, None] - r[None]) > 0.01)
                if len(pairs):
                    i, j = pairs[rng.integers(len(pairs))]
                else:
                    i, j = rng.choice(n, 2, replace=False)
                c[[i, j]] = c[[j, i]]
                s = 0.9 * min(r[i], r[j])
                r[i] = r[j] = s
                c = np.clip(c + rng.normal(0, 0.01, c.shape), 0, 1)
            c, r = _solve_once(n, rng, np.concatenate([c.ravel(), r]))
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        improved = best is None or r.sum() > best[1].sum()
        if kind >= 0:
            tries[kind] += 1
            wins[kind] += improved
        if improved:
            best = (c, r)
    return best
