def solve(n, seed, time_limit):
    rng, t0, best, slowest = np.random.default_rng(seed), time.process_time(), None, 0.0
    while True:   # random restarts until the time is used
        elapsed = time.process_time() - t0
        if elapsed >= TIME_FRACTION * time_limit or END_FRACTION * time_limit - elapsed < 1.5 * slowest:   # deadline guard
            break
        guided = best is not None and elapsed > 0.5 * TIME_FRACTION * time_limit and rng.random() < 0.5
        c, r = _solve_once(n, rng, best[0] if guided else None)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        slowest = max(slowest, time.process_time() - t0 - elapsed)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    return best
