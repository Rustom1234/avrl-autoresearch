def _perturb(c, r, rng):
    """Basin-hopping move: relocate 1 to 3 circles into the largest holes and jitter the other centres."""
    n = len(r)
    k = rng.integers(1, 4)
    idx = rng.choice(n, k, replace=False) if rng.random() < 0.3 else np.argsort(r)[:k]
    c, r = c.copy(), 0.9 * r
    obst = np.setdiff1d(np.arange(n), idx)
    oc, orad = c[obst], r[obst]
    p = rng.uniform(0, 1, (2000, 2))
    wall = np.minimum(np.minimum(p[:, 0], p[:, 1]), np.minimum(1 - p[:, 0], 1 - p[:, 1]))
    for i in idx:
        s = np.minimum(wall, (np.sqrt(((p[:, None] - oc[None]) ** 2).sum(-1)) - orad[None]).min(1))
        j = np.argmax(s)
        c[i], r[i] = p[j], max(s[j], 0)
        oc, orad = np.vstack([oc, p[j]]), np.append(orad, r[i])
    c[obst] += rng.normal(0, 0.01, (len(obst), 2))
    return c, r
