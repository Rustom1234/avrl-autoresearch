def solve(n, seed, time_limit):
    rng, t0, best, longest = np.random.default_rng(seed), time.process_time(), None, 0.0
    while True:   # random restarts until the next one would overrun the deadline
        elapsed = time.process_time() - t0
        if elapsed + 1.5 * longest > TIME_FRACTION * time_limit:
            break
        guided = best is not None and elapsed > 0.5 * TIME_FRACTION * time_limit and rng.random() < 0.5
        c, r = _solve_once(n, rng, best[0] if guided else None, 0.95 * best[1] if guided else None)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        longest = max(longest, time.process_time() - t0 - elapsed)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    return best
