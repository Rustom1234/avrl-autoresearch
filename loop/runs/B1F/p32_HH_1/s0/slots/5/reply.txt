def solve(n, seed, time_limit):
    rng, t0 = np.random.default_rng(seed), time.process_time()
    half = 0.5 * TIME_FRACTION * time_limit
    results = []
    for episode in range(2):   # two independent episodes, each half the budget
        best, start = None, time.process_time()
        while time.process_time() - t0 < (episode + 1) * half:
            guided = best is not None and time.process_time() - start > 0.5 * half and rng.random() < 0.5
            c, r = _solve_once(n, rng, best[0] if guided else None)
            c = np.clip(c, 0, 1)
            r = _repair(c, r)
            if best is None or r.sum() > best[1].sum():
                best = (c, r)
        results.append(best)
    return max(results, key=lambda b: b[1].sum())
