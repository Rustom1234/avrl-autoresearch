def tanimoto(A, B):
    # generalized (MinMax) Jaccard for count vectors: sum(min) / sum(max)
    l1a = A.sum(axis=1)[:, None]
    l1b = B.sum(axis=1)[None, :]
    summin = 0.5 * (l1a + l1b - cdist(A, B, "cityblock"))
    return summin / np.maximum(l1a + l1b - summin, 1e-9)
