def to_mol(s):
    mol = Chem.MolFromSmiles(s)
    if mol is None:
        return Chem.MolFromSmiles("C")
    try:
        frags = Chem.GetMolFrags(mol, asMols=True, sanitizeFrags=True)
        return max(frags, key=lambda f: (f.GetNumHeavyAtoms(), Descriptors.MolWt(f)))
    except Exception:
        return mol
