def smooth_scores(score, fp):
    # fp: test feature matrix; first N_BITS columns are the Morgan bits
    A = fp[:, :N_BITS].astype(np.float64)
    inter = A @ A.T
    cnt = A.sum(axis=1)
    union = cnt[:, None] + cnt[None, :] - inter
    sim = inter / np.maximum(union, 1e-12)
    np.fill_diagonal(sim, -1.0)
    n = len(score)
    k = min(SMOOTH_K, n - 1)
    if k < 1:
        return score
    out = score.copy()
    for i in range(n):
        idx = np.argsort(-sim[i])[:k]
        s = sim[i, idx]
        if s[0] < SMOOTH_MIN_SIM:
            continue
        w = np.clip(s, 0, None) ** 3
        if w.sum() <= 0:
            continue
        out[i] = SMOOTH_ALPHA * score[i] + (1 - SMOOTH_ALPHA) * np.sum(w * score[idx]) / w.sum()
    return out
