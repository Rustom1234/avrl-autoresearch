def _refine(c, r, rng):

    n = len(r)
    c_new = c.copy()
    for _ in range(20):
        i = rng.integers(0, n)
        idx = rng.integers(0, 2)
        c_new[i, idx] += rng.uniform(-0.01, 0.01)
