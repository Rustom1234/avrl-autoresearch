    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        # Only calculate constraints for pairs that are relatively close to each other
        # To keep it simple and fast, we can check the distance between centers
        # But for SLSQP, we need the full set or a very smart subset. 10% of furthest pairs is slow.
        # Let's use a spatial heuristic: only active if dist < 2 * max(r)
        # However, for 26 circles, the full set is usually manageable if the gradient is clean.
        # Let's stick to the full set but optimize the vectorization.
        d = np.linalg.norm(c[iu[0]] - c[iu[1]], axis=1) - r[iu[0]] - r[iu[1]]
