from sklearn.ensemble import RandomForestRegressor
from sklearn.model_selection import KFold
from sklearn.metrics import mean_absolute_error

# Settings
N_BITS = 2048
RADIUS = 2
SEARCH_TREES = 200
FINAL_TREES = 400


def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    return np.array([gen.GetFingerprintAsNumPy(Chem.MolFromSmiles(s)) for s in smiles], dtype=np.float32)


def fit_predict(train_smiles, train_y, test_smiles, seed):
    X = featurize(train_smiles)
    y = np.asarray(train_y, dtype=np.float64)
    
    best_mae = float('inf')
    best_params = (0.3, 1)
    
    kf = KFold(n_splits=4, shuffle=True, random_state=seed)
    for max_f in [0.1, 0.3, 0.5]:
        for min_sl in [1, 3, 5]:
            fold_maes = []
            for train_idx, val_idx in kf.split(X):
                X_t, X_v = X[train_idx], X[val_idx]
                y_t, y_v = y[train_idx], y[val_idx]
                
                m = RandomForestRegressor(n_estimators=SEARCH_TREES, max_features=max_f, 
                                         min_samples_leaf=min_sl, n_jobs=-1, random_state=seed)
                m.fit(X_t, y_t)
                fold_maes.append(mean_absolute_error(y_v, m.predict(X_v)))
            
            avg_mae = np.mean(fold_maes)
            if avg_mae < best_mae:
                best_mae = avg_mae
                best_params = (max_f, min_sl)
                
    model = RandomForestRegressor(n_estimators=FINAL_TREES, max_features=best_params[0], 
                                  min_samples_leaf=best_params[1], n_jobs=-1, random_state=seed)
    model.fit(X, y)
    return model.predict(featurize(test_smiles))
