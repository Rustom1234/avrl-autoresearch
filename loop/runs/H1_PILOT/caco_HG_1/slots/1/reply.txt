from sklearn.ensemble import RandomForestRegressor
import lightgbm as lgb

# Settings
N_BITS = 2048
RADIUS = 2
RF_TREES = 300
LGBM_TREES = 500
LGBM_LR = 0.03
MAX_FEATURES = 0.3
MIN_SAMPLES_LEAF = 1


def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    return np.array([gen.GetFingerprintAsNumPy(Chem.MolFromSmiles(s)) for s in smiles], dtype=np.float32)


def fit_predict(train_smiles, train_y, test_smiles, seed):
    X_train = featurize(train_smiles)
    X_test = featurize(test_smiles)
    y = np.asarray(train_y, dtype=np.float64)
    
    rf = RandomForestRegressor(n_estimators=RF_TREES, max_features=MAX_FEATURES, min_samples_leaf=MIN_SAMPLES_LEAF,
                                 n_jobs=-1, random_state=seed)
    rf.fit(X_train, y)
    
    lgbm = lgb.LGBMRegressor(n_estimators=LGBM_TREES, learning_rate=LGBM_LR, 
                                 feature_fraction=MAX_FEATURES, 
                                 importance_type='gain', 
                                 random_state=seed, 
                                 verbosity=-1)
    lgbm.fit(X_train, y)
    
    return (rf.predict(X_test) + lgbm.predict(X_test)) / 2.0
