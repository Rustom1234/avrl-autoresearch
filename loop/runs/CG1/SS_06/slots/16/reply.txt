def batch_cutout(inputs, size: int):
    B, _, H, W = inputs.shape
    dev = inputs.device
    cy = (torch.rand(B, 1, 1, 1, device=dev) * H).long()
    cx = (torch.rand(B, 1, 1, 1, device=dev) * W).long()
    ys = torch.arange(H, device=dev).view(1, 1, H, 1)
    xs = torch.arange(W, device=dev).view(1, 1, 1, W)
    half = size // 2
    m = (ys >= cy - half) & (ys < cy + half) & (xs >= cx - half) & (xs < cx + half)
    m = m & (torch.rand(B, 1, 1, 1, device=dev) < 0.5)
    return torch.where(m, torch.zeros_like(inputs), inputs)
