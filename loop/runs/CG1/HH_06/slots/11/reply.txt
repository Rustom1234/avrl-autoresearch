class BlendPool(nn.Module):
    def __init__(self, global_pool=False):
        super().__init__()
        self.global_pool = global_pool
        self.alpha = nn.Parameter(torch.ones(1))

    def reset_parameters(self):
        self.alpha.data.fill_(1.0)

    def forward(self, x):
        if self.global_pool:
            mx = F.adaptive_max_pool2d(x, 1)
            av = F.adaptive_avg_pool2d(x, 1)
        else:
            mx = F.max_pool2d(x, 2)
            av = F.avg_pool2d(x, 2)
        return self.alpha * mx + (1 - self.alpha) * av
