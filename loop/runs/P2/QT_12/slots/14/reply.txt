def get_lr_multiplier(progress):
    if progress < WARMUP_RATIO:
        return progress / WARMUP_RATIO if WARMUP_RATIO > 0 else 1.0
    elif progress < 1.0 - WARMDOWN_RATIO:
        # Stay at peak for a short initial period, then linear decay
        peak_period = 0.05
        if progress < peak_period:
            return 1.0
        else:
            # Linear decay from 1.0 at peak_period to FINAL_LR_FRAC at 1.0
            # The decay happens over the remaining (1.0 - peak_period)
            decay_range = 1.0 - peak_period
            current_decay = (progress - peak_period) / decay_range
            # Scale from 1.0 to FINAL_LR_FRAC
            return 1.0 - (1.0 - FINAL_LR_FRAC) * current_decay
    else:
        cooldown = (1.0 - progress) / WARMDOWN_RATIO
        return cooldown * 1.0 + (1 - cooldown) * FINAL_LR_FRAC
