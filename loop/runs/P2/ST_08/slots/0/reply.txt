class MLP(nn.Module):
    def __init__(self, config):
        super().__init__()
        # SwiGLU: 3 matrices of size dim x hidden, hidden ~ 8/3 * dim, rounded to multiple of 64
        hidden = ((int(8 * config.n_embd / 3) + 63) // 64) * 64
        self.c_fc = nn.Linear(config.n_embd, 2 * hidden, bias=False)  # gate and up projections fused
        self.c_proj = nn.Linear(hidden, config.n_embd, bias=False)

    def forward(self, x):
        gate, up = self.c_fc(x).chunk(2, dim=-1)
        x = F.silu(gate) * up
        x = self.c_proj(x)
        return x
