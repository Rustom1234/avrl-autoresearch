class MLP(nn.Module):
    def __init__(self, config):
        super().__init__()
        # SwiGLU-lite: hidden = 8/3 * n_embd rounded to a multiple of 64 (keeps params ~ 4x MLP)
        hidden = int(round(8 * config.n_embd / 3 / 64)) * 64
        self.c_fc = nn.Linear(config.n_embd, 2 * hidden, bias=False)  # gate and value in one matmul
        self.c_proj = nn.Linear(hidden, config.n_embd, bias=False)

    def forward(self, x):
        gate, value = self.c_fc(x).chunk(2, dim=-1)
        x = F.silu(gate) * value
        x = self.c_proj(x)
        return x
