@torch.no_grad()
def calculate_metrics(metrics):
    for key in metrics:
        if isinstance(metrics[key], dict):
            for spec in metrics[key].spec:
                if isinstance(spec, torch.Tensor):
                    for i, value in enumerate(spec):
                        metrics[key][spec['name']][i] = value.item()
                elif isinstance(spec, list):
                    for value in spec:
                        if isinstance(value, torch.Tensor):
                            metrics[key][spec['name']].add(value.sum().item())
                        else:
                            metrics[key][spec['name']].add(value)
            for key in metrics[key].keys():
                if key == 'val_loss':
                    metrics[key]['val_loss'] = metrics[key]['val_loss']
    metrics = metrics.copy()
    return metrics
