def get_lr_multiplier(progress):
    if progress < WARMUP_RATIO:
        return progress / WARMUP_RATIO if WARMUP_RATIO > 0 else 1.0
    
    # Cosine annealing for the main body of training
    # progress goes from 0 to 1. We want the peak at (WARMUP_RATIO) to (1 - WARMDOWN_RATIO)
    # but for simplicity, we treat the remaining time budget as the annealing period.
    progress_in_annealing = (progress - WARMUP_RATIO) / (1.0 - WARMUP_RATIO - WARMDOWN_RATIO)
    progress_in_annealing = max(0.0, min(1.0, progress_in_annealing))
    
    cosine_val = 0.5 * (1 + math.cos(math.pi * progress_in_annealing))
    
    if progress > (1.0 - WARMDOWN_RATIO):
        # Transition to final_lr_frac during warmdown
        cooldown = (1.0 - progress) / WARMDOWN_RATIO
        return cooldown * 1.0 + (1 - cooldown) * FINAL_LR_FRAC
    
    return cosine_val
