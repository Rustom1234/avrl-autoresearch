def get_lr_multiplier(progress):
    if progress < WARMUP_RATIO:
        if WARMUP_TYPE == "cosine":
            return WARMUP_FACTOR * (0.5 * (1 + math.cos(math.pi * progress / WARMUP_RATIO)))
