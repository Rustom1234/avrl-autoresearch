# Schedules (based on wall-clock time)

def get_lr_multiplier(elapsed_time):
    # lr = lr_initial / (1 + (t / t_warmup)**0.7)
    # We treat the first 10% of TIME_BUDGET as the implicit warmup period
    t_warmup = TIME_BUDGET * 0.1
    return 1.0 / (1.0 + (elapsed_time / t_warmup)**0.7)

def get_muon_momentum(step):
    frac = min(step / 300, 1)
    return (1 - frac) * 0.85 + frac * 0.95

def get_weight_decay(progress):
    # Linear decay of weight decay over the total time budget
    # We map current progress back to a 0-1 range for the old weight decay logic
    return WEIGHT_DECAY * (1.0 - (elapsed_time / TIME_BUDGET))
