class MLP(nn.Module):
    def __init__(self, config):
        super().__init__()
        # SwiGLU: hidden ~ 8/3 * n_embd keeps parameter count close to the 4x ReLU^2 MLP
        hidden = -(-(8 * config.n_embd // 3) // 128) * 128
        self.c_fc = nn.Linear(config.n_embd, 2 * hidden, bias=False)  # gate and up projections
        self.c_proj = nn.Linear(hidden, config.n_embd, bias=False)

    def forward(self, x):
        gate, up = self.c_fc(x).chunk(2, dim=-1)
        x = F.silu(gate) * up
        x = self.c_proj(x)
        return x
