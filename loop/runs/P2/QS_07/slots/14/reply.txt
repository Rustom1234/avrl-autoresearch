N_STAGES = 5
STAGE_WARMUP = 0.075    # fraction of each stage spent climbing from 0.1 back to 1.0

def get_lr_multiplier(progress):
    pos = min(progress, 0.999999) * N_STAGES
    f = pos - int(pos)  # position within the current stage
    if f < STAGE_WARMUP:
        return 0.1 + 0.9 * f / STAGE_WARMUP
    t = (f - STAGE_WARMUP) / (1 - STAGE_WARMUP)
    return FINAL_LR_FRAC + (1 - FINAL_LR_FRAC) * 0.5 * (1 + math.cos(math.pi * t))
