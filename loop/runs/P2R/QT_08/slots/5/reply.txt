def get_lr_multiplier(progress):
    if progress < WARMUP_RATIO:
        return progress / WARMUP_RATIO if WARMUP_RATIO > 0 else 1.0
    
    # Cosine annealing from 1.0 down to FINAL_LR_FRAC
    # The peak is at 1.0 - WARMDOWN_RATIO
    peak_progress = 1.0 - WARMDOWN_RATIO
    if progress < peak_progress:
        return 1.0
    
    # Map progress from [peak_progress, 1.0] to [0, 1]
    progress_in_warmdown = (progress - peak_progress) / (1.0 - peak_progress)
    # Cosine formula: 1 - 0.5 * (1 - cos(pi * x))
    # Here we interpolate from 1.0 to FINAL_LR_FRAC
    return FINAL_LR_FRAC + (1.0 - FINAL_LR_FRAC) * (0.5 * (1 + math.cos(math.pi * progress_in_warmdown)))
