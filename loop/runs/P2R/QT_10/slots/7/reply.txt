# Plateau detection state
loss_history = []
PLATEAU_THRESHOLD = 1.001
PLATEAU_PATIENCE = 10
plateau_counter = 0
plateau_decay = 0.95
plateau_mult = 1.0

def get_lr_multiplier(progress):
    global plateau_counter, plateau_mult
    if progress < WARMUP_RATIO:
        return progress / WARMUP_RATIO if WARMUP_RATIO > 0 else 1.0
    elif progress < 1.0 - WARMDOWN_RATIO:
        # Check for plateau
        if len(loss_history) > 0:
            recent_avg = sum(loss_history[-5:]) / min(len(loss_history), 5)
            if recent_avg > loss_history[-1] * PLATEAU_THRESHOLD:
                plateau_counter = 0
            else:
                plateau_counter += 1
                if plateau_counter > PLATEAU_PATIENCE:
                    plateau_mult *= plateau_decay
                    plateau_counter = 0
        return 1.0 * plateau_mult
    else:
        cooldown = (1.0 - progress) / WARMDOWN_RATIO
        return cooldown * 1.0 + (1 - cooldown) * FINAL_LR_FRAC
