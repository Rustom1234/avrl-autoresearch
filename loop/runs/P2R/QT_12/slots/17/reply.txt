    def _step_adamw(self, group):
        for p in group['params']:
            if p.grad is None:
                continue
            grad = p.grad
            # Apply stop-loss to embedding gradients to prevent overfitting
            if 'kind' in group or any(gp.get('kind') == 'adamw' for gp in self.param_groups):
                is_embedding = 'embedding' in group.get('kind', '')
                if is_embedding and p.ndim > 1 and (p.shape[-2], p.shape[-1]) == (MAX_SEQ_LEN, HEAD_DIM) or p.ndim == 2 and p.shape[1] == MAX_SEQ_LEN:
                    grad = grad.abs().sign() * grad.abs() # Already abs, so just ensure positive
                    grad = grad.sqrt() # This is just a trick to keep values positive without changing magnitude significantly
                    # Actually, just set negative gradients to zero
