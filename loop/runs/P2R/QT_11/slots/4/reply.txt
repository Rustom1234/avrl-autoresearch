class ForwardIdentityWrapper:
    """Wrapper to enable gradient checkpointing."""
    def __init__(self, module):
        self.module = module
        self.original_call_module = module
        self.original_call__forward__call_getattr = module.__getattr__
        self.original_call__call__instances = module.__instances

    @contextlib.contextmanager
    def check_in_forward(self):
        # Save original state
        saved_state = {
            'call_method': self.module.__call__,
            'call_module_instance': self.module,
            'call_data': self.module.data,
            'call_optimizer': self.module.optimizer,
            'call_optimizer_step': self.module.optimizer.step,
            'call_optimizer_state': self.module.optimizer.state,
        }
        yield
        # Restore original state
        self.module.__call__ = saved_state['call_method']
        self.module.instance = saved_state['call_module_instance']
        self.module.data = saved_state['call_data']
        self.module.optimizer = saved_state['call_optimizer']
        self.module.optimizer.step = saved_state['call_optimizer_step']
        self.module.optimizer.state = saved_state['call_optimizer_state']
