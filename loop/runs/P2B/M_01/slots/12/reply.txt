class MLP(nn.Module):
    def __init__(self, config):
        super().__init__()
        hidden = (8 * config.n_embd) // 3  # SwiGLU width: 2/3 of 4x keeps params matched
        self.c_fc = nn.Linear(config.n_embd, 2 * hidden, bias=False)
        self.c_proj = nn.Linear(hidden, config.n_embd, bias=False)

    def forward(self, x):
        gate, val = self.c_fc(x).chunk(2, dim=-1)
        x = F.silu(gate) * val
        x = self.c_proj(x)
        return x
