    # Ramp to 0.95 early, then to 0.97 by warmdown start
    if step < 60:
        return 0.85 + (step / 60) * 0.10
    if progress > 0.7:
        return 0.97
    return (1 - (progress - 0.7) / 0.3) * 0.97
