def get_lr_multiplier(progress):
    if progress < WARMUP_RATIO:
        return progress / WARMUP_RATIO if WARMUP_RATIO > 0 else 1.0
    
    # Exponential decay phase
    # Decay starts after warmup. We use a decay constant tuned for a smooth transition.
    # To reach FINAL_LR_FRAC at progress=1.0, we use an exponential form.
    # Multiplier = exp( -k * (progress - WARMUP_RATIO) / (1 - WARMUP_RATIO) )
    
    # Solving for k: FINAL_LR_FRAC = exp(-k) => k = -ln(FINAL_LR_FRAC)
    # Since FINAL_LR_FRAC is 0.0, we use a very small epsilon or a steep decay.
    # A standard 'Shakespeare' exponential decay for deep learning often uses a 
    # decay factor like 0.95 per 10% of progress.
    
    # Simple exponential decay to reach near-zero at the end:
    remaining_dist = 1.0 - WARMUP_RATIO
    # Decay factor k such that at progress=1.0, multiplier is very small
    k = 10.0 
    return math.exp(-k * (progress - WARMUP_RATIO) / remaining_dist)
