def get_muon_momentum(step):
    # Ramp from 0.85 to 0.97 over first 200 steps, then hold 0.97 until warmdown
    if step <= 200:
        p = 0.85
        w = 0.97
        ramp_frac = step / 200.0
        return p + (w - p) * ramp_frac
    else:
        # Hold at 0.97 until warmdown begins (near the end of the schedule)
        return 0.97
