class CustomOptimizer(torch.optim.AdamW):
    def __init__(self, params, lr=1e-4, betas=(0.9, 0.999), eps=1e-8, weight_decay=1e-0,
                 maxlement=0.1, method='norm', use_grad_clipping=False, clip_value=1.0):
        super().__init__(params, lr, betas, eps, weight_decay)
        self.maxlement = maxlement
        self.method = method
        self.use_grad_clipping = use_grad_clipping
        self.clip_value = clip_value

    def step(self, closure=None):
        loss = super().step(closure)
        if self.use_grad_clipping:
            grad_norm = torch.norm(torch.stack([p.grad for p in self.param_groups[0]['params']]))
            if grad_norm > self.clip_value:
                scale = self.clip_value / grad_norm
                for p in self.param_groups[0]['params']:
                    p.grad.mul_(scale)
        return loss

def get_optimal_lr_l2_norm(model):
    l2_norm_sum = 0.0
    for p in model.parameters():
        if p.grad is not None:
            l2_norm_sum += torch.norm(p.grad)**2
    l2_norm = l2_norm_sum.sqrt()
    return l2_norm
