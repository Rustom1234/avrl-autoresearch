def get_lr_multiplier(progress):
    if progress < WARMUP_RATIO:
        return (progress / WARMUP_RATIO) ** 2
    elif progress < 1.0 - WARMDOWN_RATIO:
        # Cosine decay from 1.0 to 0.0 (or FINAL_LR_FRAC)
        progress_in_decay = (progress - WARMUP_RATIO) / (1.0 - WARMDOWN_RATIO - WARMUP_RATIO)
        return FINAL_LR_FRAC + (1.0 - FINAL_LR_FRAC) * (0.5 * (1 + math.cos(math.pi * progress_in_decay)))
    else:
        cooldown = (1.0 - progress) / WARMDOWN_RATIO
        return cooldown * 1.0 + (1 - cooldown) * FINAL_LR_FRAC
