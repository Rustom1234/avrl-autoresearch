def get_lr_multiplier(progress):
    # Soft warmup: gradual increase from 0.0 to 1.0 over 5% of the budget
    soft_warmup_ratio = 0.05
    if progress < soft_warmup_ratio:
        return progress / soft_warmup_ratio
    elif progress < 1.0 - WARMDOWN_RATIO:
        return 1.0
    else:
        cooldown = (1.0 - progress) / WARMDOWN_RATIO
        return cooldown * 1.0 + (1 - cooldown) * FINAL_LR_FRAC
