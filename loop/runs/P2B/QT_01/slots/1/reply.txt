# Schedules (all based on progress = training_time / TIME_BUDGET)

def get_lr_multiplier(progress):
    if progress < WARMUP_RATIO:
        return progress / WARMUP_RATIO if WARMUP_RATIO > 0 else 1.0
    
    # Cosine decay from 1.0 to FINAL_LR_FRAC over the remaining duration
    # Adjust for warmdown_ratio if needed, but standard cosine is usually best
    # progress here is 0.0 to 1.0
    # We want to decay from 1.0 to FINAL_LR_FRAC
    # Scale progress to the decay window (1.0 - WARMDOWN_RATIO)
    decay_progress = (progress - WARMUP_RATIO) / (1.0 - WARMUP_RATIO - WARMDOWN_RATIO)
    decay_progress = max(0.0, min(1.0, decay_progress))
    
    coeff = 0.5 * (1 + math.cos(math.pi * decay_progress))
    return coeff * (1.0 - FINAL_LR_FRAC) + FINAL_LR_FRAC

def get_muon_momentum(step):
    frac = min(step / 300, 1)
    return (1 - frac) * 0.85 + frac * 0.95

def get_weight_decay(progress):
    return WEIGHT_DECAY * (1 - progress)
