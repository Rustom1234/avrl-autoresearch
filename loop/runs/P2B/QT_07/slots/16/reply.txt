def get_lr_multiplier(progress):
    if progress < WARMUP_RATIO:
        return progress / WARMUP_RATIO if WARMUP_RATIO > 0 else 1.0
    elif progress < 1.0 - WARMDOWN_RATIO:
        return 1.0
    else:
        # Cosine decay with restarts every 1/4 of the budget
        # Progress relative to the current 1/4 window
        relative_progress = (progress - (1.0 - WARMDOWN_RATIO)) / WARMDOWN_RATIO
        return FINAL_LR_FRAC + 0.5 * (1.0 - FINAL_LR_FRAC) * (1 + math.cos(math.pi * relative_progress))
