def get_muon_momentum(step):
    # Linearly warm momentum from 0.85 (step 0) to 0.95 over the first ~300 steps
    if step == 0:
        return 0.85
    interpolation progress = min(step / 300, 1.0)
    return 0.85 + interpolation progress * (0.95 - 0.85)
