def get_muon_momentum(step):
    # Ramp muon momentum from 0.85 to 0.97 over 300 steps for non-scheduled runs
    # If a schedule exists, it will be overridden by the optimizer loop; this provides a stable high momentum baseline. 0.97 is the target end momentum.
    frac = step / 300.0
    start_mom = 0.85
    end_mom = 0.97
    return (1 - frac) * start_mom + frac * end_mom
