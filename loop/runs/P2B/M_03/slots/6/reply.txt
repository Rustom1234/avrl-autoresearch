def get_lr_multiplier(progress):
    if progress < WARMUP_RATIO:
        return progress / WARMUP_RATIO if WARMUP_RATIO > 0 else 1.0
    elif progress < 1.0 - WARMDOWN_RATIO:
        return 1.0
    else:
        cooldown = (1.0 - progress) / WARMDOWN_RATIO
        return cooldown * 1.0 + (1 - cooldown) * FINAL_LR_FRAC
    # Batch size ramp: return multiplier based on current progress
    batch_scale = get_batch_scale(progress)
    return batch_scale


def get_batch_scale(progress):
    if not AT_SCHEDULE:
        return 1.0
    # Find index of current progress
    idx = 0
    for i, (head, milestones) in enumerate(AT_SCHEDULE):
        if progress >= milestones[0]:
            if i == len(AT_SCHEDULE) - 1:
                return AT_SCHEDULE[i][2] # scaled multipliers
            else:
                return AT_SCHEDULE[1]
        else:
            idx = i
    return AT_SCHEDULE[-1][2]
