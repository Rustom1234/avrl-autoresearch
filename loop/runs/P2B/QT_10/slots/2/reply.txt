# Schedules (all based on progress = training_time / TIME_BUDGET)

def get_lr_multiplier(progress):
    if progress < WARMUP_RATIO:
        return progress / WARMUP_RATIO if WARMUP_RATIO > 0 else 1.0
    
    # Cosine annealing
    # We want to go from 1.0 to FINAL_LR_FRAC over the period from (1-WARMDOWN_RATIO) to 1.0
    if progress > (1.0 - WARMDOWN_RATIO):
        # Map progress to [0, 1] range for the warmdown period
        cooldown_progress = (progress - (1.0 - WARMDOWN_RATIO)) / WARMDOWN_RATIO
        return FINAL_LR_FRAC + 0.5 * (1.0 - FINAL_LR_FRAC) * (1 + math.cos(math.pi * cooldown_progress))
    return 1.0

def get_muon_momentum(step):
    frac = min(step / 300, 1)
    return (1 - frac) * 0.85 + frac * 0.95

def get_weight_decay(progress):
    return WEIGHT_DECAY * (1 - progress)
