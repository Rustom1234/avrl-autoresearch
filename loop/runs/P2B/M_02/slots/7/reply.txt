class MLP(nn.Module):
    def __init__(self, config):
        super().__init__()
        # SwiGLU: hidden ~ 2/3 * 4 * n_embd, rounded to a multiple of 64 (keeps params near baseline)
        hidden = round(2 * 4 * config.n_embd / 3 / 64) * 64
        self.c_fc = nn.Linear(config.n_embd, 2 * hidden, bias=False)  # gate and up projections fused
        self.c_proj = nn.Linear(hidden, config.n_embd, bias=False)

    def forward(self, x):
        gate, up = self.c_fc(x).chunk(2, dim=-1)
        x = F.silu(gate) * up
        x = self.c_proj(x)
        return x
