def get_lr_multiplier(progress):
    if progress < WARMUP_RATIO:
        return progress / WARMUP_RATIO if WARMUP_RATIO > 0 else 1.0
    
    # Cosine annealing schedule
    # Progress from 0.0 to 1.0 across the main training window
    # (excluding warmup and warmdown for the cosine curve)
    main_progress = (progress - WARMUP_RATIO) / (1.0 - WARMUP_RATIO - WARMDOWN_RATIO)
    main_progress = max(0.0, min(1.0, main_progress))
    
    # Scale the cosine from 1.0 down to FINAL_LR_FRAC
    multiplier = 0.5 * (1.0 + math.cos(math.pi * main_progress))
