def get_lr_multiplier(progress):
    # Cosine annealing with restarts: k restarts per batch (e.g., 50 steps per cycle)
    k = 50.0  # steps per restart cycle
    cycles_proceeded = int(progress / (WARMUP_RATIO / k)) if WARMUP_RATIO > 0 else 0
    n = progress * k
    fr = n / k - cycles_proceeded  # fraction within current cycle

    if WARMUP_RATIO > 0:
        # Warmup phase: linear ramp from 0 to 1
        return fr if fr < WARMUP_RATIO else WARMUP_RATIO
    else:
        # Cosine decay with restarts: cos(pi * cycles + pi * fr) -> alternates between peaks and troughs
        cos_arg = (cycles_proceeded + fr / 2) * 2.0 * math.pi
        cos_val = math.cos(cos_arg)
        return 0.5 + 0.5 * cos_val
