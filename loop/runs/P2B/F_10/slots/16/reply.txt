def get_lr_multiplier(progress):
    if progress < WARMUP_RATIO:
        return progress / WARMUP_RATIO if WARMUP_RATIO > 0 else 1.0
    else:
        # Linear decay from 1.0 at WARMUP_RATIO to FINAL_LR_FRAC at progress 1.0
        decay_range = 1.0 - WARMUP_RATIO
        if decay_range <= 0: return 1.0
        current_decay_pos = (progress - WARMUP_RATIO) / decay_range
        return 1.0 + (FINAL_LR_FRAC - 1.0) * current_decay_pos
