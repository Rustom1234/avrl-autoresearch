def build_byte_weights(tokenizer, vocab_size):
    # UTF-8 byte length of each token's decoded string; fall back to uniform weights
    w = torch.ones(vocab_size, dtype=torch.float32)
    try:
        for i in range(vocab_size):
            w[i] = float(len(tokenizer.decode([i]).encode("utf-8")))
    except Exception:
        w.fill_(1.0)
    return w.to(device)
