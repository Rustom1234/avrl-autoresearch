def get_lr_multiplier(progress):
    if progress < WARMUP_RATIO:
        return progress / WARMUP_RATIO if WARMUP_RATIO > 0 else 1.0
    
    # Cosine annealing between WARMUP_RATIO and (1 - WARMDOWN_RATIO)
    # and then falling to FINAL_LR_FRAC
    if progress < 1.0 - WARMDOWN_RATIO:
        # Map [WARMUP_RATIO, 1.0 - WARMDOWN_RATIO] to [0, 1]
        phase_min = WARMUP_RATIO
        phase_max = 1.0 - WARMDOWN_RATIO
        progress_norm = (progress - phase_min) / (phase_max - phase_min)
        return 0.5 * (1.0 + math.cos(math.pi * progress_norm))
    else:
        # Final decay to FINAL_LR_FRAC
        cooldown = (1.0 - progress) / WARMDOWN_RATIO
        return cooldown * 1.0 + (1 - cooldown) * FINAL_LR_FRAC
