def adamw_step_fused(p, grad, exp_avg, exp_avg_sq, step_t, lr_t, beta1_t, beta2_t, eps_t, wd_t):
    p.mul_(1 - lr_t * wd_t)
    exp_avg.lerp_(grad, 1 - beta1_t)
    exp_avg_sq.lerp_(grad.square(), 1 - beta2_t)
    bias1 = 1 - beta1_t ** step_t
    bias2 = 1 - beta2_t ** step_t
    denom = (exp_avg_sq / bias2).sqrt() + eps_t
    step_size = lr_t / bias1

@torch.compile(dynamic=False, fullgraph=True)
def adamw_step_fused(p, grad, exp_avg, exp_avg_sq, step_t, lr_t, beta1_t, beta2_t, eps_t, wd_t):
    p.mul_(1 - lr_t * wd_t)
    exp_avg.lerp_(grad, 1 - beta1_t)
    exp_avg_sq.lerp_(grad.square(), 1 - beta2_t)
    bias1 = 1 - beta1_t ** step_t
    bias2 = 1 - beta2_t ** step_t
    denom = (exp_avg_sq / bias2).sqrt() + eps_t
    step_size = lr_t / bias1
    p.add_(exp_avg / denom, alpha=-step_size)

def ema_model_step(mo, lam):
    """Update EMA model using decay factor lam (e.g., 0.999)"""
    olist = mo.state_dict()['optimizers'][0]
    with torch.no_grad():
        params = []
        if 'params' not in olist:
            olist['params'] = []
        # Get unique parameter states and their LRs
        states = {}
        lrs = {}
        for i, p in enumerate(olist.get('params', [])):
            if p.grad is None:
                continue
            if i not in states:
                states[i] = []
                lrs[i] = mo.param_groups[i]['lr']
            data = {}
            for key, val in mo.state[p].items():
                if key != 'step':
                    data[key] = val
            states[i].append(data)

        # For QKV, FC, fc2, proj, wte, v_e, x0, scalars
        for k in list(olist.keys()):
            if k not in states:
                continue
            # Update every parameter in the group
            for i in range(len(olist.get('params', []))):
                for j, state in enumerate(states[i]):
                    if 'exp_avg' in state:
                        mo.state[olist['params'][i]].expand_as(state['exp_avg']).lerp_(state['exp_avg'], 1 - lam)
                    if 'exp_avg_sq' in state:
                        if 'exp_avg_sq' not in mo.state[olist['params'][i]]:
                            mo.state[olist['params'][i]]['exp_avg_sq'] = torch.zeros_like(mo.state[olist['params'][i]]['exp_avg'])
                        mo.state[olist['params'][i]]['exp_avg_sq'].lerp_(state['exp_avg_sq'], 1 - lam)
                    if 'momentum_buffer' in state:
                        mo.state[olist['params'][i]]['momentum_buffer'].lerp_(state['momentum_buffer'], 1 - lam)
                    if 'second_momentum_buffer' in state:
                        mo.state[olist['params'][i]]['second_momentum_buffer'].lerp_(state['second_momentum_buffer'], 1 - lam)
            return
