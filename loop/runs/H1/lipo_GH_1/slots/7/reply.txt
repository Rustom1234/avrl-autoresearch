    mols = [Chem.MolFromSmiles(s) for s in smiles]
    gen2 = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    gen3 = rdFingerprintGenerator.GetMorganGenerator(radius=3, fpSize=N_BITS)
    return np.array([np.concatenate([gen2.GetCountFingerprintAsNumPy(m), gen3.GetCountFingerprintAsNumPy(m)])
