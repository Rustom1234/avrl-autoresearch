    def obj_penalty(v):
        c = v[:2 * n].reshape(n, 2)
        r = v[2 * n:]
        # Use squared distance to avoid sqrt in the main objective gradient
        dist_sq = np.sum((c[:, None, :2] - c[None, :, :2])**2, axis=-1)
        dist = np.sqrt(np.maximum(dist_sq, 1e-9))
        overlap = r[:, None] + r[None, :] - dist
        np.fill_diagonal(overlap, 0)
        penalty = np.sum(np.maximum(0, overlap)**2)
        return -np.sum(r) + 100 * penalty
