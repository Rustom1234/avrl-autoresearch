def _solve_once(n, rng):
    # Greedy placement: place circles one by one as large as possible
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        # Try a few random spots to find the best local spot for the i-th circle
        best_r = -1
        best_pos = np.zeros(2)
        for _ in range(20):
            pos = rng.uniform(INIT_LOW, 1 - INIT_LOW, 2)
            r_limit = min(pos[0], 1 - pos[0], pos[1], 1 - pos[1])
            for j in range(i):
                dist = np.linalg.norm(pos - centers[j])
                r_limit = min(r_limit, dist - radii[j])
            if r_limit > best_r:
                best_r = r_limit
                best_pos = pos
        centers[i] = best_pos
        radii[i] = max(best_r, 0.01)

    x0 = np.concatenate([centers.flatten(), radii])
    iu = np.triu_indices(n, 1)
