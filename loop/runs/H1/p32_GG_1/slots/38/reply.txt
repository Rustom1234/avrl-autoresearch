    # Use a Halton-like sequence for low-discrepancy initialization
    # This distributes points more uniformly than a grid or pure random
    def halton_sequence(count, dim=2):
        pts = np.zeros((count, dim))
        for i in range(count):
            for d in range(dim):
                r = i + 1
                f = 1
                base = 2**d
                while r % base != 0:
                    f /= base
                    r -= base
                pts[i, d] = f * (r / base)
        return pts
