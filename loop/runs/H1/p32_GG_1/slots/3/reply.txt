def _solve_once(n, rng):
    # Area-based initialization: distribute radii to cover ~80% of area
    # Target area A = 0.8, Sum(pi * r_i^2) = 0.8
    # Use a power-law-like distribution: r_i = k * i^(-1/2)
    # To make it stochastic, we shuffle the ranks.
    indices = rng.permutation(n)
    radii = (np.arange(1, n + 1) ** (-0.5))
    # Scale radii so that sum(pi * r_i^2) is approx 0.8
    current_area = np.sum(np.pi * (radii**2))
    radii = radii * np.sqrt(0.8 / current_area)
    
    # Clamp to ensure they are reasonable
    radii = np.clip(radii, 0.01, 0.5)
