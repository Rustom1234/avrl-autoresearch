def _repair(c, r):
    """Shrink radii until the packing is valid, so that the harness never sees an overlap."""
    n = len(r)
    c = c.copy()
    r = r.copy()
    r = np.minimum(r, np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1])))
    r = np.maximum(r, 0)
    
    for _ in range(100):
        dist_sq = np.sum((c[:, None, :] - c[None, :, :])**2, axis=2)
        d = np.sqrt(dist_sq)
        over = r[:, None] + r[None, :] - d
        np.fill_diagonal(over, -1)
        
        max_over = np.max(over)
        if max_over <= 0:
            break
            
        i, j = np.unravel_index(np.argmax(over), over.shape)
        
        # Force-directed center adjustment
        diff = c[i] - c[j]
        dist = np.linalg.norm(diff)
        if dist > 1e-9:
            push = (max_over / dist) * 0.5
            c[i] += (diff / dist) * push
            c[j] -= (diff / dist) * push
            
        # Keep in bounds
        c = np.clip(c, 0, 1)
        
        # Shrink radii if still overlapping
        s = d[i, j] / (r[i] + r[j])
        r[i] *= s
        r[j] *= s
        
    return np.maximum(r - SHRINK, 0)
