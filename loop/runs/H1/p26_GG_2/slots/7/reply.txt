def _solve_once(n, rng):
    # Initialize with potential mirror symmetry across x=0.5
    half_n = n // 2
    c_init = np.zeros((n, 2))
    r_init = np.full(n, INIT_RADIUS)
    
    # Create symmetric base for first half
    c_base = rng.uniform(INIT_LOW, 0.5 - INIT_LOW, (half_n, 2))
    c_init[:half_n] = c_base
    c_init[half_n : 2*half_n] = 1.0 - c_base
    
    # Fill any odd remaining slot or remaining slots
    rem = n - 2 * half_n
    if rem > 0:
        c_init[2*half_n:] = rng.uniform(INIT_LOW, 1 - INIT_LOW, (rem, 2))
        
    x0 = np.concatenate([c_init.flatten(), r_init])
