def _repair(c, r):
    """Shrink radii and nudge centers slightly to resolve overlaps."""
    n = len(r)
    c = c.copy()
    r = r.copy()
    r = np.minimum(r, np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1])))
    r = np.maximum(r, 0)
    for _ in range(100):
        d_vec = c[:, None, :] - c[None, :, :]
        d_sq = np.sum(d_vec**2, axis=2)
        d = np.sqrt(np.maximum(d_sq, 1e-12))
        over = r[:, None] + r[None] - d
        np.fill_diagonal(over, -1)
        if np.all(over <= 0):
            break
        i, j = np.unravel_index(np.argmax(over), over.shape)
        dist = d[i, j]
        sum_r = r[i] + r[j]
        if sum_r > 0:
            s = dist / sum_r
            r[i] *= s
            r[j] *= s
            # Nudge centers away from each other
            move = (c[i] - c[j]) * 0.1 * (1 - s)
            c[i] += move
            c[j] -= move
        else:
            r[i] *= 0.99
            r[j] *= 0.99
    return np.maximum(r - SHRINK, 0)
