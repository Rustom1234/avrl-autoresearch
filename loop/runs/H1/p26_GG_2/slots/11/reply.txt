def _solve_once(n, rng):
    centers = []
    radii = []
    # Greedy placement: Place circles one by one as large as possible
    for _ in range(n):
        # Try many random positions for the next circle
        best_r = -1.0
        best_c = None
        for _ in range(50):
            cur_c = rng.uniform(0, 1, 2)
            # Max r constrained by boundaries
            cur_r = min(cur_c[0], cur_c[1], 1 - cur_c[0], 1 - cur_c[1])
            # Max r constrained by existing circles
            if centers:
                dist_sq = np.sum((np.array(centers) - cur_c)**2, axis=1)
                min_dist = np.min(np.sqrt(dist_sq))
                cur_r = min(cur_r, min_dist - radii[0] if len(radii) > 0 else min_dist)
                # More accurately: r1 + r2 <= dist => r2 <= dist - r1
                min_dist_adj = np.min(np.sqrt(dist_sq) - np.array(radii))
                cur_r = min(cur_r, max(0, min_dist_adj))
            
            if cur_r > best_r:
                best_r = cur_r
                best_c = cur_c
        centers.append(best_c)
        radii.append(best_r)

    c0 = np.array(centers)
    r0 = np.array(radii)
    x0 = np.concatenate([c0.flatten(), r0])
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
