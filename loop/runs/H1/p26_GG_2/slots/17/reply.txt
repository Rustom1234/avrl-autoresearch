def _solve_once(n, rng):
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    def run_slsqp(start_x):
        res = minimize(lambda v: -v[2 * n:].sum(), start_x, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                       options={"maxiter": MAX_ITER})
        return res.x

    best_v = run_slsqp(x0)
    c_best, r_best = best_v[:2 * n].reshape(n, 2), best_v[2 * n:]
    
    for _ in range(3):
        # Perturb centers slightly
        std = 0.02
        perturb = rng.normal(0, std, (n, 2))
        x_hop = np.concatenate([c_best + perturb, r_best])
        v_hop = run_slsqp(x_hop)
        if -v_hop[2 * n:].sum() > -best_v[2 * n:].sum():
            best_v = v_hop
            c_best, r_best = v_hop[:2 * n].reshape(n, 2), v_hop[2 * n:]
            
    return c_best, r_best
