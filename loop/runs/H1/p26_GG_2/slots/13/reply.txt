def _solve_once(n, rng):
    # Simplified Poisson Disk Sampling via Jittered Grid
    grid_size = int(np.ceil(np.sqrt(n)))
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(grid_size):
        for j in range(grid_size):
            if idx < n:
                # Center of cell is (i/grid_size, j/grid_size)
                # Jitter within the cell
                centers[idx, 0] = (i + rng.random()) / grid_size
                centers[idx, 1] = (j + rng.random()) / grid_size
                idx += 1
    
    # Ensure centers are within [INIT_LOW, 1 - INIT_LOW]
    centers = np.clip(centers, INIT_LOW, 1 - INIT_LOW)
    x0 = np.concatenate([centers.flatten(), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]

        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
