from scipy.optimize import minimize

# Settings
TIME_FRACTION = 0.8
INIT_LOW = 0.1
MAX_ITER = 200
INIT_RADIUS = 0.05
SHRINK = 1e-9


def _solve_once(n, rng):
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)

    SCALE = 10.0
    def cons(v):
        c, r_scaled = v[:2 * n].reshape(n, 2), v[2 * n:]
        r = r_scaled / SCALE
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -(v[2 * n:] / SCALE).sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:] / SCALE


def _repair(c, r):
    """Shrink radii until the packing is valid, so that the harness never sees an overlap."""
    n = len(r)
    r = np.minimum(r, np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1])))
    r = np.maximum(r, 0)
    for _ in range(50):
        d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
        over = r[:, None] + r[None] - d
        np.fill_diagonal(over, -1)
        i, j = np.unravel_index(np.argmax(over), over.shape)
        if over[i, j] <= 0:
            break
        s = d[i, j] / (r[i] + r[j])
        r[i], r[j] = r[i] * s, r[j] * s
    return np.maximum(r - SHRINK, 0)

def _solve_once(n, rng):
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)

    SCALE = 10.0
    def cons(v):
        c, r_scaled = v[:2 * n].reshape(n, 2), v[2 * n:]
        r = r_scaled / SCALE
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -(v[2 * n:] / SCALE).sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    c, r = res.x[:2 * n].reshape(n, 2), res.x[2 * n:] / SCALE

    # Sequential Linear Programming on Radii
    tr = 0.01
    for _ in range(20):
        # Objective: max sum of radii -> min -sum(r)
        c_lin = -np.ones(n)
        
        # Constraints: A @ r <= b
        # 1. Pairwise: r_i + r_j <= dist_ij
        # 2. Walls: r_i <= c_i_x, r_i <= 1-c_i_x, etc.
        A = []
        b = []
        
        # Pairwise constraints
        for i in range(n):
            for j in range(i + 1, n):
                d_ij = np.linalg.norm(c[i] - c[j])
                row = np.zeros(n)
                row[i] = 1
                row[j] = 1
                A.append(row)
                b.append(d_ij)
        
        # Wall constraints
        for i in range(n):
            r_row1 = np.zeros(n)
            r_row1[i] = 1
            A.append(r_row1)
            b.append(min(c[i, 0], c[i, 1], 1 - c[i, 0], 1 - c[i, 1]))
            
        # Bounds (Trust Region)
        low_b = np.maximum(0, r - tr)
        up_b = np.minimum(1.0, r + tr)
        
        res_lp = linprog(c_lin, A_ub=A, b_ub=b, bounds=(low_b, up_b), method="highs")
        
        if res_lp.success and np.sum(res_lp.x) > np.sum(r) - 1e-12:
            r = res_lp.x
            tr *= 0.8
        else:
            tr *= 0.5
            if tr < 1e-9: break

    return c, r

from scipy.optimize import minimize, linprog

# Settings
TIME_FRACTION = 0.8
INIT_LOW = 0.1
MAX_ITER = 200
INIT_RADIUS = 0.05
SHRINK = 1e-9


def _solve_once(n, rng):
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)

    SCALE = 10.0
    def cons(v):
        c, r_scaled = v[:2 * n].reshape(n, 2), v[2 * n:]
        r = r_scaled / SCALE
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -(v[2 * n:] / SCALE).sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    c, r = res.x[:2 * n].reshape(n, 2), res.x[2 * n:] / SCALE

    # Sequential Linear Programming on Radii
    tr = 0.01
    for _ in range(15):
        A_list = []
        b_list = []
        for i in range(n):
            # Wall constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
            for k in range(2):
                row = np.zeros(n)
                row[i] = 1
                A_list.append(row)
                b_list.append(min(c[i, k], 1 - c[i, k]))
        
        for i in range(n):
            for j in range(i + 1, n):
                d_ij = np.linalg.norm(c[i] - c[j])
                row = np.zeros(n)
                row[i] = 1
                row[j] = 1
                A_list.append(row)
                b_list.append(d_ij)
        
        A = np.array(A_list)
        b = np.array(b_list)
        
        # Objective: min -sum(r)
        c_lin = -np.ones(n)
        # Bounds on r: [r_k - tr, r_k + tr] and r >= 0
        bounds = [(max(0, r[i] - tr), r[i] + tr) for i in range(n)]
        
        sol = linprog(c_lin, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not sol.success: break
        
        new_r = sol.x
        # Check feasibility
        feasible = True
        for i in range(n):
            if new_r[i] > c[i, 0] + 1e-9 or new_r[i] > (1 - c[i, 0]) + 1e-9 or 
               new_r[i] > c[i, 1] + 1e-9 or new_r[i] > (1 - c[i, 1]) + 1e-9:
                feasible = False; break
            for j in range(i + 1, n):
                if new_r[i] + new_r[j] > np.linalg.norm(c[i] - c[j]) + 1e-9:
                    feasible = False; break
            if not feasible: break
        
        if feasible:
            r = new_r
            if np.linalg.norm(r - new_r) < 1e-12: break
        else:
            tr *= 0.5
            if tr < 1e-12: break

    return c, r
