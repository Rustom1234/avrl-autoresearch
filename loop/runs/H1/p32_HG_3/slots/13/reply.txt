def _solve_once(n, rng):
    # Latin Hypercube Sampling for centers
    l_low = INIT_LOW
    l_high = 1 - INIT_LOW
    q1 = rng.random((n, 2)) * (l_high - l_low) + l_low
    q2 = (rng.random((n, 2)) * (l_high - l_low) + l_low).T
    # Shuffle rows of q2 to break correlation
    q2 = q2[rng.permutation(n)]
    centers = np.concatenate([q1, q2], axis=1)
    x0 = np.concatenate([centers.flatten(), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)
