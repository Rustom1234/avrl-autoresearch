def _solve_once(n, rng):
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    
    def get_constraints(v, current_c):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        # Find pairs close enough to warrant a constraint (threshold 0.4)
        dist_sq = np.sum((current_c[:, None] - current_c[None, :])**2, axis=2)
        mask = (dist_sq < 0.16) & (dist_sq > 0) & np.triu(np.ones((n, n), dtype=bool), k=1)
        iu = np.where(mask)[0]
        if len(iu) == 0: return None
        
        d = np.sqrt(dist_sq[iu[0], iu[1]]) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    # First pass: use sparse neighbors
    initial_c = x0[:2*n].reshape(n, 2)
    cons_func = get_constraints(x0, initial_c)
    
    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", 
                   constraints={"type": "ineq", "fun": cons_func} if cons_func else None,
                   options={"maxiter": MAX_ITER})
