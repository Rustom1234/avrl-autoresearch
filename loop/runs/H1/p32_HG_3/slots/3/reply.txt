def _solve_once(n, rng):
    if rng.random() < 0.5:
        # Hexagonal lattice start
        # Standard hex spacing: side length s, distance between centers s*sqrt(3)
        s = 1.0 / (np.sqrt(3/2) * np.sqrt(n) + 0.1)
        # Generate indices and apply small jitter
        indices = []
        row = 0
        curr_c = 0
        while len(indices) < n:
            x = curr_c * s
            y = row * s * (np.sqrt(3)/2)
            if 0 <= x <= 1 and 0 <= y <= 1:
                indices.append([x, y])
            curr_c += 1
            if curr_c % 3 == 0: row += 1
        
        pts = np.array(indices[:n])
        pts += rng.normal(0, s * 0.1, size=(n, 2))
        c = np.clip(pts, INIT_LOW, 1 - INIT_LOW)
        r = np.full(n, INIT_RADIUS)
        x0 = np.concatenate([c.flatten(), r])
    else:
