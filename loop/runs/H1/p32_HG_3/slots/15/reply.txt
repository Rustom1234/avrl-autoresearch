    def cons_jac(v):
        c = v[:2 * n].reshape(n, 2)
        r_scaled = v[2 * n:]
        r = r_scaled / SCALE
        # Pairwise distances: sqrt(dx^2 + dy^2) - r1 - r2
        # Indices for the pairs
        idx1 = iu[0]
        idx2 = iu[1]
        diff = c[idx1] - c[idx2]
        dist_sq = np.sum(diff**2, axis=1)
        dist = np.sqrt(dist_sq)
        
        # Jacobian for distance constraints (d)
        # Grad w.r.t c[idx1]: (diff / dist) - (grad w.r.t c[idx2] with signs flipped)
        # Grad w.r.t r[idx1]: -1/SCALE, w.r.t r[idx2]: -1/SCALE
        jac = np.zeros((5 * n + len(iu[0]), 3 * n))
        
        # Partial derivatives for d
        # w.r.t centers (2*n variables)
        # c[idx1]: (diff / dist) * [1, 1] * [idx1]
        # c[idx2]: (-diff / dist) * [1, 1] * [idx2]
        # r[idx1]: -1/SCALE * [idx1]
        # r[idx2]: -1/SCALE * [idx2]
        
        for i in range(len(iu[0])):
            i1, i2 = iu[0][i], iu[1][i]
            d_val = dist[i]
            if d_val > 1e-12:
                grad_c = diff[i] / d_val
                jac[i, 2 * i1] += grad_c[0]
                jac[i, 2 * i1 + 1] += grad_c[1]
                jac[i, 2 * i2] -= grad_c[0]
                jac[i, 2 * i2 + 1] -= grad_c[1]
            
            jac[i, 2 * n + i1] -= 1.0 / SCALE
            jac[i, 2 * n + i2] -= 1.0 / SCALE
            
        # Wall constraints: c0 - r, c1 - r, 1 - c0 - r, 1 - c1 - r
        # Jacobians for variables 2*n to 4*n-1 (centers)
        # Jacobian for variables 4*n to 6*n-1 (radii)
        for i in range(n):
            # c0 - r -> grad: [-1, 0] for centers, [1] for radii
            jac[2 * n + i, 2 * i] -= 1
            jac[2 * n + i, 2 * n + i] += 1 / SCALE
            # c1 - r -> grad: [0, -1] for centers, [1] for radii
            jac[2 * n + i + n, 2 * i + 1] -= 1
            jac[2 * n + i + n, 2 * n + i] += 1 / SCALE
            # 1 - c0 - r -> grad: [1, 0] for centers, [-1] for radii
            jac[2 * n + i + 2 * n, 2 * i] += 1
            jac[2 * n + i + 2 * n, 2 * n + i] -= 1 / SCALE
            # 1 - c1 - r -> grad: [0, 1] for centers, [-1] for radii
            jac[2 * n + i + 3 * n, 2 * i + 1] += 1
            jac[2 * n + i + 3 * n, 2 * n + i] -= 1 / SCALE
            
        # Radius inequalities: r >= 0
        for i in range(n):
            jac[4 * n + i, 2 * n + i] += 1
            
        return jac

    SCALE = 10.0
    def cons(v):
        c, r_scaled = v[:2 * n].reshape(n, 2), v[2 * n:]
        r = r_scaled / SCALE
        d_vec = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1))
        d = d_vec - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    def cons_jac(v):
        c, r_scaled = v[:2 * n].reshape(n, 2), v[2 * n:]
        r = r_scaled / SCALE
        jac = np.zeros((len(cons(v)), 3 * n))
        # Distance constraints
        # d = sqrt((x1-x2)^2 + (y1-y2)^2) - r1 - r2
        dx = c[iu[0]] - c[iu[1]]
        dy = c[iu[0]] - c[iu[1]]
        # Note: The indices are rearranged/handled by the loop
        # To keep it simple and robust, we iterate over the pairs in iu
        for k in range(len(iu[0]]):
            i, j = iu[0][k], iu[1][k]
            dist = np.sqrt((c[i,0]-c[j,0])**2 + (c[i,1]-c[j,1])**2)
            if dist < 1e-12: dist = 1e-12
            # Grad w.r.t c[i] (x, y)
            jac[k, i*3:i*3+2] += (1.0/dist) * np.array([(c[i,0]-c[j,0]), (c[i,1]-c[j,1])])
            # Grad w.r.t c[j] (x, y)
            jac[k, j*3:j*3+2] -= (1.0/dist) * np.array([(c[i,0]-c[j,0]), (c[i,1]-c[j,1])])
            # Grad w.r.t r[i], r[j]
            jac[k, i*3+2] -= 1.0 / SCALE
            jac[k, j*3+2] -= 1.0 / SCALE
            
        # Wall constraints: c_x - r >= 0, c_y - r >= 0, 1-c_x-r >= 0, 1-c_y-r >= 0
        # Indices: 2n to 2n + 4n - 1
        for i in range(n):
            # c_x - r >= 0
            jac[len(iu[0])+i, i*3] = 1.0
            jac[len(iu[0])+i, i*3+2] = -1.0 / SCALE
            # c_y - r >= 0
            jac[len(iu[0])+n+i, i*3+1] = 1.0
            jac[len(iu[0])+n+i, i*3+2] = -1.0 / SCALE
            # 1 - c_x - r >= 0
            jac[len(iu[0])+2*n+i, i*3] = -1.0
            jac[len(iu[0])+2*n+i, i*3+2] = -1.0 / SCALE
            # 1 - c_y - r >= 0
            jac[len(iu[0])+3*n+i, i*3+1] = -1.0
            jac[len(iu[0])+3*n+i, i*3+2] = -1.0 / SCALE
            
        # Radius >= 0
        for i in range(n):
            jac[len(iu[0])+4*n+i, i*3+2] = 1.0
        return jac

    def obj_grad(v):
        grad = np.zeros(3 * n)
        # Objective is -(v[2n:] / SCALE).sum(), so grad is 1/SCALE for radii
        grad[2*n:] = 1.0 / SCALE
        return grad

    res = minimize(lambda v: -(v[2 * n:] / SCALE).sum(), x0, method="SLSQP", 
                    constraintsMASS={"type": "ineq", "fun": cons, "jac": cons_jac},

    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:] / SCALE
