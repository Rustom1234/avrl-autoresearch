    # Greedy max-empty-circle initialization
    num_candidates = 1000
    candidates = rng.uniform(0, 1, (num_candidates, 2))
    # Shuffle candidates to vary the greedy path across restarts
    indices = np.arange(num_candidates)
    rng.shuffle(indices)
    candidates = candidates[indices]

    centers = np.empty((n, 2))
    radii = np.zeros(n)
    
    # We use a subset of candidates to build the initial packing
    # to ensure we don't pick the same set every time
    subset_size = min(n, num_candidates)
    used_indices = [] 
    
    # We need to pick n centers greedily
    # To keep it fast, we check against centers already placed
    current_centers = []
    for i in range(subset_size):
        best_dist = -1.0
        best_idx = -1
        for j, cand in enumerate(candidates):
            # Distance to boundary
            dist_bound = min(cand[0], cand[1], 1 - cand[0], 1 - cand[1])
            # Distance to existing circles
            dist_existing = 1e9
            if current_centers:
                # This is a simplification: just distance to centers
                # A better way is dist to circle boundary, but centers are okay for init\n                dists = np.linalg.norm(np.array(current_centers) - cand, axis=1)
                dist_existing = np.min(dists)

            if d > best_dist:
                best_dist = d
                best_idx = j
        
        current_centers.append(candidates[best_idx])
        centers[i] = candidates[best_idx]
        radii[i] = best_dist * 0.5

def _solve_once(n, rng):
    # Greedy max-empty-circle initialization
    num_candidates = 600
    candidates = rng.uniform(0, 1, (num_candidates, 2))
    
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    current_centers = []
    for i in range(n):
        dist_to_existing = np.full(num_candidates, 1e9)
        if current_centers:
            current_centers = np.array(current_centers)
            dist_to_existing = np.min(np.linalg.norm(current_centers[:, None] - candidates, axis=2), axis=0)
        
        dist_to_bound = np.minimum.reduce([candidates[:, 0], candidates[:, 1], 1 - candidates[:, 0], 1 - candidates[:, 1]], axis=0)
        d = np.minimum(dist_to_bound, dist_to_existing)
        
        best_idx = rng.choice(np.where(d == np.max(d))[0])
        best_pos = candidates[best_idx]
        
        centers[i] = best_pos
        radii[i] = d[best_idx] * 0.5
        current_centers.append(best_pos)

    x0 = np.concatenate([centers, radii])
    iu = np.triu_indices(n, 1)

    SCALE = 10.0
    def cons(v):
        c, r_scaled = v[:2 * n].reshape(n, 2), v[2 * n:]
        r = r_scaled / SCALE
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -(v[2 * n:] / SCALE).sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:] / SCALE
