def featurize(smiles, variance_mask=None):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    features = []
    for s in smiles:
        mol = Chem.MolFromSmiles(s)
        if mol is None: 
            features.append(np.zeros(N_BITS + 4, dtype=np.float32))
            continue
        fp = gen.GetFingerprintAsNumPy(mol)
        descriptors = np.array([ 
            Descriptors.MolWt(mol),
            Descriptors.MolLogP(mol),
            Descriptors.NumHDonors(mol),
            Descriptors.NumHAcceptors(mol)
        ], dtype=np.float32)
        features.append(np.concatenate([fp, descriptors]))
    
    X = np.array(features, dtype=np.float32)
    if variance_mask is not None:
        # Only keep bits that are not constant, then append descriptors
        # The variance_mask is for the first N_BITS
        fp_part = X[:, :N_BITS]
        desc_part = X[:, N_BITS:]
        
        # variance_mask is a boolean array of shape (N_BITS,)
        # We want to keep columns where variance > 0
        keep_indices = np.where(variance_mask)[0]
        X = np.hstack([fp_part[:, keep_indices], desc_part])
    
    return X

def fit_predict(train_smiles, train_y, test_smiles, seed):
    X_train = featurize(train_smiles)
    
    # Feature Selection: Identify bits with non-zero variance
    # We only care about the fingerprint bits (first N_BITS)
    fp_bits = X_train[:, :N_BITS]
    variances = np.var(fp_bits, axis=0)
    variance_mask = variances > 0
    
    X_train_sel = featurize(train_smiles, variance_mask=variance_mask)
    X_test_sel = featurize(test_smiles, variance_mask=variance_mask)
    
    y_train = np.asarray(train_y, dtype=np.float64)
    
    train_data = lgb.Dataset(X_train_sel, label=y_train)
    
    params = {
        'objective': 'regression',
        'metric': 'mae',
        'boosting_type': 'gbdt',
        'learning_rate': LEARNING_RATE,
        'num_leaves': NUM_LEAVES,
        'verbosity': -1,
        'seed': seed
    }
    
    model = lgb.train(
        params,
        train_data,
        num_boost_round=N_ESTIMATORS
    )
    
    return model.predict(X_test_sel)
