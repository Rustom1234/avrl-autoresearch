def featurize(smiles):

    mols = [Chem.MolFromSmiles(s) for s in smiles]
    fps = [gen.GetFingerprintAsNumPy(m) for m in mols]
    mws = np.array([Chem.MolToMolWt(m) for m in mols], dtype=np.float32).reshape(-1, 1)
    return np.hstack([np.array(fps), mws], dtype=np.float32)
