def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    temp = 1.0
    cooling_rate = 0.995
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        
        current_sum = r.sum()
        if best is None or current_sum > best[1].sum():
            best = (c, r)
        else:
            if rng.random() < np.exp((current_sum - best[1].sum()) / temp):
                best = (c, r)
        
        temp *= cooling_rate
        if temp < 0.01: temp = 1.0
    return best
