def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    
    # Phase 1: Random restarts for 30% of the time
    while time.process_time() - t0 < 0.3 * time_limit:
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)

    # Phase 2: Local refinements for the remaining 70% of the time
    if best is not None:
        c_best, r_best = best
        x0_best = np.concatenate([c_best, r_best])
        while time.process_time() - t0 < time_limit:
            # Apply small jitter and run a quick optimization
            jitter = rng.normal(0, 1e-4, x0_best.shape)
            res = minimize(lambda v: -v[2 * n:].sum(), x0_best + jitter, 
                            method="SLSQP", 
                            constraints={"type": "ineq", "fun": _cons_func(n)}, 
                            options={"maxiter": 50})
            
            c_new = np.clip(res.x[:2 * n].reshape(n, 2), 0, 1)
            r_new = _repair(c_new, res.x[2 * n:])
            
            if r_new.sum() > best[1].sum():
                best = (c_new, r_new)
                x0_best = np.concatenate([c_new, r_new])
    return best

def _cons_func(n):
    iu = np.triu_indices(n, 1)
    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])
    return cons
