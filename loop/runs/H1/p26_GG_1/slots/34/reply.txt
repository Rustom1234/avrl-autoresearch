    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},

    v = res.x
    # Local polish: small gradient steps on tightest constraints
    for _ in range(5):
        c_all = v[:2 * n].reshape(n, 2)
        r_all = v[2 * n:]
        iu = np.triu_indices(n, 1)
        dist_vec = np.sqrt(((c_all[iu[0]] - c_all[iu[1]]) ** 2).sum(1))
        gaps = dist_vec - r_all[iu[0]] - r_all[iu[1]]
        # Focus on smallest non-negative gaps
        tight_idx = np.where(gaps < 0.01)[0]
        if len(tight_idx) == 0: break
        min_idx = tight_idx[np.argmin(gaps[tight_idx])]
        i, j = iu[0][min_idx], iu[1][min_idx]
        # Simple push-away step for the two circles involved in the tightest constraint
        diff = c_all[i] - c_all[j]
        norm = np.linalg.norm(diff)
        if norm > 1e-9:
            push = (diff / norm) * (0.001 * (-(dist_vec[min_idx] - r_all[i] - r_all[j])))
            v[i*2 : i*2+2] += push
            v[j*2 : j*2+2] -= push
    return v[:2 * n].reshape(n, 2), v[2 * n:]
