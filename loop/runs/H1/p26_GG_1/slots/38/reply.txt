def _solve_once(n, rng):
    # Physics-based initialization
    c = rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
    r = rng.uniform(INIT_RADIUS, 0.15, n)
    v = rng.standard_normal(n, 2) * 0.01
    
    for _ in range(100):
        dists = np.linalg.norm(c[:, None, :] - c[None, :, :], axis=2)
        # Repulsion force
        forces = np.zeros_like(c)
        for i in range(n):
            # Repel from boundaries
            forces[i, 0] += (0.5 - c[i, 0]) * (r[i] / 0.5)
            forces[i, 1] += (0.5 - c[i, 1]) * (r[i] / 0.5)
            
            # Repel from others
            diff = c[i] - c
            d_vec = dists[i] - r[i] - r
            mask = (d_vec < 0) & (np.arange(n) != i)
            if np.any(mask):
                active_diff = diff[mask] / (d_vec[mask, None] + 1e-6)
                forces[i] += np.sum(active_diff * 0.5, axis=0)
        
        v += forces
        v *= 0.9
        c += v
        # Keep in bounds
        c = np.clip(c, r + 0.001, 1 - r - 0.001)

    x0 = np.concatenate([c.flatten(), r])
    iu = np.triu_indices(n, 1)

    def cons(v):
        c_v, r_v = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c_v[iu[0]] - c_v[iu[1]]) ** 2).sum(1)) - r_v[iu[0]] - r_v[iu[1]]
        return np.concatenate([d, c_v[:, 0] - r_v, c_v[:, 1] - r_v, 1 - c_v[:, 0] - r_v, 1 - c_v[:, 1] - r_v, r_v])

    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
