def solve(n, seed, time_limit):
    rng = np.random.default_rng(seed)
    best = None
    # Use a fixed number of restarts to balance exploration and exploitation\n    # 10 restarts * 30s = 300s, but we only have 30s total. 10 * 3s = 30s.
    # We use a count that fits within the time_limit (e.g., 8 restarts).
    num_restarts = 8
    for _ in range(num_restarts):
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    return best
