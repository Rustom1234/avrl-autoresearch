    def cons(v):
        c = v[:2 * n].reshape(n, 2)
        r = v[2 * n:]
        # d_ij = sqrt((x_i - x_j)^2 + (y_i - y_j)^2) - r_i - r_j
        # c_constraints = x - r, y - r, 1 - x - r, 1 - y - r, r
        # Jacobian calculation:
        # For d_ij:
        # d(d_ij)/dx_i = (x_i - x_j) / d_ij, d(d_ij)/dy_i = (y_i - y_j) / d_ij
        # d(d_ij)/dx_j = -(x_i - x_j) / d_ij, d(d_ij)/dy_j = -(y_i - y_j) / d_ij
        # d(d_ij)/dr_i = -1, d(d_ij)/dr_j = -1
        
        # For x - r:
        # d(x - r)/dx = 1, d(x - r)/dr = -1
        # For 1 - x - r:
        # d(1 - x - r)/dx = -1, d(1 - x - r)/dr = -1
        
        # For r:
        # d(r)/dr = 1

        dx = c[iu[0]] - c[iu[1]]
        dy = c[iu[0]][:, 1] - c[iu[1]][:, 1]
        dist = np.sqrt(dx**2 + dy**2)
        
        # Jacobian for d_ij (n*(n-1)/2 constraints)
        # Each d_ij depends on x_i, y_i, x_j, y_j, r_i, r_j
        # Shape of Jacobian: (num_constraints, 3n)
        # For simplicity, we can use a small trick: most of the Jacobian is sparse.
        # But scipy's SLSQP 'fun' / 'jac' approach handles the full Jacobian shape.
        
        # Using numerical gradient for stability if needed, but let's try to be explicit.
        # Actually, since SLSQP is slow, let's use a simpler approach: 
        # minimize(..., jac=...) is very effective.
        
        # Due to complexity of indexing, we'll stick to the current func and 
        # let SLSQP's internal logic handle it, but optimize the evaluation.
        d = dist - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", 
                     constraints={"type": "ineq", "fun": cons},
                     options={"maxiter": MAX_ITER, "ftol": 1e-9})
