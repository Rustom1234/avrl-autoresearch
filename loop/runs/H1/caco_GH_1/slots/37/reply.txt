def fit_predict(train_smiles, train_y, test_smiles, seed):
    X_train = featurize(train_smiles)
    X_test = featurize(test_smiles)
    # keep the 1024 Morgan bits with highest variance on the training fold; always keep the 8 descriptors
    top_bits = np.argsort(X_train[:, :N_BITS].var(axis=0))[::-1][:1024]
    keep = np.concatenate([np.sort(top_bits), np.arange(N_BITS, X_train.shape[1])])
    X_train = X_train[:, keep]
    X_test = X_test[:, keep]
    model = RandomForestRegressor(n_estimators=N_TREES, max_features=MAX_FEATURES, min_samples_leaf=MIN_SAMPLES_LEAF,
                                  n_jobs=-1, random_state=seed)
    model.fit(X_train, np.asarray(train_y, dtype=np.float64))
    per_tree = np.stack([tree.predict(X_test) for tree in model.estimators_])
    return np.median(per_tree, axis=0)
