def solve(n, seed, time_limit):
    rng, t0, best, cur, start = np.random.default_rng(seed), time.process_time(), None, None, None
    budget = TIME_FRACTION * time_limit
    while time.process_time() - t0 < budget:   # restarts until the time is used
        temp = TEMP0 * (1 - (time.process_time() - t0) / budget)   # cooling temperature
        c, r = _solve_once(n, rng, start)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
        # Metropolis acceptance: always keep improvements, sometimes keep slightly worse packings
        if cur is None or r.sum() > cur[1].sum() or (temp > 0 and rng.random() < np.exp((r.sum() - cur[1].sum()) / temp)):
            cur = (c, r)
        start = cur[0] if rng.random() < 0.5 else None   # restart from the current centres half of the time
    return best
