def _poisson_init(n, rng, tries=5000):
    """Dart-throwing Poisson disc sampling: candidates closer than mind to an accepted point are rejected."""
    mind, pts = 0.5 / np.sqrt(n), []
    for _ in range(tries):
        if len(pts) == n:
            break
        p = rng.uniform(INIT_LOW, 1 - INIT_LOW, 2)
        if all(np.hypot(*(p - q)) >= mind for q in pts):
            pts.append(p)
    while len(pts) < n:
        pts.append(rng.uniform(INIT_LOW, 1 - INIT_LOW, 2))
    return np.array(pts)
