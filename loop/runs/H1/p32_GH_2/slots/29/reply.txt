def solve(n, seed, time_limit):
    rng, t0, best, start, shrink = np.random.default_rng(seed), time.process_time(), None, None, False
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # restarts until the time is used
        if shrink:   # previous solve did not improve: relax the best packing and grow again
            c, r = _solve_once(n, rng, best[0], 0.9 * best[1])
        else:
            c, r = _solve_once(n, rng, start)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        improved = best is None or r.sum() > best[1].sum() + 1e-4
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
        shrink = not improved
        start = best[0] if rng.random() < 0.5 else None   # reuse the best centres half of the time
    return best
