def _radii_lp(c, r):
    """Phase 2: fix the centres and re-optimise the radii by linear programming, then repair."""
    n = len(r)
    iu = np.triu_indices(n, 1)
    m = len(iu[0])
    d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1))
    A = np.zeros((m + n, n))
    A[np.arange(m), iu[0]] = 1
    A[np.arange(m), iu[1]] = 1
    A[m:] = np.eye(n)
    b = np.concatenate([d, np.minimum.reduce([c[:, 0], c[:, 1], 1 - c[:, 0], 1 - c[:, 1]])])
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None), method="highs")
    return _repair(c, res.x) if res.success else r
