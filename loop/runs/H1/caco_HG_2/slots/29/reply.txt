from sklearn.linear_model import Ridge
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import KFold

# Settings
N_BITS = 2048
RADIUS = 2
N_ESTIMATORS = 700
LEARNING_RATE = 0.03

def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(
        radius=RADIUS, 
        fpSize=N_BITS, 
        atomInvariantsGenerator=rdFingerprintGenerator.GetMorganFeatureAtomInvGen()
    )
    mols = [Chem.MolFromSmiles(s) for s in smiles]
    # Morgan fingerprints
    fps = np.array([gen.GetFingerprintAsNumPy(m) for m in mols], dtype=np.float32)
    # RDKit descriptors
    desc = np.array([[Descriptors.MolWt(m), Descriptors.MolLogP(m), Descriptors.NumHDonors(m), 
                               Descriptors.NumHAcceptors(m), Descriptors.TPSA(m)] 
                      for m in mols], dtype=np.float32)
    return np.hstack([fps, desc])


def fit_predict(train_smiles, train_y, test_smiles, seed):
    X_all = featurize(train_smiles)
    y_all = np.asarray(train_y, dtype=np.float64)
    
    # Split features for Ridge
    X_desc = X_all[:, -5:]
    
    # Ridge on descriptors
    scaler = StandardScaler()
    X_desc_scaled = scaler.fit_transform(X_desc)
    
    # Out-of-fold residuals for XGB
    kf = KFold(n_splits=5, shuffle=True, random_state=seed)
    oof_res = np.zeros_like(y_all)

    ridge = Ridge(alpha=1.0)
    for tr_idx, val_idx in kf.split(X_desc_scaled):
        X_train_r = X_desc_scaled[tr_idx]
        X_val_r = X_desc_scaled[val_idx]
        y_train_r = y_all[tr_idx]
        
        ridge.fit(X_train_r, y_train_r)
        oof_res[val_idx] = y_all[val_idx] - ridge.predict(X_val_r)
    
    # XGB on fingerprints + descriptors to predict residuals
    xgb_model = XGBRegressor(n_estimators=N_ESTIMATORS, learning_rate=LEARNING_RATE, 
                               max_depth=6, subsample=0.8, colsample_bytree=0.5, 
                               objective='reg:absoluteerror', n_jobs=-1, random_state=seed)
    xgb_model.fit(X_all, oof_res)
    
    # Final Prediction
    X_test = featurize(test_smiles)
    X_test_desc = X_test[:, -5:]
    ridge_pred = ridge.predict(scaler.transform(X_test_desc))
    xgb_res_pred = xgb_model.predict(X_test)
    
    return ridge_pred + xgb_res_pred

from rdkit.Chem import Descriptors
from sklearn.ensemble import RandomForestRegressor
from xgboost import XGBRegressor
from lightgbm import LGBMRegressor
from sklearn.svm import SVR
from sklearn.linear_model import Ridge
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import KFold
from rdkit import DataStructs

# Settings
N_BITS = 2048
RADIUS = 2
N_ESTIMATORS = 300

def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(
        radius=RADIUS, 
        fpSize=N_BITS, 
        atomInvariantsGenerator=rdFingerprintGenerator.GetMorganFeatureAtomInvGen()
    )
    mols = [Chem.MolFromSmiles(s) for s in smiles]
    fps = np.array([gen.GetFingerprintAsNumPy(m) for m in mols], dtype=np.float32)
    desc = np.array([[Descriptors.MolWt(m), Descriptors.MolLogP(m), Descriptors.NumHDonors(m), 
                               Descriptors.NumHAcceptors(m), Descriptors.TPSA(m)] 
                      for m in mols], dtype=np.float32)
    return np.hstack([fps, desc])

def tanimoto_kernel(x1, x2):
    # x1, x2 are arrays of Morgan fingerprints (N_BITS)
    sims = []
    for i in range(len(x1)):
        # Compute Tanimoto similarity between x1[i] and x2
        # DataStructs.TanimotoSimilarity expects bit vectors
        # Since we use GetFingerprintAsNumPy, we treat them as bits
        score = DataStructs.TanimotoSimilarity(Chem.DataStructs.GetBitVector(x1[i]), 
                                                 Chem.DataStructs.GetBitVector(x2[i]))
        sims.append(score)
    return np.array(sims)

def fit_predict(train_smiles, train_y, test_smiles, seed):
    X_all = featurize(train_smiles)
    y_all = np.asarray(train_y, dtype=np.float64)
    
    # Base Learners
    rf = RandomForestRegressor(n_estimators=N_ESTIMATORS, random_state=seed)
    lgbm = LGBMRegressor(n_estimators=N_ESTIMATORS, learning_rate=0.05, random_state=seed, verbose=-1)
    svr = SVR(kernel='linear') # SVR with Tanimoto is slow, using linear for speed
    
    # Out-of-fold predictions
    kf = KFold(n_splits=5, shuffle=True, random_state=seed)
    oof_preds = np.zeros_like(y_all)
    
    for tr_idx, val_idx in kf.split(X_all):
        X_tr, X_val = X_all[tr_idx], X_all[val_idx]
        y_tr, y_val = y_all[tr_idx], y_all[val_idx]
        
        rf.fit(X_tr, y_tr)
        lgbm.fit(X_tr, y_tr)
        # For SVR, keep it simple to stay within time limits
        svr.fit(X_tr, y_tr)
        
        oof_preds[val_idx, 0] = rf.predict(X_val)
        oof_preds[val_idx, 1] = lgbm.predict(X_val)
        oof_preds[val_idx, 2] = svr.predict(X_val)
    
    # Meta-learner (Ridge)
    meta = Ridge(alpha=1.0)
    meta.fit(oof_preds, y_all)
    
    # Final predictions
    X_test = featurize(test_smiles)
    test_preds = np.zeros((len(test_smiles), 3))
    
    # Refit base learners
    rf.fit(X_all, y_all)
    lgbm.fit(X_all, y_all)
    svr.fit(X_all, y_all)
    
    test_preds[:, 0] = rf.predict(X_test)
    test_preds[:, 1] = lgbm.predict(X_test)
    test_preds[:, 2] = svr.predict(X_test)
    
    return meta.predict(test_preds)

from rdkit.Chem import Descriptors
from sklearn.ensemble import RandomForestRegressor
from xgboost import XGBRegressor
from lightgbm import LGBMRegressor
from sklearn.svm import SVR
from sklearn.linear_model import Ridge
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import KFold
from rdkit import DataStructs

# Settings
N_BITS = 2048
RADIUS = 2
RF_ESTIMATORS = 300
LGBM_ESTIMATORS = 300
LGBM_LR = 0.05

def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(
        radius=RADIUS, 
        fpSize=N_BITS, 
        atomInvariantsGenerator=rdFingerprintGenerator.GetMorganFeatureAtomInvGen()
    )
    mols = [Chem.MolFromSmiles(s) for s in smiles]
    fps = np.array([gen.GetFingerprintAsNumPy(m) for m in mols], dtype=np.float32)
    desc = np.array([[Descriptors.MolWt(m), Descriptors.MolLogP(m), Descriptors.NumHDonors(m), 
                               Descriptors.NumHAcceptors(m), Descriptors.TPSA(m)] 
                      for m in mols], dtype=np.float32)
    return np.hstack([fps, desc])

def fit_predict(train_smiles, train_y, test_smiles, seed):
    X_all = featurize(train_smiles)
    y_all = np.asarray(train_y, dtype=np.float64)
    
    kf = KFold(n_splits=5, shuffle=True, random_state=seed)
    # oof_preds for 3 base learners
    oof_rf = np.zeros(len(y_all))
    oof_lgbm = np.zeros(len(y_all))
    oof_tan = np.zeros(len(y_all))
    
    # 1. RF on Morgan bits
    rf_base = RandomForestRegressor(n_estimators=RF_ESTIMATORS, random_state=seed, n_jobs=-1)
    for tr_idx, val_idx in kf.split(X_all):
        rf_base.fit(X_all[tr_idx], y_all[tr_idx])
        oof_rf[val_idx] = rf_base.predict(X_all[val_idx])
        
    # 2. LGBM on Morgan + Descriptors
    lgbm_base = LGBMRegressor(n_estimators=LGBM_ESTIMATORS, learning_rate=LGBM_LR, random_state=seed, n_jobs=-1)
    for tr_idx, val_idx in kf.split(X_all):
        lgbm_base.fit(X_all[tr_idx], y_all[tr_idx])
        oof_lgbm[val_idx] = lgbm_base.predict(X_all[val_idx])
        
    # 3. SVR with Tanimoto kernel
    def tanimoto_kernel(x1, x2):
        # Similarity for each pair
        sims = []
        for i in range(len(x1)):
            row_sims = []
            for j in range(len(x2)):
                sims.append(DataStructs.TanimotoSimilarity(x1[i], x2[j]))
            row_sims.append(sims)
        # This is inefficient for KFold. Use standard RBF for stability and speed.
        return np.exp(-np.linalg.norm(x1 - x2, axis=1))

    # SVR is slow with custom kernels on large data, use a simpler approach or linearize
    # Given the constraints, let's use a linear SVR or just skip complex kernels to avoid timeout
    # Substituting with a simple Ridge for the 3rd base learner to ensure it completes
    ridge_base = Ridge(alpha=1.0)
    for tr_idx, val_idx in kf.split(X_all):
        ridge_base.fit(X_all[tr_idx], y_all[tr_idx])
        oof_tan[val_idx] = ridge_base.predict(X_all[val_idx])
        
    # Meta-learner
    oof_matrix = np.column_stack([oof_rf, oof_lgbm, oof_tan])
    meta_model = Ridge(alpha=1.0)
    meta_model.fit(oof_matrix, y_all)
    
    # Final Predictions
    X_test = featurize(test_smiles)
    test_rf = rf_base.predict(X_test)
    test_lgbm = lgbm_base.predict(X_test)
    test_tan = ridge_base.predict(X_test)
    
    test_matrix = np.column_stack([test_rf, test_lgbm, test_tan])
    return meta_model.predict(test_matrix)

from sklearn.model_selection import KFold
from rdkit import DataStructs

# Settings
N_BITS = 2048
RADIUS = 2
RF_EST = 300
LGBM_EST = 300
LGBM_LR = 0.05

        radius=RADIUS, 
        fpSize=N_BITS, 
        atomInvariantsGenerator=rdFingerprintGenerator.GetMorganFeatureAtomInvGen()
    )

    # Morgan fingerprints
    fps = np.array([gen.GetFingerprintAsNumPy(m) for m in mols], dtype=np.float32)

    desc = np.array([[Descriptors.MolWt(m), Descriptors.MolLogP(m), Descriptors.NumHDonors(m),

                      for m in mols], dtype=np.float32)
    return np.hstack([fps, desc])


def fit_predict(train_smiles, train_y, test_smiles, seed):
    X_all = featurize(train_smiles)
    y_all = np.asarray(train_y, dtype=np.float64)
    
    kf = KFold(n_splits=5, shuffle=True, random_state=seed)
    
    # Base learners initialization
    rf = RandomForestRegressor(n_estimators=RF_EST, max_depth=15, random_state=seed, n_jobs=-1)
    lgbm = LGBMRegressor(n_estimators=LGBM_EST, learning_rate=LGBM_LR, max_depth=7, 
                            subsample=0.8, colsample_bytree=0.5, importance_type='gain', 
                            n_jobs=-1, random_state=seed)
    
    # To store OOF predictions
    oof_rf = np.zeros(len(y_all))
    oof_lgbm = np.zeros(len(y_all))
    
    # For SVR with a simple dot product (approximating similarity/kernel context)
    # Note: Full Tanimoto kernel is expensive, using dot product as a proxy for simplicity
    # Or just use a smaller subset of descriptors for SVR to avoid memory issues
    
    for tr_idx, val_idx in kf.split(X_all):
        X_tr, X_val = X_all[tr_idx], X_all[val_idx]
        y_tr = y_all[tr_idx]
        
        rf.fit(X_tr, y_tr)
        oof_rf[val_idx] = rf.predict(X_val)
        
        lgbm.fit(X_tr, y_tr)
        oof_lgbm[val_idx] = lgbm.predict(X_val)

    meta_model = Ridge(alpha=1.0)
    meta_model.fit(X_meta, y_all)
    
    # Final training on all data
    rf.fit(X_all, y_all)
    lgbm.fit(X_all, y_all)
    
    # Test predictions
    X_test = featurize(test_smiles)
    pred_rf = rf.predict(X_test)
    pred_lgbm = lgbm.predict(X_test)
    
    X_test_meta = np.column_stack([pred_rf, pred_lgbm])
    return meta_model.predict(X_test_meta)
