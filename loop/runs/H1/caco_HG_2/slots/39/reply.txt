    train_res = y_all - xgb_model.predict(X_all)
    
    # k=5 nearest neighbors correction
    k = 5
    nn_correction = np.zeros(len(test_smiles))
    for i in range(len(test_smiles)):
        # Tanimoto similarity: intersection / union
        # Using bitwise logic for speed
        sims = np.zeros(len(fps_train))
        for j in range(len(fps_train)):
            inter = np.logical_and(fps_test[i], fps_train[j]).sum()
            union = np.logical_or(fps_test[i], fps_train[j]).sum()
            sims[j] = inter / (union + 1e-9)

        top_k_idx = np.argsort(sims)[-k:]
        top_sims = sims[top_k_idx]
        top_res = train_res[top_k_idx]
        
        # Similarity-weighted mean: sum(sim^3 * res) / sum(sim^3)
        weights = top_sims**3
        nn_correction[i] = np.sum(weights * top_res) / (np.sum(weights) + 1e-9) * 0.5
