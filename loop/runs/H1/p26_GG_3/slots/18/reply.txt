def _solve_once(n, rng):
    # Grid-based initialization: find a grid of centers and jitter them
    # We use a square grid as a simple starting point
    side = int(np.ceil(np.sqrt(n)))
    grid_x = np.linspace(0, 1, side)
    grid_y = np.linspace(0, 1, side)
    gx, gy = np.meshgrid(grid_x, grid_y)
    pts = np.stack([gx.flatten(), gy.flatten()], axis=1)[:n]
    
    # Jitter the points slightly to break symmetry and make them unique
    pts += rng.normal(0, 0.02, size=(n, 2))
    pts = np.clip(pts, 0.05, 0.95)
    
    x0 = np.concatenate([pts.flatten(), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
