def _solve_once(n, rng):
    # Initialize radii such that sum of areas is ~0.8
    # We want sum(pi * r_i^2) = 0.8  => sum(r_i^2) = 0.8 / pi 
    target_sum_sq = 0.8 / np.pi
    # Use a power-law distribution to get a mix of small and large radii
    powers = rng.power(0.8, size=n)
    r_raw = np.sqrt(powers * (target_sum_sq / np.sum(powers)))
    
    # Ensure they are at least small but not too large for the container
    r_raw = np.clip(r_raw, 0.01, 0.4)
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), r_raw])
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
