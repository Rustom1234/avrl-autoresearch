def _solve_once(n, rng):
    # Start with a jittered grid to provide a better initial feasible structure
    grid_size = int(np.ceil(np.sqrt(n)))
    x_coords = np.linspace(0.1, 0.9, grid_size)
    y_coords = np.linspace(0.1, 0.9, grid_size)
    grid = np.stack([x_coords, y_coords], axis=-1).reshape(-1, 2)
    indices = np.arange(min(n, grid.shape[0]))
    x0 = np.concatenate([grid[indices, :].flatten(), np.full(n, INIT_RADIUS)])
    x0 += rng.normal(0, 0.02, size=2 * n)
    
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        dist_sq = np.sum((c[iu[0]] - c[iu[1]])**2, axis=1)
        d = np.sqrt(dist_sq) - (r[iu[0]] + r[iu[1]])
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP",
