def _solve_once(n, rng):
    # Hexagonal grid initialization
    pts = []
    # Generate a large enough hex grid
    # Distance between points in hex grid
    d = 0.2
    # Just generate a dense enough set of points and sample n
    for row in range(5):
        for col in range(6):
            x = 0.2 + col * d * 1.0
            y = 0.2 + row * d * (np.sqrt(3)/2)
            pts.append([x, y])
    
    # Shuffle and pick n
    indices = rng.permutation(len(pts))[:n]
    centers = np.array([pts[i] for i in indices])
    # Add jitter
    centers += rng.uniform(-0.03, 0.03, centers.shape)
    # Clamp to unit square
    centers = np.clip(centers, 0.01, 0.99)

    # Estimate initial radii
    # For n=26, a safe start radius to avoid massive overlap
    init_r = np.full(n, 0.04)
    
    x0 = np.concatenate([centers.flatten(), init_r])
    iu = np.triu_indices(n, 1)
