    c, r = _solve_once(n, rng)
    c = np.clip(c, 0, 1)
    r = _repair(c, r)
    
    # Local optimization of radii for fixed centers (greedy expansion)
    for _ in range(5): 
        for i in range(n): 
            # Ensure inside boundary
            r[i] = min(r[i], c[i,0], 1-c[i,0], c[i,1], 1-c[i,1])
            for j in range(i+1, n): 
                d = np.linalg.norm(c[i] - c[j])
                # If they overlap, shrink both to just touch
                if d < (r[i] + r[j]) - 1e-11: 
                    s = d / (r[i] + r[j])
                    r[i] *= s
                    r[j] *= s

    if best is None or r.sum() > best[1].sum():
        best = (c, r)
    return best
