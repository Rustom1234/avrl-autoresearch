def solve(n, seed, time_limit):
    rng, t0 = np.random.default_rng(seed), time.process_time()
    elites = []  # List of (centers, radii, sum_radii)
    
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        # Decide whether to start fresh or perturb an elite
        if len(elites) == 0 or rng.random() < 0.3:
            c, r = _solve_once(n, rng)
            c = np.clip(c, 0, 1)
            r = _repair(c, r)
        else:
            # Pick a random elite and perturb it slightly
            idx = rng.integers(len(elites))
            ec, er, es = elites[idx]
            c = ec + rng.normal(0, 0.02, (n, 2))
            r = er + rng.normal(0, 0.005, n)
            c = np.clip(c, 0, 1)
            r = _repair(c, r)
        
        current_sum = r.sum()
        # Maintain top 5 distinct packings
        elites.append((c, r, current_sum))
        elites.sort(key=lambda x: x[2], reverse=True)
        
        # Keep unique enough solutions (gap > 1e-6)
        unique_elites = [elites[0]]
        for i in range(1, len(elites)):
            if elites[i][2] < unique_elites[-1][2] - 1e-6:
                unique_elites.append(elites[i])
        elites = unique_elites[:5]
        
        if len(elites) > 5:
            elites = elites[:5]
            
    return elites[0][0], elites[0][1]
