def _grow(c, r, rng):
    n = len(r)
    indices = rng.permutation(n)
    for _ in range(2):
        for i in indices:
            # Distance to boundaries
            max_r = min(c[i, 0], c[i, 1], 1 - c[i, 0], 1 - c[i, 1])
            # Distance to other circles
            dists = np.linalg.norm(c - c[i], axis=1)
            # Only consider circles other than i
            mask = np.arange(n) != i
            dists = dists[mask]
            r_others = r[mask]
            max_r = min(max_r, np.min(dists - r_others))
            r[i] = max(r[i], max_r)
    return r

def _grow(c, r, rng):
    n = len(r)
    indices = rng.permutation(n)
    for _ in range(2):
        for i in indices:
            # Max radius allowed by the unit square boundaries
            max_r = min(c[i, 0], c[i, 1], 1 - c[i, 0], 1 - c[i, 1])
            # Max radius allowed by distances to all other circles
            dists = np.linalg.norm(c - c[i], axis=1)
            mask = np.arange(n) != i
            d_others = dists[mask]
            r_others = r[mask]
            max_r = min(max_r, np.min(d_others - r_others))
            r[i] = max(r[i], max_r)
    return r
