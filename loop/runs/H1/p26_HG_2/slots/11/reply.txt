def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    first_restart_time = t0 + 0.2 * time_limit

    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        if time.process_time() - t0 < first_restart_time or best is None:
            c, r = _solve_once(n, rng)
            c = np.clip(c, 0, 1)
            r = _repair(c, r)
            if best is None or r.sum() > best[1].sum():
                best = (c, r)
        else:
            # Basin hopping from best
            c_old, r_old = best
            c = np.clip(c_old + rng.standard_normal(c_old.shape) * 0.03, 0, 1)
            r_start = np.full(n, INIT_RADIUS)
            x0 = np.concatenate([c.flatten(), r_start])
            iu = np.triu_indices(n, 1)

            def cons(v):

                d = np.sqrt(((cv[iu[0]] - cv[iu[1]]) ** 2).sum(1)) - rv[iu[0]] - rv[iu[1]]
                return np.concatenate([d, cv[:, 0] - rv, cv[:, 1] - rv, 1 - cv[:, 0] - rv, 1 - cv[:, 1] - rv, rv])

            res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP",
                             constraints={"type": "ineq", "fun": cons},
                             options={"maxiter": MAX_ITER})
            
            c_new = np.clip(res.x[:2 * n].reshape(n, 2), 0, 1)
            r_new = _repair(c_new, res.x[2 * n:])
            
            if r_new.sum() > best[1].sum():
