from scipy.optimize import linprog

def _repair(c, r):
    """Maximize the sum of radii for the given centers using linear programming."""
    n = len(r)
    # Constraints: 
    # 1. r_i + r_j <= dist(c_i, c_j)
    # 2. r_i <= min(c_i_x, c_i_y, 1-c_i_x, 1-c_i_y)
    
    # Objective: minimize -sum(r_i)
    obj = -np.ones(n)
    
    # Inequality constraints: A @ r <= b
    # Distance constraints (n*(n-1)/2 pairs)
    # Boundary constraints (4*n)
    
    A = []
    b = []
    
    # Pairwise overlaps
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(c[i] - c[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A.append(row)
            b.append(d - 1e-10)
            
    # Boundary constraints
    for i in range(n):
        # r_i <= c_i_x 
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(c[i, 0] - 1e-10)
        # r_i <= c_i_y
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(c[i, 1] - 1e-10)
        # r_i <= 1 - c_i_x
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(1 - c[i, 0] - 1e-10)
        # r_i <= 1 - c_i_y
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(1 - c[i, 1] - 1e-10)

    res = linprog(obj, A_ub=np.array(A), b_ub=np.array(b), bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return r
