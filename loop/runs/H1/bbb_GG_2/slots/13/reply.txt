import lightgbm as lgb

# Settings
N_BITS = 2048
RADIUS = 2
N_TREES = 300
LEARNING_RATE = 0.1
NUM_LEAVES = 31

def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    mols = [Chem.MolFromSmiles(s) for s in smiles]
    fps = np.array([gen.GetFingerprintAsNumPy(m) for m in mols], dtype=np.float32)
    
    desc = np.array([
        [Descriptors.MolWt(m),
         Descriptors.MolLogP(m),
         Descriptors.NumHDonors(m),
         Descriptors.NumHAcceptors(m),
         Descriptors.TPSA(m)] for m in mols], dtype=np.float32)
    
    return np.hstack([fps, desc])


def fit_predict(train_smiles, train_y, test_smiles, seed):
    model = lgb.LGBMClassifier(
        n_estimators=N_TREES,
        learning_rate=LEARNING_RATE,
        num_leaves=NUM_LEAVES,
        n_jobs=-1,
        random_state=seed,
        importance_type='gain'
    )
    model.fit(featurize(train_smiles), np.asarray(train_y, dtype=np.int64))
    return model.predict_proba(featurize(test_smiles))[:, 1]

from catboost import CatBoostClassifier

# Settings
N_BITS = 2048
RADIUS = 2
N_TREES = 1000
LEARNING_RATE = 0.05

def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    mols = [Chem.MolFromSmiles(s) for s in smiles]
    fps = np.array([gen.GetFingerprintAsNumPy(m) for m in mols], dtype=np.float32)
    
    desc = np.array([
        [Descriptors.MolWt(m),
         Descriptors.MolLogP(m),
         Descriptors.NumHDonors(m),
         Descriptors.NumHAcceptors(m),
         Descriptors.TPSA(m)] for m in mols], dtype=np.float32)
    
    return np.hstack([fps, desc])


def fit_predict(train_smiles, train_y, test_smiles, seed):
    X = featurize(train_smiles)
    y = np.asarray(train_y, dtype=np.int64)
    
    # Simple internal split for early stopping
    split = int(len(X) * 0.9)
    X_tr, X_val = X[:split], X[split:]
    y_tr, y_val = y[:split], y[split:]

    model = CatBoostClassifier(
        iterations=N_TREES,
        learning_rate=LEARNING_RATE,
        depth=6,
        loss_function='Logloss',
        random_seed=seed,
        verbose=False
    )
    model.fit(X_tr, y_tr, eval_set=(X_val, y_val), early_stopping_rounds=50, use_best_model=True)
    
    return model.predict_proba(featurize(test_smiles))[:, 1]
