def fit_predict(train_smiles, train_y, test_smiles, seed):
    seeds = [seed, seed + 1, seed + 2, seed + 3, seed + 4]
    probs = []
    train_feats = featurize(train_smiles)
    test_feats = featurize(test_smiles)
    train_y_arr = np.asarray(train_y, dtype=np.int64)
    
    for s in seeds:
        model = lgb.LGBMClassifier(
            n_estimators=N_TREES,
            learning_rate=LEARNING_RATE,
            num_leaves=NUM_LEAVES,
            n_jobs=-1,
            random_state=s,
            importance_type='gain'
        )
        model.fit(train_feats, train_y_arr)
        probs.append(model.predict_proba(test_feats)[:, 1])
        
    return np.mean(probs, axis=0)

            importance_type='gain'
        )

    return np.mean(probs, axis=0)
