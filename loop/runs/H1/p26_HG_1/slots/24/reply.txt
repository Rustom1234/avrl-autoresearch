def _solve_once(n, rng):
    # Hexagonal lattice initialization
    rows = int(np.sqrt(n * 0.75)) + 1
    cols = int(np.ceil(n / rows)) + 1
    # Generate grid
    x = np.linspace(INIT_LOW, 1 - INIT_LOW, cols)
    y = np.linspace(INIT_LOW, 1 - INIT_LOW, rows)
    X, Y = np.meshgrid(x, y)
    points = np.vstack([X.ravel(), Y.ravel()]).T
    # Apply hexagonal shift to every odd row
    points[1::2, 0] += 0.1
    # Jitter and select n
    points += rng.uniform(-0.03, 0.03, (points.shape[0], 2))
    indices = rng.sample(np.arange(points.shape[0]), n)
    c_init = points[indices]
    
    x0 = np.concatenate([c_init.flatten(), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)

        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
