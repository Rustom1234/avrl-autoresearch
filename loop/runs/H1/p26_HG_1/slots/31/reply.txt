def _solve_once(n, rng):
    # Farthest-point sampling for well-spread centers
    candidates = rng.uniform(0, 1, (200, 2))
    centers = np.zeros((n, 2))
    centers[0] = candidates[rng.integers(0, 200)]
    for i in range(1, n):
        dists = np.min(np.linalg.norm(candidates - centers[:i], axis=1), axis=1)
        centers[i] = candidates[np.argmax(dists)]
    
    x0 = np.concatenate([centers, np.full(n, INIT_RADIUS)])
