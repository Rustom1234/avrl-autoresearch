def _solve_once(n, rng):
    def _constrained_solve(num_circles, initial_x, method_opts):
        iu = np.triu_indices(num_circles, 1)
        def cons(v):
            c, r = v[:2 * num_circles].reshape(num_circles, 2), v[2 * num_circles:]
            d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
            return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])
        res = minimize(lambda v: -v[2 * num_circles:].sum(), initial_x, 
                        method="SLSQP", constraints={"type": "ineq", "fun": cons}, 
                        options=method_opts)
        return res.x

    # Lift-and-drop: solve for n+1 circles
    n_plus_1 = n + 1
    x0_plus_1 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n_plus_1), 
                                  np.full(n_plus_1, INIT_RADIUS)])
    
    res_plus_1 = _constrained_solve(n_plus_1, x0_plus_1, {"maxiter": MAX_ITER})
    
    if res_plus_1 is None or not hasattr(res_plus_1, 'success'):
        # Fallback to standard n-solve
        x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
        res = _constrained_solve(n, x0, {"maxiter": MAX_ITER})\n        return res[:2 * n].reshape(n, 2), res[2 * n:]
    
    c_plus_1 = res_plus_1[:2 * n_plus_1].reshape(n_plus_1, 2)
    r_plus_1 = res_plus_1[2 * n_plus_1:]
    
    # Drop the smallest circle
    drop_idx = np.argmin(r_plus_1)
    c = np.delete(c_plus_1, drop_idx, axis=0)
    r = np.delete(r_plus_1, drop_idx)
    
    # Polish for a few iterations
    polish_opts = {"maxiter": 20}
    res_polished = _constrained_solve(n, np.concatenate([c.flatten(), r]), polish_opts)
    
    return res_polished[:2 * n].reshape(n, 2), res_polished[2 * n:]
