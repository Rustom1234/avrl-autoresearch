from scipy.optimize import minimize, linprog

# Settings
TIME_FRACTION = 0.9
INIT_LOW = 0.1
INIT_RADIUS = 0.05
SHRINK = 1e-9


def _solve_once(n, rng):
    c = rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
    r = np.full(n, INIT_RADIUS)
    
    # Alternating Optimization
    for _ in range(10):
        # 1. Solve for radii (LP)
        # Constraints: r[i] + r[j] <= dist(c[i], c[j])
        #             r[i] <= c[i][0], r[i] <= c[i][1], r[i] <= 1-c[i][0], r[i] <= 1-c[i][1]
        # Maximize sum(r_i) -> Minimize -sum(r_i)
        A_ub = []
        b_ub = []
        # Pairwise constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(c[i] - c[j])
                A_ub.append([0]*n)
                A_ub[-1][i] = 1
                A_ub[-1][j] = 1
                b_ub.append(dist)
        # Boundary constraints
        for i in range(n):
            A_ub.append([0]*n)
            A_ub[-1][i] = 1
            b_ub.append(min(c[i, 0], c[i, 1], 1 - c[i, 0], 1 - c[i, 1]))
        
        res = linprog(np.ones(n) * -1, A_ub=A_ub, b_ub=b_ub, bounds=(0, 1), method='highs')
        if not res.success: break
        r = res.x
        
        # 2. Move centers (SLSQP)
        def obj(v): return -v[n:].sum()
        def cons_func(v):
            curr_c = v[:2*n].reshape(n, 2)
            curr_r = v[n:]
            d_pair = np.linalg.norm(curr_c[:, None, :] - curr_c[None, :, :], axis=2)
            over_pair = curr_r[:, None] + curr_r[None, :] - d_pair
            np.fill_diagonal(over_pair, -1)
            # Boundary
            bound_cons = np.minimum(curr_c, 1 - curr_c) - curr_r
            return np.concatenate([over_pair.flatten(), bound_cons.flatten()])

        x0 = np.concatenate([c.flatten(), r])
        res_sq = minimize(obj, x0, method='SLSQP', constraints={'type': 'ineq', 'fun': cons_func}, 
                            options={'maxiter': 50})
        if not res_sq.success: break
        c = res_sq.x[:2*n].reshape(n, 2)
        
    return c, r
