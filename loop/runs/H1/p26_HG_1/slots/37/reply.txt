def _solve_once(n, rng):
    if rng.random() < 1/3:
        k = rng.integers(2, 5)
        ring_radii = [(j + 0.5) / (k + 0.5) * 0.5 for j in range(k)]
        counts = [int(round(n * (r / (ring_radii[-1] + 0.1))) / (sum(int(round(n * (r / (ring_radii[-1] + 0.1)))) for r in ring_radii))) 
                    for r in ring_radii]
        counts = np.array(counts)
        counts[:-1] = np.minimum(counts[:-1], counts[-1])
        counts[-1] = n - counts[:-1].sum()
        
        c = np.zeros((n, 2))
        idx = 0
        for j in range(k):
            if counts[j] > 0:
                angles = rng.uniform(0, 2 * np.pi, counts[j])
                c[idx:idx+counts[j], 0] = 0.5 + ring_radii[j] * np.cos(angles)
                c[idx:idx+counts[j], 1] = 0.5 + ring_radii[j] * np.sin(angles)
                idx += counts[j]
        c = np.clip(c, 0.05, 0.95)
        x0 = np.concatenate([c.flatten(), np.full(n, INIT_RADIUS)])
    else:
        x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
