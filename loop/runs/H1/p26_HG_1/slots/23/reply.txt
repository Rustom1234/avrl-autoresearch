def _solve_once(n, rng):
    # Rescale: centers u = (c - 0.5) * 2  (range [-1, 1])
    #           radii s = r * 10            (scale up for precision)
    # To map back: c = (u + 1) / 2, r = s / 10
    x0 = np.concatenate([rng.uniform(-1 + 2 * INIT_LOW, 1 - 2 * INIT_LOW, 2 * n), 
                           np.full(n, INIT_RADIUS * 10)])
    iu = np.triu_indices(n, 1)

    def cons(v):
        u = v[:2 * n].reshape(n, 2)
        s = v[2 * n:]
        c = (u + 1) / 2
        r = s / 10
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -v[2 * n:].sum() / 10, x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})\n    u_res = res.x[:2 * n].reshape(n, 2)
    s_res = res.x[2 * n:]
    return (u_res + 1) / 2, s_res / 10
