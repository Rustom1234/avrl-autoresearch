MAX_ITER = 200
INIT_RADIUS = 0.05

def _solve_once(n, rng):

    iu = np.triu_indices(n, 1)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]

    iu = np.triu_indices(n, 1)
    
    def objective(v, mu):
        c = v[:2 * n].reshape(n, 2)
        r = v[2 * n:]
        
        # Slack variables
        # Pairwise distances
        d_vec = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1))
        d = d_vec - r[iu[0]] - r[iu[1]]
        # Wall constraints
        w1 = c[:, 0] - r
        w2 = c[:, 1] - r
        w3 = 1 - c[:, 0] - r
        w4 = 1 - c[:, 1] - r
        
        slacks = np.concatenate([d, w1, w2, w3, w4, r])
        
        # Penalize very negative slacks (infeasibility)
        penalty = 1e6 * np.sum(np.maximum(0, -slacks) ** 2)

        # Only take logs of positive slacks to avoid -inf; use a small epsilon
        safe_slacks = np.maximum(slacks, 1e-12)
