# Settings
N_BITS = 2048
RADIUS = 2
N_TREES = 600
LEARNING_RATE = 0.03
NUM_LEAVES = 31
FEATURE_FRACTION = 0.3
BAGGING_FRACTION = 0.8
BAGGING_FREQ = 1
MIN_CHILD_SAMPLES = 10


def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    return np.array([gen.GetFingerprintAsNumPy(Chem.MolFromSmiles(s)) for s in smiles], dtype=np.float32)


def fit_predict(train_smiles, train_y, test_smiles, seed):
    model = LGBMClassifier(
        n_estimators=N_TREES,
        learning_rate=LEARNING_RATE,
        num_leaves=NUM_LEAVES,
        feature_fraction=FEATURE_FRACTION,
        bagging_fraction=BAGGING_FRACTION,
        bagging_freq=BAGGING_FREQ,
        min_child_samples=MIN_CHILD_SAMPLES,
        random_state=seed,
        n_jobs=8,
        verbose=-1
    )
    model.fit(featurize(train_smiles), np.asarray(train_y, dtype=np.int64))
    return model.predict_proba(featurize(test_smiles))[:, 1]

# Settings
N_BITS = 2048
N_TREES = 600
LEARNING_RATE = 0.03
NUM_LEAVES = 31
FEATURE_FRACTION = 0.3
BAGGING_FRACTION = 0.8
BAGGING_FREQ = 1
MIN_CHILD_SAMPLES = 10


def featurize(smiles, radius):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=radius, fpSize=N_BITS)
    return np.array([gen.GetFingerprintAsNumPy(Chem.MolFromSmiles(s)) for s in smiles], dtype=np.float32)


def fit_predict(train_smiles, train_y, test_smiles, seed):
    probs = []
    for r in [1, 2, 3]:
        model = LGBMClassifier(
            n_estimators=N_TREES,
            learning_rate=LEARNING_RATE,
            num_leaves=NUM_LEAVES,
            feature_fraction=FEATURE_FRACTION,
            bagging_fraction=BAGGING_FRACTION,
            bagging_freq=BAGGING_FREQ,
            min_child_samples=MIN_CHILD_SAMPLES,
            random_state=seed + r,
            n_jobs=8,
            verbose=-1
        )
        model.fit(featurize(train_smiles, r), np.asarray(train_y, dtype=np.int64))
        probs.append(model.predict_proba(featurize(test_smiles, r))[:, 1])
    return np.mean(probs, axis=0)
