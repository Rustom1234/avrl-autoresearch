from sklearn.ensemble import RandomForestClassifier
import lightgbm as lgb

# Settings
N_BITS = 2048
RADIUS = 2
N_TREES_RF = 300
N_TREES_LGBM = 400
MAX_FEATURES = 0.3
MIN_SAMPLES_LEAF = 1


def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    return np.array([gen.GetFingerprintAsNumPy(Chem.MolFromSmiles(s)) for s in smiles], dtype=np.float32)


def fit_predict(train_smiles, train_y, test_smiles, seed):
    x_train = featurize(train_smiles)
    y_train = np.asarray(train_y, dtype=np.int64)
    
    # RandomForest
    rf = RandomForestClassifier(n_estimators=N_TREES_RF, max_features=MAX_FEATURES, min_samples_leaf=MIN_SAMPLES_LEAF,
                                 n_jobs=-1, random_state=seed)
    rf.fit(x_train, y_train)
    
    # LightGBM
    lgbm = lgb.LGBMClassifier(n_estimators=N_TREES_LGBM, learning_rate=0.05, feature_fraction=MAX_FEATURES,
                                n_jobs=-1, random_state=seed, verbosity=-1)
    lgbm.fit(x_train, y_train)
    
    x_test = featurize(test_smiles)
    rf_probs = rf.predict_proba(x_test)[:, 1]
    lgbm_probs = lgbm.predict_proba(x_test)[:, 1]
    
    return (rf_probs + lgbm_probs) / 2.0
