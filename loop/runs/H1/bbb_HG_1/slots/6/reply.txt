from rdkit import Descriptors

def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    features = []
    for s in smiles:
        mol = Chem.MolFromSmiles(s)
        if mol is None: continue
        fp = gen.GetFingerprintAsNumPy(mol)
        
        # PhysChem Descriptors
        mw = Descriptors.MolWt(mol)
        logp = Descriptors.MolLogP(mol)
        tpsa = Descriptors.TPSA(mol)
        hbd = Descriptors.NumHDonors(mol)
        hba = Descriptors.NumHAcceptors(mol)
        rotb = Descriptors.NumRotBonds(mol)
        aroms = Descriptors.NumAromatic(mol)
        charge = Descriptors.MolContribH_Ex_Sum(mol)
        
        # sp3 fraction
        sp3_count = sum(1 for a in mol.GetAtoms() if a.GetHybridization() == Chem.rdFMIC.sp3)
        sp3_frac = sp3_count / mol.GetNumAtoms() if mol.GetNumAtoms() > 0 else 0
        
        # SMARTS counts
        car_s = Chem.MolFromSmarts('[CXO3][CXO3]')[0] if Chem.MolFromSmarts('[CXO3][CXO3]') else None
        amine_s = Chem.MolFromSmarts('[NX3;H3,H2,H1;!$(lost_charge)]')[0] if Chem.MolFromSmarts('[NX3;H3,H2,H1;!$(lost_charge)]') else None
        car_count = len(mol.GetSubstructMatches(car_s)) if car_s else 0
        amine_count = len(mol.GetSubstructMatches(amine_s)) if amine_s else 0
        
        desc = np.array([mw, logp, tpsa, hbd, hba, rotb, aroms, charge, sp3_frac, car_count, amine_count], dtype=np.float32)
        features.append(np.concatenate([fp, desc]))
    return np.array(features, dtype=np.float32)

    model = RandomForestClassifier(n_estimators=N_TREES, max_features=MAX_FEATURES, min_samples_leaf=MIN_SAMPLES_LEAF,
