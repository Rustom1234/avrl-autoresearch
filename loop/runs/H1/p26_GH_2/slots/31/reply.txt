def _greedy_init(n, rng, n_cand=100):
    """Place centres one by one at the candidate point farthest from placed centres and walls, with small noise."""
    cand = rng.uniform(0, 1, (n_cand * n, 2))
    wall = 2 * np.minimum(cand, 1 - cand).min(1)
    c = np.zeros((n, 2))
    c[0] = cand[0]
    for i in range(1, n):
        d = np.sqrt(((cand[:, None] - c[None, :i]) ** 2).sum(-1)).min(1)
        score = np.minimum(d, wall) + 0.01 * rng.standard_normal(len(cand))
        c[i] = cand[np.argmax(score)]
    return c
