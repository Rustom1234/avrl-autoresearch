def _greedy_centres(n, rng):
    """Greedy sequential placement: largest radii first, each at the first grid point (in random scan order) that fits."""
    radii = np.sort(rng.uniform(0.03, 0.1, n))[::-1]
    g = np.linspace(0, 1, GRID)
    cand = np.stack(np.meshgrid(g, g), -1).reshape(-1, 2)
    cand = cand[rng.permutation(len(cand))]
    c = np.zeros((n, 2))
    for i in range(n):
        ok = np.all((cand >= radii[i]) & (cand <= 1 - radii[i]), axis=1)
        if i:
            d = np.sqrt(((cand[:, None] - c[None, :i]) ** 2).sum(-1))
            ok &= np.all(d >= radii[i] + radii[None, :i], axis=1)
        idx = np.flatnonzero(ok)
        c[i] = cand[idx[0]] if len(idx) else rng.uniform(0, 1, 2)
    return c, radii
