def _untangle(c, r, steps=100, step=0.3):
    """Force-directed relaxation: each overlapping pair pushes its centres apart; walls keep centres inside."""
    for _ in range(steps):
        diff = c[:, None] - c[None]
        d = np.sqrt((diff ** 2).sum(-1)) + 1e-12
        over = np.maximum(0, r[:, None] + r[None] - d)
        np.fill_diagonal(over, 0)
        f = ((diff / d[..., None]) * over[..., None]).sum(1)
        c = np.clip(c + step * f, 0, 1)
    return c
