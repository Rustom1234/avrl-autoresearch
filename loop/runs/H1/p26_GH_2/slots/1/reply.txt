def _repair(c, r):
    """Move centres along the overlap gradient, then shrink radii until the packing is valid."""
    n = len(r)
    eye = np.eye(n)
    c = np.clip(c, 0, 1)
    r = np.maximum(r, 0)
    for _ in range(300):
        diff = c[:, None] - c[None]
        d = np.sqrt((diff ** 2).sum(-1)) + eye
        over = np.maximum(r[:, None] + r[None] - d, 0) * (1 - eye)
        if over.max() <= 0:
            break
