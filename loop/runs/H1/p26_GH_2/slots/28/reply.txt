def _hex_start(n, rng):
    """Pick n points of a hexagonal lattice in the unit square, then jitter them."""
    s = 0.9 * np.sqrt(2 / (np.sqrt(3) * n))
    h = s * np.sqrt(3) / 2
    pts = np.array([[j * s + (i % 2) * s / 2, i * h] for i in range(int(1 / h) + 1) for j in range(int(1 / s) + 2)])
    pts = pts[(pts[:, 0] <= 1) & (pts[:, 1] <= 1)]
    pts = pts[rng.permutation(len(pts))[:n]]
    if len(pts) < n:
        pts = np.vstack([pts, rng.uniform(0, 1, (n - len(pts), 2))])
    return np.clip(pts + rng.normal(0, 0.1 * s, pts.shape), 0, 1)
