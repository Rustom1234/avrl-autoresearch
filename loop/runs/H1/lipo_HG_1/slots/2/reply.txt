from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator, Descriptors
import lightgbm as lgb

# Settings
N_BITS = 2048
RADIUS = 2
LR = 0.05

def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    fps = []
    descs = []
    for s in smiles:
        mol = Chem.MolFromSmiles(s)
        if mol:
            fps.append(gen.GetFingerprintAsNumPy(mol))
            descs.append([Descriptors.MolWt(mol), Descriptors.LogP(mol), 
                           Descriptors.NumHDonors(mol), Descriptors.NumHAcceptors(mol)])
        else:
            fps.append(np.zeros(N_BITS))
            descs.append([0.0, 0.0, 0.0, 0.0])
    return np.hstack([np.array(fps, dtype=np.float32), np.array(descs, dtype=np.float32)])

def fit_predict(train_smiles, train_y, test_smiles, seed):
    X = featurize(train_smiles)
    y = np.asarray(train_y, dtype=np.float64)

    mols = [Chem.MolFromSmiles(s) for s in train_smiles]

    scaffolds = [s.GetScaffold() if s else None for s in scaffolds]
    
    unique_scaffs = list(set(s for s in scaffolds if s is not None))
    scaffold_map = {sc: i for i, sc in enumerate(unique_scaffs)}
    scaffold_ids = [scaffold_map[s] if s is not None else -1 for s in scaffolds]
    
    # Pick 10% of unique scaffolds for validation
    num_val_scaffs = max(1, int(len(unique_scaffs) * 0.1))
    val_scaffold_indices = np.random.choice(len(unique_scaffs), num_val_scaffs, replace=False, size=num_val_scaffs)
    # Use a fixed set of scaffold IDs for the split based on seed
    rng = np.random.default_rng(seed)
    val_scaff_ids = rng.choice(unique_scaffs, num_val_scaffs, replace=False)
    
    train_idx = []
    val_idx = []
    for i, s_id in enumerate(scaffolds):
        if s_id in val_scaff_ids:
            val_idx.append(i)
        else:
            train_idx.append(i)
            
    X_train, y_train = X[train_idx], y[train_idx]
    X_val, y_val = X[val_idx], y[val_idx]
    
    # Early stopping
    params = {
        'objective': 'l1',
        'metric': 'mae',
        'learning_rate': LR,
        'verbosity': -1,
        'boosting_type': 'gbdt',
        'random_state': seed,
        'n_jobs': -1
    }

    val_set = lgb.Dataset(X_val, label=y_val, reference=train_set)
    
    best_round = 0
    best_score = float('inf')
    
    # Manual early stopping loop to handle objective/params easily
    for round_num in range(1, 2001):
