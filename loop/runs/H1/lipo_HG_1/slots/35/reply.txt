from sklearn.neural_network import MLPRegressor
import numpy as np
from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator, Descriptors

# Settings
N_BITS = 4096

def featurize(smiles, radius):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=radius, fpSize=N_BITS)
    return np.array([gen.GetFingerprintAsNumPy(Chem.MolFromSmiles(s)) for s in smiles], dtype=np.float32)

def fit_predict(train_smiles, train_y, test_smiles, seed):
    X_train = featurize(train_smiles, 2)
    X_test = featurize(test_smiles, 2)
    
    y_train = np.asarray(train_y, dtype=np.float64)
    
    # Scale targets
    y_mean = np.mean(y_train)
    y_std = np.std(y_train)
    y_train_scaled = (y_train - y_mean) / (y_std + 1e-9)
    
    # Ensemble of 5 MLPs
    all_preds = []
    for i in range(5):
        current_seed = seed + i
        model = MLPRegressor(
            hidden_layer_sizes=(512, 128), 
            activation='relu', 
            alpha=1e-2, 
            learning_rate_init=1e-3, 
            max_iter=300, 
            early_stopping=True, 
            validation_fraction=0.1, 
            random_state=current_seed
        )
        model.fit(X_train, y_train_scaled)
        all_preds.append(model.predict(X_test))
    
    avg_preds_scaled = np.mean(all_preds, axis=0)
    
    # Invert scaling
    return (avg_preds_scaled * y_std) + y_mean
