    # Compute Tanimoto kernel: K(x,y) = sum(min(x,y)) / sum(max(x,y))
    # For binary fingerprints, this is the standard Jaccard similarity.
    # We use a simple efficient approximation: 
    # Since we have float vectors, we'll use a manual implementation of the kernel.
    # To stay within memory limits, we avoid huge intermediates.
    
    def get_tanimoto(A, B):
        # A: (n, d), B: (m, d)
        # Tanimoto = dot(A, B) / (sum(A) + sum(B) - dot(A, B))
        dot_prod = np.dot(A, B.T)
        sum_A = np.sum(A, axis=1).reshape(-1, 1)
        sum_B = np.sum(B, axis=1).reshape(1, -1)

# Settings
N_BITS = 2048
RADIUS = 2

def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    return np.array([gen.GetFingerprintAsNumPy(Chem.MolFromSmiles(s)) for s in smiles], dtype=np.float32)

def fit_predict(train_smiles, train_y, test_smiles, seed):
    X_train = featurize(train_smiles)
    X_test = featurize(test_smiles)
    
    def get_tanimoto(A, B):
        dot_prod = np.dot(A, B.T)
        sum_A = np.sum(A, axis=1).reshape(-1, 1)
        sum_B = np.sum(B, axis=1).reshape(1, -1)
        return dot_prod / (sum_A + sum_B - dot_prod + 1e-9)

    K_train = get_tanimoto(X_train, X_train)
    K_test = get_tanimoto(X_test, X_train)
    
    # Inner CV for hyperparameter tuning
    train_y = np.asarray(train_y, dtype=np.float64)
    n_train = len(train_y)
    rng = np.random.RandomState(seed)
    indices = rng.permutation(n_train)
    
    # Simple 3-fold inner split
    fold_size = n_train // 3
    val_idx = indices[fold_size:2*fold_size]
    train_idx = np.concatenate([indices[:fold_size], indices[2*fold_size:]])

    best_c, best_eps = 10.0, 0.1
    min_mae = float('inf')

    for c in [1.0, 10.0, 30.0]:
        for eps in [0.05, 0.1, 0.2]:
            # SVR with precomputed kernel requires the kernel of the subset
            # K_train[train_idx][:, train_idx] is the kernel for the training subset
            # K_train[val_idx][:, val_idx] is the kernel for the validation subset
            K_train_sub = K_train[np.ix_(train_idx, train_idx)]
            K_val_sub = K_train[np.ix_(val_idx, val_idx)]
            
            model = SVR(kernel='precomputed', C=c, epsilon=eps)
            model.fit(K_train_sub, train_y[train_idx])
            preds = model.predict(K_val_sub)
            mae = np.mean(np.abs(preds - train_y[val_idx]))
            
            if mae < min_mae:
                min_mae = mae
                best_c, best_eps = c, eps

    final_model = SVR(kernel='precomputed', C=best_c, epsilon=best_eps)
    final_model.fit(K_train, train_y)
    return final_model.predict(K_test)
