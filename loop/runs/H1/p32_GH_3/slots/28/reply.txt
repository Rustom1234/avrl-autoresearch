def _insert_init(n, rng, samples=300):
    """Place circles one by one at the candidate point that allows the largest radius."""
    c, r = np.zeros((n, 2)), np.zeros(n)
    for k in range(n):
        p = rng.uniform(0, 1, (samples, 2))
        rad = np.minimum(np.minimum(p[:, 0], p[:, 1]), np.minimum(1 - p[:, 0], 1 - p[:, 1]))
        if k > 0:
            d = np.sqrt(((p[:, None] - c[None, :k]) ** 2).sum(-1)) - r[None, :k]
            rad = np.minimum(rad, d.min(1))
        j = np.argmax(rad)
        c[k], r[k] = p[j], max(rad[j], 0)
    return c, r
