def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    cur, cur_sum = None, -np.inf
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        frac = (time.process_time() - t0) / (TIME_FRACTION * time_limit)
        if cur is not None and frac > 0.5:
            c, r = _jump(cur[0], cur[1], rng)     # jump from the current (annealed) packing
        else:
            c, r = _solve_once(n, rng)            # random restart
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        s = r.sum()
        if best is None or s > best[1].sum():
            best = (c, r)
        temp = TEMP0 * max(1 - frac, 1e-3)        # linear cooling to near zero
        if cur is None or s > cur_sum or rng.random() < np.exp((s - cur_sum) / temp):
            cur, cur_sum = (c, r), s              # Metropolis acceptance
    return best
