def _penalty(v, mu, n, iu):
    """Objective -sum(r) plus mu times squared overlap and wall violations; overlaps are allowed but penalised."""
    c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
    d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) + 1e-12
    ov = np.maximum(0, r[iu[0]] + r[iu[1]] - d)
    wall = np.maximum(0, r[:, None] - c) + np.maximum(0, c + r[:, None] - 1)
    return -r.sum() + mu * ((ov ** 2).sum() + (wall ** 2).sum())
