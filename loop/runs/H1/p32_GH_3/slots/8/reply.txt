POP_SIZE = 3
MUTATE_EVERY = 3


def solve(n, seed, time_limit):
    rng, t0, pop, it = np.random.default_rng(seed), time.process_time(), [], 0
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        it += 1
        if len(pop) >= POP_SIZE and it % MUTATE_EVERY == 0:
            best = max(pop, key=lambda p: p[1].sum())
            c, r = _jump(best[0], best[1], rng)   # mutate the best packing
        else:
            c, r = _solve_once(n, rng)            # random restart
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if len(pop) < POP_SIZE:
            pop.append((c, r))
        else:
            w = min(range(POP_SIZE), key=lambda k: pop[k][1].sum())
            if r.sum() > pop[w][1].sum():         # replace the worst member
                pop[w] = (c, r)
    return max(pop, key=lambda p: p[1].sum())
