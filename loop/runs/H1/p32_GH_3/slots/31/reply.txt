def _hex_centers(n, rng):
    """Densest-packing start: a hex lattice, randomly rotated, cut to the n points nearest the centre, fitted to the square."""
    m = int(2 * np.sqrt(n)) + 2
    i, j = np.meshgrid(np.arange(-m, m + 1), np.arange(-m, m + 1))
    p = np.stack([i + 0.5 * (j % 2), j * np.sqrt(3) / 2], -1).reshape(-1, 2)
    a = rng.uniform(0, np.pi / 3)
    rot = np.array([[np.cos(a), -np.sin(a)], [np.sin(a), np.cos(a)]])
    p = p @ rot.T
    p = p[np.argsort((p ** 2).sum(1))[:n]]
    span = (p.max(0) - p.min(0)).max()
    return (p - (p.max(0) + p.min(0)) / 2) * (1 - 2 * INIT_LOW) / span + 0.5
