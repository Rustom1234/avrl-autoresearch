def _anneal(n, rng, steps=400):
    """Short simulated annealing on the centres: maximise the minimum pairwise centre distance."""
    def score(c):
        d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
        d[np.diag_indices(n)] = np.inf
        return d.min()

    c = rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
    s = score(c)
    for k in range(steps):
        T = 0.02 * (1 - k / steps) + 1e-6
        i = rng.integers(n)
        old = c[i].copy()
        c[i] = np.clip(old + rng.normal(0, 0.1, 2), 0, 1)
        s2 = score(c)
        if s2 >= s or rng.random() < np.exp((s2 - s) / T):
            s = s2
        else:
            c[i] = old
    return c
