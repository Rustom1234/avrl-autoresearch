def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    budget = TIME_FRACTION * time_limit
    while time.process_time() - t0 < EXPLORE_FRACTION * budget:   # exploration: short random restarts
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    while time.process_time() - t0 < budget:   # exploitation: SLSQP from the best packing, slightly jittered
        x0 = np.concatenate([(best[0] + rng.normal(0, JITTER, (n, 2))).ravel(), best[1]])
        c, r = _solve_once(n, rng, x0)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if r.sum() > best[1].sum():
            best = (c, r)
    return best
