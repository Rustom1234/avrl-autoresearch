def _sequential(n, rng):
    """Greedy construction: add one circle at a time and re-optimise from the previous packing."""
    c, r = np.empty((0, 2)), np.empty(0)
    for k in range(1, n + 1):
        c = np.vstack([c, rng.uniform(INIT_LOW, 1 - INIT_LOW, 2)])
        r = np.append(r, INIT_RADIUS)
        c, r = _solve_once(k, rng, np.concatenate([c.ravel(), r]))
    return c, r
