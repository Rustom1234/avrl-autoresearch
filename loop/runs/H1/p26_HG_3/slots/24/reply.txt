def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        if best is None:
            c, r = _solve_once(n, rng)
            c = np.clip(c, 0, 1)
            r = _repair(c, r)
        else:
            # Perturb the best packing: re-randomize 3-6 circles
            c_best, r_best = best
            num_to_reset = rng.integers(3, 7)
            indices = rng.choice(n, num_to_reset, replace=False)
            c_perturbed = c_best.copy()
            c_perturbed[indices] = rng.uniform(INIT_LOW, 1 - INIT_LOW, (num_to_reset, 2))
            
            # Start SLSQP from the perturbed centers with best radii
            x0 = np.concatenate([c_perturbed, r_best])
            iu = np.triu_indices(n, 1)
            def cons_v(v):
                c_v, r_v = v[:2 * n].reshape(n, 2), v[2 * n:]
                d = np.sqrt(((c_v[iu[0]] - c_v[iu[1]]) ** 2).sum(1)) - r_v[iu[0]] - r_v[iu[1]]
                return np.concatenate([d, c_v[:, 0] - r_v, c_v[:, 1] - r_v, 1 - c_v[:, 0] - r_v, 1 - c_v[:, 1] - r_v, r_v])
            
            res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", constraintsr={"type": "ineq", "fun": cons_v}, options={"maxiter": MAX_ITER})
            c, r = res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
            c = np.clip(c, 0, 1)
            r = _repair(c, r)

        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    return best
