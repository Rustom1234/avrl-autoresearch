def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    exploit_start = t0 + 0.4 * time_limit
    
    # Phase 1: Random Restarts
    while time.process_time() < exploit_start:
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
            
    # Phase 2: Basin Hopping (Exploitation)
    if best is not None:
        while time.process_time() < TIME_FRACTION * time_limit:
            # Perturb best packing slightly
            c_best, r_best = best
            noise = rng.normal(0, 0.02, (n, 2))
            c_new = np.clip(c_best + noise, 0, 1)
            r_new = _repair(c_new, r_best)
            
            if r_new.sum() > best[1].sum():
                best = (c_new, r_new)
    return best
