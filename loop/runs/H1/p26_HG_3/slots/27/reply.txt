    def cons_grad(v):

        grad = np.zeros_like(v)
        # Pairwise constraints
        d_vec = c[iu[0]] - c[iu[1]]
        dist = np.sqrt((d_vec**2).sum(1))

            # Grad w.r.t c_i, c_j, r_i, r_j
