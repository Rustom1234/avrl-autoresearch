# Settings
TIME_FRACTION = 0.8
INIT_LOW = 0.1
MAX_ITER = 200
INIT_RADIUS = 0.05
SHRINK = 1e-9


def _solve_once(n, rng):
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]


def _repair(c, r):
    """Shrink radii until the packing is valid, so that the harness never sees an overlap."""
    n = len(r)
    r = np.minimum(r, np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1])))
    r = np.maximum(r, 0)
    for _ in range(50):
        d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
        over = r[:, None] + r[None] - d
        np.fill_diagonal(over, -1)
        i, j = np.unravel_index(np.argmax(over), over.shape)
        if over[i, j] <= 0:
            break
        s = d[i, j] / (r[i] + r[j])
        r[i], r[j] = r[i] * s, r[j] * s
    return np.maximum(r - SHRINK, 0)


def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    return best

def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    # Bandit stats: [best_sum, count]
    # 0: Uniform, 1: Edge-biased, 2: Dihedral (Warm start)
    stats = [[-1.0, 0], [-1.0, 0], [-1.0, 0]]
    
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        # UCB selection
        probs = []
        for i in range(3):
            if stats[i][1] == 0:
                probs.append(1.0 / 3.0)
            else:
                # Simple UCB-like, cap usage at 60% (approx 0.2 per slot for 3 strategies)
                score = stats[i][0] + 1.41 * np.sqrt(np.log(max(1, stats[2][1])) / (stats[i][1] + 1))
                probs.append(score)
        
        probs = np.array(probs)
        # Selection with soft-cap: if any count > 0.6 * total_calls, reduce its prob
        total_calls = sum(s[1] for s in stats)
        for i in range(3):
            if total_calls > 0 and stats[i][1] > 0.6 * total_calls:
                probs[i] *= 0.1
        
        idx = rng.choice([0, 1, 2], p=probs / probs.sum())
        
        # Generate x0
        if idx == 0:
            x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
        elif idx == 1:
            c0 = np.zeros((n, 2))
            for i in range(n):
                side = rng.randint(0, 4)
                if side == 0: c0[i] = [rng.uniform(0, 1), 0]
                elif side == 1: c0[i] = [rng.uniform(0, 1), 1]
                elif side == 2: c0[i] = [0, rng.uniform(0, 1)]
                else: c0[i] = [1, rng.uniform(0, 1)]
            x0 = np.concatenate([c0, np.full(n, INIT_RADIUS)])
        else:
            if best is not None:
                best_c, best_r = best
                # Simple dihedral flip
                c_flip = np.flip(best_c, axis=1) # Example flip
                x0 = np.concatenate([c_flip, best_r])
            else:
                x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
        
        c, r = _solve_once(n, rng, x0)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        
        current_sum = r.sum()
        stats[idx][0] = max(stats[idx][0], current_sum)
        stats[idx][1] += 1
        
        if best is None or current_sum > best[1].sum():
            best = (c, r)
            
    return best
