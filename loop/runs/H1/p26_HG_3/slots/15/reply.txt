def _solve_once(n, rng):
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]

def _solve_once(n, rng):
    c = rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
    for _ in range(30):
        diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        d2 = np.sum(diffs**2, axis=2)
        np.fill_diagonal(d2, np.inf)
        weights = 1.0 / (d2 + 1e-9)
        repulsion = np.sum(diffs * weights[..., np.newaxis], axis=1)
        norm = np.linalg.norm(repulsion, axis=1, keepdims=True)
        repulsion = np.divide(repulsion, norm, out=np.zeros_like(repulsion), where=norm > 0)
        c += 0.02 * repulsion
        c = np.clip(c, 0.05, 0.95)
    
    x0 = np.concatenate([c.flatten(), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)

    def cons(v):
        c_v, r_v = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c_v[iu[0]] - c_v[iu[1]]) ** 2).sum(1)) - r_v[iu[0]] - r_v[iu[1]]
        return np.concatenate([d, c_v[:, 0] - r_v, c_v[:, 1] - r_v, 1 - c_v[:, 0] - r_v, 1 - c_v[:, 1] - r_v, r_v])

    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
