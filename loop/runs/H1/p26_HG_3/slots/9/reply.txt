from scipy.optimize import linprog

def _repair(c, r):
    """Shrink radii until the packing is valid, so that the harness never sees an overlap."""
    n = len(c)
    # Objective: maximize sum of radii -> minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Constraints:
    # 1. r_i + r_j <= d_ij  =>  r_i + r_j - d_ij <= 0
    # 2. r_i <= min(x_i, y_i, 1-x_i, 1-y_i)
    
    A_ub = []
    b_ub = []
    
    # Pairwise constraints
    dist_matrix = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
    iu = np.triu_indices(n, k=1)
    for i, j in zip(iu[0], iu[1]):
        row = np.zeros(n)
        row[i] = 1
        row[j] = 1
        A_ub.append(row)
        b_ub.append(dist_matrix[i, j])
        
    # Boundary constraints
    bounds = []
    for i in range(n):
        max_r = min(c[i, 0], c[i, 1], 1 - c[i, 0], 1 - c[i, 1])
        bounds.append((0, max_r))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x - 1e-12
    return r * 0.999999999

    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)
