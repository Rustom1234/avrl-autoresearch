def _solve_once(n, rng):
    # Lubachevsky-Stillinger growth start
    pos = rng.random((n, 2)) * (1 - 2 * INIT_LOW) + INIT_LOW
    vel = (rng.random((n, 2)) - 0.5) * 0.1
    radii = np.full(n, 0.0)
    
    dt = 0.01
    for _ in range(500):
        # Move and grow
        pos += vel * dt
        radii += dt
        
        # Collision handling
        for i in range(n):
            # Wall collisions
            for k in range(2):
                if pos[i, k] - radii[i] < 0:
                    pos[i, k] = radii[i]
                    vel[i, k] *= -1
                if pos[i, k] + radii[i] > 1:
                    pos[i, k] = 1 - radii[i]
                    vel[i, k] *= -1
            
            # Circle collisions
            for j in range(i + 1, n):
                dist_vec = pos[i] - pos[j]
                dist = np.linalg.norm(dist_vec)
                if dist < radii[i] + radii[j]:
                    # Simple elastic reflection
                    new_vel_i = vel[i] - 2 * np.dot(vel[i] - vel[j], dist_vec / dist) * (dist_vec / dist)
                    vel[i] = new_vel_i
                    vel[j] = 2 * vel[i] - vel[j]

    x0 = np.concatenate([pos.flatten(), radii])
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
