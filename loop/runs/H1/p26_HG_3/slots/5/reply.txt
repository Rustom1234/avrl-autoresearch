    # Initialize with slightly larger radii to allow overlap for relaxation
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS * 2)])
    iu = np.triu_indices(n, 1)

    def penalty_obj(v, weight):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        # Maximize sum of radii -> minimize negative sum
        obj = -r.sum()
        # Overlap penalty
        dist_sq = ((c[:, np.newaxis, :] - c[np.newaxis, :, :]) ** 2).sum(axis=2)
        d = np.sqrt(np.maximum(dist_sq, 0))
        overlap = (r[:, np.newaxis] + r[np.newaxis, :]) - d
        np.fill_diagonal(overlap, -1)
        over_penalty = np.sum(np.maximum(0, overlap[iu[0]], axis=0)**2)
        # Wall penalty
        wall_penalty = np.sum(np.maximum(0, r - c[:, 0])**2) + 
                        np.sum(np.maximum(0, r - c[:, 1])**2) + 
                        np.sum(np.maximum(0, r - (1 - c[:, 0])))**2 +
