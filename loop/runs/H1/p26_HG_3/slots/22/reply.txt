    for i in range(k):
        angle = (2 * np.pi * i / k) + rng.uniform(0, 2 * np.pi / k)
        # Distance from center to make the ring just inside the [0, 1] square
        # and inset by a small margin (0.05)
        dist = 0.45
        x0[i, 0] = 0.5 + dist * np.cos(angle)
        x0[i, 1] = 0.5 + dist * np.sin(angle)
