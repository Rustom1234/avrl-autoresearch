    iu = np.triu_indices(n, 1)

    def cons(v):

        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]

        np.fill_diagonal(over, -1)
        i, j = np.unravel_index(np.argmax(over), over.shape)

    return np.maximum(r - SHRINK, 0)

def solve(n, seed, time_limit):
    rng, t0 = np.random.default_rng(seed), time.process_time()
    pool = []
    
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        if not pool:
            c, r = _solve_once(n, rng)
            c = np.clip(c, 0, 1)
            r = _repair(c, r)
        else:
            # Pick a random member, biased towards better ones
            idx = rng.choices(range(len(pool)), weights=[r_i.sum() for _, r_i in pool], k=1)[0]
            pc, pr = pool[idx]
            # Perturb centers slightly
            x_pert = np.concatenate([pc + rng.normal(0, 0.02, (n, 2)), pr])
            c, r = _solve_once(n, rng, x0=x_pert)
            c = np.clip(c, 0, 1)
            r = _repair(c, r)
        
        # Check for diversity and update pool
        is_diverse = True
        for pc, pr in pool:
            if np.linalg.norm(r - pr) < 1e-3:
                is_diverse = False
                break
        if is_diverse:
            pool.append((c, r))
            pool.sort(key=lambda x: x[1].sum(), reverse=True)
            if len(pool) > 5:
                pool.pop()
        
        # Also keep track of the global best
        current_best = pool[0] if pool else (None, np.array([]))
        if best is None or r.sum() > current_best[1].sum():
            best = (c, r)
            
    return best
