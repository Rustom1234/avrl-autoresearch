from sklearn.ensemble import HistGradientBoostingClassifier

# Settings
N_BITS = 2048
RADIUS = 2


def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    macc_gen = MACCSkeys.GenMACCSKeys
    feats = []
    for s in smiles:
        mol = Chem.MolFromSmiles(s)
        if mol:
            morgan = gen.GetCountFingerprintAsNumPy(mol)
            maccs = macc_gen(mol)
            # Add basic 2D descriptors: MW, LogP, TPSA, HBD, HBA
            from rdkit.Chem import Descriptors
            mw = Descriptors.MolWt(mol)
            logp = Descriptors.MolLogP(mol)
            tpsa = Descriptors.TPSA(mol)
            hbd = Descriptors.NumH Donors(mol)
            hba = Descriptors.NumH Acceptors(mol)
            desc = np.array([mw, logp, tpsa, hbd, hba], dtype=np.float32)
            
            feats.append(np.concatenate([morgan, np.array(maccs, dtype=np.float32), desc]))
        else:
            # Fallback for invalid SMILES to maintain array shape
            feats.append(np.zeros(N_BITS + 167 + 5, dtype=np.float32))
    feats = np.array(feats, dtype=np.float32)
    return np.nan_to_num(feats, nan=0.0)


def fit_predict(train_smiles, train_y, test_smiles, seed):
    model = HistGradientBoostingClassifier(
        max_iter=400, 
        learning_rate=0.05, 
        max_leaf_nodes=31, 
        l2_regularization=1.0, 
        random_state=seed)
    model.fit(featurize(train_smiles), np.asarray(train_y, dtype=np.int64))
    return model.predict_proba(featurize(test_smiles))[:, 1]
