def _solve_once(n, rng):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Start with a small radius to find positions, then expand
    # Sequential greedy approach: place circles one by one
    for i in range(n):
        # Try a few random positions for the i-th circle
        best_r = -1.0
        best_c = np.zeros(2)
        
        for _ in range(15):
            c_cand = rng.uniform(INIT_LOW, 1 - INIT_LOW, 2)
            # Calculate max possible radius at this center
            # 1. Distance to boundaries
            r_bound = np.minimum(c_cand[0], c_cand[1], 1 - c_cand[0], 1 - c_cand[1])
            
            # 2. Distance to existing circles
            if i > 0:
                dist_to_prev = np.linalg.norm(centers[:i] - c_cand, axis=1)
                r_prev = radii[:i] + 1e-9
                r_min = np.minimum(dist_to_prev - r_prev, r_bound)
                r_cand = np.min(r_min)
            else:
                r_cand = r_bound
            
            if r_cand > best_r:
                best_r = r_cand
                best_c = c_cand
        
        centers[i] = best_c
        radii[i] = best_r

    # Local refinement for all circles at once using SLSQP
    x0 = np.concatenate([centers.flatten(), radii])
    iu = np.triu_indices(n, 1)
