import time

import numpy as np
from scipy.optimize import minimize

# Settings
TIME_FRACTION = 0.8
INIT_LOW = 0.1
MAX_ITER = 200
INIT_RADIUS = 0.05
SHRINK = 1e-9


def _solve_once(n, rng):
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]


def _repair(c, r):
    """Shrink radii until the packing is valid, so that the harness never sees an overlap."""
    n = len(r)
    r = np.minimum(r, np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1])))
    r = np.maximum(r, 0)
    for _ in range(50):
        d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
        over = r[:, None] + r[None] - d
        np.fill_diagonal(over, -1)
        i, j = np.unravel_index(np.argmax(over), over.shape)
        if over[i, j] <= 0:
            break
        s = d[i, j] / (r[i] + r[j])
        r[i], r[j] = r[i] * s, r[j] * s
    return np.maximum(r - SHRINK, 0)


def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    return best

        if over[i, j] <= 0:
            break
        s = d[i, j] / (r[i] + r[j])
        r[i], r[j] = r[i] * s, r[j] * s
    return np.maximum(r - SHRINK, 0)


def _worker_task(n, seed):
    rng = np.random.default_rng(seed)
    c, r = _solve_once(n, rng)
    c = np.clip(c, 0, 1)
    r = _repair(c, r)
    return (c, r)


def solve(n, seed, time_limit):
    t0 = time.process_time()
    limit = TIME_FRACTION * time_limit
    
    # Use as many processes as possible but leave one for the main thread
    num_procs = max(1, mp.cpu_count() - 1)
    
    best = None
    
    # To mimic random restarts, we generate many seeds
    rng = np.random.default_rng(seed)
    seeds = rng.1000000
    
    # We can't easily use Pool.imap because we need to check time per solve
    # Instead, we use a pool to run batches of work
    with mp.Pool(processes=num_procs) as pool:
        batch_size = num_procs
        idx = 0
        while time.process_time() - t0 < limit:
            # Submit a batch of tasks
            # Note: To strictly respect time_limit per call, we'd need a more complex
            # mechanism, but since we have 30s total, running as many as possible
            # in the 30s window is the goal. 
            tasks = [(n, seeds[i]) for i in range(idx, min(idx + batch_size, len(seeds)))]
            results = pool.map(_worker_task, tasks)
            
            for c, r in results:
                if best is None or r.sum() > best[1].sum():
                    best = (c, r)
            
            idx += batch_size
            if idx >= len(seeds): idx = 0 # Loop seeds if needed
            
    return best

def solve(n, seed, time_limit):
    import multiprocessing
    
    rng = np.random.default_rng(seed)
    t0 = time.process_time()
    best = None
    
    # Since we cannot easily share a Pool with specific time limits per task in pure Python 
    # without complexity, we use a pool to run batches of restarts.
    with multiprocessing.Pool(processes=multiprocessing.cpu_count()) as pool:
        while time.process_time() - t0 < TIME_FRACTION * time_limit:
            # Batch of 4 restarts
            seeds = (rng.integers(0, 2**31) for _ in range(4))
            results = pool.imap(_solve_once, [(n, s) for s in seeds])
            
            for c, r in results:
                if best is None or r.sum() > best[1].sum():
                    best = (c, r)
                    
            if time.process_time() - t0 > TIME_FRACTION * time_limit:
                break
                
    return best
