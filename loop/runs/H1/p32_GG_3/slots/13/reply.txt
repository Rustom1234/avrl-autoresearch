def _repair(c, r):
    """Shrink radii until the packing is valid, so that the harness never sees an overlap."""
    n = len(r)
    # Enforce boundary constraints initially
    r = np.minimum(r, np.minimum(c, 1 - c)).max(axis=1)
    r = np.maximum(r, 0)

    for _ in range(100):
        # Distance matrix between all centers
        d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
        # Overlap matrix
        over = r[:, None] + r[None] - d
        np.fill_diagonal(over, -1)
        
        max_idx = np.argmax(over)
        if over[max_idx] <= 0:
            break
            
        i, j = np.unravel_index(max_idx, over.shape)
        
        # Shrink radii based on overlap magnitude
        dist_ij = d[i, j]
        sum_r_ij = r[i] + r[j]
        if sum_r_ij > 1e-12:
            factor = max(0, dist_ij / sum_r_ij)
            r[i] *= factor
            r[j] *= factor
        
        # Re-enforce boundary constraints
        r = np.minimum(r, np.minimum(c, 1 - c)).max(axis=1)
        r = np.maximum(r, 0)

    return np.maximum(r - SHRINK, 0)
