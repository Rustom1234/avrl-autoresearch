def _solve_once(n, rng):
    iu = np.triu_indices(n, 1)

    def cons(c, r, t):
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]] + t
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    # stage 1: overlaps may exceed zero by a shared slack t, which is penalised in the objective
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS), [SLACK0]])
    res = minimize(lambda v: -v[2 * n:3 * n].sum() + PENALTY * v[-1], x0, method="SLSQP",
                   constraints={"type": "ineq", "fun": lambda v: np.concatenate(
                       [cons(v[:2 * n].reshape(n, 2), v[2 * n:3 * n], v[-1]), [v[-1]]])},
                   options={"maxiter": MAX_ITER})
    # stage 2: slack fixed at zero, true constraints, warm-started from stage 1
    x1 = np.concatenate([res.x[:2 * n], res.x[2 * n:3 * n]])
    res = minimize(lambda v: -v[2 * n:].sum(), x1, method="SLSQP",
                   constraints={"type": "ineq", "fun": lambda v: cons(v[:2 * n].reshape(n, 2), v[2 * n:], 0.0)},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
