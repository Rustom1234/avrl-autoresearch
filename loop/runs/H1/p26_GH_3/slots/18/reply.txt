def _grid(n):
    """Fixed square-grid layout: the first n cells of a k x k grid, with equal small radii."""
    k = int(np.ceil(np.sqrt(n)))
    g = (np.arange(k) + 0.5) / k
    c = np.array([(x, y) for x in g for y in g])[:n]
    return np.concatenate([c.ravel(), np.full(n, 0.5 / k)])
