def _greedy(n, rng, n_cand=2000):
    """Place circles one by one at the candidate point with the largest gap to walls and existing circles."""
    c, r = np.zeros((n, 2)), np.zeros(n)
    for k in range(n):
        p = rng.uniform(0, 1, (n_cand, 2))
        g = np.minimum(p, 1 - p).min(1)
        if k > 0:
            d = np.sqrt(((p[:, None] - c[None, :k]) ** 2).sum(-1)) - r[None, :k]
            g = np.minimum(g, d.min(1))
        i = np.argmax(g)
        c[k], r[k] = p[i], max(g[i], 0)
    return c, r
