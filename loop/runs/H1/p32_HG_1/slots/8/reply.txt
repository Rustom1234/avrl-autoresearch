    n = len(r)
    r = np.minimum(r, np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1])))

        np.fill_diagonal(over, -1)

            break

        r[i], r[j] = r[i] * s, r[j] * s

def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    curr_c, curr_r = _solve_once(n, rng)
    curr_c = np.clip(curr_c, 0, 1)
    curr_r = _repair(curr_c, curr_r)
    
    if best is None or curr_r.sum() > best[1].sum():
        best = (curr_c, curr_r)

    # Simulated Annealing over SLSQP local optima
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        # Propose: perturb one circle's position
        idx = rng.integers(0, n)
        next_c = curr_c.copy()
        next_c[idx] = rng.uniform(0, 1, 2)
        
        # Refine perturbation to local optimum
        next_c, next_r = _solve_once(n, rng)
        next_c = np.clip(next_c, 0, 1)
        next_r = _repair(next_c, next_r)
        
        # Acceptance
        diff = next_r.sum() - curr_r.sum()
        temp = 0.01 * (1 - (time.process_time() - t0) / (TIME_FRACTION * time_limit))**5
        temp = max(temp, 1e-4)
        
        if diff > 0 or rng.random() < np.exp(diff / temp):
            curr_c, curr_r = next_c, next_r
            if curr_r.sum() > best[1].sum():
