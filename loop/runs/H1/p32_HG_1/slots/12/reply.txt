def _solve_once(n, rng):
    batch_size = 32
    # Vectorized penalty descent
    # coords: (batch, n, 2), radii: (batch, n)
    coords = rng.uniform(0.1, 0.9, (batch_size, n, 2))
    radii = rng.uniform(0.01, 0.08, (batch_size, n))
    
    lr = 0.001
    for step in range(2000):
        # Objective: sum(radii)
        # Penalty: overlap and boundary
        penalty_weight = 10 * (1.1 ** (step / 100))
        
        # Distances
        diff = coords[:, None, :] - coords[:, :, None, :]
        dist = np.linalg.norm(diff, axis=-1)
        # Overlap penalty
        over = (radii[:, None] + radii[:, None, :]) - dist
        over[np.eye(n, dtype=bool)] = -1
        
        # Gradients (simplified)
        grad_c = np.zeros_like(coords)
        grad_r = np.zeros_like(radii)
        
        # Boundary penalty
        grad_c[..., 0] += penalty_weight * (coords[..., 0] - radii[:, :, None] - 1e-8)
        grad_c[..., 1] += penalty_weight * (coords[..., 1] - radii[:, :, None] - 1e-8)
        
        # Overlap penalty
        # This is a simplified gradient for speed
        mask = over > 0
        # Note: Full gradient for overlapping circles is omitted for simplicity/speed
        # just move centers apart and shrink radii
        grad_c += penalty_weight * (coords - 0.5) * 0.01 
        
        coords -= lr * grad_c
        radii -= lr * grad_r
        
        if step % 500 == 0:
            # Project to valid space to keep stability
            coords = np.clip(coords, 0.01, 0.99)
            radii = np.clip(radii, 0, 0.2)

    # Pick top 4 and polish
    best_idx = np.argsort(radii.sum(axis=1))[::-1][:4]
    best_r_sum = -1
    best_res = (None, None)

        # Fallback to a decent starting point if SLSQP fails
        coords = coords[best_idx[0]]
        radii = radii[best_idx[0]]
        return coords, radii

    return best_res
