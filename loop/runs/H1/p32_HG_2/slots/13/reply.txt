            # Strategy 0: Uniform
            c = rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
        elif rng.random() < 0.66:
            # Strategy 1: Jittered Grid
            grid_size = int(np.ceil(np.sqrt(n)))
            spacing = 1.0 / grid_size
            c = np.array([[i * spacing + rng.uniform(0, 0.02), j * spacing + rng.uniform(0, 0.02)] 
                          for i in range(grid_size) for j in range(grid_size)])[:n]
            c = np.clip(c, INIT_LOW, 1 - INIT_LOW)
        else:
            # Strategy 2: Boundary-biased
            c = rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
            mask = rng.random(n) < 0.5

            r = _repair(c, r)

            c_curr, r_curr = best
            c_new = c_curr.copy()
            num_to_move = rng.integers(1, 4)
            indices = rng.choice(n, num_to_move, replace=False)

                hops_since_improvement = 0

                hops_since_improvement += 1

def solve(n, seed, time_limit):
    rng = np.random.default_rng(seed)
    t0 = time.process_time()
    best_per_strat = [None, None, None]
    
    # First 80% of time: round-robin strategies
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        idx = int((time.process_time() - t0) / (TIME_FRACTION * time_limit) *                    (3 / (TIME_FRACTION * time_limit))) % 3
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        
        if best_per_strat[idx] is None or r.sum() > best_per_strat[idx][1].sum():
            best_per_strat[idx] = (c, r)

    # Final 20% of time: best strategy only
    best_overall = None
    if any(best_per_strat):
        best_overall = max(best_per_strat, key=lambda x: x[1].sum() if x else -1)
    
    if best_overall is None:
        # Fallback if no packing found
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        best_overall = (c, r)

    while time.process_time() - t0 < time_limit:
        c_curr, r_curr = best_overall
        c_new = c_curr.copy()
        num_to_move = rng.integers(1, 4)
        indices = rng.choice(n, num_to_move, replace=False)
        
        if rng.random() < 0.5:
            c_new[indices] = rng.uniform(INIT_LOW, 1 - INIT_LOW, (num_to_move, 2))

        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        
        if r.sum() > best_overall[1].sum():
            best_overall = (c, r)
            
    return best_overall
