from scipy.optimize import least_squares

def _solve_once(n, rng, x0=None):
    if x0 is None:
        x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)

    def get_residuals(v):
        c = v[:2 * n].reshape(n, 2)
        r = v[2 * n:]
        d_ij = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1))
        res = np.concatenate([
            np.maximum(0, r[iu[0]] + r[iu[1]] - d_ij),
            np.maximum(0, r - c[:, 0]),
            np.maximum(0, r - (1 - c[:, 0])),
            np.maximum(0, r - c[:, 1]),
            np.maximum(0, r - (1 - c[:, 1])),
            -1e-4 * r
        ])
        return res

    x_ls = x0.copy()
    for w in [1.0, 10.0, 100.0]:
        res_ls = least_squares(lambda v: w * get_residuals(v), x_ls, method='trf', xtol=1e-4)
        x_ls = res_ls.x

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -v[2 * n:].sum(), x_ls, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
