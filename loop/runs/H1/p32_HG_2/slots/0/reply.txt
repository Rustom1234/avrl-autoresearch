def _solve_once(n, rng):
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    
    # Simple neighbor filtering to reduce constraint count
    # Only include pairs that are potentially overlapping or close
    iu = []
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = np.sum((x0[:2*n][i] - x0[:2*n][j])**2)
            if dist_sq < 0.5: # Heuristic for neighbor
                iu.append((i, j))
    iu = np.array(iu)
    
    if len(iu) == 0:
        iu = np.triu_indices(n, 1)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        c_pair = c[iu[:, 0]]
        c_pair2 = c[iu[:, 1]]
        r_pair = r[iu[:, 0]]
        r_pair2 = r[iu[:, 1]]
        d = np.sqrt(((c_pair - c_pair2) ** 2).sum(1)) - r_pair - r_pair2
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", 
                     constraints={"type": "ineq", "fun": cons},
                     options={"maxiter": MAX_ITER, "ftol": 1e-12})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
