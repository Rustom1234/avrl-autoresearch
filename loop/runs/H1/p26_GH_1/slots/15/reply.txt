def _push(c, iters=30, step=0.02):
    """Force-directed warm start: 1/d^2 repulsion between centres, kept inside the square."""
    n = len(c)
    eye = np.eye(n)
    for _ in range(iters):
        diff = c[:, None] - c[None]
        d3 = ((diff ** 2).sum(-1) + eye) ** 1.5   # |diff| / d3 has magnitude 1/d^2
        f = np.clip((diff / d3[..., None]).sum(1), -1, 1)
        c = np.clip(c + step * f, INIT_RADIUS, 1 - INIT_RADIUS)
    return c
