def _hex_init(n, rng):
    """Hexagonal lattice points, subsampled to n, squashed into the square, then jittered."""
    m = int(np.ceil(np.sqrt(n))) + 2
    pts = np.array([(j + 0.5 * (i % 2), i * np.sqrt(3) / 2) for i in range(m) for j in range(m)])
    p = pts[rng.choice(len(pts), n, replace=False)]
    p = (p - p.min(0)) / (p.max(0) - p.min(0))   # squash into [0, 1]^2
    p = INIT_LOW + (1 - 2 * INIT_LOW) * p + rng.normal(0, 0.02, p.shape)
    return np.clip(p, 0, 1)
