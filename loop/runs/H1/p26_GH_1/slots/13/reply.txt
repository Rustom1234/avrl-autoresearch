def solve(n, seed, time_limit):
    rng, t0, best, stall = np.random.default_rng(seed), time.process_time(), None, 0
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        if best is not None and stall >= 5:   # stalled: move the smallest circle of the best packing and re-solve
            c, r = best[0].copy(), best[1].copy()
            k = np.argmin(r)
            c[k] = rng.uniform(INIT_LOW, 1 - INIT_LOW, 2)
            r[k] *= 1.2
            c, r = _solve_once(n, rng, np.concatenate([c.ravel(), r]))
            stall = 0
        else:
            c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best, stall = (c, r), 0
        else:
            stall += 1
    return best
