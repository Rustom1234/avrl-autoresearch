def _shake(c, r, rng, steps=40, dt=0.05):
    """Kick the centres, then let velocity-damped repulsion from neighbours and walls settle them."""
    c = c + rng.normal(0, 0.02, c.shape)
    v = np.zeros_like(c)
    eye = np.eye(len(c))
    for _ in range(steps):
        d = c[:, None] - c[None]
        dist = np.sqrt((d ** 2).sum(-1)) + eye
        over = np.maximum(r[:, None] + r[None] - dist, 0) * (1 - eye)
        f = ((over / dist)[..., None] * d).sum(1)
        for k in range(2):
            f[:, k] += np.maximum(r - c[:, k], 0) - np.maximum(c[:, k] + r - 1, 0)
        v = 0.8 * v + dt * f
        c = np.clip(c + v, 0, 1)
    return c
