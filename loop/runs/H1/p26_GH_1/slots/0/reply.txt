def _solve_once(n, rng):
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)

    def pen(v, w):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1) + 1e-12)
        ov = np.maximum(r[iu[0]] + r[iu[1]] - d, 0)
        bd = np.maximum(np.concatenate([r - c[:, 0], r - c[:, 1], c[:, 0] + r - 1, c[:, 1] + r - 1]), 0)
        return -r.sum() + w * ((ov ** 2).sum() + (bd ** 2).sum())

    v = x0
    for w in (1e2, 1e4, 1e6, 1e8):   # staged penalty weights
        v = minimize(pen, v, args=(w,), method="L-BFGS-B", options={"maxiter": MAX_ITER}).x
    return v[:2 * n].reshape(n, 2), v[2 * n:]
