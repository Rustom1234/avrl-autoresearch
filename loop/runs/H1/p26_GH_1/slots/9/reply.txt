def _greedy_init(n, rng, m=400):
    """Place circles one by one at the candidate point with the most free space."""
    c, r = np.zeros((0, 2)), np.zeros(0)
    for _ in range(n):
        p = rng.uniform(0, 1, (m, 2))
        rad = np.minimum(p, 1 - p).min(1)   # distance to the walls
        if len(r):
            rad = np.minimum(rad, (np.sqrt(((p[:, None] - c[None]) ** 2).sum(-1)) - r[None]).min(1))
        k = np.argmax(rad)
        c = np.vstack([c, p[k]])
        r = np.append(r, max(rad[k], 0))
    return c, r
