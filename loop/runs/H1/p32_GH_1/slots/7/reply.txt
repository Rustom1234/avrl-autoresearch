def _solve_once(n, rng):
    c = rng.uniform(INIT_LOW, 1 - INIT_LOW, (1, 2))
    r = np.full(1, INIT_RADIUS)
    for m in range(1, n + 1):   # add one circle at a time and re-optimise all of them
        if m > 1:
            c = np.vstack([c, rng.uniform(INIT_LOW, 1 - INIT_LOW, (1, 2))])
            r = np.append(r, INIT_RADIUS)
        caps = RADIUS_CAPS if m == n else (1.0,)
        x = _optimize(m, np.concatenate([c.ravel(), r]), caps)
        c, r = x[:2 * m].reshape(m, 2), x[2 * m:]
    return c, r
