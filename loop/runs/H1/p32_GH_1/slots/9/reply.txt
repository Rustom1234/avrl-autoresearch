def _greedy_init(n, rng, k=200):
    """Place circles one by one at the candidate point with the largest free radius."""
    c, r = np.zeros((0, 2)), np.zeros(0)
    for _ in range(n):
        cand = rng.uniform(0, 1, (k, 2))
        free = np.minimum(np.minimum(cand, 1 - cand).min(1), 1.0)
        if len(r):
            d = np.sqrt(((cand[:, None] - c[None]) ** 2).sum(-1)) - r[None]
            free = np.minimum(free, d.min(1))
        j = np.argmax(free)
        c = np.vstack([c, cand[j]])
        r = np.append(r, max(free[j], 1e-3))
    return np.concatenate([c.ravel(), r])
