def _momentum_pass(x, n, steps=100, lr=1e-3, beta=0.9, mu=100.0):
    """Momentum gradient descent on -sum(r) + mu * (squared overlap and wall violations), analytic gradient."""
    iu = np.triu_indices(n, 1)
    vel = np.zeros_like(x)
    for _ in range(steps):
        c, r = x[:2 * n].reshape(n, 2), x[2 * n:]
        gc, gr = np.zeros((n, 2)), -np.ones(n)
        diff = c[iu[0]] - c[iu[1]]
        d = np.sqrt((diff ** 2).sum(1)) + 1e-12
        ov = np.maximum(0, r[iu[0]] + r[iu[1]] - d)
        np.add.at(gr, iu[0], 2 * mu * ov)
        np.add.at(gr, iu[1], 2 * mu * ov)
        g = (2 * mu * ov / d)[:, None] * diff
        np.add.at(gc, iu[0], -g)
        np.add.at(gc, iu[1], g)
        lo = np.maximum(0, r[:, None] - c)
        hi = np.maximum(0, r[:, None] + c - 1)
        gc += 2 * mu * (hi - lo)
        gr += 2 * mu * (lo.sum(1) + hi.sum(1)) - 2 * mu * np.maximum(0, -r)
        vel = beta * vel - lr * np.concatenate([gc.ravel(), gr])
        x = x + vel
    return x
