            Descriptors.MolLogP(mol),

            Descriptors.NumHDonors(mol),
            Descriptors.NumHAcceptors(mol),
            Descriptors.NumRotatableBonds(mol),
            Descriptors.FractionCSP3(mol),

        ]
        fps.append(np.concatenate([fp, np.array(descs, dtype=np.float32)]))
    
    return np.array(fps, dtype=np.float32)

    # To follow the prompt strictly (RF params) while maintaining the CatBoost model:
    # We will use a simple grid search for CatBoost params as it's the current architecture.

                                     loss_function='MAE', random_seed=seed, 
                                     thread_count=8, verbose=0)

        **best_params,
        loss_function='MAE',
        random_seed=seed,
        thread_count=8,

    )
    model.fit(X_train, y_train)
    
    X_test = featurize(test_smiles)
    return model.predict(X_test)
