from rdkit import Chem
from rdkit import DataStructs
from rdkit.Chem import rdFingerprintGenerator, Descriptors
from catboost import CatBoostRegressor
from scipy.stats import spearmanr

# Settings
N_BITS = 2048
RADIUS = 2
ITERATIONS = 1500
LEARNING_RATE = 0.05
DEPTH = 6


def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    fps = []
    for s in smiles:
        mol = Chem.MolFromSmiles(s)
        if mol is None: continue
        fp = gen.GetFingerprintAsNumPy(mol)
        
        # Physchem descriptors
        descs = [
            Descriptors.MolLogP(mol),
            Descriptors.MolWt(mol),
            Descriptors.TPSA(mol),
            Descriptors.NumHDonors(mol),
            Descriptors.NumHAcceptors(mol),
            Descriptors.NumRotatableBonds(mol),
            Descriptors.FractionCSP3(mol),
            Descriptors.HeavyAtomCount(mol),
            Descriptors.NumAromaticRings(mol)
        ]
        fps.append(np.concatenate([fp, np.array(descs, dtype=np.float32)]))
    
    return np.array(fps, dtype=np.float32)


def fit_predict(train_smiles, train_y, test_smiles, seed):
    X_train = featurize(train_smiles)
    y_train = np.asarray(train_y, dtype=np.float64)
    
    # Supervised bit selection
    # X_train shape is (N, N_BITS + 9)
    # Calculate Spearman correlation for the Morgan bits only
    correlations = []
    for i in range(N_BITS):
        corr, _ = spearmanr(X_train[:, i], y_train)
        correlations.append(abs(corr))
    
    # Get indices of top 400 bits
    top_indices = np.argsort(correlations)[-400:]
    
    # Construct feature index list: top bits + the last 9 descriptors
    # We need to be careful: we want to keep the relative order if possible, 
    # but simple concatenation is fine as long as it's consistent.
    final_indices = np.concatenate([top_indices, np.arange(N_BITS, N_BITS + 9)])
    
    X_train_selected = X_train[:, final_indices]
    X_test = featurize(test_smiles)
    X_test_selected = X_test[:, final_indices]
    
    model = CatBoostRegressor(
        iterations=ITERATIONS,
        learning_rate=LEARNING_RATE,
        depth=DEPTH,
        loss_function='MAE',
        random_seed=seed,
        thread_count=8,
        verbose=0
    )
    model.fit(X_train_selected, y_train)
    
    return model.predict(X_test_selected)
