from sklearn.ensemble import RandomForestRegressor
from sklearn.svm import SVR
from sklearn.linear_model import LinearRegression
from sklearn.model_selection import KFold
from lightgbm import LGBMRegressor

# Settings
N_BITS = 2048
RADIUS = 2
N_TREES = 200
MAX_FEATURES = 0.3
MIN_SAMPLES_LEAF = 1


def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    return np.array([gen.GetFingerprintAsNumPy(Chem.MolFromSmiles(s)) for s in smiles], dtype=np.float32)


def fit_predict(train_smiles, train_y, test_smiles, seed):
    X = featurize(train_smiles)
    y = np.asarray(train_y, dtype=np.float64).reshape(-1, 1)
    
    kf = KFold(n_splits=5, shuffle=True, random_state=seed)
    oof_preds = np.zeros((X.shape[0], 3))
    
    # Base models
    rf_model = RandomForestRegressor(n_estimators=N_TREES, max_features=MAX_FEATURES, min_samples_leaf=MIN_SAMPLES_LEAF, n_jobs=-1, random_state=seed)
    lgb_model = LGBMRegressor(n_estimators=N_TREES, max_depth=7, learning_rate=0.1, importance_type='gain', random_state=seed, verbose=-1)
    svr_model = SVR(kernel='linear', C=1.0)

    for train_idx, val_idx in kf.split(X):
        X_tr, X_val = X[train_idx], X[val_idx]
        y_tr, y_val = y[train_idx], y[val_idx]
        
        rf_model.fit(X_tr, y_tr)
        lgb_model.fit(X_tr, y_tr)
        svr_model.fit(X_tr, y_tr)
        
        oof_preds[val_idx, 0] = rf_model.predict(X_val)
        oof_preds[val_idx, 1] = lgb_model.predict(X_val)
        oof_preds[val_idx, 2] = svr_model.predict(X_val)

    # Meta-learner
    meta_model = LinearRegression(positive=True)
    meta_model.fit(oof_preds, y)
    
    # Final prediction
    X_test = featurize(test_smiles)
    rf_test = rf_model.predict(X_test)
    lgb_test = lgb_model.predict(X_test)
    svr_test = svr_model.predict(X_test)
    
    test_meta_input = np.column_stack([rf_test, lgb_test, svr_test])
    return meta_model.predict(test_meta_input).flatten()
