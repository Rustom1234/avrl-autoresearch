RADIUS = 2
ITERATIONS = 1500
LEARNING_RATE = 0.05
DEPTH = 6

        descs = [
            Descriptors.MolLogP(mol),

            Descriptors.TPSA(mol),
            Descriptors.NumHDonors(mol),
            Descriptors.NumHAcceptors(mol),
            Descriptors.NumRotatableBonds(mol),
            Descriptors.FractionCSP3(mol),
            Descriptors.HeavyAtomCount(mol),
            Descriptors.NumAromaticRings(mol)
        ]
        fps.append(np.concatenate([fp, np.array(descs, dtype=np.float32)]))
    
    return np.array(fps, dtype=np.float32)

    model = CatBoostRegressor(

        depth=DEPTH,
        loss_function='MAE',

    )
    model.fit(X_train, y_train)

ALPHA = 0.3

        if mol is None: continue
        fp = gen.GetFingerprintAsNumPy(mol)
        
        # Physchem descriptors
        descs = [
            Descriptors.MolLogP(mol),
            Descriptors.MolWt(mol),
            Descriptors.TPSA(mol),
            Descriptors.NumHDonors(mol),
            Descriptors.NumHAcceptors(mol),
            Descriptors.NumRotatableBonds(mol),
            Descriptors.FractionCSP3(mol),
            Descriptors.HeavyAtomCount(mol),
            Descriptors.NumAromaticRings(mol)
        ]
        fps.append(np.concatenate([fp, np.array(descs, dtype=np.float32)]))
    
    return np.array(fps, dtype=np.float32)


def get_tanimoto_kernel(fps):
    n = fps.shape[0]
    K = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            # Use only the bit part of the fingerprint for Tanimoto
            # The bits are the first N_BITS columns
            fp_i = fps[i, :N_BITS]
            fp_j = fps[j, :N_BITS]
            intersection = np.logical_and(fp_i, fp_j).sum()
            union = np.logical_or(fp_i, fp_j).sum()
            sim = intersection / union if union > 0 else 0.0
            K[i, j] = K[j, i] = sim
    return K + np.eye(n)


def fit_predict(train_smiles, train_y, test_smiles, seed):
    X_train = featurize(train_smiles)
    y_train = np.asarray(train_y, dtype=np.float64)
    
    # Extract bit features for kernel
    fps_train = X_train[:, :N_BITS]
    K = get_tanimoto_kernel(fps_train)
    
    # Kernel Ridge Regression
    # solve (K + alpha*I) coef = y_centered
    y_mean = np.mean(y_train)
    y_centered = y_train - y_mean
    
    # Regularize the kernel
    A = K + ALPHA * np.eye(K.shape[0])
    coef = np.linalg.solve(A, y_centered)
    
    X_test = featurize(test_smiles)
    fps_test = X_test[:, :N_BITS]
    
    # Build K_test
    K_test = np.zeros((X_test.shape[0], K.shape[0]))
    for i in range(X_test.shape[0]):
        for j in range(K.shape[0]):
            fp_i = fps_test[i, :]
            fp_j = fps_train[j, :]
            intersection = np.logical_and(fp_i, fp_j).sum()
            union = np.logical_or(fp_i, fp_j).sum()
            sim = intersection / union if union > 0 else 0.0
            K_test[i, j] = sim
            
    return K_test @ coef + y_mean

    bits = fps[:, :N_BITS]

        for j in range(i + 1, n):

            sim = inter / union if union > 0 else 0.0

    X_train = featurize(train_smiles)
    y_train = np.asarray(train_y, dtype=np.float64)
    
    K = get_tanimoto_kernel(X_train)
    
    # Solve (K + alpha*I) * coef = y
    # Use y - mean(y) for centering
    y_centered = y_train - np.mean(y_train)
    # Add regularization to diagonal
    K_reg = K + ALPHA * np.eye(K.shape[0])
    
    try:
        coef = np.linalg.solve(K_reg, y_centered)

    predictions = K_test @ coef + np.mean(y_train)
    return predictions
