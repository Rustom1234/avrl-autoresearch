def fit_predict(train_smiles, train_y, test_smiles, seed):
    train_fps = featurize(train_smiles)
    test_fps = featurize(test_smiles)
    
    # Calculate weights based on similarity to the test set
    # We use the mean similarity to all test points as a proxy for 'relevance'
    weights = np.ones(len(train_fps))
    if len(test_fps) > 0:
        # To stay within time/memory limits, we sample a few test points if set is large
        sample_idx = np.random.choice(len(test_fps), min(len(test_fps), 50), replace=False, random_state=seed)
        sample_test_fps = test_fps[sample_idx]
        
        for i, fp in enumerate(train_fps):
            sims = [DataStructs.SimilaritySimilarity(fp, tp) for tp in sample_test_fps]

RADIUS = 2
N_TREES = 1000
LEARNING_RATE = 0.05
NUM_LEAVES = 31

    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    data = []
    for s in smiles:
        mol = Chem.MolFromSmiles(s)
        if mol is None: continue
        fp = gen.GetFingerprintAsNumPy(mol)
        c = sum(1 for atom in mol.GetAtoms() if atom.GetSymbol() == 'C')

        s_ = sum(1 for atom in mol.GetAtoms() if atom.GetSymbol() == 'S')
        data.append(np.concatenate([fp, [c, n, o, s_]]))
    return np.array(data, dtype=np.float32)
