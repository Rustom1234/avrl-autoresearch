def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    fps = []
    for s in smiles:
        mol = Chem.MolFromSmiles(s)
        fp = gen.GetFingerprintAsNumPy(mol)
        # Add heavy atom count and hydrogen count
        heavy_count = np.array([1 for _ in range(mol.GetNumHeavyAtoms())], dtype=np.float32)
        h_count = np.array([1 for _ in range(mol.GetNumHydrogenAtoms())], dtype=np.float32)
        fps.append(np.concatenate([fp, heavy_count, h_count]))
    return np.array(fps, dtype=np.float32)

def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    fps = []
    for s in smiles:
        mol = Chem.MolFromSmiles(s)
        if mol is None: continue
        fp = gen.GetFingerprintAsNumPy(mol)
        h_count = mol.GetNumHydrogenAtoms()
        heavy_count = mol.GetNumHeavyAtoms()
        fps.append(np.concatenate([fp, [heavy_count, h_count]]))
    return np.array(fps, dtype=np.float32)
