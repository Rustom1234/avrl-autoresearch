from lightgbm import LGBMRegressor
from xgboost import XGBRegressor
from catboost import CatBoostRegressor

# Settings
N_BITS = 2048
RADIUS = 2
N_TREES = 300
FEATURE_FRACTION = 0.8
BAG_FRACTION = 0.8


def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    return np.array([gen.GetFingerprintAsNumPy(Chem.MolFromSmiles(s)) for s in smiles], dtype=np.float32)


def fit_predict(train_smiles, train_y, test_smiles, seed):
    x_train = featurize(train_smiles)
    x_test = featurize(test_smiles)
    y = np.asarray(train_y, dtype=np.float64)
    
    # LightGBM
    m1 = LGBMRegressor(n_estimators=N_TREES, feature_fraction=FEATURE_FRACTION, 
                         bag_fraction=BAG_FRACTION, random_state=seed, n_jobs=-1, verbosity=-1)
    m1.fit(x_train, y)
    
    # XGBoost
    m2 = XGBRegressor(n_estimators=N_TREES, colsample_bytree=FEATURE_FRACTION, 
                        subsample=BAG_FRACTION, random_state=seed, n_jobs=-1)
    m2.fit(x_train, y)
    
    # CatBoost
    m3 = CatBoostRegressor(iterations=N_TREES, colsample_bylevel=FEATURE_FRACTION, 
                            subsample=BAG_FRACTION, random_seed=seed, verbose=0)
    m3.fit(x_train, y)
    
    preds = (m1.predict(x_test) + m2.predict(x_test) + m3.predict(x_test)) / 3.0
    return preds
