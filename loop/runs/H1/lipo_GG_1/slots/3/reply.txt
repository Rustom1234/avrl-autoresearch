def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    fps = []
    weights = []
    for s in smiles:
        mol = Chem.MolFromSmiles(s)
        if mol:
            fps.append(gen.GetFingerprintAsNumPy(mol))
            weights.append(Chem.MolWt(mol))
    
    features = np.stack([np.array(fps), np.array(weights)], axis=1)
    return features.reshape(len(smiles), -1).astype(np.float32)
