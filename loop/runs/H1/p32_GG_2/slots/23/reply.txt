def _solve_once(n, rng):
    # Jittered grid initialization for better initial coverage
    grid_size = int(np.ceil(np.sqrt(n)))
    grid_x = np.linspace(0, 1, grid_size)
    grid_y = np.linspace(0, 1, grid_size)
    grid_coords = np.array([[x, y] for x in grid_x for y in grid_y]).reshape(-1, 2)
    
    # Shuffle and pad the grid to match n
    indices = rng.permutation(len(grid_coords))
    selected_coords = grid_coords[indices[:n]]
    
    # Add jitter
    jitter = rng.uniform(-0.05, 0.05, (n, 2))
    c_init = np.clip(selected_coords + jitter, 0, 1)
    
    x0 = np.concatenate([c_init.flatten(), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
