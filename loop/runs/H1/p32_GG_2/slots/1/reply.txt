def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    elapsed = 0
    restart_count = 0
    while elapsed < TIME_FRACTION * time_limit:
        # Dynamically decrease the time fraction allowed for each attempt\n        t_frac = max(0.1, 1.0 - (restart_count * 0.05))
        c, r = _solve_once(n, rng, t_frac)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
        elapsed = time.process_time() - t0
        restart_count += 1
    return best
