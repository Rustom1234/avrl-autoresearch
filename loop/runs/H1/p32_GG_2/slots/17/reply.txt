def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        elapsed = time.process_time() - t0
        # Progress from 0 to 1
        progress = elapsed / (TIME_FRACTION * time_limit)
        # Linear increase of iterations: 200 at start, 1000 at deadline
        current_max_iter = int(BASE_MAX_ITER + (1000 - BASE_MAX_ITER) * progress)
        c, r = _solve_once(n, rng, current_max_iter)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    return best
