def _solve_once(n, rng):
    # Jittered hexagonal grid initialization
    # Determine grid dimensions (approximate for 32 circles)
    # A square of side ~0.5 gives 32 hex cells (0.5 / (sqrt(3)/2) * 0.5 * 2 approx 30)
    # We use a slightly smaller spacing to ensure we have enough points
    spacing = 0.45
    grid_x = np.arange(0, 1.1, spacing)
    grid_y = np.arange(0, 1.1, spacing * np.sqrt(3)/2)
    pts = []
    for i, x in enumerate(grid_x):
        for j, y in enumerate(grid_y):
            pts.append([x, y])
    pts = np.array(pts)
    # Filter and jitter
    valid_pts = pts[(pts[:, 0] >= 0) & (pts[:, 0] <= 1) & (pts[:, 1] >= 0) & (pts[:, 1] <= 1)]
    if len(valid_pts) < n:
        # Fallback to random if grid is too sparse
        valid_pts = rng.random((n, 2)) * 2 - 1
    
    centers = valid_pts[:n] + rng.standard_normal((n, 2)) * 0.02
    centers = np.clip(centers, INIT_LOW, 1 - INIT_LOW)
    x0 = np.concatenate([centers.flatten(), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)
