def _solve_once(n, rng):
    centers = np.empty((n, 2))
    radii = np.empty(n)
    for i in range(n):
        # Sample potential centers and find the one that allows the largest radius
        # This is a greedy approach to find a good starting point for SLSQP
        samples = rng.random((100, 2))
        best_r = -1.0
        best_p = samples[0]
        for p in samples:
            # Distance to boundaries
            d_bound = min(p[0], 1 - p[0], p[1], 1 - p[1])
            # Distance to existing circles
            d_circles = []
            if i > 0:
                d_circles = np.linalg.norm(centers[:i] - p, axis=1) - radii[:i]
            d_circ_min = np.min(d_circles) if d_circles else 10.0
            
            r_possible = min(d_bound, d_circ_min)
            if r_possible > best_r:
                best_r = r_possible
                best_p = p
        
        centers[i] = best_p
        radii[i] = best_r * 0.95  # Leave a little room for the optimizer

    x0 = np.concatenate([centers.flatten(), radii])
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
