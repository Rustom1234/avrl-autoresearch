def _solve_once(n, rng):
    # Use a jittered grid to distribute centers more evenly than pure random
    grid_side = int(np.ceil(np.sqrt(n)))
    x0_centers = np.zeros((n, 2))
    indices = np.arange(n)
    for i, idx in enumerate(indices):
        r_idx = idx // grid_side
        c_idx = idx % grid_side
        # Map to [0, 1] with slight jitter
        x0_centers[idx, 0] = (r_idx + rng.random()) / grid_side
        x0_centers[idx, 1] = (c_idx + rng.random()) / grid_side
    
    # Ensure centers are not too close to edges for initial radii
    x0_centers = np.clip(x0_centers, INIT_LOW, 1 - INIT_LOW)
    x0 = np.concatenate([x0_centers.flatten(), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)
