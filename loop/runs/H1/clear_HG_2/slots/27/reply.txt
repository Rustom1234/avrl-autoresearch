from rdkit import Chem

from scipy.stats import rankdata, norm

# Settings
N_BITS = 2048
RADIUS = 2

def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    feats = []
    for s in smiles:
        mol = Chem.MolFromSmiles(s)
        if mol is None: continue
        fp = gen.GetFingerprintAsNumPy(mol)
        desc = np.array([Descriptors.MolWt(mol), Descriptors.MolLogP(mol), 
                          Descriptors.NumHDonors(mol), Descriptors.NumHAcceptors(mol), 
                          Descriptors.NumRotatableBonds(mol), Descriptors.TPSA(mol)])
        feats.append(np.concatenate([fp, desc]))
    return np.array(feats, dtype=np.float32)


def fit_predict(train_smiles, train_y, test_smiles, seed):
    model = XGBRegressor(n_estimators=500, max_depth=5, learning_rate=0.03, 
                           subsample=0.8, colsample_bytree=0.5, 
                           min_child_weight=3, random_state=seed, 
                           n_jobs=-1, tree_method='hist')
    X_train = featurize(train_smiles)
    y_train = np.asarray(train_y, dtype=np.float64)
    
    # Rank transformation
    ranks = rankdata(y_train)
    y_train_rank = norm.ppf((ranks - 0.5) / len(y_train))
    
    model.fit(X_train, y_train_rank)
    preds_rank = model.predict(featurize(test_smiles))
    
    # Inverse rank transformation
    # Note: we only have the ranking of the training set, so we return the rank-space prediction
    # or transform it back if we had the test ranks. 
    # Given the goal is to minimize Spearman, fitting on ranks is sufficient.
    # However, to return the correct scale, we use the training indices to map back if possible,
    # but since we only have test_smiles, we return the ranks which is standard for rank-loss.
    
    # Actually, for XGBoost, we need to return the original scale if possible. 
    # Since we can't easily get test ranks, we'll return the model's output.
    # A better way is to return the transformed values if the evaluation is rank-based.
    
    # Let's just return the predictions in rank space for now as a simple implementation.
    return preds_rank
