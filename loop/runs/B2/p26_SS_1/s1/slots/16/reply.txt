def _polish(n, c, r):
    """One tight-tolerance SLSQP run from a raw (unrepaired) packing."""
    iu = np.triu_indices(n, 1)

    def cons(v):
        c_, r_ = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c_[iu[0]] - c_[iu[1]]) ** 2).sum(1)) - r_[iu[0]] - r_[iu[1]]
        return np.concatenate([d, c_[:, 0] - r_, c_[:, 1] - r_, 1 - c_[:, 0] - r_, 1 - c_[:, 1] - r_, r_])

    x0 = np.concatenate([c.ravel(), r])
    res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": 1000, "ftol": 1e-13})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
