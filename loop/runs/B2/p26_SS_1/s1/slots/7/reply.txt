def _maximin(n, rng, m=400):
    """Farthest-point sampling: greedily pick n of m random candidates spread out over the square."""
    cand = rng.uniform(INIT_LOW, 1 - INIT_LOW, (m, 2))
    chosen = [int(rng.integers(m))]
    d = np.linalg.norm(cand - cand[chosen[0]], axis=1)
    for _ in range(n - 1):
        j = int(np.argmax(d))
        chosen.append(j)
        d = np.minimum(d, np.linalg.norm(cand - cand[j], axis=1))
    return cand[chosen]
