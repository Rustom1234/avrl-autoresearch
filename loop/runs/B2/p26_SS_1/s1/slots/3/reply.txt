def _maximin(n, rng, m=400):
    """Greedy farthest-point sampling of n centres from m random candidates."""
    cand = rng.uniform(INIT_LOW, 1 - INIT_LOW, (m, 2))
    pts = [cand[rng.integers(m)]]
    d = np.sqrt(((cand - pts[0]) ** 2).sum(1))
    for _ in range(n - 1):
        k = np.argmax(d)
        pts.append(cand[k])
        d = np.minimum(d, np.sqrt(((cand - cand[k]) ** 2).sum(1)))
    return np.array(pts)
