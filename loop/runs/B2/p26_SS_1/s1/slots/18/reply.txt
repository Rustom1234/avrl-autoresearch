def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    budget = TIME_FRACTION * time_limit
    fails, switched = 0, False
    while time.process_time() - t0 < budget:   # random restarts until the stall rule switches to perturbation
        frac = (time.process_time() - t0) / budget
        if best is not None and not switched and frac >= 0.1 and (fails >= 8 or frac >= 0.6):
            switched = True
        if switched:
            c, r = _solve_once(n, rng, best)
        else:
            c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
            fails = 0
        else:
            fails += 1
    return best
