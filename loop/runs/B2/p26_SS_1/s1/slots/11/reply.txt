def _maximin(n, rng, m=400):
    """Farthest-point sampling: greedily keep the candidate farthest from the points chosen so far."""
    cand = rng.uniform(INIT_LOW, 1 - INIT_LOW, (m, 2))
    pts = [cand[0]]
    d = np.linalg.norm(cand - cand[0], axis=1)
    for _ in range(n - 1):
        k = np.argmax(d)
        pts.append(cand[k])
        d = np.minimum(d, np.linalg.norm(cand - cand[k], axis=1))
    return np.array(pts)
