def _maximin(n, rng, m=400):
    cand = rng.uniform(0.02, 0.98, (m, 2))
    pts = [cand[0]]
    d = np.full(m, np.inf)
    for _ in range(n - 1):
        d = np.minimum(d, ((cand - pts[-1]) ** 2).sum(1))
        pts.append(cand[np.argmax(d)])
    return np.array(pts)


def _ring_start(n, rng, k=12):
    a = rng.uniform(0.03, 0.08, k)
    t = (np.arange(k) + rng.uniform(0, 1, k)) / k * 4
    side = np.floor(t).astype(int)
    u = t - side
    x = np.select([side == 0, side == 1, side == 2, side == 3], [u, 1 - a, 1 - u, a])
    y = np.select([side == 0, side == 1, side == 2, side == 3], [a, u, 1 - a, 1 - u])
    inner = rng.uniform(0.2, 0.8, (n - k, 2))
    return np.vstack([np.stack([x, y], 1), inner])


def _init_centres(n, rng):
    if rng.random() < 0.5:
        return _maximin(n, rng)
    return _ring_start(n, rng)
