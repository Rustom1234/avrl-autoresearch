def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    cur, T0, T1 = None, 0.003, 1e-5
    budget = TIME_FRACTION * time_limit
    while time.process_time() - t0 < budget:   # random restarts until the time is used
        el = time.process_time() - t0
        if best is not None and el > 0.3 * budget:
            if cur is None:
                cur = best
            T = T0 * (T1 / T0) ** ((el - 0.3 * budget) / (0.7 * budget))
            c, r = _solve_once(n, rng, cur)
        else:
            c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
        if cur is not None and rng.random() < np.exp(min(0.0, (r.sum() - cur[1].sum()) / T)):
            cur = (c, r)
    return best
