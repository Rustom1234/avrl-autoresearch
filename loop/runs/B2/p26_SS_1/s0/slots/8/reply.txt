def _maximin_centers(n, rng, n_cand=400):
    cand = rng.uniform(INIT_LOW, 1 - INIT_LOW, (n_cand, 2))
    chosen = [rng.integers(n_cand)]
    d = np.sqrt(((cand - cand[chosen[0]]) ** 2).sum(1))
    for _ in range(n - 1):
        k = int(np.argmax(d))
        chosen.append(k)
        d = np.minimum(d, np.sqrt(((cand - cand[k]) ** 2).sum(1)))
    return cand[chosen]
