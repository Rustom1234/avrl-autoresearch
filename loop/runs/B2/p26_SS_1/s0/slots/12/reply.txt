def _maximin_centres(n, rng):
    cand = rng.uniform(0.05, 0.95, (400, 2))
    chosen = [int(rng.integers(len(cand)))]
    dmin = np.linalg.norm(cand - cand[chosen[0]], axis=1)
    for _ in range(n - 1):
        k = int(np.argmax(dmin))
        chosen.append(k)
        dmin = np.minimum(dmin, np.linalg.norm(cand - cand[k], axis=1))
    return cand[chosen]
