def _maximin(n, rng):
    cand = rng.uniform(INIT_LOW, 1 - INIT_LOW, (400, 2))
    pts = [cand[0]]
    dmin = np.linalg.norm(cand - cand[0], axis=1)
    for _ in range(n - 1):
        k = np.argmax(dmin)
        pts.append(cand[k])
        dmin = np.minimum(dmin, np.linalg.norm(cand - cand[k], axis=1))
    return np.array(pts)
