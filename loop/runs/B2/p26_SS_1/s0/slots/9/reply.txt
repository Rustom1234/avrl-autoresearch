def _maximin(cand, k):
    """Farthest-point sampling: k candidates, each the one farthest from those already chosen."""
    idx = [0]
    d = np.linalg.norm(cand - cand[0], axis=1)
    for _ in range(k - 1):
        i = int(np.argmax(d))
        idx.append(i)
        d = np.minimum(d, np.linalg.norm(cand - cand[i], axis=1))
    return cand[idx]


def _fresh_centres(n, rng):
    kind = rng.integers(3)
    if kind == 0:
        return rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
    if kind == 1:
        return _maximin(rng.uniform(INIT_LOW, 1 - INIT_LOW, (400, 2)), n)
    m = n // 2
    half = _maximin(rng.uniform([INIT_LOW, INIT_LOW], [0.47, 1 - INIT_LOW], (200, 2)), m)
    mirror = np.column_stack([1 - half[:, 0], half[:, 1]])
    rest = rng.uniform(INIT_LOW, 1 - INIT_LOW, (n - 2 * m, 2))
    return np.vstack([half, mirror, rest]) + rng.normal(0, 0.005, (n, 2))
