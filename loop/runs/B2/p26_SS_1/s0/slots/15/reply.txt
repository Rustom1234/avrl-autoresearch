def _maximin(n, rng):
    cand = rng.uniform(INIT_LOW, 1 - INIT_LOW, (400, 2))
    pts = [cand[rng.integers(400)]]
    d = np.sqrt(((cand - pts[0]) ** 2).sum(1))
    for _ in range(n - 1):
        k = np.argmax(d)
        pts.append(cand[k])
        d = np.minimum(d, np.sqrt(((cand - cand[k]) ** 2).sum(1)))
    return np.array(pts)
