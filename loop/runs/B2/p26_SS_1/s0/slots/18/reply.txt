def _maximin(n, rng, m=400):
    cand = rng.uniform(INIT_LOW, 1 - INIT_LOW, (m, 2))
    pts = [cand[rng.integers(m)]]
    d = np.linalg.norm(cand - pts[0], axis=1)
    for _ in range(n - 1):
        k = np.argmax(d)
        pts.append(cand[k])
        d = np.minimum(d, np.linalg.norm(cand - cand[k], axis=1))
    return np.array(pts)


def _grow_radii(c):
    d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
    np.fill_diagonal(d, np.inf)
    wall = np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1]))
    return 0.9 * np.minimum(wall, 0.5 * d.min(1))
