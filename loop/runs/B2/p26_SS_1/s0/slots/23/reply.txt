def _fresh_centres(n, rng):
    u = rng.random()
    if u < 0.5:  # farthest-point (maximin) sampling from random candidates
        cand = rng.uniform(INIT_LOW, 1 - INIT_LOW, (400, 2))
        pts = [cand[0]]
        dmin = np.linalg.norm(cand - cand[0], axis=1)
        for _ in range(n - 1):
            k = np.argmax(dmin)
            pts.append(cand[k])
            dmin = np.minimum(dmin, np.linalg.norm(cand - cand[k], axis=1))
        return np.array(pts)
    if u < 0.75:  # 12 perimeter circles (3 per wall) plus interior circles
        t = (np.arange(3) + 0.5) / 3
        along = np.clip(np.tile(t, 4) + rng.normal(0, 0.03, 12), 0.05, 0.95)
        depth = rng.uniform(0.03, 0.12, 12)
        side = np.repeat(np.arange(4), 3)
        ring = np.empty((12, 2))
        ring[:, 0] = np.where(side < 2, along, np.where(side == 2, depth, 1 - depth))
        ring[:, 1] = np.where(side == 0, depth, np.where(side == 1, 1 - depth, along))
        return np.vstack([ring, rng.uniform(0.2, 0.8, (n - 12, 2))])
    return rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
