def _maximin(n, rng):
    m = rng.choice([60, 150, 400])
    cand = rng.uniform(INIT_LOW, 1 - INIT_LOW, (m, 2))
    k = rng.integers(m)
    chosen = [k]
    d = np.sqrt(((cand - cand[k]) ** 2).sum(1))
    for _ in range(n - 1):
        k = np.argmax(d)
        chosen.append(k)
        d = np.minimum(d, np.sqrt(((cand - cand[k]) ** 2).sum(1)))
    return cand[chosen] + rng.normal(0, 0.01, (n, 2))
