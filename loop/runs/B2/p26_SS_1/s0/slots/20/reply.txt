def _maximin_centres(n, rng, m=400):
    c = [rng.uniform(0.03, 0.97, 2)]
    for _ in range(n - 1):
        cand = rng.uniform(0.03, 0.97, (m, 2))
        d = np.sqrt(((cand[:, None] - np.array(c)[None]) ** 2).sum(-1)).min(1)
        c.append(cand[np.argmax(d)])
    return np.array(c)
