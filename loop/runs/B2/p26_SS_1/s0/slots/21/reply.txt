def _maximin(n, rng, m=400):
    cand = rng.uniform(0.1, 0.9, (m, 2))
    chosen = [cand[rng.integers(m)]]
    dmin = np.sqrt(((cand - chosen[0]) ** 2).sum(1))
    for _ in range(n - 1):
        k = np.argmax(dmin)
        chosen.append(cand[k])
        dmin = np.minimum(dmin, np.sqrt(((cand - cand[k]) ** 2).sum(1)))
    return np.array(chosen)


def _ring(n, rng):
    k = min(12, n)
    t = rng.uniform(0, 4, k)
    side = np.floor(t).astype(int)
    f = t - side
    d = rng.uniform(0.03, 0.12, k)
    x = np.select([side == 0, side == 1, side == 2, side == 3], [f, 1 - d, 1 - f, d])
    y = np.select([side == 0, side == 1, side == 2, side == 3], [d, f, 1 - d, 1 - f])
    return np.vstack([np.stack([x, y], 1), rng.uniform(0.2, 0.8, (n - k, 2))])
