def dedup(smiles, y):
    groups = {}
    for s, v in zip(smiles, y):
        groups.setdefault(Chem.MolToSmiles(Chem.MolFromSmiles(s)), []).append(float(v))
    keys = list(groups)
    return keys, np.array([np.mean(groups[k]) for k in keys], dtype=np.float64)


def fit_predict(train_smiles, train_y, test_smiles, seed):
    train_smiles, y = dedup(train_smiles, train_y)
