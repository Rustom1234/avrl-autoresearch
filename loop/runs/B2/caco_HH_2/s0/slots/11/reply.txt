def scaffold_strings(smiles):
    out = []
    for s in smiles:
        m = Chem.MolFromSmiles(s)
        generic = Chem.MolToSmiles(MurckoScaffold.MakeScaffoldGeneric(MurckoScaffold.GetScaffoldForMol(m)))
        systems = []
        for ring in m.GetRingInfo().AtomRings():
            merged = set(ring)
            keep = []
            for sy in systems:
                if sy & merged:
                    merged |= sy
                else:
                    keep.append(sy)
            systems = keep + [merged]
        out.append((generic, ",".join(sorted(str(len(sy)) for sy in systems))))
    return out
