def desc_rbf(ZA, ZB, gamma=0.5):

    sq = (ZA ** 2).sum(1)[:, None] + (ZB ** 2).sum(1)[None, :] - 2.0 * ZA @ ZB.T
    return np.exp(-gamma * np.maximum(sq, 0.0) / d)
