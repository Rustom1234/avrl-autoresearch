def fit_predict(train_smiles, train_y, test_smiles, seed):
    y = rankdata(np.asarray(train_y, dtype=np.float64)) / len(train_y)
    D_tr, D_te = desc_features(train_smiles), desc_features(test_smiles)

    # honest out-of-fold descriptor predictions on scaffold-grouped folds
    scaffolds = [MurckoScaffold.MurckoScaffoldSmiles(smiles=s) for s in train_smiles]
    groups = np.unique(scaffolds, return_inverse=True)[1]
    oof = np.zeros(len(y))
    for tr, va in GroupKFold(n_splits=5).split(D_tr, y, groups):
        oof[va] = lgbm_model(seed).fit(D_tr[tr], y[tr]).predict(D_tr[va])
    p_desc = lgbm_model(seed).fit(D_tr, y).predict(D_te)

    # fingerprint GP models only what the descriptors miss
    X_tr, X_te = bit_fingerprints(train_smiles), bit_fingerprints(test_smiles)
    K_tr = tanimoto(X_tr, X_tr)
    resid = y - oof
    s2 = choose_sigma2(K_tr, resid, train_smiles)
    p_res = gp_mean(K_tr, resid, tanimoto(X_te, X_tr), s2)

    return rankdata(p_desc + p_res) / len(p_desc)
