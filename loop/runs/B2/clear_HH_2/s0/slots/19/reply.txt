def knn_density(K, k=10):
    Kd = K.copy()
    np.fill_diagonal(Kd, -np.inf)
    return np.sort(Kd, axis=1)[:, -k:].mean(1)


def gp_mean(K_tr, y, K_te, sigma2, c):
    mu = y.mean()
    noise = sigma2 * (1 + c * (1 - knn_density(K_tr)))
    alpha = np.linalg.solve(K_tr + np.diag(noise), y - mu)
    return mu + K_te @ alpha
