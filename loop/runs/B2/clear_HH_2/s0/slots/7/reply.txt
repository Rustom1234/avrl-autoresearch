                             colsample_bytree=0.5, min_child_samples=10, random_state=seed, n_jobs=8, verbose=-1)


def fit_predict(train_smiles, train_y, test_smiles, seed):
    y = rankdata(np.asarray(train_y, dtype=np.float64)) / len(train_y)
    Xd_tr, Xd_te = desc_features(train_smiles), desc_features(test_smiles)

    lgbm = make_lgbm(seed).fit(Xd_tr, y)
    p_desc = lgbm.predict(Xd_te)

    # out-of-fold LightGBM predictions on scaffold-grouped folds, used to fit a GP on the residuals
    groups = np.unique([MurckoScaffold.MurckoScaffoldSmiles(smiles=s) for s in train_smiles], return_inverse=True)[1]
    oof = np.zeros(len(y))
    for tr, va in GroupKFold(n_splits=5).split(Xd_tr, y, groups):
        oof[va] = make_lgbm(seed).fit(Xd_tr[tr], y[tr]).predict(Xd_tr[va])

    X_tr, X_te = bit_fingerprints(train_smiles), bit_fingerprints(test_smiles)
    K_tr, K_te = tanimoto(X_tr, X_tr), tanimoto(X_te, X_tr)
    s2 = choose_sigma2(K_tr, y, train_smiles)
    p_fp = gp_mean(K_tr, y, K_te, s2)
    p_res = p_desc + gp_mean(K_tr, y - oof, K_te, s2)

    n = len(p_desc)
    blend = W_DESC * rankdata(p_desc) / n + (1 - W_DESC) * rankdata(p_fp) / n
    return 0.5 * rankdata(p_res) / n + 0.5 * rankdata(blend) / n
