def solve(n, seed, time_limit):
    rng, t0, best, top = np.random.default_rng(seed), time.process_time(), None, None
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        late = best is not None and time.process_time() - t0 > 0.4 * time_limit
        c, r = _solve_once(n, rng, best if late else None)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        tol = 1e-6 if late else 0.0
        if best is None or r.sum() > best[1].sum() - tol:
            best = (c, r)
        if top is None or r.sum() > top[1].sum():
            top = (c, r)
    return top
