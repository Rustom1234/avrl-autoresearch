def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    gain = {"fresh": 0.0, "bh": 0.0}   # total improvement of the best sum produced by each arm
    used = {"fresh": 0.0, "bh": 0.0}   # total CPU time spent by each arm
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        el = time.process_time() - t0
        if best is None or el < 0.1 * time_limit:
            arm = "fresh"
        else:
            score = {k: (gain[k] + 1e-4) / (used[k] + 1e-3) for k in gain}
            p_bh = min(max(score["bh"] / (score["bh"] + score["fresh"]), 0.2), 0.8)
            arm = "bh" if rng.random() < p_bh else "fresh"
        ts, prev = time.process_time(), (best[1].sum() if best is not None else None)
        c, r = _solve_once(n, rng, best if arm == "bh" else None)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
        if prev is not None:
            gain[arm] += max(0.0, best[1].sum() - prev)
        used[arm] += time.process_time() - ts
    return best
