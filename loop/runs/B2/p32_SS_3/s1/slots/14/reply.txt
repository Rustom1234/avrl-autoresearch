def _repel(c, steps=20, step=0.02):
    """Spread centres apart with 1/d^2 repulsion; the largest displacement per step is capped at `step`."""
    for _ in range(steps):
        diff = c[:, None] - c[None]
        d = np.sqrt((diff ** 2).sum(-1))
        np.fill_diagonal(d, np.inf)
        f = (diff / d[..., None] ** 3).sum(1)
        m = np.sqrt((f ** 2).sum(1)).max()
        if m > 0:
            c = c + f * (step / m)
        c = np.clip(c, 0.02, 0.98)
    return c
