def _repel(c, steps=20):
    """Push random centres apart with pair and wall 1/d^2 repulsion, capping each move at 0.02."""
    for _ in range(steps):
        diff = c[:, None] - c[None]
        d = np.sqrt((diff ** 2).sum(-1)) + np.eye(len(c))
        f = (diff / d[..., None] ** 3).sum(1)
        f += 1 / c ** 2 - 1 / (1 - c) ** 2
        step = np.minimum(1, 0.02 / (np.linalg.norm(f, axis=1) + 1e-12))
        c = np.clip(c + f * step[:, None], 0.05, 0.95)
    return c

def solve(n, seed, time_limit):
    rng, t0, best, cur = np.random.default_rng(seed), time.process_time(), None, None
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        late = best is not None and time.process_time() - t0 > 0.4 * time_limit
        if late and cur is None:
            cur = best
        c, r = _solve_once(n, rng, cur if late else None)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
        if late:
            gap = r.sum() - cur[1].sum()
            if gap > 0 or rng.random() < np.exp(gap / 0.002):
                cur = (c, r)
    return best
