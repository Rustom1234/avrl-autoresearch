def _greedy_centres(n, rng, cap):
    """Place centres one at a time at the candidate with the largest allowed radius (capped)."""
    cand = rng.uniform(0, 1, (2000, 2))
    allowed = np.minimum(np.minimum(cand[:, 0], cand[:, 1]), np.minimum(1 - cand[:, 0], 1 - cand[:, 1]))
    centres = []
    for _ in range(n):
        score = np.minimum(allowed, cap)
        k = np.argmax(score)
        centres.append(cand[k])
        allowed = np.minimum(allowed, np.sqrt(((cand - cand[k]) ** 2).sum(1)) - score[k])
    return np.array(centres)
