def charge_feats(m):
    from rdkit.Chem import AllChem
    AllChem.ComputeGasteigerCharges(m)
    q = np.array([a.GetDoubleProp('_GasteigerCharge') for a in m.GetAtoms()], dtype=np.float64)
    q = np.nan_to_num(q, nan=0.0, posinf=0.0, neginf=0.0)
    if q.size == 0:
        return [0.0] * 7
    return [q.max(), q.min(), np.abs(q).sum(), q[q > 0].sum(), q[q < 0].sum(),
            q.max() - q.min(), float((np.abs(q) > 0.2).sum())]
