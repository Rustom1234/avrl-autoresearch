def ring_feats(smiles):
    rows = []
    for s in smiles:
        m = Chem.MolFromSmiles(s)
        n = max(m.GetNumAtoms(), 1)
        scaf = MurckoScaffold.GetScaffoldForMol(m)
        ri = m.GetRingInfo()
        rings = [set(r) for r in ri.AtomRings()]
        systems = []
        for r in rings:
            new = set(r)
            for sy in [sy for sy in systems if sy & r]:
                new |= sy
                systems.remove(sy)
            systems.append(new)
        largest = max([len(sy) for sy in systems], default=0)
        fused = sum(1 for a in range(m.GetNumAtoms()) if ri.NumAtomRings(a) >= 2)
        spiro = 0
        for i in range(len(rings)):
            for j in range(i + 1, len(rings)):
                if len(rings[i] & rings[j]) == 1:
                    spiro += 1
        acyclic = sum(1 for a in range(m.GetNumAtoms()) if ri.NumAtomRings(a) == 0)
        rows.append([len(systems), largest, fused, spiro, scaf.GetNumAtoms() / n, acyclic])
    return np.array(rows, dtype=np.float64)
