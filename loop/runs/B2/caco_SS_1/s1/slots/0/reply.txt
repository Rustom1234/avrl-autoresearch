def ion_feats(smiles):
    quat = Chem.MolFromSmarts('[N+;!$(N=O);!$(N-[O-])]')
    rows = []
    for s in smiles:
        m = Chem.MolFromSmiles(s)
        n_acid = len(m.GetSubstructMatches(ACID))
        n_amine = len(m.GetSubstructMatches(AMINE))
        n_quat = len(m.GetSubstructMatches(quat))
        pen = 3.0 * n_acid + min(n_amine, 1)
        rows.append([Crippen.MolLogP(m) - pen, pen, n_quat, Chem.GetFormalCharge(m)])
    return np.array(rows, dtype=np.float64)
