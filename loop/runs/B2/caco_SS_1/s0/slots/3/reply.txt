def fit_predict(train_smiles, train_y, test_smiles, seed):
    y = np.asarray(train_y, dtype=np.float64)
    import lightgbm as lgb
    from sklearn.model_selection import KFold
    nd = len(Descriptors.descList) + 2
    Xtr = featurize(train_smiles)[:, :nd]
    Xte = featurize(test_smiles)[:, :nd]

    def make():
        return lgb.LGBMRegressor(objective='huber', n_estimators=500, learning_rate=0.03, num_leaves=15,
                                 subsample=0.8, subsample_freq=1, colsample_bytree=0.5,
                                 min_child_samples=10, random_state=seed, n_jobs=8, verbose=-1)

    oof = np.zeros(len(y))
    for tr, va in KFold(5, shuffle=True, random_state=seed).split(Xtr):
        m = make()
        m.fit(Xtr[tr], y[tr])
        oof[va] = m.predict(Xtr[va])
    model = make()
    model.fit(Xtr, y)
    pred = model.predict(Xte)
    resid_pred = knn_predict(train_smiles, y - oof, test_smiles, k=7)
    return pred + 0.5 * resid_pred
