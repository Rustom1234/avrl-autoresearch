def fit_predict(train_smiles, train_y, test_smiles, seed):
    y = np.asarray(train_y, dtype=np.float64)
    Xtr = featurize(train_smiles)
    Xte = featurize(test_smiles)
    model = ExtraTreesRegressor(n_estimators=300, max_features=0.3, min_samples_leaf=3,
                                random_state=seed, n_jobs=8)
    model.fit(Xtr, y)
    per_tree = np.stack([t.predict(Xte) for t in model.estimators_])
    tree_pred = np.median(per_tree, axis=0)
    knn = knn_predict(train_smiles, y, test_smiles)
    return 0.75 * tree_pred + 0.25 * knn
