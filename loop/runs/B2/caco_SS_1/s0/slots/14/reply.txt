def clean_mol(s):
    # Keep the largest fragment (drops counter-ions); fall back to the original molecule on any failure.
    try:
        m = Chem.MolFromSmiles(s)
        best = max(Chem.GetMolFrags(m, asMols=True), key=lambda f: f.GetNumHeavyAtoms())
        out = Chem.MolFromSmiles(Chem.MolToSmiles(best))
        return out if out is not None else m
    except Exception:
        return Chem.MolFromSmiles(s)
