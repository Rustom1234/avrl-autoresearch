def prune_cols(X, thr=0.95):
    X = X.astype(np.float64)
    sd = X.std(0)
    ok = sd > 1e-12
    Z = (X - X.mean(0)) / np.where(ok, sd, 1.0)
    C = np.abs(Z.T @ Z) / len(Z)
    C[~ok, :] = 0.0
    C[:, ~ok] = 0.0
    kept = []
    for j in range(X.shape[1]):
        if ok[j] and all(C[j, k] <= thr for k in kept):
            kept.append(j)
    return np.array(kept, dtype=int)
