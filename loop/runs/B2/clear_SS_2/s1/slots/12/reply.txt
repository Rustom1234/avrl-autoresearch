def mol(s):
    from rdkit.Chem import rdmolops
    m = Chem.MolFromSmiles(s)
    frags = rdmolops.GetMolFrags(m, asMols=True)
    return max(frags, key=lambda f: f.GetNumHeavyAtoms())
