def avalon(smiles):
    from rdkit.Avalon.pyAvalonTools import GetAvalonCountFP
    out = np.zeros((len(smiles), 1024), dtype=np.float32)
    for i, s in enumerate(smiles):
        fp = GetAvalonCountFP(Chem.MolFromSmiles(s), nBits=1024)
        for j, v in fp.GetNonzeroElements().items():
            out[i, j] = v
    return out
