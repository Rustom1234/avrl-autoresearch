def estate(smiles):
    from rdkit.Chem.EState import Fingerprinter as EF, EState as ES
    rows = []
    for s in smiles:
        m = Chem.MolFromSmiles(s)
        try:
            counts = np.asarray(EF.FingerprintMol(m)[0], dtype=np.float64)
            if counts.shape[0] != 79:
                counts = np.zeros(79)
        except Exception:
            counts = np.zeros(79)
        try:
            idx = np.asarray(ES.EStateIndices(m), dtype=np.float64)
            agg = [idx.sum(), idx.max()] if idx.size else [0.0, 0.0]
        except Exception:
            agg = [0.0, 0.0]
        rows.append(np.concatenate([counts, agg]))
    return np.nan_to_num(np.array(rows, dtype=np.float64), nan=0.0, posinf=0.0, neginf=0.0).astype(np.float32)
