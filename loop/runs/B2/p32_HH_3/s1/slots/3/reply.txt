def _sigma(c, r):
    """Per-circle perturbation sd: loose circles (fewer than 3 contacts) move more than jammed ones."""
    d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
    gap = d - r[:, None] - r[None]
    np.fill_diagonal(gap, np.inf)
    k = (gap < 1e-6).sum(1)
    return np.where(k >= 3, 0.01, 0.04)[:, None]
