def _score(c, r):
    """Sum of radii minus total overlap and wall violation, so that partial solves can be compared."""
    d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
    over = np.maximum(r[:, None] + r[None] - d, 0)
    np.fill_diagonal(over, 0)
    walls = np.maximum(r[:, None] - c, 0).sum() + np.maximum(c + r[:, None] - 1, 0).sum()
    return r.sum() - over.sum() / 2 - walls
