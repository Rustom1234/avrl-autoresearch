import numpy as np
from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator, Descriptors
from rdkit.Chem.MolStandardize import rdMolStandardize
from scipy.stats import rankdata
from sklearn.ensemble import RandomForestClassifier
from sklearn.neural_network import MLPClassifier
from sklearn.preprocessing import QuantileTransformer
from lightgbm import LGBMClassifier
from catboost import CatBoostClassifier

# Settings
N_BITS = 2048
RADIUS = 2
N_TREES = 300
MAX_FEATURES = 0.3
MIN_SAMPLES_LEAF = 1
N_NEIGHBORS = 15
NOISE_SD = 0.05
N_NOISY_COPIES = 2
N_DESC = len(Descriptors.descList)

_LFC = rdMolStandardize.LargestFragmentChooser()
_UNCHARGER = rdMolStandardize.Uncharger()


def standardize(smi):
    m0 = Chem.MolFromSmiles(smi)
    try:
        m = _UNCHARGER.uncharge(_LFC.choose(m0))
        if m is None or m.GetNumAtoms() == 0:
            return m0
        return m
    except Exception:
        return m0


def augment(X, y, seed):
    # Gaussian noise on descriptor columns only; labels unchanged
    rng = np.random.default_rng(seed)
    sd = X[:, :N_DESC].std(axis=0)
    Xs, ys = [X], [y]
    for _ in range(N_NOISY_COPIES):
        Xn = X.copy()
        noise = rng.normal(0.0, 1.0, size=(len(X), N_DESC))
        Xn[:, :N_DESC] += (noise * (NOISE_SD * sd)).astype(np.float32)
        Xs.append(Xn)
        ys.append(y)
    return np.vstack(Xs), np.concatenate(ys)


def tanimoto_knn(A_bits, ytr, B_bits, k):
    A = A_bits.astype(np.float32)
    B = B_bits.astype(np.float32)
    inter = B @ A.T
    union = B.sum(1)[:, None] + A.sum(1)[None, :] - inter
    sim = inter / np.maximum(union, 1e-9)
    idx = np.argpartition(-sim, k, axis=1)[:, :k]
    return ytr[idx].mean(axis=1)

def fit_predict(train_smiles, train_y, test_smiles, seed):
    X = featurize(train_smiles)
    Xt = featurize(test_smiles)
    y = np.asarray(train_y, dtype=np.int64)
    Xa, ya = augment(X, y, seed)
    models = [
        RandomForestClassifier(n_estimators=N_TREES, max_features=MAX_FEATURES, min_samples_leaf=MIN_SAMPLES_LEAF,
                               n_jobs=-1, random_state=seed),
        LGBMClassifier(n_estimators=300, learning_rate=0.05, num_leaves=31, subsample=0.8, subsample_freq=1,
                       colsample_bytree=0.3, n_jobs=8, random_state=seed, verbose=-1),
        CatBoostClassifier(iterations=500, learning_rate=0.05, depth=6, rsm=0.2, thread_count=8,
                           random_seed=seed, verbose=0),
    ]
    preds = []
    for m in models:
        m.fit(Xa, ya)
        preds.append(rankdata(m.predict_proba(Xt)[:, 1]) / len(Xt))
    qt = QuantileTransformer(n_quantiles=min(1000, len(X)), random_state=seed)
    qt.fit(X)
    mlp = MLPClassifier(hidden_layer_sizes=(256, 64), early_stopping=True, max_iter=500, random_state=seed)
    mlp.fit(qt.transform(Xa), ya)
