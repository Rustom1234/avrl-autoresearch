def standardize(smiles):
    mol = Chem.MolFromSmiles(smiles)
    try:
        m = rdMolStandardize.LargestFragmentChooser().choose(mol)
        return rdMolStandardize.Uncharger().uncharge(m)
    except Exception:
        return mol
