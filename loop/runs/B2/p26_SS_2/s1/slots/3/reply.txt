def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    sd = 0.02   # one-fifth success rule: jitter scale for perturbation children
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        perturb = best is not None and time.process_time() - t0 > 0.3 * time_limit
        if perturb:
            c, r = _solve_once(n, rng, best, sd)
        else:
            c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            if perturb:
                sd = min(sd * 1.5, 0.1)
            best = (c, r)
        elif perturb:
            sd = max(sd * 0.97, 0.002)
    return best
