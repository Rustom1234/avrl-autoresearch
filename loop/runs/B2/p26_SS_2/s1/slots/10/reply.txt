def _greedy_start(n, rng):
    """Place circles one at a time at the candidate point with the largest clearance."""
    c, r = np.zeros((n, 2)), np.zeros(n)
    for k in range(n):
        p = rng.uniform(0, 1, (400, 2))
        clr = np.minimum(p, 1 - p).min(1)
        if k > 0:
            d = np.sqrt(((p[:, None] - c[None, :k]) ** 2).sum(-1)) - r[None, :k]
            clr = np.minimum(clr, d.min(1))
        j = np.argmax(clr)
        c[k] = p[j]
        r[k] = min(max(clr[j], 0), 0.15)
    return c, r
