def solve(n, seed, time_limit):
    rng, t0, best, cur = np.random.default_rng(seed), time.process_time(), None, None
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        late = best is not None and time.process_time() - t0 > 0.3 * time_limit
        if late:
            if cur is None:
                cur = best
            c, r = _solve_once(n, rng, cur, rng.choice([0.005, 0.01, 0.02]))
        else:
            c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
        if late and r.sum() >= cur[1].sum() - 0.002:
            cur = (c, r)
    return best
