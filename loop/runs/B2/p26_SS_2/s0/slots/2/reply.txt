def _hex_centres(n, rng):
    """Row-structured start: 5 or 6 rows with random counts per row, evenly spaced, jittered by 0.02."""
    k = rng.choice([5, 6])
    cuts = np.sort(rng.choice(np.arange(1, n), k - 1, replace=False))
    counts = np.diff(np.concatenate([[0], cuts, [n]]))
    pts = []
    for row, m in enumerate(counts):
        xs = (np.arange(m) + 0.5) / m * 0.8 + 0.1
        pts.append(np.stack([xs, np.full(m, 0.1 + 0.8 * row / (k - 1))], 1))
    return np.vstack(pts) + rng.uniform(-0.02, 0.02, (n, 2))
