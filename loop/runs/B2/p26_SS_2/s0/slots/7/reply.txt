def _ring_centers(n, rng):
    """Centres on three concentric rings around the square centre, with jittered counts and noise."""
    counts = np.maximum(np.array([1, 8, 17]) + rng.integers(-2, 3, 3), 1)
    counts[2] = n - counts[0] - counts[1]
    pts = []
    for k, rad in zip(counts, [0.0, 0.2, 0.4]):
        th = rng.uniform(0, 2 * np.pi) + 2 * np.pi * np.arange(k) / k
        pts.append(np.stack([0.5 + rad * np.cos(th), 0.5 + rad * np.sin(th)], 1))
    c = np.concatenate(pts) + rng.normal(0, 0.02, (n, 2))
    return np.clip(c, INIT_LOW, 1 - INIT_LOW)
