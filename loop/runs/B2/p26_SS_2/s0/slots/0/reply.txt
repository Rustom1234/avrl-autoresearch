def _contacts(c, r):
    """Number of touching neighbours and walls per circle (gap below 1e-4)."""
    d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
    gap = d - r[:, None] - r[None]
    np.fill_diagonal(gap, 1)
    wall = np.stack([c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r], 1)
    return (gap < 1e-4).sum(1) + (wall < 1e-4).sum(1)
