def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    sds = [0.005, 0.01, 0.02, 0.05]
    tries, wins = np.ones(4), np.ones(4)
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        k, prev = None, (best[1].sum() if best is not None else -np.inf)
        if best is not None and time.process_time() - t0 > 0.3 * time_limit:
            p = wins / tries
            k = rng.choice(4, p=p / p.sum())
            c, r = _solve_once(n, rng, best, sds[k])
        else:
            c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if k is not None:
            tries[k] += 1
            if r.sum() > prev + 1e-7:
                wins[k] += 1
        if best is None or r.sum() > prev:
            best = (c, r)
    return best
