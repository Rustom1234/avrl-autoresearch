def _lloyd(n, rng, iters=5, g=60):
    """Spread n uniform random sites with Lloyd iterations on a g x g grid of sample points."""
    pts = rng.uniform(0, 1, (n, 2))
    gx = (np.arange(g) + 0.5) / g
    grid = np.stack(np.meshgrid(gx, gx), -1).reshape(-1, 2)
    for _ in range(iters):
        lab = np.argmin(((grid[:, None] - pts[None]) ** 2).sum(-1), 1)
        for k in range(n):
            m = lab == k
            if m.any():
                pts[k] = grid[m].mean(0)
    return pts
