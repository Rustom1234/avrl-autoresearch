def solve(n, seed, time_limit):
    rng, t0, best, fails = np.random.default_rng(seed), time.process_time(), None, 0
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        escalate = False
        if best is not None and time.process_time() - t0 > 0.3 * time_limit:
            if fails < 6:
                c, r = _solve_once(n, rng, best, rng.choice([0.005, 0.01, 0.02]))
            else:
                escalate = True
                start_c = best[0].copy()
                start_c[np.argsort(best[1])[:2]] = rng.uniform(0.1, 0.9, (2, 2))
                c, r = _solve_once(n, rng, (start_c, best[1]), 0.05)
        else:
            c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best, fails = (c, r), 0
        else:
            fails += 1
        if escalate:
            fails = 3
    return best
