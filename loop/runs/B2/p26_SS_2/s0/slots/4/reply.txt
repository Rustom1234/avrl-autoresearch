def _halton(n, base, offset):
    out = np.zeros(n)
    for i in range(n):
        f, r, k = 1.0, 0.0, i + 1 + offset
        while k > 0:
            f /= base
            r += f * (k % base)
            k //= base
        out[i] = r
    return out
