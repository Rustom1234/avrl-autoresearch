def _largest_hole(c, r, skip):
    """Grid point with the largest clearance to the walls and to all circles except circle `skip`."""
    g = np.linspace(0.02, 0.98, 60)
    p = np.stack(np.meshgrid(g, g), -1).reshape(-1, 2)
    others = np.arange(len(r)) != skip
    d = np.sqrt(((p[:, None] - c[None, others]) ** 2).sum(-1)) - r[None, others]
    clear = np.minimum(np.minimum(p, 1 - p).min(1), d.min(1))
    return p[np.argmax(clear)]
