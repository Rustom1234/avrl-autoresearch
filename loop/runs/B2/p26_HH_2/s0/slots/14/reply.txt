def _lp_radii(c):
    """Exact radii for fixed centres: maximise sum r subject to r_i + r_j <= d_ij and wall bounds."""
    n = len(c)
    iu = np.triu_indices(n, 1)
    d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1))
    m = len(d)
    A = np.zeros((m, n))
    A[np.arange(m), iu[0]] = 1
    A[np.arange(m), iu[1]] = 1
    wall = np.maximum(np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1])), 0)
    res = linprog(-np.ones(n), A_ub=A, b_ub=d, bounds=np.column_stack([np.zeros(n), wall]), method="highs")
    return res.x if res.status == 0 else None


def _slsqp_polish(c, r):
    n = len(r)
    iu = np.triu_indices(n, 1)

    def cons(v):
        cc, rr = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((cc[iu[0]] - cc[iu[1]]) ** 2).sum(1)) - rr[iu[0]] - rr[iu[1]]
        return np.concatenate([d, cc[:, 0] - rr, cc[:, 1] - rr, 1 - cc[:, 0] - rr, 1 - cc[:, 1] - rr, rr])

    x = np.concatenate([c.ravel(), r])
    res = minimize(lambda v: -v[2 * n:].sum(), x, method="SLSQP",
                   constraints={"type": "ineq", "fun": cons}, options={"maxiter": 50})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
