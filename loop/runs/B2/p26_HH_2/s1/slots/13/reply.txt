def _grow(c, r, rng):
    """Greedily grow each radius to the largest value that keeps it inside the square and clear of the others."""
    n = len(r)
    wall = np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1]))
    total = r.sum()
    for _ in range(5):
        for i in rng.permutation(n):
            d = np.sqrt(((c - c[i]) ** 2).sum(1))
            d[i] = np.inf
            r[i] = max(0.0, min(wall[i], np.min(d - r)))
        new = r.sum()
        if new - total <= 1e-12:
            break
        total = new
    return r
