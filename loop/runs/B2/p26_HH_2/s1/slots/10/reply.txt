def solve(n, seed, time_limit):
    rng, t0, best, k = np.random.default_rng(seed), time.process_time(), None, 0
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        k += 1
        if best is not None and k > 5 and k % 3 == 0:
            # restart from a random dihedral image of the best centres, with small noise
            bc = best[0]
            x, y = bc[:, 0], bc[:, 1]
            imgs = [(x, y), (1 - x, y), (x, 1 - y), (1 - x, 1 - y), (y, x), (1 - y, x), (y, 1 - x), (1 - y, 1 - x)]
            cc = np.stack(imgs[rng.integers(8)], axis=1) + rng.normal(0, 0.01, bc.shape)
            c, r = _solve_once(n, rng, np.concatenate([cc.ravel(), best[1]]), [0.4, 1.0])
        else:
            c, r = _solve_once(n, rng)
