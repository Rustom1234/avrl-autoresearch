def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    stall, switched = 0, False
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        explore = (time.process_time() - t0 < 0.3 * TIME_FRACTION * time_limit or best is None) and not switched
        c, r = _solve_once(n, rng, None if explore else best)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if explore:
            gain = r.sum() - best[1].sum() if best is not None else np.inf
            stall = 0 if gain > 1e-4 else stall + 1
            switched = stall >= 8
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    return best
