def solve(n, seed, time_limit):
    rng, t0, top = np.random.default_rng(seed), time.process_time(), []
    k = 0
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        explore = time.process_time() - t0 < 0.2 * TIME_FRACTION * time_limit or not top
        base = None if explore else top[k % len(top)]
        k += 1
        c, r = _solve_once(n, rng, base)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        pool = sorted(top + [(c, r)], key=lambda p: -p[1].sum())
        top = [pool[0]]
        for p in pool[1:]:
            if len(top) < 2 and abs(p[1].sum() - top[0][1].sum()) > 1e-4:
                top.append(p)
    return top[0]
