def _greedy(n, rng):
    """Greedy max-clearance start: place circles at the candidate points with the most room."""
    cand = rng.uniform(0, 1, (600, 2))
    cap = rng.uniform(0.08, 0.13)
    c, r = np.zeros((n, 2)), np.zeros(n)
    for k in range(n):
        clr = np.minimum(cand, 1 - cand).min(1)
        if k > 0:
            clr = np.minimum(clr, (np.sqrt(((cand[:, None] - c[None, :k]) ** 2).sum(-1)) - r[None, :k]).min(1))
        j = np.argsort(-clr)[:10][rng.integers(10)]
        if clr[j] > 0:
            c[k], r[k] = cand[j], min(cap, clr[j]) * 0.9
        else:
            c[k], r[k] = rng.uniform(0, 1, 2), 0.005
    return np.concatenate([c.ravel(), r])
