def _swap(pool, rng):
    """Restart type: swap the centres of two neighbouring circles with different radii, shrink radii by 10%."""
    c, r = pool[rng.integers(len(pool))]
    c, r = c.copy(), r.copy()
    for _ in range(100):
        i, j = rng.choice(len(r), 2, replace=False)
        if r[i] != r[j] and np.linalg.norm(c[i] - c[j]) < 1.2 * (r[i] + r[j]):
            c[[i, j]] = c[[j, i]]
            break
    return np.concatenate([c.ravel(), 0.9 * r])
