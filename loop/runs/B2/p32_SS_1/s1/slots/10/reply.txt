def _relax(c, target=0.09):
    """Push centres apart with a smooth overlap/wall penalty before SLSQP."""
    n = len(c)
    iu = np.triu_indices(n, 1)

    def energy(v):
        x = v.reshape(n, 2)
        diff = x[iu[0]] - x[iu[1]]
        d = np.sqrt((diff ** 2).sum(1)) + 1e-12
        ov = np.maximum(0, 2 * target - d)
        lo = np.maximum(0, target - x)
        hi = np.maximum(0, x - (1 - target))
        g = np.zeros((n, 2))
        gp = (-2 * ov / d)[:, None] * diff
        np.add.at(g, iu[0], gp)
        np.add.at(g, iu[1], -gp)
        g += -2 * lo + 2 * hi
        e = (ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()
        return e, g.ravel()

    return minimize(energy, c.ravel(), jac=True, method="L-BFGS-B", options={"maxiter": 50}).x.reshape(n, 2)
