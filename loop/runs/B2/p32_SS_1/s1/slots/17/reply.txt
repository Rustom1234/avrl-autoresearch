def _greedy_centres(n, rng):
    cand = rng.uniform(INIT_LOW, 1 - INIT_LOW, (2000, 2))
    wall = np.minimum(np.minimum(cand[:, 0], cand[:, 1]), np.minimum(1 - cand[:, 0], 1 - cand[:, 1]))
    c = [cand[np.argmax(wall)]]
    d = np.full(len(cand), np.inf)
    for _ in range(n - 1):
        d = np.minimum(d, np.sqrt(((cand - c[-1]) ** 2).sum(1)))
        c.append(cand[np.argmax(np.minimum(d / 2, wall))])
    return np.array(c)
