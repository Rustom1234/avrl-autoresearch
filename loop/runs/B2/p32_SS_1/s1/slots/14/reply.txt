def _halton(n):
    out = np.zeros((n, 2))
    for j, b in enumerate((2, 3)):
        for i in range(n):
            f, k, x = 1.0, i + 1, 0.0
            while k > 0:
                f /= b
                x += f * (k % b)
                k //= b
            out[i, j] = x
    return out
