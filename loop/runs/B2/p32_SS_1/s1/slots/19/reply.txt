def _relax(c, r, steps=50):
    """Relax centres of fixed-radius circles by L-BFGS-B on an overlap and wall penalty energy."""
    n = len(c)
    iu = np.triu_indices(n, 1)

    def energy(x):
        p = x.reshape(n, 2)
        diff = p[iu[0]] - p[iu[1]]
        d = np.sqrt((diff ** 2).sum(1)) + 1e-12
        ov = np.maximum(0, r[iu[0]] + r[iu[1]] - d)
        lo = np.maximum(0, r[:, None] - p)
        hi = np.maximum(0, p + r[:, None] - 1)
        g = np.zeros((n, 2))
        gp = (-2 * ov / d)[:, None] * diff
        np.add.at(g, iu[0], gp)
        np.add.at(g, iu[1], -gp)
        g += -2 * lo + 2 * hi
        return (ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum(), g.ravel()

    res = minimize(energy, c.ravel(), jac=True, method="L-BFGS-B", options={"maxiter": steps})
    return res.x.reshape(n, 2)

def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    pool = []
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # restarts until the time is used
        x0 = _relocate(pool, n, rng) if pool and rng.random() < 0.7 else _fresh(n, rng)
        c, r = _solve_once(n, rng, x0)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
        if all(np.linalg.norm(np.sort(r) - np.sort(p[1])) >= DIV_TOL for p in pool):
            pool.append((c, r))
            pool.sort(key=lambda p: -p[1].sum())
            pool = pool[:POOL_SIZE]
    return best
