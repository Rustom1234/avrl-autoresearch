def _energy(v, n):
    c = v.reshape(n, 2)
    iu = np.triu_indices(n, 1)
    diff = c[iu[0]] - c[iu[1]]
    d = np.sqrt((diff ** 2).sum(1)) + 1e-12
    ov = np.maximum(0, 2 * RELAX_TARGET - d)
    lo = np.maximum(0, RELAX_TARGET - c)
    hi = np.maximum(0, c - (1 - RELAX_TARGET))
    f = (ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()
    coef = (-2 * ov / d)[:, None] * diff
    g = np.zeros((n, 2))
    np.add.at(g, iu[0], coef)
    np.add.at(g, iu[1], -coef)
    g += -2 * lo + 2 * hi
    return f, g.ravel()
