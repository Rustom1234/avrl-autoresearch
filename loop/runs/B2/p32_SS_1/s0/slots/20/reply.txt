def _band_slide(pool, n, rng):
    c, r = pool[rng.integers(len(pool))]
    c, r = c.copy(), r.copy()
    ax = rng.integers(2)
    other = 1 - ax
    w = rng.uniform(0.2, 0.4)
    a = rng.uniform(0, 1 - w)
    m = (c[:, other] >= a) & (c[:, other] <= a + w)
    c[m, ax] += rng.uniform(-0.08, 0.08)
    r = r * 0.95
    c = np.clip(c, r[:, None], 1 - r[:, None])
    return np.concatenate([c.ravel(), r])
