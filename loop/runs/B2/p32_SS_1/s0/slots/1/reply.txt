def _swap(pool, n, rng):

    d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
    gap = d - r[:, None] - r[None]
    i, j = np.triu_indices(n, 1)
    big = np.maximum(r[i], r[j])
    ok = (gap[i, j] < 1e-4) & (np.abs(r[i] - r[j]) >= 0.2 * big)
    if not ok.any():
        return None
    k = rng.choice(np.flatnonzero(ok))
    a, b = i[k], j[k]
    r = r.copy()
    r[a], r[b] = 0.9 * r[b], 0.9 * r[a]
    return np.concatenate([c.ravel(), r])
