def _relax(n, rng):
    """Relax random centres under a penalty energy at a random target radius R (overlaps and walls)."""
    R = rng.uniform(0.07, 0.11)
    iu = np.triu_indices(n, 1)

    def energy(x):
        c = x.reshape(n, 2)
        diff = c[iu[0]] - c[iu[1]]
        d = np.sqrt((diff ** 2).sum(1)) + 1e-12
        ov = np.maximum(2 * R - d, 0)
        lo = np.maximum(R - c, 0)
        hi = np.maximum(c - (1 - R), 0)
        pair = (-2 * ov / d)[:, None] * diff
        g = np.zeros((n, 2))
        np.add.at(g, iu[0], pair)
        np.add.at(g, iu[1], -pair)
        g += -2 * lo + 2 * hi
        return (ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum(), g.ravel()

    x = minimize(energy, rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), jac=True, method="L-BFGS-B",
                 options={"maxiter": 200}).x
    return x.reshape(n, 2), np.full(n, 0.8 * R)
