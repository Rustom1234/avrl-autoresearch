def _relax(n, rng, steps=300):
    """Fresh start: centres relaxed by L-BFGS-B on a penalty energy with fixed radius, so SLSQP starts near feasible."""
    iu = np.triu_indices(n, 1)
    rr = 0.08

    def energy(x):
        c = x.reshape(n, 2)
        diff = c[iu[0]] - c[iu[1]]
        d = np.sqrt((diff ** 2).sum(1)) + 1e-12
        ov = np.maximum(0, 2 * rr - d)
        lo = np.maximum(0, rr - c)
        hi = np.maximum(0, c - (1 - rr))
        e = (ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()
        coef = (2 * ov / d)[:, None] * diff
        g = np.zeros((n, 2))
        np.add.at(g, iu[0], -coef)
        np.add.at(g, iu[1], coef)
        g += -2 * lo + 2 * hi
        return e, g.ravel()

    x = minimize(energy, rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), jac=True, method="L-BFGS-B",
                 options={"maxiter": steps}).x
    return np.concatenate([np.clip(x, 0, 1), np.full(n, INIT_RADIUS)])
