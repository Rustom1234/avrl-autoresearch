_CACHE = {}


def featurize(smiles):
    new = [s for s in dict.fromkeys(smiles) if s not in _CACHE]
    if new:
        for s, x in zip(new, featurize_raw(new)):
            _CACHE[s] = x
    return np.array([_CACHE[s] for s in smiles])
