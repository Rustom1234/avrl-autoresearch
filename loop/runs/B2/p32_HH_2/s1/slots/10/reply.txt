def _contacts(c, r):
    """Number of near-touching pairs and walls per circle (gap below 1e-6)."""
    n = len(r)
    iu = np.triu_indices(n, 1)
    gap = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
    touch = gap < 1e-6
    cnt = np.bincount(iu[0][touch], minlength=n) + np.bincount(iu[1][touch], minlength=n)
    walls = (c - r[:, None] < 1e-6) | (1 - c - r[:, None] < 1e-6)
    return cnt + walls.sum(1)
