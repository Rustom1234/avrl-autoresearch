def _ring_start(n, rng):
    """Centres on 2 or 3 concentric square rings with random counts, offset, phase and jitter."""
    k = int(rng.integers(2, 4))
    counts = rng.multinomial(n, np.full(k, 1 / k))
    offset = rng.uniform(0, 0.05)
    pts = []
    for i, m in enumerate(counts):
        t = offset + i * (0.5 - offset) / k
        L = 1 - 2 * t
        s = (rng.uniform() + np.arange(m) / m) * 4
        side, f = np.floor(s).astype(int), s % 1
        x = np.select([side == 0, side == 1, side == 2], [t + f * L, 1 - t + 0 * f, 1 - t - f * L], t + 0 * f)
        y = np.select([side == 0, side == 1, side == 2], [t + 0 * f, t + f * L, 1 - t + 0 * f], 1 - t - f * L)
        pts.append(np.stack([x, y], 1))
    c = np.vstack(pts) + rng.normal(0, 0.01, (n, 2))
    return np.clip(c, 0.01, 0.99).ravel()
