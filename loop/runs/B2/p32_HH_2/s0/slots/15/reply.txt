def _start(rng, n):
    """Random start: half the draws are point-symmetric about (0.5, 0.5), the rest uniform."""
    if rng.random() < 0.5:
        m = n // 2
        h = rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * m).reshape(m, 2)
        c = np.vstack([h, 1 - h])
        if n % 2:

        return c.ravel()
