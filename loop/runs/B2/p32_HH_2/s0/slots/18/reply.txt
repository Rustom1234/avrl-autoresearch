    budget = TIME_FRACTION * time_limit
    longest = 0.0
    def left():
        return budget - (time.process_time() - t0)
    while time.process_time() - t0 < HOP_START * budget and left() >= 1.5 * longest:   # random restarts first
        pool = sorted((_relax(rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), n) for _ in range(POOL)), key=lambda t: t[0])
