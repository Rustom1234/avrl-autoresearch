def _start(rng, n, mirror):
    """Random centres; if mirror, the first n//2 are drawn in the left half and mirrored to (1 - x, y)."""
    if not mirror:
        return rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2)).ravel()
    h = n // 2
    half = rng.uniform(INIT_LOW, 0.5, (h, 2))
    rest = rng.uniform(INIT_LOW, 1 - INIT_LOW, (n - 2 * h, 2))
    return np.vstack([half, half * [-1, 1] + [1, 0], rest]).ravel()
