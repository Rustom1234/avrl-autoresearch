def fit_predict(train_smiles, train_y, test_smiles, seed):
    def make_lgb():
        return LGBMRegressor(n_estimators=500, learning_rate=0.03, num_leaves=15, subsample=0.8, subsample_freq=1,
                             colsample_bytree=0.5, min_child_samples=10, verbose=-1, random_state=seed, n_jobs=8)
    Xtr, Xte = featurize(train_smiles), featurize(test_smiles)
    y = np.asarray(train_y, dtype=np.float64)
    # out-of-fold LightGBM predictions on the training set -> residuals
    oof = np.zeros(len(y))
    for tr_idx, va_idx in KFold(5, shuffle=True, random_state=seed).split(Xtr):
        oof[va_idx] = make_lgb().fit(Xtr[tr_idx], y[tr_idx]).predict(Xtr[va_idx])
    resid = y - oof
    p_lgb = make_lgb().fit(Xtr, y).predict(Xte)
    keep = Xtr.std(axis=0) > 0
    qt = QuantileTransformer(output_distribution='normal', n_quantiles=min(200, len(Xtr)), random_state=seed)
    Str = qt.fit_transform(Xtr[:, keep])
    Ste = qt.transform(Xte[:, keep])
    svr = SVR(kernel='rbf', C=3, epsilon=0.1, gamma='scale').fit(Str, resid)
    return p_lgb + 0.7 * svr.predict(Ste)
