_CHOOSER = rdMolStandardize.LargestFragmentChooser()
_UNCHARGER = rdMolStandardize.Uncharger()


def standardize(m):
    if m is None:
        return m
    try:
        out = _UNCHARGER.uncharge(_CHOOSER.choose(m))
        return out if out is not None else m
    except Exception:
        return m
