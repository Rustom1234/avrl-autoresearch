def _optimise(n, x0):
    iu = np.triu_indices(n, 1)
    pi, pj = iu
    m = len(pi)
    ar = np.arange(n)
    rows = np.arange(m)
    obj_jac = np.concatenate([np.zeros(2 * n), -np.ones(n)])

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[pi] - c[pj]) ** 2).sum(1)) - r[pi] - r[pj]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    def cons_jac(v):
        c = v[:2 * n].reshape(n, 2)
        diff = c[pi] - c[pj]
        dist = np.maximum(np.sqrt((diff ** 2).sum(1)), 1e-12)
        u = diff / dist[:, None]
        J = np.zeros((m + 5 * n, 3 * n))
        J[rows, 2 * pi] = u[:, 0]
        J[rows, 2 * pi + 1] = u[:, 1]
        J[rows, 2 * pj] = -u[:, 0]
        J[rows, 2 * pj + 1] = -u[:, 1]
        J[rows, 2 * n + pi] = -1
        J[rows, 2 * n + pj] = -1
        k = m
        J[k + ar, 2 * ar] = 1
        J[k + ar, 2 * n + ar] = -1
        k += n
        J[k + ar, 2 * ar + 1] = 1
        J[k + ar, 2 * n + ar] = -1
        k += n
        J[k + ar, 2 * ar] = -1
        J[k + ar, 2 * n + ar] = -1
        k += n
        J[k + ar, 2 * ar + 1] = -1
        J[k + ar, 2 * n + ar] = -1
        k += n
        J[k + ar, 2 * n + ar] = 1
        return J

    res = minimize(lambda v: -v[2 * n:].sum(), x0, jac=lambda v: obj_jac, method="SLSQP",
                   constraints={"type": "ineq", "fun": cons, "jac": cons_jac}, options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
