def solve(n, seed, time_limit):
    rng, t0 = np.random.default_rng(seed), time.process_time()
    budget = TIME_FRACTION * time_limit
    found = []
    while time.process_time() - t0 < RESTART_FRACTION * budget or not found:   # short random-restart phase
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        found.append((c, _repair(c, r)))
    found.sort(key=lambda p: -p[1].sum())
    chains = [found[0]]
    for p in found[1:]:   # second chain: best restart packing whose sum differs from the first
        if abs(p[1].sum() - found[0][1].sum()) > 1e-4:
            chains.append(p)
            break

    def step(c, r):   # remove-and-reinsert the smallest circle
        c2, r2 = c.copy(), r.copy()
        i = np.argmin(r2)
        c2[i] = rng.uniform(0, 1, 2)
        r2[i] = 0
        c2, r2 = _optimise(n, np.concatenate([c2.ravel(), r2]))
        c2 = np.clip(c2, 0, 1)
        r2 = _repair(c2, r2)
        return (c2, r2) if r2.sum() > r.sum() else (c, r)

    if len(chains) == 1:
        c, r = chains[0]
        while time.process_time() - t0 < budget:
            c, r = step(c, r)
        return c, r
    t_mid = time.process_time() + (budget - (time.process_time() - t0)) / 2   # half the remaining budget
    while time.process_time() < t_mid:   # alternate one step on each chain
        chains = [step(c, r) for c, r in chains]
    c, r = max(chains, key=lambda p: p[1].sum())   # drop the weaker chain
    while time.process_time() - t0 < budget:
        c, r = step(c, r)
    return c, r
