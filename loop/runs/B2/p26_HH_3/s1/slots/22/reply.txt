def _lp_radii(c, r):
    """Exact radii for fixed centres: maximise sum r subject to pair and wall constraints (LP)."""
    n = len(r)
    i, j = np.triu_indices(n, 1)
    A = np.zeros((len(i), n))
    A[np.arange(len(i)), i] = 1
    A[np.arange(len(i)), j] = 1
    b = np.sqrt(((c[i] - c[j]) ** 2).sum(1))
    wall = np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1]))
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=np.c_[np.zeros(n), np.maximum(wall, 0)], method="highs")
    if res.status == 0 and res.x.sum() > r.sum():
        return res.x
    return r
