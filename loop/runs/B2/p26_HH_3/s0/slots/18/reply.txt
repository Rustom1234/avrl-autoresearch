def _solve_once(n, rng):
    k = rng.integers(3, 6)
    cluster_centres = rng.uniform(0.2, 0.8, (k, 2))
    idx = rng.integers(0, k, n)
    pts = np.clip(cluster_centres[idx] + rng.normal(0, 0.15, (n, 2)), INIT_LOW, 1 - INIT_LOW)
    x0 = np.concatenate([pts.ravel(), np.full(n, INIT_RADIUS)])
    return _optimise(n, x0)
