def _optimise(n, x0):
    iu = np.triu_indices(n, 1)
    m = len(iu[0])
    k = np.arange(n)
    p = np.arange(m)
    grad = np.concatenate([np.zeros(2 * n), -np.ones(n)])

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    def cons_jac(v):
        c = v[:2 * n].reshape(n, 2)
        i, j = iu
        delta = c[i] - c[j]
        u = delta / np.maximum(np.sqrt((delta ** 2).sum(1)), 1e-12)[:, None]
        J = np.zeros((m + 5 * n, 3 * n))
        J[p, 2 * i] = u[:, 0]
        J[p, 2 * i + 1] = u[:, 1]
        J[p, 2 * j] = -u[:, 0]
        J[p, 2 * j + 1] = -u[:, 1]
        J[p, 2 * n + i] = -1
        J[p, 2 * n + j] = -1
        b = m
        J[b + k, 2 * k] = 1
        J[b + n + k, 2 * k + 1] = 1
        J[b + 2 * n + k, 2 * k] = -1
        J[b + 3 * n + k, 2 * k + 1] = -1
        for s in range(4):
            J[b + s * n + k, 2 * n + k] = -1
        J[b + 4 * n + k, 2 * n + k] = 1
        return J

    res = minimize(lambda v: -v[2 * n:].sum(), x0, jac=lambda v: grad, method="SLSQP",
                   constraints={"type": "ineq", "fun": cons, "jac": cons_jac}, options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
