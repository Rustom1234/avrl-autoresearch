def _greedy_init(n, rng):
    """Place circles one at a time in the largest hole of a fixed 40x40 candidate grid."""
    g = (np.arange(40) + 0.5) / 40
    cand = np.stack(np.meshgrid(g, g), -1).reshape(-1, 2)
    wall = np.minimum(np.minimum(cand[:, 0], cand[:, 1]), np.minimum(1 - cand[:, 0], 1 - cand[:, 1]))
    centers, radii = [], []
    for _ in range(n):
        allowed = wall
        if centers:
            C, Rr = np.array(centers), np.array(radii)
            d = np.sqrt(((cand[:, None] - C[None]) ** 2).sum(-1))
            allowed = np.minimum(wall, (d - Rr[None]).min(1))
        allowed = np.minimum(allowed, rng.uniform(0.07, 0.12))
        k = np.argsort(-allowed)[:5][rng.integers(5)]
        centers.append(cand[k])
        radii.append(max(allowed[k], 0.0))
    return np.array(centers), np.array(radii)
