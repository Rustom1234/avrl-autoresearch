def _hex_init(n, rng):
    R = int(rng.integers(5, 8))
    m = int(np.ceil(n / R))
    pts = np.array([((j + 0.5 + 0.5 * (i % 2)) / (m + 0.5), (i + 0.5) / R) for i in range(R) for j in range(m)])
    pts = pts[rng.permutation(len(pts))[:n]] + rng.normal(0, 0.02, (n, 2))
    return np.clip(pts, 0.03, 0.97)


def _score(pts):
    """Feasible radius estimate: sum of min(wall distance, half nearest-neighbour distance)."""
    wall = np.minimum(pts, 1 - pts).min(1)
    d = np.sqrt(((pts[:, None] - pts[None]) ** 2).sum(-1))
    np.fill_diagonal(d, np.inf)
    return np.minimum(wall, 0.5 * d.min(1)).sum()
