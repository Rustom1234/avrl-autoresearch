def descriptors(smiles):
    base = featurize(smiles)
    gast = []
    for s in smiles:
        m = Chem.MolFromSmiles(s)
        rdPartialCharges.ComputeGasteigerCharges(m)
        c = np.array([m.GetAtomWithIdx(a.GetIdx()).GetDoubleProp('_GasteigerCharge')
                      for a in m.GetAtoms() if a.GetAtomicNum() > 1])
        c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
        c = np.clip(c, -0.5, 0.5)
        hist, _ = np.histogram(c, bins=12, range=(-0.5, 0.5))
        gast.append(np.concatenate([hist, [c[c > 0].sum(), c[c < 0].sum()]]))
    return np.hstack([base, np.array(gast, dtype=np.float32)])
