def fit_predict(train_smiles, train_y, test_smiles, seed):
    y = np.asarray(train_y, dtype=np.int64)
    train_sets = fingerprints(train_smiles)
    test_sets = fingerprints(test_smiles)

    # Inner 4-fold stratified CV on training data only: score each feature set by out-of-fold AUROC
    inner = StratifiedKFold(n_splits=4, shuffle=True, random_state=seed)
    aucs = []
    for x_tr in train_sets:
        oof = np.zeros(len(y))
        for a, b in inner.split(x_tr, y):
            m = XGBClassifier(n_estimators=300, max_depth=5, learning_rate=0.05, subsample=0.8, colsample_bytree=0.3,
                              min_child_weight=2, eval_metric='logloss', random_state=seed, n_jobs=8)
            m.fit(x_tr[a], y[a])
            oof[b] = m.predict_proba(x_tr[b])[:, 1]
        aucs.append(roc_auc_score(y, oof))
    w = np.maximum(np.array(aucs) - 0.5, 0.0)
    w = w / w.sum() if w.sum() > 0 else np.ones(len(w)) / len(w)

    total = np.zeros(len(test_smiles))
    for wi, x_tr, x_te in zip(w, train_sets, test_sets):
        model = XGBClassifier(n_estimators=500, max_depth=5, learning_rate=0.05, subsample=0.8, colsample_bytree=0.3,
                              min_child_weight=2, eval_metric='logloss', random_state=seed, n_jobs=8)
        model.fit(x_tr, y)
        total += wi * rankdata(model.predict_proba(x_te)[:, 1])
    total = (total - total.min()) / (total.max() - total.min() + 1e-12)
    return total
