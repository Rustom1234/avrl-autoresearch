def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    budget = TIME_FRACTION * time_limit
    while time.process_time() - t0 < budget:
        if best is None:
            x0 = None   # first run: random start
        else:           # basin hopping: jitter the incumbent, sd decays geometrically with elapsed budget
            frac = (time.process_time() - t0) / budget
            sigma = JITTER_START * (JITTER_END / JITTER_START) ** frac
            c0 = best[0] + rng.normal(0, sigma, best[0].shape)
            x0 = np.concatenate([c0.ravel(), best[1]])
        c, r = _solve_once(n, rng, x0)
        c = np.clip(c, 0, 1)
