def _grow(c, r):
    """Add one circle at the largest hole found on a grid; its radius is the clearance there."""
    g = np.linspace(0, 1, 101)
    p = np.stack(np.meshgrid(g, g), -1).reshape(-1, 2)
    d = np.sqrt(((p[:, None] - c[None]) ** 2).sum(-1)) - r[None]
    clear = np.minimum(d.min(1), np.minimum(np.minimum(p[:, 0], p[:, 1]), np.minimum(1 - p[:, 0], 1 - p[:, 1])))
    k = np.argmax(clear)
    return np.vstack([c, p[k]]), np.append(r, max(clear[k], 0))


def _best_restarts(n, rng, budget):
    t0, best = time.process_time(), None
    while time.process_time() - t0 < budget:   # random restarts until the budget is used
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    return best


def solve(n, seed, time_limit):
    rng, t0 = np.random.default_rng(seed), time.process_time()
    budget = TIME_FRACTION * time_limit
    best = None
    if n > 1:   # incremental start: packing of n-1 circles plus one circle in the largest hole
        cp, rp = _best_restarts(n - 1, rng, 0.3 * budget)
        c0, r0 = _grow(cp, rp)
        c, r = _solve_once(n, rng, np.concatenate([c0.ravel(), r0]))
        c = np.clip(c, 0, 1)
        best = (c, _repair(c, r))
