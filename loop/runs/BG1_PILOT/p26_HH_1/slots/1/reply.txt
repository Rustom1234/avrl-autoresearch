def _solve_once(n, rng):
    v = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    iu = np.triu_indices(n, 1)

    def cons(v, s):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r, s * 0.5 - r])

    for s in STAGES:   # each stage starts from the previous stage's solution
        res = minimize(lambda v: -v[2 * n:].sum(), v, method="SLSQP",
                       constraints={"type": "ineq", "fun": lambda v: cons(v, s)}, options={"maxiter": MAX_ITER})
        v = res.x
    return v[:2 * n].reshape(n, 2), v[2 * n:]
