def _penalty(v, n, iu, w=1e3):
    """Minus the sum of radii plus w times the squared overlap and wall violations, with its gradient."""
    c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
    diff = c[iu[0]] - c[iu[1]]
    dist = np.sqrt((diff ** 2).sum(1)) + 1e-12
    ov = np.maximum(0, r[iu[0]] + r[iu[1]] - dist)
    lo = np.maximum(0, r[:, None] - c)
    hi = np.maximum(0, c + r[:, None] - 1)
    f = -r.sum() + w * ((ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum())
    u, gp = diff / dist[:, None], 2 * w * ov
    gc, gr = np.zeros((n, 2)), -np.ones(n)
    np.add.at(gc, iu[0], -gp[:, None] * u)
    np.add.at(gc, iu[1], gp[:, None] * u)
    np.add.at(gr, iu[0], gp)
    np.add.at(gr, iu[1], gp)
    gc += 2 * w * (hi - lo)
    gr += 2 * w * (lo.sum(1) + hi.sum(1))
    return f, np.concatenate([gc.ravel(), gr])
