    """With centres fixed, the max-sum radii is an LP. Falls back to r if the LP fails."""
    n = len(r)
    i, j = np.triu_indices(n, 1)
    d = np.sqrt(((c[i] - c[j]) ** 2).sum(1))
    A = np.zeros((len(i), n))
    A[np.arange(len(i)), i] = 1
    A[np.arange(len(i)), j] = 1
    wall = np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1]))
    wall = np.maximum(wall - 1e-9, 0)
    res = linprog(-np.ones(n), A_ub=A, b_ub=d - 1e-9, bounds=list(zip(np.zeros(n), wall)), method="highs")
    if res.status != 0:
        return r
    return np.maximum(res.x - 1e-12, 0)
