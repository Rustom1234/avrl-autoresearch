def _rattlers(c, r, tol=1e-6):
    """Indices of circles with no constraint of slack below tol (not load-bearing)."""
    n = len(r)
    loaded = np.zeros(n, dtype=bool)
    iu, ju = np.triu_indices(n, 1)
    dd = np.hypot(c[iu, 0] - c[ju, 0], c[iu, 1] - c[ju, 1]) - (r[iu] + r[ju])
    act = dd < tol
    loaded[iu[act]] = True
    loaded[ju[act]] = True
    wall = np.minimum(np.minimum(c[:, 0] - r, 1 - c[:, 0] - r), np.minimum(c[:, 1] - r, 1 - c[:, 1] - r))
    loaded |= wall < tol
    return np.where(~loaded)[0]


def _largest_hole(centers, radii, rng, m=2000):
    """Sample m seeded points; return the one with the greatest clearance to circles and walls."""
    p = rng.uniform(0.0, 1.0, size=(m, 2))
    dc = np.hypot(p[:, None, 0] - centers[None, :, 0], p[:, None, 1] - centers[None, :, 1]) - radii[None, :]
    wall = np.minimum(np.minimum(p[:, 0], 1 - p[:, 0]), np.minimum(p[:, 1], 1 - p[:, 1]))
    score = np.minimum(dc.min(axis=1), wall)
    return p[int(np.argmax(score))]
