def _fast_radii(c):
    """Vectorized feasibility projection: radii for fixed centers via pairwise shrink."""
    n = c.shape[0]
    cc = np.clip(c, 0.0, 1.0)
    r = np.maximum(np.minimum(np.minimum(cc[:, 0], cc[:, 1]),
                              np.minimum(1 - cc[:, 0], 1 - cc[:, 1])), 0.0)
    iu, ju = np.triu_indices(n, 1)
    d = np.sqrt(np.sum((cc[iu] - cc[ju]) ** 2, axis=1))
    for _ in range(60):
        s = r[iu] + r[ju]
        bad = s > d * (1 - 1e-10)
        if not np.any(bad):
            break
        f = np.where(bad, d * (1 - 1e-10) / np.maximum(s, 1e-12), 1.0)
        fac = np.ones(n)
        np.minimum.at(fac, iu, f)
        np.minimum.at(fac, ju, f)
        r = r * fac
    return np.maximum(r * (1 - 1e-9), 0.0)
