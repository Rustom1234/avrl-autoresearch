def _optimize_log(centers0, radii0):
    """Joint SLSQP with r = exp(u): positivity built in, radius bounds dropped,
    Jacobian columns for u chain-ruled by multiplying the r-columns by r."""
    n = len(radii0)
    iu, ju = np.triu_indices(n, 1)
    m = len(iu)
    idx = np.arange(n)

    def unpack(z):
        return z[:n], z[n:2 * n], np.exp(z[2 * n:])

    def obj(z):
        return -np.sum(np.exp(z[2 * n:]))

    def obj_grad(z):
        g = np.zeros_like(z)
        g[2 * n:] = -np.exp(z[2 * n:])
        return g

    def cons(z):
        x, y, r = unpack(z)
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        pair = dx * dx + dy * dy - (r[iu] + r[ju]) ** 2
        bnd = np.concatenate([x - r, 1 - x - r, y - r, 1 - y - r])
        return np.concatenate([pair, bnd])

    def cons_jac(z):
        x, y, r = unpack(z)
        J = np.zeros((m + 4 * n, 3 * n))
        rows = np.arange(m)
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        s = r[iu] + r[ju]
        J[rows, iu] = 2 * dx
        J[rows, ju] = -2 * dx
        J[rows, n + iu] = 2 * dy
        J[rows, n + ju] = -2 * dy
        J[rows, 2 * n + iu] = -2 * s * r[iu]
        J[rows, 2 * n + ju] = -2 * s * r[ju]
        J[m + idx, idx] = 1
        J[m + idx, 2 * n + idx] = -r
        J[m + n + idx, idx] = -1
        J[m + n + idx, 2 * n + idx] = -r
        J[m + 2 * n + idx, n + idx] = 1
        J[m + 2 * n + idx, 2 * n + idx] = -r
        J[m + 3 * n + idx, n + idx] = -1
        J[m + 3 * n + idx, 2 * n + idx] = -r
        return J

    z0 = np.concatenate([centers0[:, 0], centers0[:, 1],
                         np.log(np.maximum(np.asarray(radii0, dtype=float), 1e-3))])
    bounds = [(0.0, 1.0)] * (2 * n) + [(None, None)] * n
    res = minimize(obj, z0, jac=obj_grad, method="SLSQP",
                   constraints=[{"type": "ineq", "fun": cons, "jac": cons_jac}],
                   bounds=bounds, options={"maxiter": 300, "ftol": 1e-12})
    x, y, r = unpack(res.x)
    return np.column_stack([x, y]), r
