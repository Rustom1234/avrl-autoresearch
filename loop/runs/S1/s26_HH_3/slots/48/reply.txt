def _pair_set(centers, n):
    """Delaunay edges plus close pairs (distance < 0.35) as the constraint index set."""
    from scipy.spatial import Delaunay
    cs = set()
    try:
        tri = Delaunay(np.asarray(centers, dtype=float))
        for simp in tri.simplices:
            for a in range(3):
                for b2 in range(a + 1, 3):
                    p, q = sorted((int(simp[a]), int(simp[b2])))
                    cs.add((p, q))
    except Exception:
        pass
    for i in range(n):
        for j in range(i + 1, n):
            if np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1]) < 0.35:
                cs.add((i, j))
    if not cs:
        return np.triu_indices(n, 1)
    pairs = sorted(cs)
    return np.array([p[0] for p in pairs]), np.array([p[1] for p in pairs])
