from scipy.optimize import linprog


def _lp_radii(centers):
    """Maximize sum(r) at fixed centers via LP (pair and wall constraints)."""
    n = centers.shape[0]
    iu, ju = np.triu_indices(n, 1)
    m = len(iu)
    d = np.hypot(centers[iu, 0] - centers[ju, 0], centers[iu, 1] - centers[ju, 1])
    A = np.zeros((m, n))
    A[np.arange(m), iu] = 1.0
    A[np.arange(m), ju] = 1.0
    x, y = centers[:, 0], centers[:, 1]
    wall = np.maximum(np.minimum(np.minimum(x, y), np.minimum(1 - x, 1 - y)), 0.0)
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=d,
                      bounds=[(0.0, float(w)) for w in wall], method="highs")
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    return np.maximum(res.x, 0.0) * (1 - 1e-9)


def _es_search(c0, deadline, gens=150, lam=12, seed=2024):
    """(1+lambda) evolution strategy over centers; radii from LP; one-fifth-style sigma rule."""
    rng = np.random.default_rng(seed)
    n = c0.shape[0]
    parent = np.array(c0, dtype=float)
    r0 = _lp_radii(parent)
    if r0 is None:
        return parent, -1.0
    f_par = float(np.sum(r0))
    sigma = 0.05
    for _ in range(gens):
        if time.time() > deadline:
            break
        best_child, best_f = None, -1.0
        for _ in range(lam):
            child = np.clip(parent + rng.normal(0.0, sigma, size=(n, 2)), 0.0, 1.0)
            rc = _lp_radii(child)
            if rc is None:
                continue
            fc = float(np.sum(rc))
            if fc > best_f:
                best_child, best_f = child, fc
        if best_child is not None and best_f > f_par + 1e-12:
            parent, f_par = best_child, best_f
            sigma *= 1.2
        else:
            sigma *= 0.9
    return parent, f_par
