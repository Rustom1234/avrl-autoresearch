def _spiral_start(rng, n):
    """Archimedean spiral centres r = a + b*theta, winding outward from the centre,
    rescaled into [0.05, 0.95]^2, with a decreasing radius profile (large inside, small at rim)."""
    a = rng.uniform(0.0, 0.05)
    b = rng.uniform(0.02, 0.06)
    off = rng.uniform(0.0, 2 * np.pi)
    th = np.linspace(0.0, 4 * np.pi, n)
    rr = a + b * th
    pts = np.column_stack([rr * np.cos(th + off), rr * np.sin(th + off)])
    lo = pts.min(axis=0)
    hi = pts.max(axis=0)
    pts = 0.05 + 0.9 * (pts - lo) / np.maximum(hi - lo, 1e-9)
    radii = np.linspace(0.13, 0.06, n) * rng.uniform(0.9, 1.1, size=n)
    return pts, radii
