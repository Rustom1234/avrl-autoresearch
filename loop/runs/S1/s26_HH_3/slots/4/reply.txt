def _inward_pull(centers):
    """Staged quadratic penalty k*||c-0.5||^2 pulling centers inward; k decays to zero."""
    c = np.clip(np.asarray(centers, dtype=float).copy(), 0.0, 1.0)
    for k in (0.05, 0.02, 0.005, 0.0):
        for _ in range(10):
            c = c - 0.5 * (2.0 * k) * (c - 0.5)
    return np.clip(c, 0.0, 1.0)
