def _tr_slp(centers0, radii0, max_iter=60):
    """Trust-region sequential LP over (x, y, r). Pair distances are linearised (the
    linearisation of the convex distance is conservative), containment is exact and
    linear. Step accepted if the true max violation <= 1e-9 and sum(r) improves."""
    from scipy.optimize import linprog
    n = len(radii0)
    iu, ju = np.triu_indices(n, 1)
    m = len(iu)
    c = np.clip(np.asarray(centers0, dtype=float).copy(), 0.0, 1.0)
    r = _repair(c, np.asarray(radii0, dtype=float).copy())

    def viol(cc, rr):
        dx = cc[iu, 0] - cc[ju, 0]
        dy = cc[iu, 1] - cc[ju, 1]
        d = np.sqrt(dx * dx + dy * dy)
        pv = float(np.max(rr[iu] + rr[ju] - d)) if m else 0.0
        bv = float(np.max(np.concatenate([rr - cc[:, 0], cc[:, 0] + rr - 1.0,
                                          rr - cc[:, 1], cc[:, 1] + rr - 1.0])))
        return max(pv, bv, 0.0)

    cur = float(np.sum(r))
    delta = 0.01
    cost = np.zeros(3 * n)
    cost[2 * n:] = -1.0
    for _ in range(max_iter):
        if delta < 1e-7:
            break
        dx = c[iu, 0] - c[ju, 0]
        dy = c[iu, 1] - c[ju, 1]
        d = np.maximum(np.sqrt(dx * dx + dy * dy), 1e-12)
        ux = dx / d
        uy = dy / d
        A = np.zeros((m + 4 * n, 3 * n))
        b = np.zeros(m + 4 * n)
        rows = np.arange(m)
        A[rows, iu] = -ux
        A[rows, ju] = ux
        A[rows, n + iu] = -uy
        A[rows, n + ju] = uy
        A[rows, 2 * n + iu] = 1.0
        A[rows, 2 * n + ju] = 1.0
        b[:m] = d - r[iu] - r[ju]
        idx = np.arange(n)
        A[m + idx, idx] = -1.0
        A[m + idx, 2 * n + idx] = 1.0
        b[m:m + n] = c[:, 0] - r
        A[m + n + idx, idx] = 1.0
        A[m + n + idx, 2 * n + idx] = 1.0
        b[m + n:m + 2 * n] = 1.0 - c[:, 0] - r
        A[m + 2 * n + idx, n + idx] = -1.0
        A[m + 2 * n + idx, 2 * n + idx] = 1.0
        b[m + 2 * n:m + 3 * n] = c[:, 1] - r
        A[m + 3 * n + idx, n + idx] = 1.0
        A[m + 3 * n + idx, 2 * n + idx] = 1.0
        b[m + 3 * n:] = 1.0 - c[:, 1] - r
        b = np.maximum(b, 0.0)
        bounds = [(-delta, delta)] * (2 * n) + [(max(-delta, -float(ri)), delta) for ri in r]
        try:
            res = linprog(cost, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        except Exception:
            break
        if res.status != 0 or res.x is None:
            delta *= 0.5
            continue
        dz = res.x
        cn = c + np.column_stack([dz[:n], dz[n:2 * n]])
        rn = np.maximum(r + dz[2 * n:], 0.0)
        s_new = float(np.sum(rn))
        if viol(cn, rn) <= 1e-9 and s_new > cur + 1e-12:
            c, r, cur = cn, rn, s_new
            delta = min(delta * 1.5, 0.2)
        else:
            delta *= 0.5
    return c, r
