def _gravity_start(rng, n):
    """Biased compaction start: circles sink toward y=0 under a seeded per-start
    bias direction, with a per-circle seeded horizontal shear; clipped each step."""
    c = rng.uniform(0.05, 0.95, size=(n, 2))
    ang = rng.uniform(-0.6, 0.6)
    g = np.array([np.sin(ang), -np.cos(ang)])
    off = rng.uniform(-1.0, 1.0, size=n)
    for _ in range(30):
        c[:, 0] = c[:, 0] + 0.02 * g[0] + 0.005 * off
        c[:, 1] = c[:, 1] + 0.02 * g[1]
        c = np.clip(c, 0.0, 1.0)
    return c
