def _pinned_start(rng):
    """Edge-pinned start: 4 to 6 seeded circles per wall are placed in contact with it
    (radius equal to wall distance); the rest are uniform random."""
    n = 26
    c = rng.uniform(0.05, 0.95, size=(n, 2))
    r = rng.uniform(0.06, 0.1, size=n)
    pins = []
    for ax in (0, 1):
        for val in (0, 1):
            k = int(rng.integers(4, 7))
            for i in rng.choice(n, size=k, replace=False):
                i = int(i)
                if any(p[0] == i and p[1] == ax for p in pins):
                    continue
                pins.append((i, ax, val))
                c[i, ax] = r[i] if val == 0 else 1 - r[i]
    return np.clip(c, 0.0, 1.0), r, pins
