def _maxmin_start(centers0):
    """Max-min homotopy: maximize one shared radius t over centers, then return centers and t."""
    n = centers0.shape[0]
    iu, ju = np.triu_indices(n, 1)
    m = len(iu)
    idx = np.arange(n)

    def obj(z):
        return -z[2 * n]

    def obj_grad(z):
        g = np.zeros_like(z)
        g[2 * n] = -1.0
        return g

    def cons(z):
        x, y, t = z[:n], z[n:2 * n], z[2 * n]
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        pair = dx * dx + dy * dy - 4.0 * t * t
        bnd = np.concatenate([x - t, 1 - x - t, y - t, 1 - y - t])
        return np.concatenate([pair, bnd])

    def cons_jac(z):
        x, y, t = z[:n], z[n:2 * n], z[2 * n]
        J = np.zeros((m + 4 * n, 2 * n + 1))
        rows = np.arange(m)
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        J[rows, iu] = 2 * dx
        J[rows, ju] = -2 * dx
        J[rows, n + iu] = 2 * dy
        J[rows, n + ju] = -2 * dy
        J[rows, 2 * n] = -8 * t
        J[m + idx, idx] = 1
        J[m + idx, 2 * n] = -1
        J[m + n + idx, idx] = -1
        J[m + n + idx, 2 * n] = -1
        J[m + 2 * n + idx, n + idx] = 1
        J[m + 2 * n + idx, 2 * n] = -1
        J[m + 3 * n + idx, n + idx] = -1
        J[m + 3 * n + idx, 2 * n] = -1
        return J

    t0 = 0.01
    z0 = np.concatenate([centers0[:, 0], centers0[:, 1], [t0]])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)]
    res = minimize(obj, z0, jac=obj_grad, method="SLSQP",
                   constraints=[{"type": "ineq", "fun": cons, "jac": cons_jac}],
                   bounds=bounds, options={"maxiter": 200, "ftol": 1e-12})
    x = np.clip(res.x[:n], 0.0, 1.0)
    y = np.clip(res.x[n:2 * n], 0.0, 1.0)
    return np.column_stack([x, y]), max(float(res.x[2 * n]), 0.0)
