def _hex_start(rng):
    """Staggered rows of 5,6,5,5,5 circles, odd rows offset by half a spacing, plus Gaussian jitter."""
    counts = [5, 6, 5, 5, 5]
    nrows = len(counts)
    pts = []
    for ri, cnt in enumerate(counts):
        y = 0.1 + 0.8 * ri / (nrows - 1)
        off = 0.5 if ri % 2 else 0.0
        for ci in range(cnt):
            x = 0.1 + 0.8 * (ci + 0.5 + off) / cnt
            pts.append([x, y])
    c = np.array(pts, dtype=float)
    c = c + rng.normal(0.0, 0.02, size=c.shape)
    return np.clip(c, 0.01, 0.99)
