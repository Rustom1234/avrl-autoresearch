def _warp_search(rng):
    """Smooth-warp start: x_i = u_i + sum_k a_k P_k(u_i, v_i), y likewise, with
    tensor-product Legendre P_0..P_2 in each coordinate (9 terms per axis).
    Coefficients start at zero; jointly optimised with radii by SLSQP."""
    n = 26
    nb = 9
    u = rng.uniform(0.05, 0.95, size=n)
    v = rng.uniform(0.05, 0.95, size=n)

    def leg(x):
        return [np.ones_like(x), x, 1.5 * x * x - 0.5]

    Lu = leg(2 * u - 1)
    Lv = leg(2 * v - 1)
    Phi = np.column_stack([Lu[i] * Lv[j] for i in range(3) for j in range(3)])
    nv = 2 * nb + n
    A = np.zeros((3 * n, nv))
    b = np.zeros(3 * n)
    idx = np.arange(n)
    A[:n, :nb] = Phi
    b[:n] = u
    A[n:2 * n, nb:2 * nb] = Phi
    b[n:2 * n] = v
    A[2 * n + idx, 2 * nb + idx] = 1.0
    iu, ju = np.triu_indices(n, 1)
    mp = len(iu)

    def expand(w):
        return A @ w + b

    def obj(w):
        return -np.sum(expand(w)[2 * n:])

    def obj_grad(w):
        return -A[2 * n:].sum(axis=0)

    def cons(w):
        z = expand(w)
        x, y, r = z[:n], z[n:2 * n], z[2 * n:]
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        pair = dx * dx + dy * dy - (r[iu] + r[ju]) ** 2
        bnd = np.concatenate([x - r, 1 - x - r, y - r, 1 - y - r])
        return np.concatenate([pair, bnd])

    def cons_jac(w):
        z = expand(w)
        x, y, r = z[:n], z[n:2 * n], z[2 * n:]
        J = np.zeros((mp + 4 * n, 3 * n))
        rows = np.arange(mp)
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        s = r[iu] + r[ju]
        J[rows, iu] = 2 * dx
        J[rows, ju] = -2 * dx
        J[rows, n + iu] = 2 * dy
        J[rows, n + ju] = -2 * dy
        J[rows, 2 * n + iu] = -2 * s
        J[rows, 2 * n + ju] = -2 * s
        J[mp + idx, idx] = 1
        J[mp + idx, 2 * n + idx] = -1
        J[mp + n + idx, idx] = -1
        J[mp + n + idx, 2 * n + idx] = -1
        J[mp + 2 * n + idx, n + idx] = 1
        J[mp + 2 * n + idx, 2 * n + idx] = -1
        J[mp + 3 * n + idx, n + idx] = -1
        J[mp + 3 * n + idx, 2 * n + idx] = -1
        return J @ A

    w0 = np.concatenate([np.zeros(2 * nb), np.full(n, 0.07)])
    bounds = [(-0.6, 0.6)] * (2 * nb) + [(0.0, 0.5)] * n
    res = minimize(obj, w0, jac=obj_grad, method="SLSQP",
                   constraints=[{"type": "ineq", "fun": cons, "jac": cons_jac}],
                   bounds=bounds, options={"maxiter": 300, "ftol": 1e-12})
    z = expand(res.x)
    centers = np.clip(np.column_stack([z[:n], z[n:2 * n]]), 0.0, 1.0)
    radii = np.maximum(z[2 * n:], 0.0)
    return centers, radii
