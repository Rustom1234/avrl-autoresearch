def _torus_start(rng):
    """Periodic soft-repulsion packing on a unit torus, mapped into the square with 0.9 shrink."""
    n = 26
    rho = 0.1
    p = rng.uniform(0.0, 1.0, size=(n, 2))
    iu, ju = np.triu_indices(n, 1)
    step = 0.05
    for _ in range(300):
        d = p[iu] - p[ju]
        d = d - np.round(d)
        dist = np.sqrt(np.sum(d * d, axis=1)) + 1e-12
        ov = np.maximum(0.0, 2 * rho - dist)
        if not np.any(ov > 0):
            break
        push = (ov / dist)[:, None] * d
        disp = np.zeros((n, 2))
        np.add.at(disp, iu, push)
        np.add.at(disp, ju, -push)
        p = (p + step * disp) % 1.0
    q = 0.5 + 0.9 * (p - 0.5)
    c = rho + (1.0 - 2.0 * rho) * q
    return np.clip(c, 0.0, 1.0)
