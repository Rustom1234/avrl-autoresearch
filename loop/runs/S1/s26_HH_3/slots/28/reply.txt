def _al_polish(centers0, radii0, outer=10):
    """Augmented-Lagrangian polish of sum(r) subject to pair and wall constraints g >= 0."""
    n = len(radii0)
    iu, ju = np.triu_indices(n, 1)
    m = len(iu)
    idx = np.arange(n)

    def unpack(z):
        return z[:n], z[n:2 * n], z[2 * n:]

    def cons(z):
        x, y, r = unpack(z)
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        pair = dx * dx + dy * dy - (r[iu] + r[ju]) ** 2
        bnd = np.concatenate([x - r, 1 - x - r, y - r, 1 - y - r])
        return np.concatenate([pair, bnd])

    def cons_jac(z):
        x, y, r = unpack(z)
        J = np.zeros((m + 4 * n, 3 * n))
        rows = np.arange(m)
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        s = r[iu] + r[ju]
        J[rows, iu] = 2 * dx
        J[rows, ju] = -2 * dx
        J[rows, n + iu] = 2 * dy
        J[rows, n + ju] = -2 * dy
        J[rows, 2 * n + iu] = -2 * s
        J[rows, 2 * n + ju] = -2 * s
        J[m + idx, idx] = 1
        J[m + idx, 2 * n + idx] = -1
        J[m + n + idx, idx] = -1
        J[m + n + idx, 2 * n + idx] = -1
        J[m + 2 * n + idx, n + idx] = 1
        J[m + 2 * n + idx, 2 * n + idx] = -1
        J[m + 3 * n + idx, n + idx] = -1
        J[m + 3 * n + idx, 2 * n + idx] = -1
        return J

    z = np.concatenate([centers0[:, 0], centers0[:, 1], radii0])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    lam = np.zeros(m + 4 * n)
    rho = 10.0
    prev_viol = np.inf
    for _ in range(outer):
        def f(zz, lam=lam, rho=rho):
            g = cons(zz)
            t = np.maximum(0.0, lam / rho - g)
            val = -np.sum(zz[2 * n:]) + 0.5 * rho * np.sum(t * t) - np.sum(lam * lam) / (2 * rho)
            grad = np.zeros_like(zz)
            grad[2 * n:] = -1.0
            grad -= rho * (t @ cons_jac(zz))
            return val, grad

        res = minimize(f, z, jac=True, method="L-BFGS-B", bounds=bounds,
                       options={"maxiter": 200})
        z = res.x
        g = cons(z)
        lam = np.maximum(0.0, lam - rho * g)
        viol = max(0.0, float(-g.min()))
        if viol > prev_viol / 4.0:
            rho *= 3.0
        prev_viol = viol
    x, y, r = unpack(z)
    return np.column_stack([x, y]), r
