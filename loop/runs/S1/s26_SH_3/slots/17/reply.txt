def corner_jitter(rng):
    """Corner-anchored init: 4 corner circles, 4 edge midpoints, 18 interior lattice points (jittered)."""
    pts = [[0.13, 0.13], [0.87, 0.13], [0.13, 0.87], [0.87, 0.87],
           [0.5, 0.05], [0.05, 0.5], [0.95, 0.5], [0.5, 0.95]]
    rows = [5, 4, 5, 4]
    span = float(rng.uniform(0.34, 0.40))
    R = len(rows)
    for k, cnt in enumerate(rows):
        y = 0.5 + span * (k / (R - 1) - 0.5)
        for j in range(cnt):
            x = 0.5 + 0.5 * span * (j - (cnt - 1) / 2.0) / 2.5
            pts.append([x, y])
    pts = np.array(pts)
    pts = pts + rng.normal(0.0, 0.02, pts.shape)
    return np.clip(pts, 0.05, 0.95)
