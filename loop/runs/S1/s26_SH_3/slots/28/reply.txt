def sa_centers(c, rng, t0, steps=6000, T0=0.01, T1=0.0001):
    """Simulated annealing over centers; objective is LP-optimal sum of radii."""
    import time
    import math
    cur = c.copy()
    cur_r = lp_radii(cur)
    cur_v = float(np.sum(cur_r))
    best, best_v = cur.copy(), cur_v
    for s in range(steps):
        if time.time() - t0 > 60:
            break
        frac = s / steps
        T = T0 * (T1 / T0) ** frac
        sigma = 0.003 + 0.02 * (T / T0)
        if rng.random() < 0.5:
            k = int(np.argsort(cur_r)[int(rng.integers(5))])
        else:
            k = int(rng.integers(c.shape[0]))
        cand = cur.copy()
        cand[k] = np.clip(cand[k] + rng.normal(0.0, sigma, 2), 0.02, 0.98)
        cand_r = lp_radii(cand)
        v = float(np.sum(cand_r))
        if v >= cur_v or rng.random() < math.exp((v - cur_v) / T):
            cur, cur_v, cur_r = cand, v, cand_r
            if v > best_v:
                best, best_v = cand.copy(), v
    return best
