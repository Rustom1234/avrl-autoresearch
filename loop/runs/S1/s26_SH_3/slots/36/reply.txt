    def optimize(z0):
        from scipy.optimize import NonlinearConstraint
        from scipy.sparse import csr_matrix
        # near-neighbour pairs (d < 0.4) at the current start, plus all boundary rows
        dd = np.sqrt((z0[I] - z0[Jp]) ** 2 + (z0[n + I] - z0[n + Jp]) ** 2)
        sel = np.concatenate([np.where(dd < 0.4)[0], np.arange(m, m + 4 * n)])

        def cons_s(z):
            return cons(z)[sel]

        def cjac_s(z):
            return csr_matrix(cjac(z)[sel])

        nlc = NonlinearConstraint(cons_s, 0.0, np.inf, jac=cjac_s)
        res = minimize(lambda z: -np.sum(z[2 * n:]), z0, jac=lambda z: obj_grad,
                       constraints=[nlc],
                       bounds=[(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n,
                       method="trust-constr",
                       options={"maxiter": 150, "gtol": 1e-9, "xtol": 1e-10})
        z = res.x
