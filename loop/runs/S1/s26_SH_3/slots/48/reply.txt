def big_center_init(rng):
    """Hex-row layout with 1-4 big circles near the centre; returns centres and starting radii."""
    c = hex_jitter(rng)
    k = int(rng.integers(1, 5))
    c[:k] = rng.uniform(0.35, 0.65, (k, 2))
    d = np.sqrt(np.sum((c - 0.5) ** 2, axis=1))
    r0 = 0.16 - 0.10 * np.minimum(d / 0.5, 1.0)
    return c, r0
