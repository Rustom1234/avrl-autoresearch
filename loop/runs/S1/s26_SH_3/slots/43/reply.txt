def radial_init(rng):
    """Big centre, small edges: centres on rings with jitter plus 4 corners; radii fall off with distance from centre."""
    pts = [[0.5, 0.5]]
    for rr, cnt in [(0.2, 6), (0.33, 8), (0.43, 7)]:
        off = rng.random() * np.pi
        for j in range(cnt):
            a = off + 2 * np.pi * j / cnt
            pts.append([0.5 + rr * np.cos(a), 0.5 + rr * np.sin(a)])
    pts += [[0.07, 0.07], [0.93, 0.07], [0.07, 0.93], [0.93, 0.93]]
    c = np.array(pts) + rng.normal(0.0, 0.01, (26, 2))
    c = np.clip(c, 0.05, 0.95)
    d = np.sqrt(np.sum((c - 0.5) ** 2, axis=1))
    r = 0.14 - 0.07 * np.minimum(d / 0.6, 1.0)
    return c, r
