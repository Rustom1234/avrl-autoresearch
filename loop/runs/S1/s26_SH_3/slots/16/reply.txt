    def sig_of(r):

    for start in range(4):

        c0 = hex_jitter(rng)

        z = np.concatenate([c0[:, 0], c0[:, 1], lp_radii(c0)])
        cur = optimize(z)
        cur_val = np.sum(cur[2 * n:])
        # tabu: skip restarts that converge to an already found optimum
        if sig_of(cur[2 * n:]) in tabu_sigs:
            continue
        tabu_sigs.append(sig_of(cur[2 * n:]))
        # shrink-and-grow cycles, perturbation magnitude grows with cycle index
        for cyc in range(8):
            if time.time() - t0 > 180:
                break
            scale = 0.02 * (1.0 + 0.5 * cyc)
            zs = cur.copy()
            zs[2 * n:] *= 0.9
            zs[:2 * n] = np.clip(zs[:2 * n] + rng.normal(0, scale, 2 * n), 0.0, 1.0)
