    def pen_obj(z):
        r = z[2 * n:]
        return -np.sum(r) + 0.5 * np.sum((r - r.mean()) ** 2)

    def pen_grad(z):
        r = z[2 * n:]
        g = np.zeros(3 * n)
        g[2 * n:] = -1.0 + (r - r.mean())
        return g

    def optimize(z0):
        bnds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
        cons_list = [{"type": "ineq", "fun": cons, "jac": cjac}]
