def _pool_update(pool, v, st):
    """Keep up to 3 distinct states (max centre difference > 0.05) with the largest values."""
    for k, (pv, ps) in enumerate(pool):
        if np.max(np.abs(ps - st)) <= 0.05:
            if v > pv:
                pool[k] = (v, st.copy())
            return pool
    if len(pool) < 3:
        pool.append((v, st.copy()))
    else:
        k = int(np.argmin([p[0] for p in pool]))
        if v > pool[k][0]:
            pool[k] = (v, st.copy())
    return pool
