    import time
    from scipy.optimize import minimize

    t0 = time.time()
    n = 26
    I, Jp = np.triu_indices(n, 1)
    m = len(I)
    rows = np.arange(m)
    ar = np.arange(n)

    def cons(z):
        x = z[:n]
        y = z[n:2 * n]
        r = z[2 * n:]
        dx = x[I] - x[Jp]
        dy = y[I] - y[Jp]
        rs = r[I] + r[Jp]
        return np.concatenate([dx * dx + dy * dy - rs * rs, x - r, 1 - x - r, y - r, 1 - y - r])

    def cjac(z):
        x = z[:n]
        y = z[n:2 * n]
        r = z[2 * n:]
        dx = x[I] - x[Jp]
        dy = y[I] - y[Jp]
        rs = r[I] + r[Jp]
        Jm = np.zeros((m + 4 * n, 3 * n))
        Jm[rows, I] = 2 * dx
        Jm[rows, Jp] = -2 * dx
        Jm[rows, n + I] = 2 * dy
        Jm[rows, n + Jp] = -2 * dy
        Jm[rows, 2 * n + I] = -2 * rs
        Jm[rows, 2 * n + Jp] = -2 * rs
        b = m
        Jm[b + ar, ar] = 1
        Jm[b + ar, 2 * n + ar] = -1
        Jm[b + n + ar, ar] = -1
        Jm[b + n + ar, 2 * n + ar] = -1
        Jm[b + 2 * n + ar, n + ar] = 1
        Jm[b + 2 * n + ar, 2 * n + ar] = -1
        Jm[b + 3 * n + ar, n + ar] = -1
        Jm[b + 3 * n + ar, 2 * n + ar] = -1
        return Jm

    obj_grad = np.concatenate([np.zeros(2 * n), -np.ones(n)])

    def optimize(z0):
        res = minimize(lambda z: -np.sum(z[2 * n:]), z0, jac=lambda z: obj_grad,
                       constraints=[{"type": "ineq", "fun": cons, "jac": cjac}],
                       bounds=[(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n,
                       method="SLSQP", options={"maxiter": 300, "ftol": 1e-12})
        z = res.x
        if not np.all(np.isfinite(z)):
            return z0
        return z

    def safe_radii(c, r):
        r = np.minimum(r, np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1])))
        r = np.maximum(r, 0.0)
        for _ in range(500):
            bad = False
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                    s = r[i] + r[j]
                    if s > d:
                        f = d / s
                        r[i] *= f
                        r[j] *= f
                        bad = True
            if not bad:
                break
        return r * (1 - 1e-9)
