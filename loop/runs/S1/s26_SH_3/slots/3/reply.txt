def hex_jitter(rng):
    """Hex-like row lattice for 26 points, with Gaussian jitter (sigma 0.02)."""
    rows = [5, 5, 5, 5, 6] if rng.random() < 0.5 else [6, 5, 6, 5, 4]
    pts = []
    R = len(rows)
    for k, cnt in enumerate(rows):
        y = (k + 0.5) / R
        shift = 0.5 if k % 2 else 0.0
        for j in range(cnt):
            x = (j + 0.5 + shift) / (cnt + 0.5)
            pts.append([x, y])
    pts = np.array(pts)
    pts = pts + rng.normal(0.0, 0.02, pts.shape)
    return np.clip(pts, 0.05, 0.95)

def hex_jitter(rng):
    """5x5 square grid plus one extra circle at a random gap, with strong jitter (sigma 0.05)."""
    pts = []
    for j in range(5):
        for i in range(5):
            pts.append([(i + 0.5) / 5.0, (j + 0.5) / 5.0])
    pts.append([rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)])
    pts = np.array(pts)
    pts = pts + rng.normal(0.0, 0.05, pts.shape)
    return np.clip(pts, 0.05, 0.95)
