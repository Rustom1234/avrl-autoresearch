def ring_init(rng):
    """Ring-based start for 26 points: 1 centre, 6 on r~0.2, 12 on r~0.35, 7 on r~0.46, jittered."""
    pts = [[0.5, 0.5]]
    for k in range(6):
        a = 2 * np.pi * k / 6 + rng.normal(0.0, 0.05)
        r = 0.2 + rng.normal(0.0, 0.02)
        pts.append([0.5 + r * np.cos(a), 0.5 + r * np.sin(a)])
    for k in range(12):
        a = 2 * np.pi * k / 12 + rng.normal(0.0, 0.05)
        r = 0.35 + rng.normal(0.0, 0.02)
        pts.append([0.5 + r * np.cos(a), 0.5 + r * np.sin(a)])
    for k in range(7):
        a = 2 * np.pi * k / 7 + rng.normal(0.0, 0.05)
        r = 0.46 + rng.normal(0.0, 0.02)
        pts.append([0.5 + r * np.cos(a), 0.5 + r * np.sin(a)])
    pts = np.array(pts)
    return np.clip(pts, 0.05, 0.95)
