def lattice_from_params(p):
    """Hex-row lattice from 16 params in [0,1]: row pattern, y offsets, shifts, stretches."""
    rows = [5, 5, 5, 5, 6] if p[0] < 0.5 else [6, 5, 6, 5, 4]
    R = len(rows)
    pts = []
    for k, cnt in enumerate(rows):
        y = (k + 0.5) / R + (p[1 + k] - 0.5) * 0.1
        base = 0.5 if k % 2 else 0.0
        shift = base + (p[6 + k] - 0.5) * 0.5
        s = 0.8 + 0.4 * p[11 + k]
        for j in range(cnt):
            x = (j + 0.5 + shift) / (cnt + 0.5)
            x = 0.5 + (x - 0.5) * s
            pts.append([x, y])
    return np.clip(np.array(pts), 0.05, 0.95)
