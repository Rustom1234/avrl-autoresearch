def sa_centers(c, rng, t0, steps=2500, T0=0.02, T1=0.0005):
    """Simulated annealing over centers; objective is LP-optimal sum of radii.
    Moves: Gaussian single-circle (default), relocate smallest-radius circle,
    and swap two nearby circles with jitter."""
    import time
    import math
    cur = c.copy()
    cur_r = lp_radii(cur)
    cur_v = float(np.sum(cur_r))
    best, best_v = cur.copy(), cur_v
    for s in range(steps):
        if time.time() - t0 > 120:
            break
        frac = s / steps
        T = T0 * (T1 / T0) ** frac
        sigma = 0.005 + 0.045 * (T / T0)
        k = int(rng.integers(c.shape[0]))
        cand = cur.copy()
        u = rng.random()
        if u < 0.2:
            m = int(np.argmin(cur_r))
            cand[m] = rng.uniform(0.02, 0.98, 2)
        elif u < 0.3:
            d = np.sqrt(np.sum((cur - cur[k]) ** 2, axis=1))
            nb = np.where((d < 0.25) & (d > 0))[0]
            if len(nb) > 0:
                j = int(rng.choice(nb))
                cand[[k, j]] = cur[[j, k]]
                cand[[k, j]] += rng.normal(0.0, 0.01, (2, 2))
                cand = np.clip(cand, 0.02, 0.98)
            else:
                cand[k] = np.clip(cand[k] + rng.normal(0.0, sigma, 2), 0.02, 0.98)
        else:
            cand[k] = np.clip(cand[k] + rng.normal(0.0, sigma, 2), 0.02, 0.98)
        cand_r = lp_radii(cand)
        v = float(np.sum(cand_r))
        if v >= cur_v or rng.random() < math.exp((v - cur_v) / T):
            cur, cur_v, cur_r = cand, v, cand_r
            if v > best_v:
                best, best_v = cand.copy(), v
    return best
