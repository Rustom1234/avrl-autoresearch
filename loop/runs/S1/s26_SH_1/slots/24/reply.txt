def _sym_seed(rng):
    # 4 circles on the diagonal x=y plus 11 mirrored pairs (about x=y): 4 + 22 = 26
    t = rng.uniform(0.05, 0.95, size=4)
    diag = np.column_stack([t, t])
    P = rng.uniform(0.05, 0.95, size=(11, 2))
    M = np.column_stack([P[:, 1], P[:, 0]])
    Q = np.vstack([diag, P, M])
    return Q[:, 0].copy(), Q[:, 1].copy()
