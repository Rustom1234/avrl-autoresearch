def _run_stages(z, caps, maxiter):
    cons = [{"type": "ineq", "fun": _cons_fun, "jac": _cons_jac}]
    for cap in caps:
        bnds = [(None, None)] * (2 * _N) + [(0.0, cap)] * _N
        try:
            res = minimize(_obj, z, jac=_obj_grad, constraints=cons,
                           bounds=bnds, method="SLSQP",
                           options={"maxiter": maxiter, "ftol": 1e-12})
            if np.all(np.isfinite(res.x)):
                z = res.x
        except Exception:
            pass
    return z
