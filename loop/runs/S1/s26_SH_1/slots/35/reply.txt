def _reg_pair(lam):
    """Objective -sum(r) - lam*sum exp(-gap/0.01) over pairs, with analytic gradient."""
    I, J = _I, _J

    def _unpack(z):
        x = z[:_N]
        y = z[_N:2 * _N]
        r = z[2 * _N:]
        dx = x[I] - x[J]
        dy = y[I] - y[J]
        d = np.sqrt(dx * dx + dy * dy) + 1e-12
        gap = d - (r[I] + r[J])
        return dx, dy, d, gap

    def f(z):
        _, _, _, gap = _unpack(z)
        e = np.exp(-gap / 0.01)
        return -np.sum(z[2 * _N:]) - lam * np.sum(e)

    def g(z):
        dx, dy, d, gap = _unpack(z)
        e = np.exp(-gap / 0.01)
        w = lam * e / 0.01
        ux = dx / d
        uy = dy / d
        gr = np.zeros(3 * _N)
        gr[2 * _N:] = -1.0
        gr[:_N] += np.bincount(I, w * ux, _N) - np.bincount(J, w * ux, _N)
        gr[_N:2 * _N] += np.bincount(I, w * uy, _N) - np.bincount(J, w * uy, _N)
        gr[2 * _N:] += -np.bincount(I, w, _N) - np.bincount(J, w, _N)
        return gr

    return f, g
