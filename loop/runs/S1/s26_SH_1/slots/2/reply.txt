def _swap_move(x, y, r, rng):
    """Swap centers of a large circle and a neighbouring smaller circle."""
    order = np.argsort(-r)
    big = order[rng.integers(0, _N // 3)]
    d = np.hypot(x - x[big], y - y[big])
    d[big] = np.inf
    nb = np.argsort(d)[:4]
    cands = [int(j) for j in nb if r[j] < r[big]]
    if not cands:
        return x.copy(), y.copy(), r.copy()
    j = cands[rng.integers(0, len(cands))]
    x2, y2, r2 = x.copy(), y.copy(), r.copy()
    x2[big], x2[j] = x[j], x[big]
    y2[big], y2[j] = y[j], y[big]
    r2[big], r2[j] = r[j], r[big]
    return x2, y2, r2
