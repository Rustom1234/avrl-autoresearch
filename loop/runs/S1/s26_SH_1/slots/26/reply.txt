def _valid(x, y, r):
    """Explicit double-precision check of borders and pair non-overlap."""
    I, J = _I, _J
    if np.any(r < 0):
        return False
    if np.any(x - r < 0) or np.any(x + r > 1) or np.any(y - r < 0) or np.any(y + r > 1):
        return False
    d = np.sqrt((x[I] - x[J]) ** 2 + (y[I] - y[J]) ** 2)
    return bool(np.all(r[I] + r[J] <= d))
