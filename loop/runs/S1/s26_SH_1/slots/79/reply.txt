def _set_n(n):
    global _N, _I, _J
    _N = n
    _I, _J = np.triu_indices(n, 1)


def _multistart(n, rng, tlimit):
    _set_n(n)
    best = None
    t0 = time.time()
    while time.time() - t0 < tlimit:
        P = rng.uniform(0.05, 0.95, size=(n, 2))
        x, y = _optimize(P[:, 0].copy(), P[:, 1].copy(), np.full(n, 0.05))
        r = _lp_radii(x, y)
        if r is None:
            continue
        s = float(np.sum(r))
        if best is None or s > best[0]:
            best = (s, x.copy(), y.copy(), r)
    _set_n(26)
    return best


def _add_in_hole(x, y):
    g = np.linspace(0.0, 1.0, 201)
    GX, GY = np.meshgrid(g, g)
    G = np.column_stack([GX.ravel(), GY.ravel()])
    C = np.column_stack([x, y])
    d = np.sqrt(((G[:, None, :] - C[None, :, :]) ** 2).sum(-1)).min(axis=1)
    b = np.minimum(np.minimum(G[:, 0], 1 - G[:, 0]), np.minimum(G[:, 1], 1 - G[:, 1]))
    score = np.minimum(d, 2 * b)
    k = int(np.argmax(score))
    return np.append(x, G[k, 0]), np.append(y, G[k, 1])
