def _lp_local_search(x, y, deadline, rng):
    """Random local search on centers; objective = LP-optimal sum of radii."""
    x = x.copy()
    y = y.copy()
    r = _lp_radii(x, y)
    cur = float(np.sum(r)) if r is not None else -1.0
    best_x, best_y, best_r, best_s = x.copy(), y.copy(), r, cur
    t_start = time.time()
    total = max(deadline - t_start, 1e-6)
    while time.time() < deadline:
        frac = min((time.time() - t_start) / total, 1.0)
        sigma = 0.05 * (0.002 / 0.05) ** frac
        k = int(rng.integers(1, 4))
        idx = rng.choice(_N, size=k, replace=False)
        nx = x.copy()
        ny = y.copy()
        nx[idx] = np.clip(nx[idx] + rng.normal(0.0, sigma, k), 0.0, 1.0)
        ny[idx] = np.clip(ny[idx] + rng.normal(0.0, sigma, k), 0.0, 1.0)
        rn = _lp_radii(nx, ny)
        if rn is None:
            continue
        sn = float(np.sum(rn))
        if sn > cur:
            x, y, r, cur = nx, ny, rn, sn
            if sn > best_s:
                best_x, best_y, best_r, best_s = x.copy(), y.copy(), r, sn
    return best_x, best_y, best_r
