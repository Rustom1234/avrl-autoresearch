    d = rng.uniform(0.12, 0.19)
    pts = None
    while d > 0.02 and pts is None:

        while len(cand) < _N and tries < 4000:
            tries += 1
            p = rng.uniform(0.05, 0.95, size=2)

            d *= 0.95
    if pts is None:
        pts = rng.uniform(0.05, 0.95, size=(_N, 2))
    D = np.sqrt(((pts[:, None, :] - pts[None, :, :]) ** 2).sum(-1))
