def _cons_fun(z):

    x = z[:n]

    dx = x[I] - x[J]

    d = np.sqrt(dx * dx + dy * dy + 1e-30)
    pair = d - r[I] - r[J]

    return np.concatenate([border, pair])


def _cons_jac(z):
    n = _N
    I, J = _I, _J
    m = len(I)
    x = z[:n]
    y = z[n:2 * n]
    Jm = np.zeros((4 * n + m, 3 * n))
    k = np.arange(n)
    c = 1.0 / _RS
    Jm[k, k] = 1.0
    Jm[k, 2 * n + k] = -c
    Jm[n + k, k] = -1.0
    Jm[n + k, 2 * n + k] = -c
    Jm[2 * n + k, n + k] = 1.0
    Jm[2 * n + k, 2 * n + k] = -c
    Jm[3 * n + k, n + k] = -1.0
    Jm[3 * n + k, 2 * n + k] = -c
    rows = 4 * n + np.arange(m)
    dx = x[I] - x[J]
    dy = y[I] - y[J]
    d = np.sqrt(dx * dx + dy * dy + 1e-30)
    ux = dx / d
    uy = dy / d
    Jm[rows, I] = ux
    Jm[rows, J] = -ux
    Jm[rows, n + I] = uy
    Jm[rows, n + J] = -uy
    Jm[rows, 2 * n + I] = -c
    Jm[rows, 2 * n + J] = -c
    return Jm
