    # 4 corner circles at r~0.10; 22 centres from a jittered 5x5 grid;
    # initial radii 0.12 near the centre falling to 0.06 near the edges
    corners = np.array([[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]])
    g = np.linspace(0.25, 0.75, 5)

    P = np.vstack([corners, G])
    P = np.clip(P, 0.02, 0.98)
    t = np.clip(np.max(np.abs(P - 0.5), axis=1) * 2.0, 0.0, 1.0)
