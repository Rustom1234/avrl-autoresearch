def _hex_seed(rng, rows):
    # perturbed hexagonal-like rows; odd rows shifted by half a spacing
    pts = []
    m = len(rows)
    for i, k in enumerate(rows):
        yy = 0.1 + 0.8 * i / (m - 1)
        shift = 0.5 * (0.8 / max(k - 1, 1)) * (i % 2)
        xs = np.linspace(0.1, 0.9, k) + shift
        for xx in xs:
            pts.append((xx, yy))
    P = np.array(pts)
    P = P + rng.normal(0.0, 0.02, size=P.shape)
    return np.clip(P, 0.05, 0.95)
