def construct_packing():
    seeds = [1, 7, 99, 2024]
    rngs = [np.random.default_rng(s) for s in seeds]
    rng = rngs[0]
    spent = [0.0] * len(seeds)
    best = None
    tprev = time.time()
    prev = 0
    for _k in range(100000):
        now = time.time()
        spent[prev] += now - tprev
        tprev = now

        prev = q
