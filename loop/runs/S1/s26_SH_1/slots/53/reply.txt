def _screen_one(z0):
    # single capped SLSQP stage (radii <= 0.10, maxiter 100) used for screening
    cons = [{"type": "ineq", "fun": _cons_fun, "jac": _cons_jac}]
    bnds = [(None, None)] * (2 * _N) + [(0.0, 0.10)] * _N
    try:
        res = minimize(_obj, z0, jac=_obj_grad, constraints=cons, bounds=bnds,
                       method="SLSQP", options={"maxiter": 100, "ftol": 1e-12})
        z = res.x if np.all(np.isfinite(res.x)) else z0
    except Exception:
        z = z0
    return z


def _finish_from(z):
    # remaining staged schedule: caps 0.12 then uncapped, maxiter 300
    cons = [{"type": "ineq", "fun": _cons_fun, "jac": _cons_jac}]
    for cap in (0.12, None):
        bnds = [(None, None)] * (2 * _N) + [(0.0, cap)] * _N
        try:
            res = minimize(_obj, z, jac=_obj_grad, constraints=cons, bounds=bnds,
                           method="SLSQP", options={"maxiter": 300, "ftol": 1e-12})
            if np.all(np.isfinite(res.x)):
                z = res.x
        except Exception:
            pass
    return z


def construct_packing():
    rng = np.random.default_rng(12345)
    best = None
    t0 = time.time()
    pool = []
    # screening phase: many cheap starts, keep LP-scored pool
    for _k in range(600):
