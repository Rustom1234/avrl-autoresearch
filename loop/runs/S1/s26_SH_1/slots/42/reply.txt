def _soft_relax(P):
    """Spread random centers with 50 L-BFGS-B steps on a Riesz-like energy plus wall repulsion."""
    n = _N
    eye = np.eye(n, dtype=bool)

    def fg(v):
        Q = v.reshape(n, 2)
        D = Q[:, None, :] - Q[None, :, :]
        d2 = (D ** 2).sum(-1) + 1e-4
        d2[eye] = np.inf
        inv = 1.0 / d2
        E = 0.5 * np.sum(inv)
        G = -2.0 * np.sum(D * (inv * inv)[:, :, None], axis=1)
        xc = np.clip(Q, 1e-3, 1.0 - 1e-3)
        E += np.sum(1.0 / xc ** 2 + 1.0 / (1.0 - xc) ** 2)
        G += -2.0 / xc ** 3 + 2.0 / (1.0 - xc) ** 3
        return E, G.ravel()

    res = minimize(fg, np.clip(P, 0.0, 1.0).ravel(), jac=True, method="L-BFGS-B",
                   bounds=[(0.0, 1.0)] * (2 * n), options={"maxiter": 50})
    if not np.all(np.isfinite(res.x)):
        return P
    return np.clip(res.x.reshape(n, 2), 0.0, 1.0)
