def _pm_obj(p):
    # power-mean objective: maximize sum(r^p), i.e. minimize -sum(r^p)
    def f(z):
        r = np.maximum(z[2 * _N:], 1e-6)
        return -np.sum(r ** p)

    def g(z):
        gr = np.zeros(3 * _N)
        r = np.maximum(z[2 * _N:], 1e-6)
        gr[2 * _N:] = -p * r ** (p - 1.0)
        return gr
    return f, g
