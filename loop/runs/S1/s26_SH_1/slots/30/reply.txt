def _template_start(rng):
    # corners and mid-edge circles in fixed structure; 14 interior centers from a
    # jittered 4x4 grid with 2 random points dropped; SLSQP is free to move all of them
    pts = [[0.13, 0.13], [0.87, 0.13], [0.13, 0.87], [0.87, 0.87]]
    rad = [0.13] * 4
    for t in (0.3, 0.7):
        pts += [[t, 0.08], [t, 0.92], [0.08, t], [0.92, t]]
        rad += [0.09] * 4
    g = np.linspace(0.25, 0.75, 4)
    grid = np.array([(gx, gy) for gx in g for gy in g])
    grid = grid + rng.uniform(-0.04, 0.04, size=grid.shape)
    n_int = _N - len(pts)
    keep = rng.permutation(len(grid))[:n_int]
    P = np.vstack([np.array(pts), grid[keep]])
    R = np.concatenate([np.array(rad), np.full(n_int, 0.04)])
    R = np.minimum(R, 0.08)
    P = np.clip(P, 0.05, 0.95)
    return P[:, 0].copy(), P[:, 1].copy(), R
