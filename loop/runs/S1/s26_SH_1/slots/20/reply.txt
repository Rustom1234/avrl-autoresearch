def _sa_refine(centers, rng, t_budget):
    # simulated annealing over center moves; objective = LP max sum of radii
    x = centers[:, 0].copy()
    y = centers[:, 1].copy()

    def score(xx, yy):
        rr = _lp_radii(xx, yy)
        return -1.0 if rr is None else float(np.sum(rr))

    cur = score(x, y)
    best_s = cur
    best_xy = (x.copy(), y.copy())
    T0, T1 = 0.01, 1e-5
    t_start = time.time()
    while True:
        el = time.time() - t_start
        if el > t_budget:
            break
        frac = el / t_budget
        T = T0 * (T1 / T0) ** frac
        step = 0.05 * (1.0 - frac) + 0.002
        k = int(rng.integers(_N))
        nx = x.copy()
        ny = y.copy()
        nx[k] = np.clip(x[k] + rng.normal(0.0, step), 0.0, 1.0)
        ny[k] = np.clip(y[k] + rng.normal(0.0, step), 0.0, 1.0)
        s = score(nx, ny)
        if s >= cur or rng.random() < np.exp((s - cur) / T):
            x, y, cur = nx, ny, s
            if cur > best_s:
                best_s = cur
                best_xy = (x.copy(), y.copy())
    return best_xy
