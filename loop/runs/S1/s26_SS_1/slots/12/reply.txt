def _rowinit(rng):
    while True:
        nr = int(rng.integers(4, 7))
        cnt = rng.integers(3, 8, nr)
        if cnt.sum() == _N:
            break
    h = rng.uniform(0.8, 1.2, nr)
    ys = (np.cumsum(h) - h / 2) / h.sum()
    pts = []
    for k in range(nr):
        m = int(cnt[k])
        sh = rng.uniform(-0.5, 0.5) * 0.5 / m
        xs = (np.arange(m) + 0.5) / m + sh * (1 if k % 2 else -1)
        for x in xs:
            pts.append([x, ys[k]])
    c = np.array(pts, float)
    if rng.random() < 0.5:
        c = c[:, ::-1].copy()
    c = c + rng.normal(0, rng.uniform(0.003, 0.02), c.shape)
    return np.clip(c, 0.03, 0.97)
