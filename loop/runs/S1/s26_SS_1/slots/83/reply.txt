def _shearlat(rng):
    s = rng.uniform(0.17, 0.21)
    th = rng.uniform(0, np.pi)
    sh = rng.uniform(-0.3, 0.3)
    a1 = np.array([s, 0.0])
    a2 = np.array([s / 2, s * np.sqrt(3) / 2])
    S = np.array([[1, sh], [0, 1]])
    R = np.array([[np.cos(th), -np.sin(th)], [np.sin(th), np.cos(th)]])
    M = R @ S
    off = rng.uniform(0, s, 2)
    pts = []
    for a in range(-12, 13):
        for b in range(-12, 13):

            if 0.03 <= p[0] <= 0.97 and 0.03 <= p[1] <= 0.97:
                pts.append(p)
    c = np.array(pts, float).reshape(-1, 2)
    if len(c) > _N:
        c = c[rng.permutation(len(c))[:_N]]

        c = np.vstack([c, rng.uniform(0.1, 0.9, (_N - len(c), 2))])
