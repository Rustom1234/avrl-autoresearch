def _sym_build(rng):
    n = _N
    k = int(rng.choice([0, 2, 4]))
    m = (n - k) // 2
    nq = 3 * m + 2 * k
    M = np.zeros((3 * n, nq))
    # q: px[m], py[m], t[k], pr[m], dr[k]
    for a in range(m):
        M[a, a] = 1
        M[n + a, m + a] = 1
        M[m + a, m + a] = 1
        M[n + m + a, a] = 1
        M[2 * n + a, 2 * m + k + a] = 1
        M[2 * n + m + a, 2 * m + k + a] = 1
    for b in range(k):
        M[2 * m + b, 2 * m + b] = 1
        M[n + 2 * m + b, 2 * m + b] = 1
        M[2 * n + 2 * m + b, 3 * m + k + b] = 1
    q0 = np.concatenate([rng.uniform(0.05, 0.95, 2 * m), rng.uniform(0.05, 0.95, k),
                         np.full(m + k, 0.05)])
    return M, q0
