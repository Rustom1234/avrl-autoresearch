def _set_n(n):
    global _N, _IU
    _N = n
    _IU = np.triu_indices(n, 1)


def _add_hole(c, r):
    g = (np.arange(100) + 0.5) / 100
    P = np.stack(np.meshgrid(g, g), -1).reshape(-1, 2)
    d = np.sqrt(((P[:, None, :] - c[None, :, :]) ** 2).sum(2)) - r[None, :]
    sc = np.minimum(d.min(1), np.minimum(np.minimum(P[:, 0], P[:, 1]), np.minimum(1 - P[:, 0], 1 - P[:, 1])))
    k = int(sc.argmax())
    return P[k], max(sc[k], 0.005) * 0.5


def _grow_start(rng):
    try:
        _set_n(24)

            p, rr = _add_hole(c, r)
