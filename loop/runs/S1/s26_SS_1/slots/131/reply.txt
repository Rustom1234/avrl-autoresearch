        for q in range(-25, 26):
            x = p + 0.5 * q + sh[0]
            y = q * 0.8660254 + sh[1]
            xr = ct * x - st * y + W / 2
            yr = st * x + ct * y + H / 2
            if 0 <= xr <= W and 0 <= yr <= H:
                pts.append([xr / W, yr / H])
    c = np.array(pts, float).reshape(-1, 2)
