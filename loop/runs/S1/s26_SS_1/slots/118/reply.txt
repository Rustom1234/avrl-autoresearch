def _soft(c, rng):

    t = rng.lognormal(0, 0.3, n)
    t *= 2.6 / t.sum()
    ti = t[i] + t[j]

        P = x.reshape(n, 2)
        dv = P[i] - P[j]
        d = np.sqrt((dv ** 2).sum(1)) + 1e-12
        o = np.maximum(ti - d, 0)
        val = (o ** 2).sum()
        g = np.zeros((n, 2))
        w = (-2 * o / d)[:, None] * dv
        np.add.at(g, i, w)
        np.add.at(g, j, -w)
        a = np.maximum(t[:, None] - P, 0)
        b = np.maximum(P + t[:, None] - 1, 0)
        val += (a ** 2).sum() + (b ** 2).sum()

        return val, g.ravel()
