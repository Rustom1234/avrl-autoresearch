def _twoclass(rng):
    n = _N
    q = rng.uniform(1.3, 2.2)
    k = int(rng.integers(8, 15))
    s = np.sqrt(0.78 / (np.pi * (k * q * q + (n - k))))
    r = np.concatenate([np.full(k, q * s), np.full(n - k, s)])
    c = np.zeros((n, 2))
    for a in range(n):
        cand = rng.uniform(r[a], 1 - r[a], (300, 2))
        slack = np.minimum(cand.min(1), (1 - cand).min(1)) - r[a]
        if a > 0:
            d = np.sqrt(((cand[:, None, :] - c[None, :a, :]) ** 2).sum(2))
            slack = np.minimum(slack, (d - r[a] - r[None, :a]).min(1))
        c[a] = cand[np.argmax(slack)]
    return np.clip(c, 0.02, 0.98), r
