def _pen_fg(x, mu):
    n = _N
    i, j = _IU
    X, Y, R = x[:n], x[n:2 * n], x[2 * n:]
    dx = X[i] - X[j]
    dy = Y[i] - Y[j]
    d = np.sqrt(dx * dx + dy * dy + 1e-18)
    o = np.maximum(0, R[i] + R[j] - d)
    f = -R.sum() + mu * (o * o).sum()
    gX = np.zeros(n)
    gY = np.zeros(n)
    gR = -np.ones(n)
    t = 2 * mu * o
    np.add.at(gR, i, t)
    np.add.at(gR, j, t)
    np.add.at(gX, i, -t * dx / d)
    np.add.at(gX, j, t * dx / d)
    np.add.at(gY, i, -t * dy / d)
    np.add.at(gY, j, t * dy / d)
    for P, G, sgn in ((X, gX, 1), (Y, gY, 1)):
        v1 = np.maximum(0, R - P)
        v2 = np.maximum(0, R + P - 1)
        f += mu * ((v1 * v1).sum() + (v2 * v2).sum())
        gR += 2 * mu * (v1 + v2)
        G += 2 * mu * (-v1 + v2)
    return f, np.concatenate([gX, gY, gR])
