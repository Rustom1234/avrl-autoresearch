def _symm(rng):
    k = int(rng.choice([0, 2, 4]))
    p = (_N - k) // 2
    a = rng.uniform(0.05, 0.95, (p, 2))
    if rng.random() < 0.5:
        b = a[:, ::-1]
    else:
        b = 1 - a[:, ::-1]
    t = rng.uniform(0.05, 0.95, k)
    d = np.stack([t, t], 1) if k else np.zeros((0, 2))
    c = np.vstack([a, b, d])
    if rng.random() < 0.5:
        c = c.copy()
        c[:, 0] = 1 - c[:, 0]
    c = c + rng.normal(0, 0.005, c.shape)
    return np.clip(c, 0.03, 0.97)
