    Z = np.zeros((n, n))
    wj = np.vstack([
        np.hstack([I, Z, -I]),
        np.hstack([-I, Z, -I]),
        np.hstack([Z, I, -I]),
        np.hstack([Z, -I, -I]),
    ])

    def cons(x):
        X, Y, R = x[:n], x[n:2 * n], x[2 * n:]
        dx = X[i] - X[j]
        dy = Y[i] - Y[j]
        pair = dx * dx + dy * dy - (R[i] + R[j]) ** 2
        return np.concatenate([pair, X - R, 1 - X - R, Y - R, 1 - Y - R])

    def consj(x):
        X, Y, R = x[:n], x[n:2 * n], x[2 * n:]

        J[k, 2 * n + i] = -2 * s
        J[k, 2 * n + j] = -2 * s
