def _gridinit(rng):
    k = int(rng.integers(5))
    pts = []
    for a in range(5):
        m = 6 if a == k else 5
        for b in range(m):
            pts.append([(b + 0.5) / m, (a + 0.5) / 5])
    c = np.array(pts, float)
    c = c + rng.normal(0, 0.02, c.shape)
    if rng.random() < 0.5:
        c = c[:, ::-1].copy()
    return np.clip(c, 0.03, 0.97)
