def _cornerinit(rng):
    pts = []
    k = rng.uniform(0.12, 0.16)
    for a in (k, 1 - k):
        for b in (k, 1 - k):
            pts.append([a, b])
    m = rng.uniform(0.35, 0.65, 4)
    e = rng.uniform(0.05, 0.1)
    pts += [[m[0], e], [m[1], 1 - e], [e, m[2]], [1 - e, m[3]]]
    for row in range(3):
        y = 0.25 + 0.25 * row
        off = 0.04 * (1 if row % 2 else -1)
        for j in range(6):
            pts.append([(j + 0.5) / 6 + off * 0.5, y])
    c = np.array(pts, float)
    if rng.random() < 0.5:
        c = c[:, ::-1].copy()
    c = c + rng.normal(0, rng.uniform(0.01, 0.03), c.shape)
    return np.clip(c, 0.03, 0.97)
