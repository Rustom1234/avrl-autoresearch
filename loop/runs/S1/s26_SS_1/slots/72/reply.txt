def _shear(rng):
    s = rng.uniform(0.17, 0.21)
    t = rng.uniform(0, np.pi / 3)
    sh = rng.uniform(-0.15, 0.15)
    a = s * np.array([np.cos(t), np.sin(t)])
    b = s * np.array([np.cos(t + np.pi / 3 + sh), np.sin(t + np.pi / 3 + sh)])
    o = rng.uniform(0, 1, 2)
    pts = []
    for p in range(-12, 13):
        for q in range(-12, 13):
            v = o + p * a + q * b
            if 0.04 <= v[0] <= 0.96 and 0.04 <= v[1] <= 0.96:
                pts.append(v)
    c = np.array(pts, float).reshape(-1, 2)

            dd = np.sqrt(((c - p) ** 2).sum(1)).min() if len(c) else 1
            if dd > bd:
