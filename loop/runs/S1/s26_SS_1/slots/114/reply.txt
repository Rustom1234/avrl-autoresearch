def _shear(rng):

    t = rng.uniform(0, np.pi)
    phi = np.deg2rad(rng.uniform(55, 90))
    a = s * np.array([np.cos(t), np.sin(t)])
    b = s * np.array([np.cos(t + phi), np.sin(t + phi)])
    o = rng.uniform(0, 1, 2)
    pts = []

    c = np.array(pts, float).reshape(-1, 2)
