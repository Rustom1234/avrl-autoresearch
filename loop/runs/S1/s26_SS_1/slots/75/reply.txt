def _polish_sub(c, r, k=6, maxiter=30):
    n = _N
    D = np.sqrt(((c[:, None, :] - c[None, :, :]) ** 2).sum(2))
    np.fill_diagonal(D, 9)
    nb = np.argsort(D, 1)[:, :k]
    s = set()
    for a in range(n):
        for b in nb[a]:
            s.add((min(a, int(b)), max(a, int(b))))
    pr = np.array(sorted(s))
    i = pr[:, 0]
    j = pr[:, 1]
    m = len(i)
    x0 = np.concatenate([c[:, 0], c[:, 1], r])
    I = np.eye(n)
    Z = np.zeros((n, n))
    wj = np.vstack([
        np.hstack([I, Z, -I]),
        np.hstack([-I, Z, -I]),
        np.hstack([Z, I, -I]),
        np.hstack([Z, -I, -I]),
    ])

    def objg(x):
        g = np.zeros(3 * n)
        g[2 * n:] = -1
        return g

    def cons(x):
        X, Y, R = x[:n], x[n:2 * n], x[2 * n:]
        dx = X[i] - X[j]
        dy = Y[i] - Y[j]
        pair = dx * dx + dy * dy - (R[i] + R[j]) ** 2
        return np.concatenate([pair, X - R, 1 - X - R, Y - R, 1 - Y - R])

    def consj(x):
        X, Y, R = x[:n], x[n:2 * n], x[2 * n:]
        dx = X[i] - X[j]
        dy = Y[i] - Y[j]
        sm = R[i] + R[j]
        J = np.zeros((m, 3 * n))
        q = np.arange(m)
        J[q, i] = 2 * dx
        J[q, j] = -2 * dx
        J[q, n + i] = 2 * dy
        J[q, n + j] = -2 * dy
        J[q, 2 * n + i] = -2 * sm
        J[q, 2 * n + j] = -2 * sm
        return np.vstack([J, wj])

    res = minimize(lambda x: -x[2 * n:].sum(), x0, jac=objg, method="SLSQP",
                   constraints=[{"type": "ineq", "fun": cons, "jac": consj}],
                   bounds=[(0, 1)] * (2 * n) + [(0, 0.5)] * n,
                   options={"maxiter": maxiter, "ftol": 1e-10})
    x = res.x
    return np.stack([x[:n], x[n:2 * n]], 1)
