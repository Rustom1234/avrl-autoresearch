def _slsqp_general(pts, rad):
    """SLSQP polish for an arbitrary number of circles."""
    n = pts.shape[0]
    iu, ju = np.triu_indices(n, 1)
    m = iu.size
    z0 = np.concatenate([pts[:, 0], pts[:, 1], rad])
    k = np.arange(n)

    def obj(z):
        return -np.sum(z[2 * n:])

    def obj_jac(z):
        g = np.zeros_like(z)
        g[2 * n:] = -1.0
        return g

    def cons(z):
        x = z[:n]
        y = z[n:2 * n]
        r = z[2 * n:]
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        rs = r[iu] + r[ju]
        pair = dx * dx + dy * dy - rs * rs
        bnd = np.concatenate([x - r, 1 - x - r, y - r, 1 - y - r])
        return np.concatenate([pair, bnd])

    def cons_jac(z):
        x = z[:n]
        y = z[n:2 * n]
        r = z[2 * n:]
        J = np.zeros((m + 4 * n, 3 * n))
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        rs = r[iu] + r[ju]
        rows = np.arange(m)
        J[rows, iu] = 2 * dx
        J[rows, ju] = -2 * dx
        J[rows, n + iu] = 2 * dy
        J[rows, n + ju] = -2 * dy
        J[rows, 2 * n + iu] = -2 * rs
        J[rows, 2 * n + ju] = -2 * rs
        b = m
        J[b + k, k] = 1.0
        J[b + k, 2 * n + k] = -1.0
        J[b + n + k, k] = -1.0
        J[b + n + k, 2 * n + k] = -1.0
        J[b + 2 * n + k, n + k] = 1.0
        J[b + 2 * n + k, 2 * n + k] = -1.0
        J[b + 3 * n + k, n + k] = -1.0
        J[b + 3 * n + k, 2 * n + k] = -1.0
        return J

    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    try:
        res = minimize(obj, z0, jac=obj_jac, method="SLSQP", bounds=bounds,
                       constraints=[{"type": "ineq", "fun": cons, "jac": cons_jac}],
                       options={"maxiter": 500, "ftol": 1e-12})
        z = res.x
    except Exception:
        z = z0
    centers = np.column_stack([np.clip(z[:n], 0.0, 1.0), np.clip(z[n:2 * n], 0.0, 1.0)])
    radii = _repair(centers, np.maximum(z[2 * n:], 0.0))
    return centers, radii


def _delete_grow(c, r, seed):
    """Remove a small circle (25 left, re-optimise), insert at biggest hole, re-optimise to 26."""
    n = c.shape[0]
    rng = np.random.default_rng(seed)
    order = np.argsort(r)
    drop = order[rng.integers(0, 3)]
    keep = np.array([i for i in range(n) if i != drop])
    c25, r25 = _slsqp_general(c[keep], np.maximum(r[keep], 1e-3))
    g = np.linspace(0.0, 1.0, 61)
    gx, gy = np.meshgrid(g, g)
    P = np.column_stack([gx.ravel(), gy.ravel()])
    d = np.sqrt(((P[:, None, :] - c25[None, :, :]) ** 2).sum(-1)) - r25[None, :]
    border = np.minimum.reduce([P[:, 0], 1 - P[:, 0], P[:, 1], 1 - P[:, 1]])
    clear = np.minimum(d.min(axis=1), border)
    hole = P[np.argmax(clear)]
    c26 = np.vstack([c25, hole[None, :]])
    r26 = np.concatenate([r25, [max(float(clear.max()), 1e-3)]])
    return _slsqp_general(c26, r26)


def construct_packing():
    """Multi-start SLSQP, then delete-and-grow refinement of the best packing."""
    t0 = time.time()
    best = None
    best_sum = -1.0
    for seed in range(10000):
        if seed > 0 and time.time() - t0 > 120.0:
            break
        c, r = _pack_once(seed)
        s = float(np.sum(r))
        if s > best_sum:
            best_sum = s
            best = (c, r)
    it = 0
    while time.time() - t0 < 190.0:
        try:
            c, r = _delete_grow(best[0], best[1], it)
        except Exception:
            c, r = best
        s = float(np.sum(r))
        if s > best_sum:
            best_sum = s
            best = (c, r)
        it += 1
    return best
