    def barrier(z, t):
        x = z[:n]
        y = z[n:2 * n]
        r = z[2 * n:]
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        d = np.sqrt(dx * dx + dy * dy + 1e-18)
        sp = d - r[iu] - r[ju]
        sb = np.concatenate([x - r, 1 - x - r, y - r, 1 - y - r])
        if np.any(sp <= 0) or np.any(sb <= 0):
            return 1e12, np.zeros_like(z)
        f = -np.sum(r) - t * (np.sum(np.log(sp)) + np.sum(np.log(sb)))
        g = np.zeros_like(z)
        g[2 * n:] = -1.0
        w = -t / sp
        ux = dx / d
        uy = dy / d
        np.add.at(g[:n], iu, w * ux)
        np.add.at(g[:n], ju, -w * ux)
        np.add.at(g[n:2 * n], iu, w * uy)
        np.add.at(g[n:2 * n], ju, -w * uy)
        np.add.at(g[2 * n:], iu, -w)
        np.add.at(g[2 * n:], ju, -w)
        wb = -t / sb
        wl = wb[:n]
        wr = wb[n:2 * n]
        wbt = wb[2 * n:3 * n]
        wtp = wb[3 * n:]
        g[:n] += wl - wr
        g[n:2 * n] += wbt - wtp
        g[2 * n:] -= (wl + wr + wbt + wtp)
        return f, g
