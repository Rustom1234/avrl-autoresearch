def _lp_radii(centers):
    """Optimal radii for fixed centers via LP (maximize sum r)."""
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    wall = np.maximum(np.minimum.reduce([x, 1 - x, y, 1 - y]), 0.0)
    iu, ju = np.triu_indices(n, 1)
    d = np.hypot(x[iu] - x[ju], y[iu] - y[ju])
    m = iu.size
    A = np.zeros((m, n))
    A[np.arange(m), iu] = 1.0
    A[np.arange(m), ju] = 1.0
    res = linprog(-np.ones(n), A_ub=A, b_ub=d,
                  bounds=[(0.0, float(w)) for w in wall], method="highs")
    if res.status != 0:
        return None
    return np.maximum(res.x, 0.0)


def _anneal(centers, radii, t_end, seed):
    """Simulated annealing on centers; energy = -sum of LP radii."""
    rng = np.random.default_rng(seed)
    n = centers.shape[0]
    cur = np.clip(centers, 0.0, 1.0)
    cur_r = _lp_radii(cur)
    if cur_r is None:
        return centers, radii
    E = -float(np.sum(cur_r))
    best_c, best_r, best_E = cur.copy(), cur_r, E
    t_start = time.time()
    span = max(t_end - t_start, 1e-6)
    while True:
        frac = (time.time() - t_start) / span
        if frac >= 1.0:
            break
        T = 3e-3 * (1e-5 / 3e-3) ** frac
        sig = 0.05 * (0.002 / 0.05) ** frac
        cand = cur.copy()
        k = int(rng.integers(n))
        cand[k] = np.clip(cand[k] + rng.normal(0.0, sig, size=2), 0.0, 1.0)
        rc = _lp_radii(cand)
        if rc is None:
            continue
        Ec = -float(np.sum(rc))
        if Ec < E or rng.random() < np.exp(-(Ec - E) / T):
            cur, E, cur_r = cand, Ec, rc
            if E < best_E:
                best_c, best_r, best_E = cur.copy(), rc, E
    return best_c, best_r


def _polish(centers, radii):
    """SLSQP over centers and radii jointly, started from the given packing."""
    n = 26
    iu, ju = np.triu_indices(n, 1)
    z0 = np.concatenate([centers[:, 0], centers[:, 1], radii])

    def obj(z):
        return -np.sum(z[2 * n:])

    def obj_jac(z):
        g = np.zeros_like(z)
        g[2 * n:] = -1.0
        return g

    def cons(z):
        x = z[:n]
        y = z[n:2 * n]
        r = z[2 * n:]
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        rs = r[iu] + r[ju]
        return np.concatenate([dx * dx + dy * dy - rs * rs,
                               x - r, 1 - x - r, y - r, 1 - y - r])

    try:
        res = minimize(obj, z0, jac=obj_jac, method="SLSQP",
                       bounds=[(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n,
                       constraints=[{"type": "ineq", "fun": cons}],
                       options={"maxiter": 200, "ftol": 1e-12})
        z = res.x
    except Exception:
        z = z0
    c = np.column_stack([np.clip(z[:n], 0.0, 1.0), np.clip(z[n:2 * n], 0.0, 1.0)])
    r = _repair(c, np.maximum(z[2 * n:], 0.0))
    return c, r


def construct_packing():
    """Multi-start (100 s) + LP-energy simulated annealing (to 200 s) + SLSQP polish."""
    t0 = time.time()
    best = None
    best_sum = -1.0
    for seed in range(10000):
        if seed > 0 and time.time() - t0 > 100.0:
            break
        c, r = _pack_once(seed)
        s = float(np.sum(r))
        if s > best_sum:
            best_sum = s
            best = (c, r)
    c, r = _anneal(best[0], best[1], t0 + 200.0, 0)
    r = _repair(c, r)
    try:
        c2, r2 = _polish(c, r)
        if np.sum(r2) > np.sum(r):
            c, r = c2, r2
    except Exception:
        pass
    return c, r
