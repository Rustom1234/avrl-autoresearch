def _polish(centers, radii):
    """Re-optimise centers and radii jointly with SLSQP from a given start."""
    n = 26
    iu, ju = np.triu_indices(n, 1)
    m = iu.size
    k = np.arange(n)
    z0 = np.concatenate([centers[:, 0], centers[:, 1], np.maximum(radii, 0.0)])

    def cons(z):
        x = z[:n]
        y = z[n:2 * n]
        r = z[2 * n:]
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        rs = r[iu] + r[ju]
        pair = dx * dx + dy * dy - rs * rs
        bnd = np.concatenate([x - r, 1 - x - r, y - r, 1 - y - r])
        return np.concatenate([pair, bnd])

    def cons_jac(z):
        x = z[:n]
        y = z[n:2 * n]
        r = z[2 * n:]
        J = np.zeros((m + 4 * n, 3 * n))
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        rs = r[iu] + r[ju]
        rows = np.arange(m)
        J[rows, iu] = 2 * dx
        J[rows, ju] = -2 * dx
        J[rows, n + iu] = 2 * dy
        J[rows, n + ju] = -2 * dy
        J[rows, 2 * n + iu] = -2 * rs
        J[rows, 2 * n + ju] = -2 * rs
        b = m
        J[b + k, k] = 1.0
        J[b + k, 2 * n + k] = -1.0
        J[b + n + k, k] = -1.0
        J[b + n + k, 2 * n + k] = -1.0
        J[b + 2 * n + k, n + k] = 1.0
        J[b + 2 * n + k, 2 * n + k] = -1.0
        J[b + 3 * n + k, n + k] = -1.0
        J[b + 3 * n + k, 2 * n + k] = -1.0
        return J

    def obj(z):
        return -np.sum(z[2 * n:])

    def obj_jac(z):
        g = np.zeros_like(z)
        g[2 * n:] = -1.0
        return g

    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    try:
        res = minimize(obj, z0, jac=obj_jac, method="SLSQP", bounds=bounds,
                       constraints=[{"type": "ineq", "fun": cons, "jac": cons_jac}],
                       options={"maxiter": 300, "ftol": 1e-12})
        z = res.x
    except Exception:
        z = z0
    c2 = np.column_stack([np.clip(z[:n], 0.0, 1.0), np.clip(z[n:2 * n], 0.0, 1.0)])
    r2 = _repair(c2, np.maximum(z[2 * n:], 0.0))
    return c2, r2


def _reinsert(centers, radii):
    """Remove the smallest circle and put it at the largest empty hole (grid search)."""
    n = centers.shape[0]
    k = int(np.argmin(radii))
    others = np.delete(np.arange(n), k)
    g = np.linspace(0.0, 1.0, 201)
    X, Y = np.meshgrid(g, g)
    P = np.column_stack([X.ravel(), Y.ravel()])
    clear = np.minimum.reduce([P[:, 0], 1 - P[:, 0], P[:, 1], 1 - P[:, 1]])
    d = np.sqrt(((P[:, None, :] - centers[None, others, :]) ** 2).sum(-1))
    d = d - radii[others][None, :]
    clear = np.minimum(clear, d.min(axis=1))
    j = int(np.argmax(clear))
    c = centers.copy()
    r = radii.copy()
    c[k] = P[j]
    r[k] = max(float(clear[j]), 0.0)
    return c, r


def construct_packing():
    """Short multi-start, then remove-and-reinsert of the smallest circle with SLSQP polish."""
    t0 = time.time()
    best = None
    best_sum = -1.0
    for seed in range(6):
        c, r = _pack_once(seed)
        s = float(np.sum(r))
        if s > best_sum:
            best_sum = s
            best = (c, r)
        if time.time() - t0 > 90.0:
            break
    cur_c, cur_r = best
    stall = 0
    while time.time() - t0 < 180.0 and stall < 40:
        c, r = _reinsert(cur_c, cur_r)
        c, r = _polish(c, r)
        s = float(np.sum(r))
        if s > best_sum + 1e-12:
            best_sum = s
            best = (c, r)
            cur_c, cur_r = c, r
            stall = 0
        else:
            stall += 1
    return best
