    def viol(z):
        x = z[:n]
        y = z[n:2 * n]
        r = z[2 * n:]
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        d = np.sqrt(dx * dx + dy * dy + 1e-18)
        ov = np.maximum(r[iu] + r[ju] - d, 0.0)
        wv = np.concatenate([np.maximum(r - x, 0.0), np.maximum(r + x - 1.0, 0.0),
                             np.maximum(r - y, 0.0), np.maximum(r + y - 1.0, 0.0)])
        return ov, wv

    def alm(z, lp, lw, mu):
        x = z[:n]
        y = z[n:2 * n]
        r = z[2 * n:]
        dx = x[iu] - x[ju]
        dy = y[iu] - y[ju]
        d = np.sqrt(dx * dx + dy * dy + 1e-18)
        ov = np.maximum(r[iu] + r[ju] - d, 0.0)
        wv = np.concatenate([np.maximum(r - x, 0.0), np.maximum(r + x - 1.0, 0.0),
                             np.maximum(r - y, 0.0), np.maximum(r + y - 1.0, 0.0)])
        f = -np.sum(r) + np.sum(lp * ov + mu * ov * ov) + np.sum(lw * wv + mu * wv * wv)
        gw = lw + 2.0 * mu * wv
        gp = lp + 2.0 * mu * ov
        g = np.zeros_like(z)
        g[2 * n:] = -1.0
        np.add.at(g[2 * n:], iu, gp)
        np.add.at(g[2 * n:], ju, gp)
        ux = dx / d
        uy = dy / d
        np.add.at(g[:n], iu, -gp * ux)
        np.add.at(g[:n], ju, gp * ux)
        np.add.at(g[n:2 * n], iu, -gp * uy)
        np.add.at(g[n:2 * n], ju, gp * uy)
        gl, gr, gb, gt = gw[:n], gw[n:2 * n], gw[2 * n:3 * n], gw[3 * n:]
        g[2 * n:] += gl + gr + gb + gt
        g[:n] += gr - gl
        g[n:2 * n] += gt - gb
        return f, g
