def _perim_pts(k, rng):
    """k centers evenly spaced on the square inset 0.09, random offset, corners forced."""
    a, b = 0.09, 0.91
    L = b - a
    P = 4.0 * L
    s = np.arange(k) * P / k + rng.uniform(0.0, P / k)
    out = []
    for t in s:
        u = t % P
        side = int(u // L)
        f = u - side * L
        if side == 0:
            out.append([a + f, a])
        elif side == 1:
            out.append([b, a + f])
        elif side == 2:
            out.append([b - f, b])
        else:
            out.append([a, b - f])
    out = np.array(out)
    for c in np.array([[a, a], [b, a], [b, b], [a, b]]):
        j = np.argmin(np.sum((out - c) ** 2, axis=1))
        out[j] = c
    return out
