def _relax(pts, rng, steps=300):
    """Soft-disk repulsion: overlaps of growing disks push centers apart; walls clip."""
    n = pts.shape[0]
    iu2, ju2 = np.triu_indices(n, 1)
    p = np.clip(pts + rng.normal(0.0, 0.01, size=pts.shape), 0.0, 1.0)
    for s in range(steps):
        R = 0.09 * (s + 1) / steps
        dx = p[iu2, 0] - p[ju2, 0]
        dy = p[iu2, 1] - p[ju2, 1]
        d = np.sqrt(dx * dx + dy * dy) + 1e-12
        ov = np.maximum(2.0 * R - d, 0.0)
        fx = ov * dx / d
        fy = ov * dy / d
        disp = np.zeros_like(p)
        np.add.at(disp[:, 0], iu2, 0.3 * fx)
        np.add.at(disp[:, 0], ju2, -0.3 * fx)
        np.add.at(disp[:, 1], iu2, 0.3 * fy)
        np.add.at(disp[:, 1], ju2, -0.3 * fy)
        p = np.clip(p + disp, R, 1.0 - R)
    return p

def _relax(pts, rng, steps=800):
    """Soft-disk repulsion: grow target radius to 0.11, then shrink to 0.085; step size decays."""
    n = pts.shape[0]
    iu2, ju2 = np.triu_indices(n, 1)
    p = np.clip(pts + rng.normal(0.0, 0.01, size=pts.shape), 0.0, 1.0)
    for s in range(steps):
        u = (s + 1) / steps
        if u <= 0.7:
            R = 0.11 * u / 0.7
        else:
            R = 0.11 - 0.025 * (u - 0.7) / 0.3
        rate = 0.3 - 0.2 * s / (steps - 1)
        dx = p[iu2, 0] - p[ju2, 0]
        dy = p[iu2, 1] - p[ju2, 1]
        d = np.sqrt(dx * dx + dy * dy) + 1e-12
        ov = np.maximum(2.0 * R - d, 0.0)
        fx = ov * dx / d
        fy = ov * dy / d
        disp = np.zeros_like(p)
        np.add.at(disp[:, 0], iu2, rate * fx)
        np.add.at(disp[:, 0], ju2, -rate * fx)
        np.add.at(disp[:, 1], iu2, rate * fy)
        np.add.at(disp[:, 1], ju2, -rate * fy)
        p = np.clip(p + disp, R, 1.0 - R)
    return p
