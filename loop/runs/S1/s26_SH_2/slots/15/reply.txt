def _rsa_centers(rng, rad, tries=200):
    """Random sequential placement: each disk goes where it overlaps earlier disks least."""
    placed = []
    for r in rad:
        best_c = None
        best_gap = -np.inf
        for _ in range(tries):
            c = rng.uniform(r, 1.0 - r, size=2)
            gap = np.inf
            for q, qr in placed:
                gap = min(gap, np.hypot(c[0] - q[0], c[1] - q[1]) - (r + qr))
            if gap > 0:
                best_c = c
                break
            if gap > best_gap:
                best_gap = gap
                best_c = c
        placed.append((best_c, r))
    return np.array([c for c, _ in placed])
