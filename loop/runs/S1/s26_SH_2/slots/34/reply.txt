def construct_packing():
    """Cheap screening of many starts, then full SLSQP refinement of the top 8 distinct ones."""
    t0 = time.time()
    scored = []
    seed = 0
    while time.time() - t0 < 100.0 and seed < 150:
        c, r = _pack_once(seed, cheap=True)
        scored.append((float(np.sum(r)), seed))
        seed += 1
    scored.sort(reverse=True)
    picked = []
    for s, sd in scored:
        if all(abs(s - ps) > 1e-6 for ps, _ in picked):
            picked.append((s, sd))
        if len(picked) == 8:
            break
    best = None
    best_sum = -1.0
    for _, sd in picked:
        if best is not None and time.time() - t0 > 200.0:
            break
        c, r = _pack_once(sd)
        s = float(np.sum(r))
        if s > best_sum:
            best_sum = s
            best = (c, r)
    if best is None:
        best = _pack_once(0)
    return best
