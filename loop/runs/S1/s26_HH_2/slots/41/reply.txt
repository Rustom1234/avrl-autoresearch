def _dual_anneal_stage(c0, deadline):
    """Dual annealing over the 52-d center vector, scored by the exact LP sum of radii."""
    from scipy.optimize import dual_annealing
    best = [-1.0, None]

    def obj(z):
        if _time.time() > deadline:
            return 0.0
        c = np.asarray(z, dtype=float).reshape(_N, 2)
        _rv, s = _score(c)
        if s > best[0]:
            best[0] = s
            best[1] = c.copy()
        return -s

    def cb(x, f, context):
        return _time.time() > deadline

    try:
        dual_annealing(obj, bounds=[(0.05, 0.95)] * (2 * _N), x0=np.asarray(c0, dtype=float).ravel(),
                       maxiter=1000, no_local_search=True, seed=4242, callback=cb)
    except Exception:
        pass
    return best[1]
