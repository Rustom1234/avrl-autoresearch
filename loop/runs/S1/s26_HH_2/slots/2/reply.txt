def _energy(c, r):
    """Penalised objective: half the sum of ordered squared pair overlaps plus squared wall violations."""
    n = r.shape[0]
    diff = c[:, None, :] - c[None, :, :]
    d = np.sqrt(np.sum(diff ** 2, axis=-1) + np.eye(n))
    gap = d - (r[:, None] + r[None, :])
    np.fill_diagonal(gap, np.inf)
    ov = np.maximum(-gap, 0.0)
    wall = (np.maximum(r - c[:, 0], 0.0) ** 2 + np.maximum(r - 1.0 + c[:, 0], 0.0) ** 2
            + np.maximum(r - c[:, 1], 0.0) ** 2 + np.maximum(r - 1.0 + c[:, 1], 0.0) ** 2)
    return float(0.5 * np.sum(ov ** 2) + np.sum(wall))
