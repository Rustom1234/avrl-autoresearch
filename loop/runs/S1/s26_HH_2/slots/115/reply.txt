def _slsqp_refine(c, r, maxiter=150):
    """Joint SLSQP over centers and radii with annealed overlap slack (2e-3 -> 5e-4 -> 0)."""
    n = c.shape[0]
    iu, ju = np.triu_indices(n, 1)

    def obj(z):
        return -np.sum(z[2 * n:])

    def obj_grad(z):
        g = np.zeros_like(z)
        g[2 * n:] = -1.0
        return g

    def make_cons(eps):
        def cons(z):
            cc = z[:2 * n].reshape(n, 2)
            rr = z[2 * n:]
            d = np.sqrt(np.sum((cc[iu] - cc[ju]) ** 2, axis=1) + 1e-18)
            pair = d - rr[iu] - rr[ju] + eps
            x, y = cc[:, 0], cc[:, 1]
            wall = np.concatenate([x - rr + eps, y - rr + eps,
                                   1.0 - x - rr + eps, 1.0 - y - rr + eps])
            return np.concatenate([pair, wall])
        return cons

    z = np.concatenate([c.ravel(), np.maximum(r, 0.0)])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    m_stage = max(maxiter // 3, 1)
    for eps in (2e-3, 5e-4, 0.0):
        try:
            res = _minimize(obj, z, jac=obj_grad, method="SLSQP", bounds=bounds,
                            constraints=[{"type": "ineq", "fun": make_cons(eps)}],
                            options={"maxiter": m_stage, "ftol": 1e-12})
            if np.all(np.isfinite(res.x)):
                z = res.x
        except Exception:
            pass
    cc = np.clip(z[:2 * n].reshape(n, 2), 0.0, 1.0)
    rr = np.maximum(z[2 * n:], 0.0)
    return cc, rr
