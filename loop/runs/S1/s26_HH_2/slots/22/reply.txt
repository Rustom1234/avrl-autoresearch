    base_c = centers.copy()
    base_r = np.asarray(radii, dtype=float).copy()

    rng = np.random.RandomState(7)
    K = 4
    temps = 0.0005 * (10.0 ** (np.arange(K) / (K - 1.0)))
    chains = [[base_c.copy(), base_r.copy(), base_s] for _ in range(K)]
    while _time.time() - T0 < 215:
