def _diag_search(seconds=20.0):
    """Diagonal-reflection starts: 12 free centres mirrored to (y, x) plus 2 diagonal points; symmetry released in relax/SLSQP."""
    rng = np.random.RandomState(4321)
    t0 = _time.time()
    best_c, best_r, best_s = None, None, -1.0
    while _time.time() - t0 < seconds:
        free = rng.uniform(0.05, 0.95, size=(12, 2))
        t = rng.uniform(0.05, 0.95, size=(2, 1))
        diag = np.hstack([t, t])
        c = np.vstack([free, free[:, ::-1], diag])
        c, _r = _relax(c, np.full(_N, 0.005), 400, 0.3, 0.05)
        c, _r = _slsqp_refine(c, _r)
        rv, s = _score(c)
        if rv is not None and s > best_s:
            best_s, best_c, best_r = s, c.copy(), rv
    return best_c, best_r, best_s
