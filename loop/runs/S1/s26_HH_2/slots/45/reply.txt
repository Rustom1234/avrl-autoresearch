def _grid_start(rg):
    """5x5 square grid plus one random defect center, with small jitter."""
    g = (np.arange(5) + 0.5) / 5.0
    gx, gy = np.meshgrid(g, g)
    pts = np.column_stack([gx.ravel(), gy.ravel()])
    extra = rg.uniform(0.2, 0.8, size=(1, 2))
    pts = np.vstack([pts, extra])
    pts = pts + rg.normal(0.0, 0.01, size=pts.shape)
    return np.clip(pts, 0.05, 0.95)
