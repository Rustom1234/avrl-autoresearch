def _sym_search():
    """180-degree symmetric search: 13 free centres p, mirrors at (1-x, 1-y). Keep top 3, then polish unsymmetrised."""
    rng = np.random.RandomState(31337)
    t0 = _time.time()
    m = _N // 2

    def full(p):
        return np.vstack([p, 1.0 - p])

    cands = []
    while _time.time() - t0 < 20:
        p = rng.uniform(0.1, 0.9, size=(m, 2))
        c = full(p)
        c, _r = _relax(c, np.full(_N, 0.01), 300, 0.3, 0.05)
        p = 0.5 * (c[:m] + (1.0 - c[m:]))
        c = full(p)
        rv, s = _score(c)
        if rv is not None:
            cands.append((s, c.copy()))
            cands.sort(key=lambda q: -q[0])
            del cands[3:]
    best_s, best_c, best_r = -1.0, None, None
    for s0, c0 in cands:
        c, r = _relax(c0.copy(), np.full(_N, 0.005), 600, 0.05, 0.01, 0.5, c0)
        c, r = _slsqp_refine(c, r)
        rv, s = _score(c)
        if rv is not None and s > best_s:
            best_s, best_c, best_r = s, c.copy(), rv
    if best_c is None:
        return None, None, -1.0
    return best_c, best_r, best_s
