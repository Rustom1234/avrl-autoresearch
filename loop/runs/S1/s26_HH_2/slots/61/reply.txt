def _seed25(rng, budget=25.0):
    """Two-stage seed: search 25 circles, insert the 26th at the best grid hole, then relax and refine."""
    t1 = _time.time()
    best_c, best_s = None, -1.0
    while _time.time() - t1 < budget:
        c = rng.uniform(0.1, 0.9, size=(_N - 1, 2))
        c, _r = _relax(c, np.full(_N - 1, 0.01), 400, 0.3, 0.05)
        rv, s = _score(c)
        if rv is not None and s > best_s:
            best_s, best_c = s, c.copy()
    if best_c is None:
        return None
    g = np.linspace(0.0, 1.0, 80)
    G = np.array(np.meshgrid(g, g)).reshape(2, -1).T
    dd = np.sqrt(((G[:, None, :] - best_c[None, :, :]) ** 2).sum(-1)).min(axis=1)
    wall = np.minimum(np.minimum(G[:, 0], G[:, 1]), np.minimum(1 - G[:, 0], 1 - G[:, 1]))
    hole = np.minimum(dd, 2.0 * wall)
    c26 = np.vstack([best_c, G[np.argmax(hole)][None, :]])
    c, r = _relax(c26, np.full(_N, 0.005), 400, 0.1, 0.01, 0.3, c26.copy())
    c, r = _slsqp_refine(c, r)
    rv, s = _score(c)
    if rv is None:
        return None
    return (s, c.copy())
