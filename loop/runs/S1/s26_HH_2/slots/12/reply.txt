def _ring_start():
    """Mixed-size concentric-ring start: one large centre circle, smaller circles on outer rings."""
    counts = [1, 6, 9, 10]
    ring_rad = [0.0, 0.2, 0.36, 0.46]
    init_r = [0.12, 0.10, 0.08, 0.06]
    pts = []
    rad = []
    for k, m in enumerate(counts):
        if k == 0:
            pts.append([0.5, 0.5])
            rad.append(init_r[0])
            continue
        for j in range(m):
            a = 2 * np.pi * j / m + 0.3 * k
            pts.append([0.5 + ring_rad[k] * np.cos(a), 0.5 + ring_rad[k] * np.sin(a)])
            rad.append(init_r[k])
    c = np.clip(np.array(pts), 0.02, 0.98)
    return c, np.array(rad, dtype=float)
