def _smooth_val(c):
    """Mean LP score over 6 fixed-seed Gaussian jitters (sd 1e-3) of the centres."""
    sr = np.random.RandomState(99)
    vals = []
    for _ in range(6):
        cj = np.clip(c + sr.normal(0.0, 1e-3, size=c.shape), 0.0, 1.0)
        rj, sj = _score(cj)
        if rj is not None:
            vals.append(sj)
    return float(np.mean(vals)) if vals else -1.0
