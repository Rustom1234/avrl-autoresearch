def _slp_refine(c, r, iters=8):
    """Sequential LP with trust region on joint centres and radii; accepts steps only if exact score improves."""
    n = c.shape[0]
    iu, ju = np.triu_indices(n, 1)
    m = iu.shape[0]
    c = np.clip(np.asarray(c, dtype=float), 0.0, 1.0).copy()
    rv, cur_s = _score(c)
    if rv is None:
        return c, np.maximum(np.asarray(r, dtype=float), 0.0)
    r = rv.copy()
    delta = 0.02
    idx = np.arange(n)
    rows_p = np.arange(m)
    for _ in range(iters):
        if delta < 1e-7:
            break
        diff = c[iu] - c[ju]
        d = np.sqrt(np.sum(diff ** 2, axis=1) + 1e-18)
        g = diff / d[:, None]
        A = np.zeros((m, 3 * n))
        A[rows_p, 2 * iu] = -g[:, 0]
        A[rows_p, 2 * iu + 1] = -g[:, 1]
        A[rows_p, 2 * ju] = g[:, 0]
        A[rows_p, 2 * ju + 1] = g[:, 1]
        A[rows_p, 2 * n + iu] = 1.0
        A[rows_p, 2 * n + ju] = 1.0
        b_p = d - r[iu] - r[ju]
        W = np.zeros((4 * n, 3 * n))
        x, y = c[:, 0], c[:, 1]
        W[idx, 2 * idx] = -1.0
        W[idx, 2 * n + idx] = 1.0
        W[n + idx, 2 * idx + 1] = -1.0
        W[n + idx, 2 * n + idx] = 1.0
        W[2 * n + idx, 2 * idx] = 1.0
        W[2 * n + idx, 2 * n + idx] = 1.0
        W[3 * n + idx, 2 * idx + 1] = 1.0
        W[3 * n + idx, 2 * n + idx] = 1.0
        b_w = np.concatenate([x - r, y - r, 1.0 - x - r, 1.0 - y - r])
        A_ub = np.vstack([A, W])
        b_ub = np.concatenate([b_p, b_w])
        cobj = np.concatenate([np.zeros(2 * n), -np.ones(n)])
        bounds = [(-delta, delta)] * (2 * n) + [(max(-ri, -delta), delta) for ri in r]
        res = linprog(cobj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
        if res.status != 0 or res.x is None:
            delta *= 0.5
            continue
        dz = res.x
        new_c = np.clip(c + dz[:2 * n].reshape(n, 2), 0.0, 1.0)
        rv2, s2 = _score(new_c)
        if rv2 is not None and s2 > cur_s + 1e-12:
            c = new_c
            r = rv2.copy()
            cur_s = s2
            delta = min(delta * 1.5, 0.2)
        else:
            delta *= 0.5
    return c, _make_valid(c, r)
