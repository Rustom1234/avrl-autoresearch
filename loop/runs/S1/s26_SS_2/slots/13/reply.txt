    def sub_solve(v, k, iters):
        act = np.concatenate([np.arange(k), n + np.arange(k), 2 * n + np.arange(k)])
        rows = np.concatenate([np.flatnonzero((I < k) & (J < k))] +
                              [m + q * n + np.arange(k) for q in range(4)])

        def full(z):
            w = v.copy(); w[act] = z
            return w
        res = minimize(lambda z: -z[2 * k:].sum(), v[act],
                       jac=lambda z: np.concatenate([np.zeros(2 * k), -np.ones(k)]),
                       method="SLSQP", bounds=[(0, 1)] * (2 * k) + [(0, 0.5)] * k,
                       constraints=[{"type": "ineq",
                                     "fun": lambda z: cons(full(z))[rows],
                                     "jac": lambda z: cjac(full(z))[rows][:, act]}],
                       options={"maxiter": iters, "ftol": 1e-12})
        return full(res.x)
