    def strict_ok(c, r):
        if np.any(r < 0):
            return False
        if np.any(c[:, 0] - r < 0) or np.any(c[:, 0] + r > 1) or np.any(c[:, 1] - r < 0) or np.any(c[:, 1] + r > 1):
            return False
        for i, j in zip(I, J):
            d1 = np.sqrt(np.sum((c[i] - c[j]) ** 2))
            d2 = np.hypot(*(c[i] - c[j]))
            if r[i] + r[j] > min(d1, d2):
                return False
        return True
