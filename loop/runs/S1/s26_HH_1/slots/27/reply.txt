def _trust_slp(c, iters=50):
    # Trust-region sequential LP over centres and radii. The linearised distance
    # constraint underestimates d (d is convex), so accepted steps stay feasible.
    n = 26
    I, J = np.triu_indices(n, 1)
    m = I.size
    x = np.clip(c[:, 0], 0.0, 1.0).copy()
    y = np.clip(c[:, 1], 0.0, 1.0).copy()
    r = _lp_radii(np.stack([x, y], axis=1))
    idx = np.arange(n)
    rows = np.arange(m)
    B = np.zeros((4 * n, 3 * n))
    B[idx, 2 * n + idx] = 1.0
    B[idx, idx] = -1.0
    B[n + idx, 2 * n + idx] = 1.0
    B[n + idx, idx] = 1.0
    B[2 * n + idx, 2 * n + idx] = 1.0
    B[2 * n + idx, n + idx] = -1.0
    B[3 * n + idx, 2 * n + idx] = 1.0
    B[3 * n + idx, n + idx] = 1.0
    cost = np.concatenate([np.zeros(2 * n), -np.ones(n)])

    def slack(xx, yy, rr):
        d = np.sqrt((xx[I] - xx[J]) ** 2 + (yy[I] - yy[J]) ** 2)
        ps = d - rr[I] - rr[J]
        bs = np.concatenate([xx - rr, 1 - xx - rr, yy - rr, 1 - yy - rr, rr])
        return min(float(ps.min()), float(bs.min()))

    delta = 0.02
    for _ in range(iters):
        if delta < 1e-7:
            break
        dx = x[I] - x[J]
        dy = y[I] - y[J]
        d = np.sqrt(dx * dx + dy * dy) + 1e-12
        ux = dx / d
        uy = dy / d
        A_pair = np.zeros((m, 3 * n))
        A_pair[rows, 2 * n + I] = 1.0
        A_pair[rows, 2 * n + J] = 1.0
        A_pair[rows, I] = -ux
        A_pair[rows, n + I] = -uy
        A_pair[rows, J] = ux
        A_pair[rows, n + J] = uy
        b_pair = np.maximum(d - r[I] - r[J], 0.0)
        b_bord = np.concatenate([x - r, 1 - x - r, y - r, 1 - y - r])
        A = np.vstack([A_pair, B])
        b = np.concatenate([b_pair, np.maximum(b_bord, 0.0)])
        lb = np.concatenate([np.full(2 * n, -delta), np.maximum(-delta, -r)])
        ub = np.full(3 * n, delta)
        try:
            res = linprog(cost, A_ub=A, b_ub=b, bounds=list(zip(lb, ub)), method="highs")
        except Exception:
            break
        if res.status != 0:
            delta *= 0.5
            continue
        step = res.x
        xn = x + step[:n]
        yn = y + step[n:2 * n]
        rn = np.maximum(r + step[2 * n:], 0.0)
        if slack(xn, yn, rn) >= -1e-9 and float(np.sum(rn)) > float(np.sum(r)) + 1e-12:
            x, y, r = xn, yn, rn
            delta = min(delta * 1.5, 0.1)
        else:
            delta *= 0.5
    return np.clip(np.stack([x, y], axis=1), 0.0, 1.0)
