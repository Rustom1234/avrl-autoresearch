def _maximin_start(rng, jitter=0.01):
    # Seeded farthest-point sampling: greedily pick 26 of 2000 random candidates,
    # each maximising its min distance to chosen points and the square's border.
    cand = rng.uniform(0.0, 1.0, (2000, 2))
    border = np.minimum(np.minimum(cand[:, 0], cand[:, 1]), np.minimum(1 - cand[:, 0], 1 - cand[:, 1]))
    pts = [int(np.argmax(border))]
    d_min = border.copy()
    for _ in range(25):
        dj = np.sqrt(((cand - cand[pts[-1]]) ** 2).sum(1))
        d_min = np.minimum(d_min, dj)
        d_min[pts] = -1.0
        pts.append(int(np.argmax(d_min)))
    c = cand[pts] + rng.uniform(-jitter, jitter, (26, 2))
    return np.clip(c, 0.0, 1.0)
