def _cluster_start(rng):
    # Gaussian-mixture clustered start: k cluster centres in [0.2,0.8]^2,
    # circles assigned by a seeded multinomial, per-cluster Gaussian scatter, clipped to the square.
    n = 26
    k = int(rng.integers(3, 6))
    centres = rng.uniform(0.2, 0.8, (k, 2))
    sig = rng.uniform(0.05, 0.15, k)
    counts = rng.multinomial(n, np.ones(k) / k)
    pts = []
    for j in range(k):
        pts.append(centres[j] + rng.normal(0.0, sig[j], (counts[j], 2)))
    return np.clip(np.vstack(pts), 0.0, 1.0)
