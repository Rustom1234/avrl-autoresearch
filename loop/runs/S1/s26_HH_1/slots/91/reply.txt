def _cosine_grid_start(rng):
    # 6x5 tensor grid with cosine spacing (points crowd toward the walls), drop 4 at random, jitter, clip.
    i6 = np.arange(6)
    j5 = np.arange(5)
    xs = 0.5 - 0.5 * np.cos(np.pi * (i6 + 0.5) / 6)
    ys = 0.5 - 0.5 * np.cos(np.pi * (j5 + 0.5) / 5)
    gx, gy = np.meshgrid(xs, ys)
    grid = np.stack([gx.ravel(), gy.ravel()], axis=1)
    keep = rng.choice(grid.shape[0], 26, replace=False)
    c = grid[keep] + rng.uniform(-0.02, 0.02, (26, 2))
    return np.clip(c, 0.0, 1.0)
