def _quantised_escape(c, r, best_s, rng, deadline):
    # Snap 6 random centres to a 0.02 grid, nudge them, re-optimise, keep only if the sum improves.
    n = 26
    if time.perf_counter() > deadline:
        return c, r, best_s
    cand = np.array(c, dtype=float)
    idx = rng.choice(n, 6, replace=False)
    snapped = np.round(cand[idx] / 0.02) * 0.02
    snapped = np.clip(snapped + rng.uniform(-0.01, 0.01, (6, 2)), 0.0, 1.0)
    cand[idx] = snapped
    cc, rr = _local_opt(cand)
    s = float(np.sum(rr))
    if s > best_s + 1e-9 and _valid_packing(cc, rr):
        return cc, rr, s
    return c, r, best_s
