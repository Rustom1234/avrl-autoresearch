def _sub_opt(c0, m, h):
    # Maximise the sum of radii for m circles in the square [0, h] x [0, h] (SLSQP on centres and radii),
    # then recompute exact radii with the LP for the fixed centres (valid by construction).
    I, J = np.triu_indices(m, 1)
    z0 = np.concatenate([c0[:, 0], c0[:, 1], np.full(m, 0.02)])

    def obj(z):
        return -np.sum(z[2 * m:])

    def obj_grad(z):
        g = np.zeros_like(z)
        g[2 * m:] = -1.0
        return g

    def cons(z):
        x = z[:m]
        y = z[m:2 * m]
        r = z[2 * m:]
        dx = x[I] - x[J]
        dy = y[I] - y[J]
        rs = r[I] + r[J]
        pair = dx * dx + dy * dy - rs * rs
        bnd = np.concatenate([x - r, h - x - r, y - r, h - y - r])
        return np.concatenate([pair, bnd])

    bounds = [(0.0, h)] * (2 * m) + [(0.0, h)] * m
    try:
        res = minimize(obj, z0, jac=obj_grad, method="SLSQP", bounds=bounds,
                       constraints=[{"type": "ineq", "fun": cons}],
                       options={"maxiter": 300, "ftol": 1e-12})
        z = res.x
        cl = np.clip(np.stack([z[:m], z[m:2 * m]], axis=1), 0.0, h)
    except Exception:
        cl = np.clip(c0, 0.0, h)
    rl = _lp_radii(cl / h) * h
    return cl, rl


def _quadrant_start(rng):
    # Split the square into four 0.5 quadrants; counts (7, 6, 7, 6) assigned in a seeded random order.
    # Each quadrant's circles are optimised as a sub-packing in its own box, then translated back.
    counts = rng.permutation([7, 6, 7, 6])
    origins = [(0.0, 0.0), (0.5, 0.0), (0.0, 0.5), (0.5, 0.5)]
    h = 0.5
    centers = np.zeros((26, 2))
    row = 0
    for q in range(4):
        m = int(counts[q])
        best_c, best_s = None, -1.0
        for _ in range(3):
            cl, rl = _sub_opt(rng.uniform(0.05, 0.45, (m, 2)), m, h)
            s = float(np.sum(rl))
            if s > best_s:
                best_c, best_s = cl, s
        # Sub-radii lie inside each quadrant box, so neighbouring quadrants cannot overlap;
        # _local_opt below recomputes the real radii for the combined centres.
        centers[row:row + m] = best_c + np.array(origins[q])
        row += m
    return np.clip(centers, 0.0, 1.0)
