def _shear_move(c, rng):
    # Anisotropic affine basin-escape: shear x by a*y, scale y by s, refit to [0.02, 0.98]^2.
    a = rng.uniform(-0.3, 0.3)
    s = rng.uniform(0.85, 1.15)
    x = c[:, 0] + a * c[:, 1]
    y = c[:, 1] * s
    w = max(x.max() - x.min(), y.max() - y.min(), 1e-9)
    x = 0.02 + 0.96 * (x - x.min()) / w
    y = 0.02 + 0.96 * (y - y.min()) / w
    return np.clip(np.stack([x, y], axis=1), 0.0, 1.0)
