def _near_pairs(c, thr=0.35):
    # Cell-list style pruning: keep only pairs whose centres are within thr of each other.
    n = c.shape[0]
    I, J = np.triu_indices(n, 1)
    d = np.sqrt(np.sum((c[I] - c[J]) ** 2, axis=1))
    m = d < thr
    return I[m], J[m]
