def _dual_weights(centers):
    # Per-circle total LP dual weight (pair and border constraints touching it).
    # Small weight means the circle is loosely constrained.
    n = centers.shape[0]
    x = np.clip(centers[:, 0], 0.0, 1.0)
    y = np.clip(centers[:, 1], 0.0, 1.0)
    border = np.maximum(np.minimum(np.minimum(x, y), np.minimum(1 - x, 1 - y)), 0.0)
    I, J = np.triu_indices(n, 1)
    d = np.sqrt((x[I] - x[J]) ** 2 + (y[I] - y[J]) ** 2)
    A = np.zeros((len(I), n))
    A[np.arange(len(I)), I] = 1.0
    A[np.arange(len(I)), J] = 1.0
    bounds = [(0.0, float(b)) for b in border]
    res = linprog(-np.ones(n), A_ub=A, b_ub=d, bounds=bounds, method="highs")
    w = np.zeros(n)
    if res.status != 0:
        return w
    pair_m = np.abs(np.asarray(res.ineqlin.marginals, dtype=float))
    w += np.bincount(I, weights=pair_m, minlength=n)
    w += np.bincount(J, weights=pair_m, minlength=n)
    w += np.abs(np.asarray(res.upper.marginals, dtype=float))
    return w
