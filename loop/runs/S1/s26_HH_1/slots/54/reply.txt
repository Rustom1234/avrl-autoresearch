    # Stage 1: six seeded short searches, 20 s each, ranked by LP sum.
    stage1 = []
    for k in range(6):
        remaining = total_budget - (time.perf_counter() - t_start)
        if remaining <= 5.0:
            break
        start = None
        if k > 0:
            amp = 0.01 + 0.09 * ((k - 1) % 4) / 3.0
            start = _lattice_start(np.random.default_rng(999 + k), amp)

        s = float(np.sum(r))
