def _spectral_start(rng):
    # Laplacian-eigenmap start: 4-NN graph on random points, eigenvectors 2 and 3 as (x, y).
    n = 26
    p = rng.uniform(0.0, 1.0, (n, 2))
    d = np.sqrt(((p[:, None, :] - p[None, :, :]) ** 2).sum(-1))
    np.fill_diagonal(d, np.inf)
    nn = np.argsort(d, axis=1)[:, :4]
    A = np.zeros((n, n))
    for i in range(n):
        A[i, nn[i]] = 1.0
    A = np.maximum(A, A.T)
    L = np.diag(A.sum(axis=1)) - A
    _, V = np.linalg.eigh(L)

    def rescale(v):
        span = v.max() - v.min()
        if span < 1e-12:
            return np.full(n, 0.5)
        return 0.08 + 0.84 * (v - v.min()) / span

    c = np.stack([rescale(V[:, 1]), rescale(V[:, 2])], axis=1)
    c = c + rng.uniform(-0.01, 0.01, (n, 2))
    return np.clip(c, 0.0, 1.0)
