def _two_tier_start(rng):
    # Tier 1: k large circles (radius 0.14 to 0.20) at random non-overlapping interior points.
    # Tier 2: remaining small circles greedily placed at the grid point with most clearance.
    n = 26
    k = int(rng.integers(3, 6))
    rad = rng.uniform(0.14, 0.20, k)
    centers = []
    radii_list = []
    for i in range(k):
        p = np.array([0.5, 0.5])
        for _ in range(500):
            q = rng.uniform(rad[i], 1.0 - rad[i], 2)
            if all(np.hypot(q[0] - c[0], q[1] - c[1]) >= rad[i] + rr for c, rr in zip(centers, radii_list)):
                p = q
                break
        centers.append(p)
        radii_list.append(rad[i])
    g = np.linspace(0.0, 1.0, 51)
    gx, gy = np.meshgrid(g, g)
    grid = np.stack([gx.ravel(), gy.ravel()], axis=1)
    border = np.minimum(np.minimum(grid[:, 0], grid[:, 1]), np.minimum(1 - grid[:, 0], 1 - grid[:, 1]))
    for _ in range(n - k):
        C = np.array(centers)
        R = np.array(radii_list)
        dist = np.sqrt(((grid[:, None, :] - C[None, :, :]) ** 2).sum(-1))
        clear = np.minimum(border, np.min(dist - R[None, :], axis=1))
        p = grid[int(np.argmax(clear))]
        centers.append(p)
        radii_list.append(0.03)
    return np.clip(np.array(centers, dtype=float), 0.0, 1.0)
