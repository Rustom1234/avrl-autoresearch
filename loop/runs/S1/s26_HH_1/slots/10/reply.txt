def _sym_start(rng):
    # 180-degree rotational symmetry about (0.5, 0.5): 13 free centres, mirrors at (1-x, 1-y),
    # shared radius per mirrored pair. Optimised with SLSQP; the caller polishes unconstrained.
    n = 26
    m = 13
    I, J = np.triu_indices(n, 1)
    p = rng.uniform(0.1, 0.9, (m, 2))
    z0 = np.concatenate([p[:, 0], p[:, 1], np.full(m, 0.05)])

    def unpack(z):
        x = z[:m]
        y = z[m:2 * m]
        r = z[2 * m:]
        X = np.concatenate([x, 1.0 - x])
        Y = np.concatenate([y, 1.0 - y])
        R = np.concatenate([r, r])
        return X, Y, R

    def obj(z):
        return -np.sum(z[2 * m:])

    def obj_grad(z):
        g = np.zeros_like(z)
        g[2 * m:] = -1.0
        return g

    def cons(z):
        X, Y, R = unpack(z)
        dx = X[I] - X[J]
        dy = Y[I] - Y[J]
        rs = R[I] + R[J]
        pair = dx * dx + dy * dy - rs * rs
        bnd = np.concatenate([X - R, 1 - X - R, Y - R, 1 - Y - R])
        return np.concatenate([pair, bnd])

    bounds = [(0.0, 1.0)] * (2 * m) + [(0.0, 0.5)] * m
    try:
        res = minimize(obj, z0, jac=obj_grad, method="SLSQP", bounds=bounds,
                       constraints=[{"type": "ineq", "fun": cons}],
                       options={"maxiter": 300, "ftol": 1e-12})
        X, Y, _ = unpack(res.x)
    except Exception:
        X, Y, _ = unpack(z0)
    return np.clip(np.stack([X, Y], axis=1), 0.0, 1.0)
