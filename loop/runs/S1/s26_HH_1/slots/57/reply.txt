def _warp_start(c, rng):
    # Smooth non-affine sinusoidal warp of the incumbent centres, re-clipped into the square.
    k = rng.uniform(-0.5, 0.5)
    amp = rng.uniform(0.02, 0.08)
    x = c[:, 0]
    y = c[:, 1]
    nx = x + k * np.sin(np.pi * y) * amp
    ny = y + k * np.sin(np.pi * x) * amp
    return np.clip(np.stack([nx, ny], axis=1), 0.0, 1.0)
