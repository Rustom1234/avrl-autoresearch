def _profile_start(rng):
    # Jittered 6x5 grid, keep the 26 points nearest the centre (largest room in the middle, small near edges).
    gx, gy = np.meshgrid((np.arange(6) + 0.5) / 6.0, (np.arange(5) + 0.5) / 5.0)
    g = np.stack([gx.ravel(), gy.ravel()], axis=1)
    dist = np.sqrt(np.sum((g - 0.5) ** 2, axis=1))
    g = g[np.argsort(dist)[:26]]
    c = g + rng.normal(0.0, 0.02, g.shape)
    return np.clip(c, 0.05, 0.95)
