def _pool_insert(pool, c, r, s, size=6):
    # Keep the best `size` distinct packings (duplicates: sum within 1e-6).
    if any(abs(s - p[2]) < 1e-6 for p in pool):
        return
    pool.append((c, r, s))
    pool.sort(key=lambda p: -p[2])
    del pool[size:]


def _crossover_child(pool, rng):
    # Cut two parents with a random line; take one side from A, the other from B.
    n = 26
    i, j = rng.choice(len(pool), 2, replace=False)
    A = pool[i][0]
    B = pool[j][0]
    th = rng.uniform(0.0, 2.0 * np.pi)
    nv = np.array([np.cos(th), np.sin(th)])
    p0 = rng.uniform(0.2, 0.8, 2)
    side_a = (A - p0) @ nv >= 0.0
    side_b = (B - p0) @ nv < 0.0
    child = np.vstack([A[side_a], B[side_b]])
    if len(child) > n:
        child = child[rng.choice(len(child), n, replace=False)]
    while len(child) < n:
        child = np.vstack([child, rng.uniform(0.05, 0.95, (1, 2))])
    return np.clip(child, 0.0, 1.0)


def jitter_anneal_search(seed=12345, budget=150.0, start=None):
    n = 26
    rng = np.random.default_rng(seed)
    t0 = time.time()
    c = _ls_growth(rng) if start is None else start
    best_c, best_r = _local_opt(c)
    best_s = float(np.sum(best_r))
    pool = [(best_c, best_r, best_s)]
    K = 20
    for k in range(K):
        if time.time() - t0 > budget:
            break
        frac = k / (K - 1)
        sigma = 0.1 * (0.002 / 0.1) ** frac
        if k % 2 == 1 and len(pool) >= 2:
            cand = _crossover_child(pool, rng)
        else:
            cand = np.clip(best_c + rng.normal(0.0, sigma, (n, 2)), 0.0, 1.0)
        cc, rr = _local_opt(cand)
        s = float(np.sum(rr))
        if _valid_packing(cc, rr):
            _pool_insert(pool, cc, rr, s)
        if s > best_s:
            best_c, best_r, best_s = cc, rr, s
    return best_c, best_r
