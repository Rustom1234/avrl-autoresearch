def _reinsert(c, r, rng):
    """Remove the smallest circle and put it at the largest sampled hole; returns a SLSQP start vector."""
    k = int(np.argmin(r))
    keep = np.arange(len(r)) != k
    p = rng.random((20000, 2))
    d = np.sqrt(((p[:, None] - c[keep][None]) ** 2).sum(-1)) - r[keep]
    score = np.minimum(d.min(1), np.minimum(p, 1 - p).min(1))
    c2, r2 = c.copy(), r.copy()
    c2[k] = p[np.argmax(score)]
    r2[k] = 0.01
    return np.concatenate([c2.ravel(), r2])
