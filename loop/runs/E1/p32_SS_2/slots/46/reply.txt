def _child(n, rng, pool):
    a, b = rng.choice(len(pool), 2, replace=False)
    (ca, ra), (cb, rb) = pool[a][1:], pool[b][1:]
    t = rng.uniform(0.35, 0.65)
    ma, mb = ca[:, 0] < t, cb[:, 0] >= t
    c = np.concatenate([ca[ma], cb[mb]])
    r = np.concatenate([ra[ma], rb[mb]])
    if len(r) > n:
        keep = np.sort(np.argsort(-r)[:n])
        c, r = c[keep], r[keep]
    elif len(r) < n:
        k = n - len(r)
        c = np.concatenate([c, rng.uniform(0.05, 0.95, (k, 2))])
        r = np.concatenate([r, np.full(k, 0.01)])
    return np.concatenate([c.ravel(), np.maximum(r, 0.005)])


def solve(n, seed, time_limit):
    rng, t0, pool = np.random.default_rng(seed), time.process_time(), []
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        if time.process_time() - t0 < 0.6 * time_limit or len(pool) < 2:
            c, r = _solve_once(n, rng)
        else:
            c, r = _solve_once(n, rng, _child(n, rng, pool))
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        s = r.sum()
        if all(abs(s - p[0]) > 1e-6 for p in pool):
            pool.append((s, c, r))
            pool.sort(key=lambda p: -p[0])

    return pool[0][1], pool[0][2]
