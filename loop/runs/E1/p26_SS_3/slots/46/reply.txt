def _hole_start(best, rng):
    c0, r0 = best[0].copy(), best[1].copy()
    k = rng.choice(np.argsort(r0)[:3])
    mask = np.arange(len(r0)) != k
    p = rng.uniform(0, 1, (2000, 2))
    d = np.sqrt(((p[:, None] - c0[mask][None]) ** 2).sum(-1)) - r0[mask][None]
    wall = np.minimum(np.minimum(p[:, 0], p[:, 1]), np.minimum(1 - p[:, 0], 1 - p[:, 1]))
    f = np.minimum(d.min(1), wall)
    j = np.argmax(f)
    c0[k] = p[j]
    r0[k] = 0.5 * max(f[j], 1e-3)
    return np.concatenate([c0.ravel(), r0])
