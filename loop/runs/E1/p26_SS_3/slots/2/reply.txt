_CACHE = {}


def _scaffold(n):
    if n not in _CACHE:
        iu = np.triu_indices(n, 1)
        P = len(iu[0])
        base = np.zeros((P + 5 * n, 3 * n))
        rows = np.arange(P)
        base[rows, 2 * n + iu[0]] = -1
        base[rows, 2 * n + iu[1]] = -1
        k = np.arange(n)
        base[P + k, 2 * k] = 1
        base[P + n + k, 2 * k + 1] = 1
        base[P + 2 * n + k, 2 * k] = -1
        base[P + 3 * n + k, 2 * k + 1] = -1
        base[P + k, 2 * n + k] = -1
        base[P + n + k, 2 * n + k] = -1
        base[P + 2 * n + k, 2 * n + k] = -1
        base[P + 3 * n + k, 2 * n + k] = -1
        base[P + 4 * n + k, 2 * n + k] = 1
        _CACHE[n] = (iu, P, rows, base)
    return _CACHE[n]


def _solve_once(n, rng):
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    iu, P, rows, base = _scaffold(n)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    def cjac(v):
        c = v[:2 * n].reshape(n, 2)
        diff = c[iu[0]] - c[iu[1]]
        u = diff / np.maximum(np.sqrt((diff ** 2).sum(1)), 1e-12)[:, None]
        J = base.copy()
        J[rows, 2 * iu[0]] = u[:, 0]
        J[rows, 2 * iu[0] + 1] = u[:, 1]
        J[rows, 2 * iu[1]] = -u[:, 0]
        J[rows, 2 * iu[1] + 1] = -u[:, 1]
        return J

    g = np.zeros(3 * n)
    g[2 * n:] = -1
    res = minimize(lambda v: -v[2 * n:].sum(), x0, jac=lambda v: g, method="SLSQP",
                   constraints={"type": "ineq", "fun": cons, "jac": cjac},
                   options={"maxiter": MAX_ITER})
