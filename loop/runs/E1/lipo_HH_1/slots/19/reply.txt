_SHAPE_CACHE = {}


def shape_vector(smiles):
    # One ETKDGv3 conformer per SMILES (cached); zeros if embedding fails.
    if smiles not in _SHAPE_CACHE:
        mol = Chem.AddHs(Chem.MolFromSmiles(smiles))
        params = rdDistGeom.ETKDGv3()
        params.randomSeed = 42
        params.maxIterations = 50
        vec = np.zeros(6)
        if rdDistGeom.EmbedMolecule(mol, params) >= 0:
            vec = np.array([rdMolDescriptors.CalcPBF(mol), rdMolDescriptors.CalcNPR1(mol),
                            rdMolDescriptors.CalcNPR2(mol), rdMolDescriptors.CalcRadiusOfGyration(mol),
                            rdMolDescriptors.CalcInertialShapeFactor(mol), rdMolDescriptors.CalcEccentricity(mol)],
                           dtype=np.float64)
        _SHAPE_CACHE[smiles] = vec
    return _SHAPE_CACHE[smiles]


def shape_matrix(smiles):
    return np.nan_to_num(np.array([shape_vector(s) for s in smiles]), nan=0.0, posinf=0.0, neginf=0.0)
