def brics_counts(smiles):
    # Hashed BRICS fragment counts; zlib.crc32 is deterministic across processes (unlike hash()).
    out = np.zeros((len(smiles), N_FRAG), dtype=np.float64)
    for i, s in enumerate(smiles):
        for frag in BRICS.BRICSDecompose(Chem.MolFromSmiles(s)):
            out[i, zlib.crc32(frag.encode()) % N_FRAG] += 1
    return out

def _frag_hash(frag):
    # Deterministic string hash (unlike hash(), it does not depend on the process's hash seed).
    h = 0
    for ch in frag:
        h = (h * 31 + ord(ch)) & 0xFFFFFFFF
