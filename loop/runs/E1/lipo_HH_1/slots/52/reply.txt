def tanimoto(A, B):
    dot = A @ B.T
    na = (A * A).sum(axis=1)[:, None]
    nb = (B * B).sum(axis=1)[None, :]
    return dot / np.maximum(na + nb - dot, 1e-9)

def minmax(A, B):
    # Generalised Tanimoto on clipped counts: sum_i min(a_i, b_i) = sum_k [a>=k][b>=k], so each level is one matmul.
    smin = np.zeros((A.shape[0], B.shape[0]))
    for k in range(1, MM_LEVELS + 1):
        smin += (A >= k).astype(np.float64) @ (B >= k).astype(np.float64).T
    smax = A.sum(axis=1)[:, None] + B.sum(axis=1)[None, :] - smin
    return smin / np.maximum(smax, 1e-9)
