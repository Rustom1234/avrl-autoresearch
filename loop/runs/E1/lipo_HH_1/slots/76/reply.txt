def l1_dist(A, B):
    # sum|a-b| = sum a + sum b - 2 * sum_k [a>=k][b>=k]  (exact for non-negative integer counts)
    D = A.sum(axis=1)[:, None] + B.sum(axis=1)[None, :]
    top = int(max(A.max(), B.max()))
    for k in range(1, top + 1):
        D -= 2.0 * ((A >= k).astype(np.float64) @ (B >= k).astype(np.float64).T)
    return D
