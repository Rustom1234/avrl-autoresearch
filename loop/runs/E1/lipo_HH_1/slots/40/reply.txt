def wl_counts(smiles):
    # Hashed Weisfeiler-Lehman subtree counts over the heavy-atom graph (iterations 0..WL_ITERS).
    rows = np.zeros((len(smiles), WL_BUCKETS), dtype=np.float32)
    for i, s in enumerate(smiles):
        mol = Chem.MolFromSmiles(s)
        labels = [f"{a.GetSymbol()}|{a.GetDegree()}|{a.GetFormalCharge()}|{int(a.GetIsAromatic())}|{a.GetTotalNumHs()}"
                  for a in mol.GetAtoms()]
        for it in range(WL_ITERS + 1):
            for lab in labels:
                rows[i, zlib.crc32(f"{it}|{lab}".encode()) % WL_BUCKETS] += 1
            new = []
            for a, lab in zip(mol.GetAtoms(), labels):
                nb = sorted(labels[n.GetIdx()] for n in a.GetNeighbors())
                new.append(str(zlib.crc32((lab + "#" + ",".join(nb)).encode())))
            labels = new
    return rows
