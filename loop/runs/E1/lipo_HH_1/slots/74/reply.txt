def leaf_kernel(La, Lb):
    # Random-forest proximity: fraction of trees in which two rows share a leaf.
    rows_a, cols_a, rows_b, cols_b = [], [], [], []
    off = 0
    for t in range(La.shape[1]):
        uniq, inv = np.unique(La[:, t], return_inverse=True)
        pos = np.minimum(np.searchsorted(uniq, Lb[:, t]), len(uniq) - 1)
        hit = uniq[pos] == Lb[:, t]
        rows_a.append(np.arange(len(La)))
        cols_a.append(off + inv.ravel())
        rows_b.append(np.nonzero(hit)[0])
        cols_b.append(off + pos[hit])
        off += len(uniq)
    T = La.shape[1]
    Ma = sparse.csr_matrix((np.ones(sum(len(r) for r in rows_a)), (np.concatenate(rows_a), np.concatenate(cols_a))), shape=(len(La), off))
    Mb = sparse.csr_matrix((np.ones(sum(len(r) for r in rows_b)), (np.concatenate(rows_b), np.concatenate(cols_b))), shape=(len(Lb), off))
    return (Ma @ Ma.T).toarray() / T, (Mb @ Ma.T).toarray() / T
