def _solve_once(n, rng):
    c = np.empty((n, 2))
    for i in range(n):
        cand = rng.uniform(0, 1, (2000, 2))
        score = np.minimum(np.minimum(cand, 1 - cand).min(1), 0.5)
        if i > 0:
            d = np.sqrt(((cand[:, None] - c[None, :i]) ** 2).sum(-1)).min(1)
            score = np.minimum(score, d)
        c[i] = cand[np.argmax(score)]
    x0 = np.concatenate([c.ravel(), np.full(n, INIT_RADIUS)])
    return _optimise(n, x0)
