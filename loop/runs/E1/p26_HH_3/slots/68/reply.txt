def _rattlers(c, r):
    """Circles with fewer than 2 active contacts (pair gap or wall tightness under 1e-6)."""
    d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
    touch = d - r[:, None] - r[None] < 1e-6
    np.fill_diagonal(touch, False)
    walls = (c - r[:, None] < 1e-6) | (1 - c - r[:, None] < 1e-6)
    return np.flatnonzero(touch.sum(1) + walls.sum(1) < 2)


def _relocate(c, r, rng):
    """Move each rattler to the sampled point with the largest clearance from all other circles and walls."""
    for i in _rattlers(c, r):
        keep = np.arange(len(r)) != i
        pts = rng.uniform(0, 1, (400, 2))
        gap = np.sqrt(((pts[:, None] - c[keep][None]) ** 2).sum(-1)) - r[keep][None]
        free = np.minimum(gap.min(1), np.minimum(pts, 1 - pts).min(1))
        c[i], r[i] = pts[np.argmax(free)], 0
    return c, r
