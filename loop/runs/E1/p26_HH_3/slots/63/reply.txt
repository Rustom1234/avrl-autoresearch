def _radii(pop):
    """Greedy radii for a batch of centre sets (P, n, 2): valid by construction, no SLSQP needed."""
    d = np.sqrt(((pop[:, :, None] - pop[:, None]) ** 2).sum(-1))
    i = np.arange(pop.shape[1])
    d[:, i, i] = np.inf
    return np.minimum(np.minimum(pop, 1 - pop).min(2), d.min(2) / 2)


def _de(n, rng, seed_c, seconds):
    """DE/rand/1/bin over centres only, fitness = greedy radius sum. Returns the top 3 centre sets."""
    P = 15 * n
    pop = rng.uniform(0, 1, (P, n, 2))
    if seed_c is not None:
        pop[0] = seed_c
    fit = _radii(pop).sum(1)
    t0 = time.process_time()
    while time.process_time() - t0 < seconds:
        o = np.argsort(rng.random((P, P)) + 2 * np.eye(P), axis=1)[:, :3]
        mut = np.clip(pop[o[:, 0]] + 0.6 * (pop[o[:, 1]] - pop[o[:, 2]]), 0, 1)
        cross = rng.random((P, 2 * n)) < 0.9
        cross[np.arange(P), rng.integers(0, 2 * n, P)] = True
        trial = np.where(cross.reshape(P, n, 2), mut, pop)
        tf = _radii(trial).sum(1)
        better = tf >= fit
        pop[better], fit[better] = trial[better], tf[better]
    return pop[np.argsort(-fit)[:3]]
