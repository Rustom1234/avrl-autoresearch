def _crossover(n, a, b, rng):
    """Child of two elite packings: circles left of a random vertical cut from a, the rest from b, count fixed to n."""
    cut = rng.uniform(0.3, 0.7)
    (ca, ra), (cb, rb) = a, b
    la, lb = ca[:, 0] < cut, cb[:, 0] >= cut
    c = np.vstack([ca[la], cb[lb]])
    r = np.concatenate([ra[la], rb[lb]])
    while len(r) < n:
        pts = rng.uniform(0, 1, (2000, 2))
        dc = np.sqrt(((pts[:, None] - c[None]) ** 2).sum(-1)).min(1) if len(c) else np.ones(2000)
        dw = np.minimum(pts, 1 - pts).min(1)
        c = np.vstack([c, pts[np.argmax(np.minimum(dc, dw))]])
        r = np.append(r, 0.0)
    keep = np.argsort(-np.abs(c[:, 0] - cut))[:n]
    return _optimise(n, np.concatenate([c[keep].ravel(), r[keep]]))
