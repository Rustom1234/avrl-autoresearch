def _lattice_start(n, rng):
    """Rotated square-lattice start: spacing matched to about n points in [0.1, 0.9]^2, rotated about the centre."""
    theta = rng.uniform(0, np.pi / 4)
    s = np.sqrt(0.64 / n)
    m = int(0.5 / s) + 2
    gx = np.arange(-m, m + 1) * s + rng.uniform(0, s)
    gy = np.arange(-m, m + 1) * s + rng.uniform(0, s)
    p = np.stack(np.meshgrid(gx, gy), -1).reshape(-1, 2)
    p = p[np.abs(p).max(1) <= 0.4]
    R = np.array([[np.cos(theta), -np.sin(theta)], [np.sin(theta), np.cos(theta)]])
    p = p @ R.T
    p = p[np.argsort((p ** 2).sum(1))][:n]
    if len(p) < n:
        p = np.vstack([p, rng.uniform(-0.4, 0.4, (n - len(p), 2))])
    return np.clip(p + 0.5, 0, 1)


def _solve_once(n, rng):
    if rng.random() < 0.5:
        c = _lattice_start(n, rng)
        return _optimise(n, np.concatenate([c.ravel(), np.full(n, INIT_RADIUS)]))
