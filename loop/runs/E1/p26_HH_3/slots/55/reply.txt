def _solve_once(n, rng):

    corners = np.array([[r0, r0], [1 - r0, r0], [r0, 1 - r0], [1 - r0, 1 - r0]])
    k = min(4, n)
    c = np.vstack([corners[:k], rng.uniform(INIT_LOW, 1 - INIT_LOW, (n - k, 2))])
    c, r = _optimise(n, np.concatenate([c.ravel(), np.full(n, INIT_RADIUS)]), pins=k)
