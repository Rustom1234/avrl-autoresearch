def _lattice(n, rng):
    """Row-partition seed: R rows with 3 to 7 circles each, evenly spaced over [0.1, 0.9], plus Gaussian jitter."""
    R = rng.integers(4, 8)
    sizes = rng.integers(3, 8, R)
    while sizes.sum() != n:
        sizes = rng.integers(3, 8, R)
    ys = np.linspace(0.1, 0.9, R)
    pts = [np.column_stack([np.linspace(0.1, 0.9, m), np.full(m, y)]) for m, y in zip(sizes, ys)]
    return np.vstack(pts) + rng.normal(0, 0.02, (n, 2))
