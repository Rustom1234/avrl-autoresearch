def _hex_start(n, rng):
    """Random subset of a hexagonal lattice scaled into the square, with Gaussian jitter."""
    k = int(np.ceil(np.sqrt(n * 2 / np.sqrt(3)))) + 2
    g = np.array([(i + 0.5 * (j % 2), j * np.sqrt(3) / 2) for j in range(k) for i in range(k)])
    g = (g - g.min(0)) / (g.max(0) - g.min(0))
    g = INIT_LOW + (1 - 2 * INIT_LOW) * g
    c = g[rng.choice(len(g), n, replace=False)] + rng.normal(0, 0.02, (n, 2))
    return c.ravel()
