def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    budget = TIME_FRACTION * time_limit
    elites = []
    while time.process_time() - t0 < RESTART_FRACTION * budget or best is None:   # short random-restart phase
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        elites.append((r.sum(), c, r))
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    elites.sort(key=lambda e: -e[0])
    e1, e2 = elites[0][1:], elites[min(1, len(elites) - 1)][1:]
    c, r = best
    while time.process_time() - t0 < budget:
        if rng.uniform() < CROSS_PROB:   # crossover of the two elite packings, second one aligned to the first
            cut = rng.uniform(0.3, 0.7)
            _, cols = linear_sum_assignment(np.linalg.norm(e1[0][:, None] - e2[0][None], axis=-1))
            left = e1[0][:, 0] < cut
            c2 = np.where(left[:, None], e1[0], e2[0][cols])
            r2 = _repair(c2, np.where(left, e1[1], e2[1][cols]))
        else:   # remove-and-reinsert the smallest circle
