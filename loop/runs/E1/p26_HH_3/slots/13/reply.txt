def _optimise_equal(n, c0):
    """Max-min stage: maximise a common radius t with all circles of radius t."""
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, t = v[:2 * n].reshape(n, 2), v[-1]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - 2 * t
        return np.concatenate([d, c[:, 0] - t, c[:, 1] - t, 1 - c[:, 0] - t, 1 - c[:, 1] - t])

    res = minimize(lambda v: -v[-1], np.append(c0.ravel(), 0.0), method="SLSQP",
                   constraints={"type": "ineq", "fun": cons}, options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[-1]
