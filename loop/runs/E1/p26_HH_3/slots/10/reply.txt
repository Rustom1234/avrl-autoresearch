def _solve_once(n, rng):
    c = rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
    k = min(int(rng.integers(6, 13)), n)   # k centres pushed onto the walls
    w, t = rng.integers(0, 4, k), rng.uniform(0, 1, k)
    x = np.where(w < 2, t, np.where(w == 2, 0.0, 1.0))
    y = np.where(w < 2, np.where(w == 0, 0.0, 1.0), t)
    c[:k] = np.stack([x, y], 1)
    x0 = np.concatenate([c.ravel(), np.full(n, INIT_RADIUS)])
    return _optimise(n, x0)
