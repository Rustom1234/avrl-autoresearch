def _tied(n, rng):
    """First stage: circles in three random size classes share one radius each (2n + 3 variables)."""
    cls = np.empty(n, int)
    a, b = round(0.3 * n), round(0.4 * n)
    cls[rng.permutation(n)] = np.repeat([0, 1, 2], [a, b, n - a - b])
    cnt = np.bincount(cls, minlength=3)
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, R = v[:2 * n].reshape(n, 2), v[2 * n:]
        r = R[cls]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(3, INIT_RADIUS)])
    res = minimize(lambda v: -(cnt @ v[2 * n:]), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), np.maximum(res.x[2 * n:][cls], 0)


def _solve_once(n, rng):
    c, r = _tied(n, rng)
    return _optimise(n, np.concatenate([c.ravel(), r]))
