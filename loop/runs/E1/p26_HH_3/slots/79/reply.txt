def _mixture_centres(n, rng):
    """Draw n centres from a random Gaussian mixture (2 or 3 components), rejecting samples outside the box."""
    k = rng.integers(2, 4)
    means = rng.uniform(0.2, 0.8, (k, 2))
    widths = rng.uniform(0.15, 0.35, k)
    w = rng.dirichlet(np.ones(k))
    out = []
    while len(out) < n:
        j = rng.choice(k, p=w)
        p = means[j] + widths[j] * rng.standard_normal(2)
        if INIT_LOW <= p[0] <= 1 - INIT_LOW and INIT_LOW <= p[1] <= 1 - INIT_LOW:
            out.append(p)
    return np.array(out)
