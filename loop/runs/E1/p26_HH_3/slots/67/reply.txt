def _solve_once(n, rng):
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    alpha = rng.uniform(0.2, 0.6)
    p = np.arange(1, n + 1) ** -alpha
    target = p * (n * INIT_RADIUS / p.sum())
    for w in (10.0, 6.7, 3.3):   # prior weight decays linearly to zero over the first 60 percent of iterations
        c, r = _optimise(n, x0, target, w, 40)
        x0 = np.concatenate([c.ravel(), r])
    return _optimise(n, x0)
