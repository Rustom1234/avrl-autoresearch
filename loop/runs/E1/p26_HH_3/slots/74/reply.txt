def _langevin(n, c, rng, steps=300, h=0.01, s0=0.02):
    """Overdamped Langevin relaxation of the centres with fixed radii INIT_RADIUS."""
    r = INIT_RADIUS
    eye = np.eye(n)
    for t in range(steps):
        s = s0 * (1 - t / steps)
        diff = c[:, None] - c[None]
        d = np.sqrt((diff ** 2).sum(-1)) + eye
        ov = np.maximum(0, 2 * r - d)
        np.fill_diagonal(ov, 0)
        g = -2 * ((ov / d)[:, :, None] * diff).sum(1)
        g += -2 * np.maximum(0, r - c) + 2 * np.maximum(0, c + r - 1)
        c = np.clip(c - h * g + s * rng.normal(size=c.shape), 0, 1)
    return c
