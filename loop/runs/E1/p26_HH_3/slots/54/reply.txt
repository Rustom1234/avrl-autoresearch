def solve(n, seed, time_limit):
    rng, t0 = np.random.default_rng(seed), time.process_time()
    budget = TIME_FRACTION * time_limit
    pop_size, n_cull = 8, 2
    pop = []
    for _ in range(pop_size):   # independent chains from random packings
        c, r = _grow(n, rng) if rng.random() < 0.5 else _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        pop.append((c, _repair(c, r)))
    while time.process_time() - t0 < budget:   # lockstep rounds
        for k in range(pop_size):   # advance every chain by one remove-and-reinsert step
            c, r = pop[k]
            c2, r2 = c.copy(), r.copy()
            i = np.argmin(r2)
            c2[i] = rng.uniform(0, 1, 2)
            r2[i] = 0
            c2, r2 = _optimise(n, np.concatenate([c2.ravel(), r2]))
            c2 = np.clip(c2, 0, 1)
            r2 = _repair(c2, r2)
            if r2.sum() > r.sum():
                pop[k] = (c2, r2)
        order = sorted(range(pop_size), key=lambda k: pop[k][1].sum())
        bc, br = pop[order[-1]]
        for k in order[:n_cull]:   # replace the worst chains with perturbed copies of the best
            c = np.clip(bc + rng.normal(0, 0.02, bc.shape), 0, 1)
            pop[k] = (c, _repair(c, br.copy()))
    return max(pop, key=lambda p: p[1].sum())
