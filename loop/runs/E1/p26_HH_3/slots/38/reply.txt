def _solve_once(n, rng, mixed=False):
    centres = rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n)
    if mixed:
        small = rng.uniform(0.02, 0.06, n)
        large = rng.uniform(0.08, 0.12, n)
        radii = np.where(rng.uniform(size=n) < 0.7, small, large)
    else:
        radii = np.full(n, INIT_RADIUS)
    return _optimise(n, np.concatenate([centres, radii]))
