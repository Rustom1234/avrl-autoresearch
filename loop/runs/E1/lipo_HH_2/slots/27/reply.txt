def minmax_kernel(C, kmax=8):
    # MinMax similarity sum(min)/sum(max) on count vectors (counts clipped at kmax).
    # sum(min(a,b)) = sum_k [a>=k][b>=k], so it reduces to binary matmuls (fast BLAS).
    C = np.minimum(C, kmax)
    S = C.sum(1)
    mn = np.zeros((len(C), len(C)), dtype=np.float32)
    for k in range(1, kmax + 1):
        B = (C >= k).astype(np.float32)
        mn += B @ B.T
    mx = S[:, None] + S[None, :] - mn
    return mn / np.maximum(mx, 1e-9)
