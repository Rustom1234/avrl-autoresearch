def clean_smiles(s):
    try:
        m = Chem.MolFromSmiles(s)
        m = rdMolStandardize.LargestFragmentChooser().choose(m)
        m = rdMolStandardize.Uncharger().uncharge(m)
        out = Chem.MolToSmiles(m)
        if Chem.MolFromSmiles(out) is None:

        return out
