def crippen_atoms(smiles):
    # per-molecule list of (logP, MR) contributions for each heavy atom; the pair identifies the Crippen atom type
    return [rdMolDescriptors._CalcCrippenContribs(Chem.MolFromSmiles(s)) for s in smiles]


def crippen_block(atoms_tr, atoms_te):
    def keys(atoms):
        return [(round(a, 3), round(b, 3)) for a, b in atoms]
    vocab = sorted({k for at in atoms_tr for k in keys(at)})
    idx = {k: i for i, k in enumerate(vocab)}

    def counts(atoms):
        C = np.zeros((len(atoms), len(vocab) + 1), dtype=np.float32)
        for r, at in enumerate(atoms):
            for k in keys(at):
                if k in idx:
                    C[r, idx[k]] += 1
            C[r, -1] = sum(b for _, b in at)
        return C
    return counts(atoms_tr), counts(atoms_te)
