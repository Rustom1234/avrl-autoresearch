_EG_CACHE = {}


def estate_gast_matrix(smiles):
    # EState fingerprint (counts and index sums) plus mean/min/max Gasteiger charge, cached per SMILES
    missing = [s for s in set(smiles) if s not in _EG_CACHE]
    for s in missing:
        m = Chem.MolFromSmiles(s)
        counts, sums = Fingerprinter.FingerprintMol(m)
        rdPartialCharges.ComputeGasteigerCharges(m)
        q = np.array([a.GetDoubleProp("_GasteigerCharge") for a in m.GetAtoms()])
        q = np.nan_to_num(q, nan=0.0, posinf=0.0, neginf=0.0)
        g = [q.mean(), q.min(), q.max()] if len(q) else [0.0, 0.0, 0.0]
        _EG_CACHE[s] = np.concatenate([np.asarray(counts, dtype=np.float64), np.asarray(sums, dtype=np.float64), g])
    return np.array([_EG_CACHE[s] for s in smiles])
