def frag_catalog():
    if "fcat" not in _FCAT:
        fp = rdfragcatalog.FragCatParams(1, 6, os.path.join(RDConfig.RDDataDir, "FunctionalGroups.txt"))
        _FCAT["fcat"] = rdfragcatalog.FragCatalog(fp)
        _FCAT["gen"] = rdfragcatalog.FragFPGenerator()
    return _FCAT["fcat"], _FCAT["gen"]


def frag_bits(smiles):
    fcat, gen = frag_catalog()
    out = np.zeros((len(smiles), fcat.GetFPLength()), dtype=np.float32)
    for i, s in enumerate(smiles):
        for b in gen.GetFPForMol(Chem.MolFromSmiles(s), fcat).GetOnBits():
            out[i, b] = 1.0
    return out


def frag_fit(smiles):
    F = frag_bits(smiles)
    keep = np.where(F.sum(0) >= 5)[0]
    return F[:, keep], keep


def frag_apply(smiles, keep):
    return frag_bits(smiles)[:, keep]
