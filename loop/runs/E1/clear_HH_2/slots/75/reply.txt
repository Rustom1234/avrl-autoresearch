def minmax(A, B, chunk=8):
    out = np.zeros((len(A), len(B)))
    for i in range(0, len(A), chunk):
        a = A[i:i + chunk, None, :]
        mn = np.minimum(a, B[None, :, :]).sum(2)
        mx = np.maximum(a, B[None, :, :]).sum(2)
        out[i:i + chunk] = mn / np.maximum(mx, 1e-9)
    return out
