def gp_mean(K_tr, y, K_te, sigma2):
    mu = y.mean()
    w = np.ones(len(y))
    for _ in range(3):
        alpha = np.linalg.solve(K_tr + np.diag(sigma2 / w), y - mu)
        r = y - (mu + K_tr @ alpha)
        c = 1.345 * max(np.median(np.abs(r)), 1e-6)
        w = np.maximum(np.minimum(1.0, c / np.maximum(np.abs(r), 1e-12)), 1e-3)
    return mu + K_te @ alpha
