def estate_features(smiles):

    for s in smiles:
        if s not in _ESTATE_CACHE:
            m = Chem.MolFromSmiles(s)
            fc, fs = FingerprintMol(m)
            idx = np.asarray(EStateIndices(m), dtype=np.float64)
            if idx.size == 0:
                idx = np.zeros(1)
            _ESTATE_CACHE[s] = np.concatenate([fc, fs, [idx.sum(), idx.max(), idx.min()]])
        rows.append(_ESTATE_CACHE[s])
    X = np.array(rows, dtype=np.float64)
    X[~np.isfinite(X)] = 0.0
