def adam_ridge(X, r, lam, steps=500, lr0=0.01, lr1=0.0005):
    n, d = X.shape
    p = np.zeros(d + 1)  # weights, then intercept
    m, v = np.zeros(d + 1), np.zeros(d + 1)
    for t in range(1, steps + 1):
        lr = lr1 + 0.5 * (lr0 - lr1) * (1 + np.cos(np.pi * (t - 1) / (steps - 1)))
        w, b = p[:d], p[d]
        e = X @ w + b - r
        g = np.empty(d + 1)
        g[:d] = 2.0 / n * (X.T @ e) + 2 * lam * w
        g[d] = 2.0 / n * e.sum()
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        p = p - lr * (m / (1 - 0.9 ** t)) / (np.sqrt(v / (1 - 0.999 ** t)) + 1e-8)
    return p[:d], p[d]


def choose_lam(X, r, smiles):
    scaffolds = [MurckoScaffold.MurckoScaffoldSmiles(smiles=s) for s in smiles]
    groups = np.unique(scaffolds, return_inverse=True)[1]
    best_score, best_lam = -np.inf, 1.0
    for lam in (0.1, 1.0, 10.0):
        pred = np.zeros(len(r))
        for tr, va in GroupKFold(n_splits=5).split(X, r, groups):
            w, b = adam_ridge(X[tr], r[tr], lam)
            pred[va] = X[va] @ w + b
        score = spearmanr(pred, r).correlation
        if score > best_score:
            best_score, best_lam = score, lam
    return best_lam
