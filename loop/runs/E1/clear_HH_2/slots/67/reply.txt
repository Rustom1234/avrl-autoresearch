def fit_pairwise(Z, y, seed, n_pairs=20000, l2=1e-3):
    rng = np.random.default_rng(seed)
    n = len(y)
    i = rng.integers(0, n, n_pairs)
    j = rng.integers(0, n, n_pairs)
    keep = (i != j) & (y[i] != y[j])
    i, j = i[keep], j[keep]
    D = Z[i] - Z[j]
    t = (y[i] > y[j]).astype(np.float64)
    sign = np.where(t == 1, 1.0, -1.0)

    def loss(w):
        s = D @ w
        f = np.logaddexp(0, -sign * s).mean() + l2 * np.sum(w * w)
        g = D.T @ (expit(s) - t) / len(t) + 2 * l2 * w
        return f, g

    return minimize(loss, np.zeros(Z.shape[1]), jac=True, method='L-BFGS-B').x
