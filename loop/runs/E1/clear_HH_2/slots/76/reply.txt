def bit_encoder(X_tr, r, X_te, k=5):
    cnt = X_tr.sum(0)
    enc = (X_tr.T @ r) / (cnt + k)  # mean residual per bit shrunk by n/(n+k)
    seen = (cnt > 0).astype(np.float64)
    num = X_te @ enc
    den = X_te @ seen
    return np.where(den > 0, num / np.maximum(den, 1e-9), 0.0)
