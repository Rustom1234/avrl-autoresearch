def smooth_targets(K, y, k=5, thr=0.4):
    K = K.copy()
    np.fill_diagonal(K, -1.0)
    out = y.copy()
    for i in range(len(y)):
        idx = np.argsort(-K[i])[:k]
        idx = idx[K[i, idx] > thr]
        if len(idx) == 0:
            continue
        w = K[i, idx] / K[i, idx].sum()
        out[i] = 0.5 * y[i] + 0.5 * np.dot(w, y[idx])
    return out
