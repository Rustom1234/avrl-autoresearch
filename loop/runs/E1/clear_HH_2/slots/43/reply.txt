ROUNDS = (100, 200, 400, 600, 800, 1000)


def choose_n_rounds(X, y, smiles, seed):
    scaffolds = [MurckoScaffold.MurckoScaffoldSmiles(smiles=s) for s in smiles]
    groups = np.unique(scaffolds, return_inverse=True)[1]
    scores = {r: [] for r in ROUNDS}
    for tr, va in GroupKFold(n_splits=5).split(X, y, groups):
        m = lgb.LGBMRegressor(n_estimators=max(ROUNDS), learning_rate=0.03, num_leaves=31, subsample=0.8,
                              subsample_freq=1, colsample_bytree=0.5, min_child_samples=10, random_state=seed,
                              n_jobs=8, verbose=-1)
        m.fit(X[tr], y[tr])
        for r in ROUNDS:
            scores[r].append(spearmanr(m.predict(X[va], num_iteration=r), y[va]).correlation)
    return max(ROUNDS, key=lambda r: np.mean(scores[r]))
