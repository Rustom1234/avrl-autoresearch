def minmax_kernel(A):
    sa = A.sum(1)
    def rows(i):
        return np.minimum(A[i:i + 4, None, :], A[None, :, :]).sum(2)
    with ThreadPoolExecutor(8) as ex:
        S = np.vstack(list(ex.map(rows, range(0, len(A), 4))))
    return S / (sa[:, None] + sa[None, :] - S + 1e-9)
