def knn_feats(Ftr, ytr, Fte, loo):
    S = tanimoto(Fte, Ftr)
    if loo:
        np.fill_diagonal(S, -1.0)
    idx = np.argpartition(-S, 5, axis=1)[:, :5]
    s = np.take_along_axis(S, idx, 1)
    w = np.maximum(s, 0) + 1e-6
    return np.column_stack([(w * ytr[idx]).sum(1) / w.sum(1), s.max(1)]).astype(np.float32)
