def sparse_counts(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS)
    return [gen.GetSparseCountFingerprint(Chem.MolFromSmiles(s)).GetNonzeroElements() for s in smiles]


def folded_bits(counts, size):
    B = np.zeros((len(counts), size))
    for i, c in enumerate(counts):
        for k in c:
            B[i, k % size] = 1.0
    return B
