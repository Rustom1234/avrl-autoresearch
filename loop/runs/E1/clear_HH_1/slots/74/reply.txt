    rows = []
    for s in smiles:
        m = Chem.MolFromSmiles(s)
        rows.append(gen.GetCountFingerprintAsNumPy(m) / max(m.GetNumHeavyAtoms(), 1))
    return np.array(rows, dtype=np.float64)


def minmax(A, B):
    out = np.empty((A.shape[0], B.shape[0]))
    sb = B.sum(axis=1)
    for i in range(A.shape[0]):
        mn = np.minimum(A[i][None, :], B).sum(axis=1)
        mx = A[i].sum() + sb - mn
        out[i] = mn / np.maximum(mx, 1e-9)
    return out
