def tanimoto(A, B):
    # A, B are already sqrt(idf)-scaled, so squared row sums give the weighted norms
    inter = A @ B.T
    na = (A ** 2).sum(axis=1)[:, None]
    nb = (B ** 2).sum(axis=1)[None, :]
    return inter / np.maximum(na + nb - inter, 1e-9)
