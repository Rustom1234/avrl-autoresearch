def median_sq_dist(D, seed):
    rng = np.random.RandomState(seed)
    S = D[rng.permutation(len(D))[:500]]
    sq = (S ** 2).sum(axis=1)[:, None] + (S ** 2).sum(axis=1)[None, :] - 2.0 * S @ S.T
    return np.median(np.maximum(sq[np.triu_indices(len(S), 1)], 0.0)) + 1e-9


def alignment(K, y):
    # centered kernel-target alignment; y is already centered-able
    yc = y - y.mean()
    Kc = K - K.mean(axis=0)[None, :] - K.mean(axis=1)[:, None] + K.mean()
    return (yc @ K @ yc) / (np.linalg.norm(Kc) * (yc @ yc) + 1e-12)
