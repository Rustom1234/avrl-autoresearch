def featurize(smiles):
    rows = []
    for s in smiles:
        m = Chem.MolFromSmiles(s)
        inv = [a.GetAtomicNum() * 10 + a.GetDegree() for a in m.GetAtoms()]
        fp = rdMolDescriptors.GetMorganFingerprintAsBitVect(m, RADIUS, nBits=N_BITS, invariants=inv)
        rows.append(np.array(list(fp), dtype=np.float32))
    return np.array(rows, dtype=np.float32)
