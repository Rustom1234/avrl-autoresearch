def minmax(A, B, chunk=16):
    out = np.zeros((A.shape[0], B.shape[0]))
    for i in range(0, A.shape[0], chunk):
        a = A[i:i + chunk, None, :]
        mn = np.minimum(a, B[None, :, :]).sum(axis=2)
        mx = np.maximum(a, B[None, :, :]).sum(axis=2)
        out[i:i + chunk] = mn / np.maximum(mx, 1e-9)
    return out
