def _rotate_patch(c, rng):
    """Rotate the centres within a random disk of the square about a random point by a random angle."""
    c = c.copy()
    p, R = rng.uniform(0, 1, 2), rng.uniform(0.2, 0.35)
    t = rng.uniform(-np.pi / 3, np.pi / 3)
    rot = np.array([[np.cos(t), -np.sin(t)], [np.sin(t), np.cos(t)]])
    m = np.sqrt(((c - p) ** 2).sum(1)) < R
    c[m] = (c[m] - p) @ rot.T + p
    return np.clip(c, 0, 1)
