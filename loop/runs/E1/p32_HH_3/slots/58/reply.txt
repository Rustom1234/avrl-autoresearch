def _solve_sub(c, r, idx, rng):
    """Optimise only the circles idx with all other circles fixed as obstacles."""
    m = len(idx)
    fixed = np.setdiff1d(np.arange(len(r)), idx)
    cf, rf = c[fixed], r[fixed]
    iu = np.triu_indices(m, 1)

    def cons(v):
        cs, rs = v[:2 * m].reshape(m, 2), v[2 * m:]
        d = np.sqrt(((cs[iu[0]] - cs[iu[1]]) ** 2).sum(1)) - rs[iu[0]] - rs[iu[1]]
        df = np.sqrt(((cs[:, None] - cf[None]) ** 2).sum(-1)) - rs[:, None] - rf[None]
        return np.concatenate([d, df.ravel(), cs[:, 0] - rs, cs[:, 1] - rs, 1 - cs[:, 0] - rs, 1 - cs[:, 1] - rs, rs])

    x0 = np.concatenate([c[idx].ravel(), r[idx]])
    res = minimize(lambda v: -v[2 * m:].sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    cn, rn = c.copy(), r.copy()
    cn[idx] = res.x[:2 * m].reshape(m, 2)
    rn[idx] = res.x[2 * m:]
    return cn, rn
