def _solve_once(n, rng, x0=None, extra=0):
    m = n + extra
    if x0 is None:
        x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * m), np.full(m, INIT_RADIUS)])
    iu = np.triu_indices(m, 1)

    def cons(v):
        c, r = v[:2 * m].reshape(m, 2), v[2 * m:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    w = np.zeros(m)
    w[:n] = 1.0   # phantom circles carry zero weight in the objective
    res = minimize(lambda v: -(v[2 * m:] * w).sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    c, r = res.x[:2 * m].reshape(m, 2), res.x[2 * m:]
    if extra:   # drop the phantoms, keep the n largest, and re-solve the n-circle problem from that state
        keep = np.argsort(-r)[:n]
        return _solve_once(n, rng, np.concatenate([c[keep].ravel(), r[keep]]))
    return c, r
