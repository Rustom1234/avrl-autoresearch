def _init_centres(n, rng):
    """Uniform random centres, or (with prob 0.5) a mirror-symmetric set about x = 0.5 or the diagonal."""
    if rng.random() < 0.5:
        return rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
    half = rng.uniform(INIT_LOW, 1 - INIT_LOW, ((n + 1) // 2, 2))
    if rng.random() < 0.5:
        half[:, 0] *= 0.5
        mirror = half * [-1, 1] + [1, 0]
    else:
        mirror = half[:, ::-1]
    c = np.vstack([half, mirror])
    if n % 2:
        c = np.delete(c, rng.integers(len(c)), axis=0)
    return c
