def solve(n, seed, time_limit):
    rngs = [np.random.default_rng([seed, k]) for k in range(CHAINS)]
    t0 = time.process_time()
    chains = []
    for rng in rngs:
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        chains.append((c, _repair(c, r)))
    for epoch in range(1, EPOCHS + 1):
        end = TIME_FRACTION * time_limit * epoch / EPOCHS
        while time.process_time() - t0 < end:   # basin hopping in every chain
            for i, rng in enumerate(rngs):
                if time.process_time() - t0 >= end:
                    break
                bc, br = chains[i]
                c0 = np.clip(bc + rng.normal(0, 0.02, bc.shape), 0, 1)
                r0 = np.maximum(br + rng.normal(0, 0.02, br.shape), 0)
                c0, r0 = _relocate(c0, r0, rng)
                c, r = _solve_once(n, rng, np.concatenate([c0.ravel(), r0]))
                c = np.clip(c, 0, 1)
                r = _repair(c, r)
                if r.sum() > br.sum():
                    chains[i] = (c, r)
        if epoch < EPOCHS:   # cull the 3 worst chains and clone the 3 best
            chains.sort(key=lambda ch: -ch[1].sum())
            chains = chains[:CHAINS // 2] * 2
    return max(chains, key=lambda ch: ch[1].sum())
