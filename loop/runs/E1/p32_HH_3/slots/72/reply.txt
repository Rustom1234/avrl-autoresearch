def solve(n, seed, time_limit):
    rng, t0 = np.random.default_rng(seed), time.process_time()
    budget = TIME_FRACTION * time_limit

    def run(x0, maxiter):
        c, r = _solve_once(n, rng, x0, maxiter)
        c = np.clip(c, 0, 1)
        return c, r, _repair(c, r)

    def warm(c, r):
        return np.concatenate([c.ravel(), r])

    # Round 1: short SLSQP on random starts, keep the top 10
    pool = [run(None, 25)]
    while len(pool) < 40 and time.process_time() - t0 < 0.3 * budget:
        pool.append(run(None, 25))
    pool.sort(key=lambda s: -s[2].sum())
    # Round 2: continue the top 10 to 80 iterations, keep the top 3
    pool2 = []
    for c, r, _ in pool[:10]:
        if pool2 and time.process_time() - t0 > 0.6 * budget:
            break
        pool2.append(run(warm(c, r), 80))
    pool2.sort(key=lambda s: -s[2].sum())
    # Round 3: full MAX_ITER budget on the top 3
    pool3 = []
    for c, r, _ in pool2[:3]:
        if pool3 and time.process_time() - t0 > 0.8 * budget:
            break
        pool3.append(run(warm(c, r), MAX_ITER))
    c, r, rep = max(pool3, key=lambda s: s[2].sum())
    best = (c, rep)
