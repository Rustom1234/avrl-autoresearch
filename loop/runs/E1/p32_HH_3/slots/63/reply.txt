def solve(n, seed, time_limit):
    rng, t0 = np.random.default_rng(seed), time.process_time()
    c, r = _solve_once(n, rng)
    c = np.clip(c, 0, 1)
    best = (c, _repair(c, r))
    second = None   # second elite packing, distinct from best
    step = 0
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        step += 1
        relink = second is not None and step % 10 == 0
        if relink:   # path relinking between the two elites, matched by x-sorted index
            t = rng.uniform(0.2, 0.8)
            ia, ib = np.argsort(best[0][:, 0]), np.argsort(second[0][:, 0])
            c0 = (1 - t) * best[0][ia] + t * second[0][ib]
            r0 = 0.5 * (best[1][ia] + second[1][ib])
        else:   # basin hopping from the incumbent
            bc, br = best
            c0 = np.clip(bc + rng.normal(0, 0.02, bc.shape), 0, 1)
            r0 = np.maximum(br + rng.normal(0, 0.02, br.shape), 0)
            c0, r0 = _relocate(c0, r0, rng)
        c, r = _solve_once(n, rng, np.concatenate([c0.ravel(), r0]))
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if r.sum() > best[1].sum():
            second, best = best, (c, r)
        elif not relink and r.sum() < best[1].sum() - 1e-9 and (second is None or r.sum() > second[1].sum()):
            second = (c, r)
    return best
