def _solve_equal(n, rng):
    """Phase 1: all radii equal to a shared variable t, maximised under the packing constraints."""
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, t = v[:2 * n].reshape(n, 2), v[2 * n]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - 2 * t
        return np.concatenate([d, c[:, 0] - t, c[:, 1] - t, 1 - c[:, 0] - t, 1 - c[:, 1] - t, [t]])

    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), [INIT_RADIUS]])
    res = minimize(lambda v: -v[2 * n], x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return np.concatenate([res.x[:2 * n], np.full(n, res.x[2 * n])])
