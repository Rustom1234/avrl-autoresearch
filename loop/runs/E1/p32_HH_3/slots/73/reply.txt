def _spring_start(n, rng, steps=400):
    """Physical start: random centres relaxed by pair springs, wall repulsion and clearance-capped radii."""
    c = rng.uniform(0, 1, (n, 2))
    r = np.full(n, 0.06)
    eye = np.eye(n, dtype=bool)
    for s in range(steps):
        step = 0.01 * (1 - s / steps)
        d = c[:, None] - c[None]
        dist = np.sqrt((d ** 2).sum(-1))
        safe = np.where(eye, 1.0, dist)
        mag = np.where(eye, 0.0, (safe - 1.6 * (r[:, None] + r[None])) / safe)
        f = -(mag[..., None] * d).sum(1)
        f += np.maximum(r[:, None] - c, 0) - np.maximum(r[:, None] - (1 - c), 0)
        c = np.clip(c + step * f, 0, 1)
        d = c[:, None] - c[None]
        dist = np.sqrt((d ** 2).sum(-1))
        gap = np.where(eye, np.inf, dist - r[None])
        clear = np.minimum(np.minimum(c, 1 - c).min(1), gap.min(1))
        r = np.maximum(np.minimum(r + 0.05 * (clear - r), clear), 0)
    return c, r
