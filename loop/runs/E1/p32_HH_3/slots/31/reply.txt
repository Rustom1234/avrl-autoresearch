def _mc_start(n, rng, sweeps=2000):
    """Hard-disk Monte Carlo compression: random moves that keep disks valid, then radii grown to the maximum."""
    c = rng.uniform(0, 1, (n, 2))
    r = np.full(n, 0.001)
    step = 1.0
    for s in range(sweeps):
        for i in range(n):
            p = c[i] + rng.normal(0, 0.05 * step, 2)
            if p.min() < r[i] or (1 - p).min() < r[i]:
                continue
            d = np.sqrt(((c - p) ** 2).sum(1))
            d[i] = np.inf
            if np.any(d < r + r[i]):
                continue
            c[i] = p
        d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
        np.fill_diagonal(d, np.inf)
        rmax = np.minimum(np.minimum(c, 1 - c).min(1), (d - r[None]).min(1))
        r = np.maximum(0, np.minimum(rmax, 0.999 * rmax + rng.uniform(0, 1e-4)))
        if (s + 1) % 50 == 0:
            step *= 0.99
    return np.concatenate([c.ravel(), r])
