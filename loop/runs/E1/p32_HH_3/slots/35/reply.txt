def _wall_slide(c, r, rng):
    """Pick a random circle touching a wall and slide its centre along that wall to a uniform random position."""
    c, r = c.copy(), r.copy()
    gap = np.stack([c[:, 0] - r, 1 - c[:, 0] - r, c[:, 1] - r, 1 - c[:, 1] - r], 1)
    touch = np.argwhere(np.abs(gap) < 1e-6)
    if len(touch) == 0:
        return c, r
    i, side = touch[rng.integers(len(touch))]
    if side < 2:
        c[i, 0] = r[i] if side == 0 else 1 - r[i]
        c[i, 1] = rng.uniform(r[i], 1 - r[i])
    else:
        c[i, 1] = r[i] if side == 2 else 1 - r[i]
        c[i, 0] = rng.uniform(r[i], 1 - r[i])
    return c, r
