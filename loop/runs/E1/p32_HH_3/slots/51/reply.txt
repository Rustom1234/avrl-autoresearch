def _sigma(c, r):
    """Per-circle kick scale from contact count: loose circles (likely rattlers) move more, rigid core less."""
    d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
    touch = d < r[:, None] + r[None] + 1e-3
    np.fill_diagonal(touch, False)
    wall = (np.stack([c[:, 0], c[:, 1], 1 - c[:, 0], 1 - c[:, 1]], 1) < 1e-3).sum(1)
    k = touch.sum(1) + wall
    return np.where(k <= 1, 0.03, np.where(k >= 4, 0.01, 0.02))
