def _lloyd(c, r, rng, iters=3, m=4000):
    """Move centres toward power-diagram centroids (blended 50/50 with current position)."""
    c = c.copy()
    for _ in range(iters):
        p = rng.uniform(0, 1, (m, 2))
        pw = ((p[:, None] - c[None]) ** 2).sum(-1) - r[None] ** 2
        a = np.argmin(pw, 1)
        for i in range(len(c)):
            pts = p[a == i]
            if len(pts):
                c[i] = 0.5 * c[i] + 0.5 * pts.mean(0)
    return np.clip(c, 0, 1)
