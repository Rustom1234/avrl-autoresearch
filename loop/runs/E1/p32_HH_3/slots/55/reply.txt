def _slp(c, r, iters=60):
    """Sequential LP: linearise the constraints, maximise sum(dr) over a box trust region, accept good steps."""
    n = len(r)
    iu0, iu1 = np.triu_indices(n, 1)
    t = np.arange(n)

    def viol(c, r):
        d = np.sqrt(((c[iu0] - c[iu1]) ** 2).sum(1))
        v = np.maximum(r[iu0] + r[iu1] - d, 0).sum()
        w = np.concatenate([r - c[:, 0], r - c[:, 1], c[:, 0] + r - 1, c[:, 1] + r - 1])
        return v + np.maximum(w, 0).sum()

    cost = np.zeros(3 * n)
    cost[2 * n:] = -1.0
    delta = 0.02
    for _ in range(iters):
        if delta < 1e-7:
            break
        d = np.sqrt(((c[iu0] - c[iu1]) ** 2).sum(1))
        keep = d < 2 * (r[iu0] + r[iu1]) + 0.1
        i, j, dk = iu0[keep], iu1[keep], np.maximum(d[keep], 1e-12)
        u = (c[i] - c[j]) / dk[:, None]
        m = len(i)
        k = np.arange(m)
        A = np.zeros((m + 5 * n, 3 * n))
        b = np.zeros(m + 5 * n)
        A[k, 2 * i] = -u[:, 0]
        A[k, 2 * i + 1] = -u[:, 1]
        A[k, 2 * j] = u[:, 0]
        A[k, 2 * j + 1] = u[:, 1]
        A[k, 2 * n + i] = 1
        A[k, 2 * n + j] = 1
        b[:m] = dk - r[i] - r[j]
        o = m
        A[o + t, 2 * t] = -1
        A[o + t, 2 * n + t] = 1
        b[o:o + n] = c[:, 0] - r
        A[o + n + t, 2 * t] = 1
        A[o + n + t, 2 * n + t] = 1
        b[o + n:o + 2 * n] = 1 - c[:, 0] - r
        A[o + 2 * n + t, 2 * t + 1] = -1
        A[o + 2 * n + t, 2 * n + t] = 1
        b[o + 2 * n:o + 3 * n] = c[:, 1] - r
        A[o + 3 * n + t, 2 * t + 1] = 1
        A[o + 3 * n + t, 2 * n + t] = 1
        b[o + 3 * n:o + 4 * n] = 1 - c[:, 1] - r
        A[o + 4 * n + t, 2 * n + t] = -1
        b[o + 4 * n:] = r
        res = linprog(cost, A_ub=A, b_ub=b, bounds=(-delta, delta), method="highs")
        if res.status != 0:
            delta *= 0.5
            continue
        c2 = c + res.x[:2 * n].reshape(n, 2)
        r2 = np.maximum(r + res.x[2 * n:], 0)
        if viol(c2, r2) <= viol(c, r) + 1e-12 and r2.sum() > r.sum() + 1e-12:
            c, r = c2, r2
            delta = min(delta * 1.5, 0.1)
        else:
            delta *= 0.5
    return c, r
