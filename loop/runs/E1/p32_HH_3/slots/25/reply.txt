def _merge_once(n, rng):
    """Solve two halves in half-width sub-squares, then re-solve the whole set from their union."""
    m, horiz = n // 2, rng.random() < 0.5
    cs, rs = [], []
    for k, off in ((m, 0.0), (n - m, 0.5)):
        ck, rk = _solve_once(k, rng)
        ck = np.clip(ck, 0, 1)
        rk = _repair(ck, rk)
        shift = np.array([off, 0.0]) if horiz else np.array([0.0, off])
        cs.append(ck * 0.5 + shift)
        rs.append(rk * 0.5)
    c, r = np.concatenate(cs), np.concatenate(rs)
    c = np.clip(c, 0, 1)
    r = np.maximum(np.minimum(r, np.minimum(c, 1 - c).min(1)), 0)
    c, r = _solve_once(n, rng, np.concatenate([c.ravel(), r]))
    c = np.clip(c, 0, 1)
    return c, _repair(c, r)
