def _structured(n, rng):
    """Jittered 5x5 grid + 1 random circle, or a 5-5-6-5-5 row layout."""
    if rng.random() < 0.5:
        g = 0.1 + 0.2 * np.arange(5)
        pts = np.array([(x, y) for x in g for y in g])
        pts = np.vstack([pts, rng.uniform(0.1, 0.9, (1, 2))])
    else:
        pts = np.array([((j + 0.5) / m, (k + 0.5) / 5) for k, m in enumerate([5, 5, 6, 5, 5]) for j in range(m)])
    pts = np.clip(pts + rng.normal(0, 0.02, pts.shape), 0.05, 0.95)
    return np.concatenate([pts.ravel(), np.full(n, 0.08)])
