def _lp_radii(c, r):
    """Optimal radii for fixed centres (LP)."""
    n = len(r)
    iu = np.triu_indices(n, 1)
    m = len(iu[0])
    A = np.zeros((m, n))
    A[np.arange(m), iu[0]] = 1
    A[np.arange(m), iu[1]] = 1
    b = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1))
    wall = np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1]))
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=np.column_stack([np.zeros(n), np.maximum(wall, 0)]),
                  method="highs")
    return res.x if res.status == 0 else r
