    c, r = best
    n = len(r)
    k = rng.choice(np.argsort(r)[:5])
    keep = np.arange(n) != k
    c2, r2 = _solve_once(n - 1, rng, (c[keep], r[keep]), 0.0, 100, 1.0)
    c2, r2 = np.clip(c2, 0, 1), np.maximum(r2, 0)
    g = (np.arange(100) + 0.5) / 100
    P = np.stack(np.meshgrid(g, g), -1).reshape(-1, 2)
    d = np.sqrt(((P[:, None] - c2[None]) ** 2).sum(-1)) - r2[None]
    wall = np.minimum(np.minimum(P[:, 0], P[:, 1]), np.minimum(1 - P[:, 0], 1 - P[:, 1]))
    p = P[np.argmax(np.minimum(d.min(1), wall))]
    return _solve_once(n, rng, (np.vstack([c2, p]), np.append(r2, 0.02)), 0.0, 200, 1.0)
