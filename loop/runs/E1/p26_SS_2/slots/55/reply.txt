def _grow(n, rng, c, r):
    """Insert one circle at the largest empty hole of a (n-1)-circle packing and re-optimise."""
    g = (np.arange(50) + 0.5) / 50
    p = np.stack(np.meshgrid(g, g), -1).reshape(-1, 2)
    d = np.sqrt(((p[:, None] - c[None]) ** 2).sum(-1)) - r[None]
    d = np.minimum(d.min(1), np.minimum(np.minimum(p[:, 0], p[:, 1]), np.minimum(1 - p[:, 0], 1 - p[:, 1])))
    q = p[np.argmax(d)]
    x0 = np.concatenate([np.vstack([c, q]).ravel(), r * 0.9, [0.03]])
    return _solve_once(n, rng, x0=x0)
