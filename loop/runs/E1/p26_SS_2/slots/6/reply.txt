def _relax(c0, n):
    """Pre-relaxation: L-BFGS-B on the overlap penalty with growing radii (100 steps in total)."""
    a, b = np.triu_indices(n, 1)

    def pen(v, rad):
        c = v.reshape(n, 2)
        g = np.zeros_like(c)
        diff = c[a] - c[b]
        d = np.maximum(np.sqrt((diff ** 2).sum(1)), 1e-12)
        ov = np.maximum(2 * rad - d, 0)
        f = (ov ** 2).sum()
        gp = (-2 * ov / d)[:, None] * diff
        np.add.at(g, a, gp)
        np.add.at(g, b, -gp)
        lo, hi = np.maximum(rad - c, 0), np.maximum(c - (1 - rad), 0)
        f += (lo ** 2).sum() + (hi ** 2).sum()
        g += -2 * lo + 2 * hi
        return f, g.ravel()

    v = c0
    for rad in (0.05, 0.07, 0.09, 0.1):
        v = minimize(pen, v, args=(rad,), jac=True, method="L-BFGS-B", options={"maxiter": 25}).x
    return np.clip(v, 0, 1)
