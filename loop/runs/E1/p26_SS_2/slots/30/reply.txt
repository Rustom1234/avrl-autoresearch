def _insert(pool, c, r):
    s = r.sum()
    if any(abs(s - p[1].sum()) <= 1e-4 for p in pool):
        return
    pool.append((c, r))
    pool.sort(key=lambda p: -p[1].sum())
    del pool[5:]


def _child(n, rng, pool):
    a, b = rng.choice(len(pool), 2, replace=len(pool) < 2)
    ax, t = rng.integers(2), rng.uniform(0.3, 0.7)
    ca, cb = pool[a][0], pool[b][0]
    c = np.vstack([ca[ca[:, ax] < t], cb[cb[:, ax] >= t]])
    if len(c) > n:
        c = c[rng.permutation(len(c))[:n]]
    elif len(c) < n:
        c = np.vstack([c, rng.uniform(0.05, 0.95, (n - len(c), 2))])
    return c


def solve(n, seed, time_limit):
    rng, t0, pool = np.random.default_rng(seed), time.process_time(), []
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        if pool and time.process_time() - t0 > 0.3 * time_limit:
            c, r = _solve_once(n, rng, (_child(n, rng, pool), np.full(n, 0.1)), 0.005)
        else:
            c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        _insert(pool, c, r)
    return pool[0]
