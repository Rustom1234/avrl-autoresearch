def _penalty(v, mu, n, iu):
    c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
    gc, gr = np.zeros((n, 2)), -np.ones(n)
    diff = c[iu[0]] - c[iu[1]]
    d = np.sqrt((diff ** 2).sum(1)) + 1e-12
    o = np.maximum(0, r[iu[0]] + r[iu[1]] - d)
    val = -r.sum() + mu * (o ** 2).sum()
    w = 2 * mu * o
    np.add.at(gr, iu[0], w)
    np.add.at(gr, iu[1], w)
    gd = (-w / d)[:, None] * diff
    np.add.at(gc, iu[0], gd)
    np.add.at(gc, iu[1], -gd)
    for k in range(2):
        lo = np.maximum(0, r - c[:, k])
        hi = np.maximum(0, c[:, k] + r - 1)
        val += mu * (lo ** 2 + hi ** 2).sum()
        gr += 2 * mu * (lo + hi)
        gc[:, k] += 2 * mu * (hi - lo)
    return val, np.concatenate([gc.ravel(), gr])
