def standardize(m):
    try:
        frags = Chem.rdmolops.GetMolFrags(m, asMols=True)
        if len(frags) > 1:
            m = max(frags, key=lambda f: f.GetNumAtoms())
        m = rdMolStandardize.Uncharger().uncharge(m)
        Chem.SanitizeMol(m)
    except Exception:
        pass
    return m
