def knn_scores(X, y, Xt, k=5):
    A = (Xt[:, N_DESC:] > 0).astype(np.float32)
    B = (X[:, N_DESC:] > 0).astype(np.float32)
    inter = A @ B.T
    sim = inter / np.maximum(A.sum(1)[:, None] + B.sum(1)[None, :] - inter, 1e-6)
    idx = np.argsort(-sim, axis=1)[:, :k]
    w = np.take_along_axis(sim, idx, axis=1) + 1e-6
    return (w * y[idx]).sum(1) / w.sum(1)


def mlp_scores(X, y, Xt, seed):
    qt = QuantileTransformer(n_quantiles=200, output_distribution='normal', random_state=seed)
    D = qt.fit_transform(X[:, :N_DESC])
    Dt = qt.transform(Xt[:, :N_DESC])
    m = MLPClassifier(hidden_layer_sizes=(128,), alpha=1e-2, max_iter=200, early_stopping=True, random_state=seed)
    m.fit(D, y)
    return m.predict_proba(Dt)[:, 1]


def member_scores(X, y, Xt, seed):
    """Return list of rank-normalized scores, one per ensemble member."""
    models = [
        RandomForestClassifier(n_estimators=N_TREES, max_features=MAX_FEATURES, min_samples_leaf=MIN_SAMPLES_LEAF,
                               n_jobs=-1, random_state=seed),
        LGBMClassifier(n_estimators=300, learning_rate=0.05, num_leaves=31, subsample=0.8, subsample_freq=1,
                       colsample_bytree=0.3, n_jobs=8, random_state=seed, verbose=-1),
        CatBoostClassifier(iterations=500, learning_rate=0.05, depth=6, rsm=0.2, thread_count=8,
                           random_seed=seed, verbose=0),
    ]
    out = []
    for m in models:
        m.fit(X, y)
        out.append(m.predict_proba(Xt)[:, 1])
    out.append(knn_scores(X, y, Xt))
    out.append(mlp_scores(X, y, Xt, seed))
    return [rankdata(p) / len(Xt) for p in out]


def scaffold(s):
    try:
        return MurckoScaffold.MurckoScaffoldSmiles(smiles=s)
    except Exception:
        return s


def fit_predict(train_smiles, train_y, test_smiles, seed):
    X = featurize(train_smiles)
    Xt = featurize(test_smiles)
    y = np.asarray(train_y, dtype=np.int64)
    groups = np.unique([scaffold(s) for s in train_smiles], return_inverse=True)[1] if False else \
        np.unique(np.array([scaffold(s) for s in train_smiles]), return_inverse=True)[1]
    oof = np.zeros((5, len(y)))
    for tr, va in GroupKFold(n_splits=5).split(X, y, groups):
        P = member_scores(X[tr], y[tr], X[va], seed)
        for j, p in enumerate(P):
            oof[j, va] = p
    rng = np.random.RandomState(seed)
    cands = np.vstack([np.ones(5) / 5, rng.dirichlet(np.ones(5), size=200)])
    aucs = [roc_auc_score(y, w @ oof) for w in cands]
    w = 0.5 * cands[int(np.argmax(aucs))] + 0.5 * np.ones(5) / 5
    P = member_scores(X, y, Xt, seed)
    return w @ np.array(P)
