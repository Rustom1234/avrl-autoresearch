    def minmax(P, Q):
        inter = sum((P[:, nd:] >= t).astype(np.float32) @ (Q[:, nd:] >= t).astype(np.float32).T for t in range(1, 6))
        sp = np.minimum(P[:, nd:], 5).sum(1)
        sq = np.minimum(Q[:, nd:], 5).sum(1)
        return inter / np.maximum(sp[:, None] + sq[None, :] - inter, 1e-9)
