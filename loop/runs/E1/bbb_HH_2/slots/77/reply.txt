def spectral_coords(all_smiles, k=K_NN, n_comp=N_SPECTRAL):
    mols = [Chem.MolFromSmiles(s) for s in all_smiles]
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    fps = [gen.GetFingerprint(m) for m in mols]
    n = len(fps)
    sim = np.zeros((n, n))
    for i in range(n):
        sim[i] = DataStructs.BulkTanimotoSimilarity(fps[i], fps)
    np.fill_diagonal(sim, 0.0)
    idx = np.argsort(-sim, axis=1)[:, :k]
    rows = np.repeat(np.arange(n), k)
    cols = idx.ravel()
    W = np.zeros((n, n))
    W[rows, cols] = sim[rows, cols]
    W = np.maximum(W, W.T)
    d_inv = 1.0 / np.sqrt(W.sum(axis=1) + 1e-12)
    L = np.eye(n) - d_inv[:, None] * W * d_inv[None, :]
    _, vecs = np.linalg.eigh(L)
    return vecs[:, 1:n_comp + 1].astype(np.float32)
