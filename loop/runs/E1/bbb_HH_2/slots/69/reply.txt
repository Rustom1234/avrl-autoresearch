def importance_weights(x_tr, x_te, seed):
    x = np.vstack([x_tr, x_te])
    d = np.concatenate([np.zeros(len(x_tr)), np.ones(len(x_te))])
    clf = LogisticRegression(C=0.1, max_iter=1000, random_state=seed)
    clf.fit(x, d)
    p = np.clip(clf.predict_proba(x_tr)[:, 1], 1e-6, 1 - 1e-6)
    w = np.clip(p / (1 - p), 0.1, 10)
    return w / w.mean()
