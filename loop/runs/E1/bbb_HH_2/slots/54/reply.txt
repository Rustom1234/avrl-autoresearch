def butina_labels(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    fps = [gen.GetFingerprint(Chem.MolFromSmiles(s)) for s in smiles]
    dists = []
    for i in range(1, len(fps)):
        sims = DataStructs.BulkTanimotoSimilarity(fps[i], fps[:i])
        dists.extend(1.0 - np.asarray(sims))
    clusters = Butina.ClusterData(dists, len(fps), CLUSTER_DIST, isDistData=True)
    labels = np.zeros(len(fps), dtype=np.int64)
    for c_id, members in enumerate(clusters):
        labels[list(members)] = c_id
    return labels, len(clusters)


def fit_predict(train_smiles, train_y, test_smiles, seed):
    y = np.asarray(train_y, dtype=np.int64)
    train_sets = fingerprints(train_smiles)
    test_sets = fingerprints(test_smiles)
    labels, n_clusters = butina_labels(train_smiles)
    total = np.zeros(len(test_smiles))
    for bag in range(N_BAGS):
        rng = np.random.RandomState(seed * 100 + bag)
        dropped = rng.rand(n_clusters) < DROP_FRAC
        keep = ~dropped[labels]
        for x_tr, x_te in zip(train_sets, test_sets):
            model = XGBClassifier(n_estimators=300, max_depth=5, learning_rate=0.05, subsample=0.8, colsample_bytree=0.3,
                                  min_child_weight=2, eval_metric='logloss', random_state=seed + bag, n_jobs=8)
            model.fit(x_tr[keep], y[keep])
            total += rankdata(model.predict_proba(x_te)[:, 1])
    total = (total - total.min()) / (total.max() - total.min() + 1e-12)
    return total
