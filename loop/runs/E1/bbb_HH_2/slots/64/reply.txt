def spectral_embedding(x, k=10, dim=16):
    # Unsupervised: kNN Tanimoto graph over all molecules, normalized Laplacian eigenvectors.
    x = x.astype(np.float64)
    inter = x @ x.T
    cnt = x.sum(axis=1)
    sim = inter / (cnt[:, None] + cnt[None, :] - inter + 1e-12)
    np.fill_diagonal(sim, 0.0)
    nn = np.argsort(-sim, axis=1)[:, :k]
    rows = np.repeat(np.arange(len(x)), k)
    cols = nn.ravel()
    w = np.zeros_like(sim)
    w[rows, cols] = sim[rows, cols]
    w = np.maximum(w, w.T)
    d = w.sum(axis=1)
    dinv = 1.0 / np.sqrt(np.maximum(d, 1e-12))
    lap = np.eye(len(x)) - dinv[:, None] * w * dinv[None, :]
    _, vecs = np.linalg.eigh(lap)
    return vecs[:, 1:dim + 1].astype(np.float32)
