def oob_rounds(x, y, seed):
    rng = np.random.RandomState(seed)
    n = len(y)
    idx = rng.randint(0, n, n)
    oob = np.setdiff1d(np.arange(n), idx)
    model = XGBClassifier(n_estimators=1500, max_depth=5, learning_rate=0.05, subsample=0.8, colsample_bytree=0.3,
                          min_child_weight=2, eval_metric='logloss', random_state=seed, n_jobs=8)
    model.fit(x[idx], y[idx], eval_set=[(x[oob], y[oob])], verbose=False)
    ll = np.array(model.evals_result()['validation_0']['logloss'])
    return int(np.clip(np.argmin(ll) + 1, 150, 800))
