def compact(smiles):
    from rdkit.Chem import Crippen, rdMolDescriptors, MACCSkeys
    rows = []
    for s in smiles:
        m = Chem.MolFromSmiles(s)
        pos = sum(a.GetFormalCharge() > 0 for a in m.GetAtoms())
        neg = sum(a.GetFormalCharge() < 0 for a in m.GetAtoms())
        d = [Crippen.MolLogP(m), rdMolDescriptors.CalcTPSA(m), Descriptors.MolWt(m),
             rdMolDescriptors.CalcNumHBD(m), rdMolDescriptors.CalcNumHBA(m),
             rdMolDescriptors.CalcNumRotatableBonds(m), rdMolDescriptors.CalcNumAromaticRings(m),
             rdMolDescriptors.CalcFractionCSP3(m), pos, neg]
        rows.append(d + list(MACCSkeys.GenMACCSKeys(m)))
    return np.array(rows, dtype=np.float32)
