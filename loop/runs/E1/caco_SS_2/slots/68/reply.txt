def clean_mol(s):
    m0 = Chem.MolFromSmiles(s)
    try:
        frags = Chem.rdmolops.GetMolFrags(m0, asMols=True)
        m = max(frags, key=lambda f: f.GetNumHeavyAtoms())
        m = rdMolStandardize.Uncharger().uncharge(m)
        Chem.SanitizeMol(m)
        return m
    except Exception:
        return m0
