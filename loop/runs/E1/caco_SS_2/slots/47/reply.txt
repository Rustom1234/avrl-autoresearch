def minmax_kernel(A, B):
    K = np.empty((len(A), len(B)), dtype=np.float64)
    for i, a in enumerate(A):
        K[i] = np.minimum(a, B).sum(1) / np.maximum(np.maximum(a, B).sum(1), 1e-9)
    return K
