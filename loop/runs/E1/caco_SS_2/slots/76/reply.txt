def ap_svd(train_smiles, test_smiles, seed, k=20):
    from sklearn.decomposition import TruncatedSVD
    gen = rdFingerprintGenerator.GetAtomPairGenerator(fpSize=2048)
    allsm = list(train_smiles) + list(test_smiles)
    A = np.array([np.log1p(gen.GetCountFingerprintAsNumPy(Chem.MolFromSmiles(s)).astype(np.float32)) for s in allsm])
    Z = TruncatedSVD(n_components=k, random_state=seed).fit_transform(A).astype(np.float32)
    return Z[:len(train_smiles)], Z[len(train_smiles):]
