def hb_counts(m):
    counts = []
    masked = set()
    for p in HB_PATS:
        matches = m.GetSubstructMatches(p)
        counts.append(len(matches))
        for t in matches:
            masked.update(t)
    polar = [a.GetIdx() for a in m.GetAtoms() if a.GetSymbol() in ('N', 'O')]
    counts.append(sum(1 for i in polar if i not in masked))
    return np.array(counts, dtype=np.float32)
