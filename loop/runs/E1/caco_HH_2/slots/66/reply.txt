def poly_design(B, mu, sd):
    Z = (B - mu) / sd
    cols = [Z, Z ** 2]
    for i in range(Z.shape[1]):
        for j in range(i + 1, Z.shape[1]):
            cols.append((Z[:, i] * Z[:, j])[:, None])
    return np.hstack(cols)
