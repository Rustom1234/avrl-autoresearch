def fit_predict(train_smiles, train_y, test_smiles, seed):
    y = np.asarray(train_y, dtype=np.float64)
    A_tr = np.c_[np.ones(len(y)), base_features(train_smiles)]
    A_te = np.c_[np.ones(len(test_smiles)), base_features(test_smiles)]
    coef = np.linalg.lstsq(A_tr, y, rcond=None)[0]
    X_tr = featurize(train_smiles)
    X_te = featurize(test_smiles)
    resid = y - A_tr @ coef
    rng = np.random.default_rng(seed)
    preds = []
    for i in range(6):
        cols = np.sort(rng.choice(X_tr.shape[1], size=int(0.4 * X_tr.shape[1]), replace=False))
        model = CatBoostRegressor(loss_function='MAE', depth=5, learning_rate=0.1, iterations=300,
                                  bootstrap_type='Bernoulli', subsample=0.8,
                                  random_seed=seed * 10 + i, verbose=0, thread_count=8)
        model.fit(X_tr[:, cols], resid)
        preds.append(model.predict(X_te[:, cols]))
    return np.mean(preds, axis=0) + A_te @ coef
