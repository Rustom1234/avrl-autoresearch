def pipeline(train_smiles, y, test_smiles, seed):

    A_te = np.c_[np.ones(len(test_smiles)), base_features(test_smiles)]

def fit_predict(train_smiles, train_y, test_smiles, seed):
    y = np.asarray(train_y, dtype=np.float64)
    tr_smi = np.asarray(train_smiles, dtype=object)
    scaf = [MurckoScaffold.MurckoScaffoldSmiles(s) for s in tr_smi]
    groups = np.unique(scaf, return_inverse=True)[1]
    oof = np.zeros(len(y))
    for tr, va in GroupKFold(n_splits=4).split(np.zeros(len(y)), y, groups):
        oof[va] = pipeline(list(tr_smi[tr]), y[tr], list(tr_smi[va]), seed)
    mae = lambda p: np.mean(np.abs(y - (p[0] + p[1] * oof)))
    a, b = minimize(mae, x0=np.array([0.0, 1.0]), method='Nelder-Mead').x
    return a + b * pipeline(list(train_smiles), y, list(test_smiles), seed)
