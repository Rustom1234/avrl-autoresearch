def _init_centres(n, rng):
    t = rng.integers(3)
    if t == 0:
        return rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
    if t == 1:
        cand = rng.uniform(0.02, 0.98, (400, 2))
        i = 0
        idx = [i]
        d = np.sqrt(((cand - cand[i]) ** 2).sum(1))
        for _ in range(n - 1):
            i = int(np.argmax(d))
            idx.append(i)
            d = np.minimum(d, np.sqrt(((cand - cand[i]) ** 2).sum(1)))
        return cand[idx]
    k = int(rng.integers(5, 7))
    counts = np.full(k, n // k)
    counts[rng.choice(k, n - counts.sum(), replace=False)] += 1
    pts = []
    for i, m in enumerate(counts):
        for j in range(m):
            pts.append(((j + 0.5) / m, (i + 0.5) / k))
    pts = np.array(pts) + rng.normal(0, 0.02, (n, 2))
    return np.clip(pts, 0.02, 0.98)
