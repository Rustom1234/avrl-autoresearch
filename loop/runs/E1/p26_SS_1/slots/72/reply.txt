def _child(n, rng, pool):
    i, j = rng.choice(len(pool), 2, replace=False)
    t = rng.uniform(0.35, 0.65)
    a, b = pool[i], pool[j]
    ma, mb = a[0][:, 0] < t, b[0][:, 0] >= t
    c = np.concatenate([a[0][ma], b[0][mb]])
    r = np.concatenate([a[1][ma], b[1][mb]])
    if len(r) > n:
        keep = np.sort(np.argsort(-r)[:n])
        c, r = c[keep], r[keep]
    while len(r) < n:
        c = np.vstack([c, rng.uniform(0.05, 0.95, (1, 2))])
        r = np.append(r, 0.03)
    return c, r

    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
