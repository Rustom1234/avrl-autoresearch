def _lp_radii(c):
    """Best radii for fixed centres: maximise the sum of radii under pair and wall constraints."""
    n = len(c)
    i, j = np.triu_indices(n, 1)
    d = np.sqrt(((c[i] - c[j]) ** 2).sum(1))
    A = np.zeros((len(i), n))
    A[np.arange(len(i)), i] = 1
    A[np.arange(len(i)), j] = 1
    wall = np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1]))
    res = linprog(-np.ones(n), A_ub=A, b_ub=d, bounds=list(zip(np.zeros(n), np.maximum(wall, 0))), method="highs")
    return np.maximum(res.x, 0) if res.status == 0 else None
