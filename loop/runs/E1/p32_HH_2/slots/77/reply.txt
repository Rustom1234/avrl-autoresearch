def _diag_start(n, rng):
    """Start symmetric under reflection across y = x: n//2 centres in the lower triangle mirrored to (y, x)."""
    m = n // 2
    lower = np.sort(rng.uniform(INIT_LOW, 1 - INIT_LOW, (m, 2)), axis=1)[:, ::-1]
    parts = [lower, lower[:, ::-1]]
    if n % 2:
        t = rng.uniform(INIT_LOW, 1 - INIT_LOW)
        parts.append(np.array([[t, t]]))
    return np.concatenate(parts).ravel()
