def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    top, stale, hop = 0.0, 0, False
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        if stale >= STAGNATION:   # random sampling has stagnated: basin hop from the incumbent
            hop = True
        if hop:
            starts = [np.clip(best[0] + rng.normal(0, HOP_SCALE, (n, 2)), 0, 1).ravel()]
        else:   # random restarts
            pool = sorted((_relax(rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), n) for _ in range(POOL)), key=lambda t: t[0])
            starts = [p for _, p in pool[:KEEP]]
        for p in starts:
            c, r = _solve_once(n, p)
            c = np.clip(c, 0, 1)
            r = _repair(c, r)
            if best is None or r.sum() > best[1].sum():
                best = (c, r)
        stale = 0 if best[1].sum() > top + 1e-9 else stale + 1
        top = best[1].sum()
    return best
