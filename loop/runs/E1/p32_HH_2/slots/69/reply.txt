def _cma(n, rng, deadline, gens=4, lam=12, mu=4):
    """Small CMA-ES-style search over the centres; fitness is the relaxed penalty energy."""
    pool = [_relax(rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), n) for _ in range(POOL)]
    m = min(pool, key=lambda t: t[0])[1]
    sig = np.full(2 * n, 0.15)
    w = np.arange(mu, 0, -1, dtype=float)
    w /= w.sum()
    best = min(pool, key=lambda t: t[0])
    for _ in range(gens):
        if time.process_time() > deadline:
            break
        X = np.clip(m + sig * rng.standard_normal((lam, 2 * n)), 0, 1)
        res = sorted((_relax(x, n) for x in X), key=lambda t: t[0])
        best = min(best, res[0], key=lambda t: t[0])
        m = w @ np.array([p for _, p in res[:mu]])
        sig *= 0.9
    return best[1]
