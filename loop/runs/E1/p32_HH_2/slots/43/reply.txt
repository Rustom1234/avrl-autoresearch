def _relax(p, n):
    """Riesz p-energy continuation (p = 6, 12, 24, 48) with a soft wall; returns (energy, centres)."""
    iu = np.triu_indices(n, 1)
    a = (0.5 / np.sqrt(n)) ** 2

    def energy(q, s):
        c = q.reshape(n, 2)
        diff = c[iu[0]] - c[iu[1]]
        d2 = (diff ** 2).sum(1) + 1e-8
        x = (a / d2) ** (s / 2)
        lo, hi = np.maximum(0, -c), np.maximum(0, c - 1)
        g = (-s * (x / d2))[:, None] * diff
        grad = np.zeros((n, 2))
        np.add.at(grad, iu[0], g)
        np.add.at(grad, iu[1], -g)
        grad += 200 * (hi - lo)
        return x.sum() + 100 * ((lo ** 2).sum() + (hi ** 2).sum()), grad.ravel()

    c, f = p, None
    for s in (6, 12, 24, 48):
        res = minimize(energy, c, args=(s,), jac=True, method="L-BFGS-B", options={"maxiter": 200})
        c, f = res.x, res.fun
    return f, c
