def _polish(n, p, best):
    c, r = _solve_once(n, p)
    c = np.clip(c, 0, 1)
    r = _repair(c, r)
    if best is None or r.sum() > best[1].sum():
        best = (c, r)
    return best

    pop = [rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2)) for _ in range(DE_POP)]

    fit = np.array([e for e, _ in scored])
    pop = [x.reshape(n, 2) for _, x in scored]
