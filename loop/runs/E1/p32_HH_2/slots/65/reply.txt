def _power_start(n, rng):
    """Power-diagram start: sites weighted by squared target radii; cell centroids (grid approximation) are the centres."""
    sites = rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
    target = 0.05 + 0.2 * rng.beta(2, 3, n)
    w = target ** 2
    g = (np.arange(GRID) + 0.5) / GRID
    gx, gy = np.meshgrid(g, g)
    pts = np.stack([gx.ravel(), gy.ravel()], 1)
    owner = np.argmin(((pts[:, None] - sites[None]) ** 2).sum(-1) - w[None], 1)
    centers = sites.copy()
    for i in range(n):
        m = owner == i
        if m.any():
            centers[i] = pts[m].mean(0)
    r = np.minimum(target, np.minimum(np.minimum(centers[:, 0], centers[:, 1]), np.minimum(1 - centers[:, 0], 1 - centers[:, 1])))
    return centers.ravel(), np.maximum(r, 0)
