def _md_start(rng, n):
    """Thermal quench: velocity Verlet on soft repulsion with a decaying velocity scale; returns flat centres."""
    r0 = 0.5 / np.sqrt(n)
    iu = np.triu_indices(n, 1)
    dt, steps = 0.01, 300

    def force(c):
        diff = c[iu[0]] - c[iu[1]]
        d = np.sqrt((diff ** 2).sum(1) + 1e-12)
        ov = np.maximum(0, 2 * r0 - d)
        f = (2 * ov / d)[:, None] * diff
        F = np.zeros((n, 2))
        np.add.at(F, iu[0], f)
        np.add.at(F, iu[1], -f)
        return F + 2 * (np.maximum(0, r0 - c) - np.maximum(0, c - (1 - r0)))

    x = rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
    v = np.zeros((n, 2))
    F = force(x)
    for t in range(steps):
        v += 0.5 * dt * F
        x += dt * v
        F = force(x)
        v += 0.5 * dt * F
        v *= 1.5 * (1 - (t + 1) / steps)
    return x.ravel()
