def _compress(n, rng):
    """Monte Carlo compression: random moves accepted only if overlap-free, radii grown to fit; returns centres."""
    r0 = 0.5 / np.sqrt(n)
    c = rng.uniform(r0, 1 - r0, (n, 2))
    r = np.full(n, 0.5 * r0)
    for s in range(SWEEPS):
        step = 0.2 * r0 * (1 - s / SWEEPS) + 1e-4
        for i in rng.permutation(n):
            q = np.clip(c[i] + rng.normal(0, step, 2), r[i], 1 - r[i])
            d = np.sqrt(((c - q) ** 2).sum(1))
            d[i] = np.inf
            if np.all(d >= r + r[i]):
                c[i] = q
        for i in range(n):

            d[i] = np.inf
            r[i] = min(c[i].min(), (1 - c[i]).min(), (d - r).min())
    return c
