def _disperse(p, n):
    """Max-min-like dispersion: minimise a log-sum-exp of -(pair distances and wall distances) in stages of rising beta."""
    iu = np.triu_indices(n, 1)
    m_pairs = len(iu[0])

    def energy(p, beta):
        c = p.reshape(n, 2)
        diff = c[iu[0]] - c[iu[1]]
        d = np.sqrt((diff ** 2).sum(1) + 1e-12)
        t = np.concatenate([d, c[:, 0], 1 - c[:, 0], c[:, 1], 1 - c[:, 1]])
        z = -beta * t
        m = z.max()
        e = np.exp(z - m)
        s = e.sum()
        w = e / s
        g = np.zeros((n, 2))
        u = (-w[:m_pairs] / d)[:, None] * diff
        np.add.at(g, iu[0], u)
        np.add.at(g, iu[1], -u)
        gw = -w[m_pairs:]
        g[:, 0] += gw[:n] - gw[n:2 * n]
        g[:, 1] += gw[2 * n:3 * n] - gw[3 * n:]
        return (m + np.log(s)) / beta, g.ravel()

def _start_radii(p, n):
    """Initial SLSQP radii: half the nearest-neighbour distance, clipped to the walls."""
    c = p.reshape(n, 2)
    d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
    np.fill_diagonal(d, np.inf)
    walls = np.minimum(c, 1 - c).min(1)
    return np.maximum(np.minimum(d.min(1) / 2, walls), 0)
