def _repair(c, r):
    """Single in-order pass: each radius is capped by its wall distance and by the gap to every other circle."""
    n = len(r)
    r = np.maximum(np.minimum(r, np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1]))), 0)
    for i in range(n):
        d = np.sqrt(((c - c[i]) ** 2).sum(1))
        d[i] = np.inf
        r[i] = max(0.0, min(r[i], (d - r).min()))
    return r
