def _grow(p, n, max_steps=40):
    """Jamming-style growth: grow a common radius from 0.6 of the relaxed value while the overlap energy stays ~0."""
    iu = np.triu_indices(n, 1)

    def energy(q, R):
        c = q.reshape(n, 2)
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1) + 1e-12)
        ov = np.maximum(0, 2 * R - d)
        lo = np.maximum(0, R - c)
        hi = np.maximum(0, c - (1 - R))
        return (ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()

    c, R, f = p.reshape(n, 2), 0.6 * 0.5 / np.sqrt(n), 1.01
    for _ in range(max_steps):
        res = minimize(energy, c.ravel(), args=(R * f,), method="L-BFGS-B", options={"maxiter": 100})
        if res.fun < 1e-10:
            c, R = res.x.reshape(n, 2), R * f
        elif f > 1.001:
            f = 1.001
        else:
            break
    r = np.maximum(np.minimum(R, np.minimum(c, 1 - c).min(1)), 0)
    return c, r
