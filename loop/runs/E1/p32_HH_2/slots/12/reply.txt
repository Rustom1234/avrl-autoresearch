def _lattice_start(n, rng):
    """Jittered hexagonal-offset grid over [0.1, 0.9]^2: rows alternately shifted by half a spacing."""
    m = int(np.ceil(np.sqrt(n)))
    g = np.linspace(0.1, 0.9, m)
    xs, ys = np.meshgrid(g, g)
    pts = np.stack([xs.ravel(), ys.ravel()], 1)
    pts[:, 0] += (np.arange(m * m) // m) % 2 * (g[1] - g[0]) / 2
    pts = pts[rng.permutation(len(pts))[:n]] + rng.normal(0, 0.01, (n, 2))
    return np.clip(pts, 0.1, 0.9).ravel()
