def _fps(rng, n, m=2000):
    """Farthest-point sampling: greedily pick n of m uniform candidates, each maximising its min distance to those chosen."""
    cand = rng.uniform(INIT_LOW, 1 - INIT_LOW, (m, 2))
    chosen = [cand[0]]
    dmin = np.sqrt(((cand - cand[0]) ** 2).sum(1))
    for _ in range(n - 1):
        k = np.argmax(dmin)
        chosen.append(cand[k])
        dmin = np.minimum(dmin, np.sqrt(((cand - cand[k]) ** 2).sum(1)))
    return np.array(chosen).ravel()
