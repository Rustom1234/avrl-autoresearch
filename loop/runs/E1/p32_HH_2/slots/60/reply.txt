def _strat(rng, n):
    """Stratified starts: one jittered centre per cell of a ceil(sqrt(n)) grid, cells chosen by random permutation."""
    g = int(np.ceil(np.sqrt(n)))
    cells = rng.permutation(g * g)[:n]
    j = rng.uniform(0.1, 0.9, (n, 2))
    return ((np.stack([cells % g, cells // g], 1) + j) / g).ravel()
