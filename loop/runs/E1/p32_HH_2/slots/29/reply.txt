def _grow(c, r, rng):
    """Greedily grow each radius in random order to the largest feasible value, until no radius increases."""
    r = r.copy()
    for _ in range(100):
        improved = False
        for i in rng.permutation(len(r)):
            d = np.sqrt(((c - c[i]) ** 2).sum(1))
            d[i] = np.inf
            lim = min(c[i, 0], c[i, 1], 1 - c[i, 0], 1 - c[i, 1], (d - r).min()) - SHRINK
            if lim > r[i] + 1e-12:
                r[i] = lim
                improved = True
        if not improved:
            break
    return r
