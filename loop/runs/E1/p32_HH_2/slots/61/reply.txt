def _equal_radius(n, p):
    """Max-min pre-stage: maximise one shared radius t subject to non-overlap and wall constraints."""
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, t = v[:2 * n].reshape(n, 2), v[-1]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - 2 * t
        return np.concatenate([d, c[:, 0] - t, c[:, 1] - t, 1 - c[:, 0] - t, 1 - c[:, 1] - t])

    x0 = np.concatenate([p, [0.01]])
    res = minimize(lambda v: -v[-1], x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), max(res.x[-1], 0.0)
