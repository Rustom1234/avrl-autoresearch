def _penalty(v, n, iu, w):
    c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
    diff = c[iu[0]] - c[iu[1]]
    d = np.sqrt((diff ** 2).sum(1)) + 1e-12
    vi = np.maximum(0, r[iu[0]] + r[iu[1]] - d)
    a = 2 * w * vi
    gr = -np.ones(n)
    np.add.at(gr, iu[0], a)
    np.add.at(gr, iu[1], a)
    gc = np.zeros((n, 2))
    np.add.at(gc, iu[0], -(a / d)[:, None] * diff)
    np.add.at(gc, iu[1], (a / d)[:, None] * diff)
    B = np.maximum(0, r[:, None] - c)
    T = np.maximum(0, c + r[:, None] - 1)
    f = -r.sum() + w * ((vi @ vi) + (B ** 2).sum() + (T ** 2).sum())
    gr += 2 * w * (B.sum(1) + T.sum(1))
    gc += 2 * w * (T - B)
    return f, np.concatenate([gc.ravel(), gr])
