def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    archive, k = [], 0   # archive holds (sum, centres) of the best distinct packings
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        c0 = None
        if k >= WARM and archive:
            base = archive[rng.integers(len(archive))][1].copy()
            base = base + rng.normal(0, JITTER, base.shape)
            if rng.random() < REPLACE_P:
                base[rng.integers(n)] = rng.uniform(0.05, 0.95, 2)
            c0 = np.clip(base, 0.05, 0.95)
        c, r = _solve_once(n, rng, c0)
        k += 1
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
        if all(abs(r.sum() - s) > 1e-9 for s, _ in archive):
            archive.append((r.sum(), c))
            archive.sort(key=lambda a: -a[0])
            del archive[ARCHIVE:]
    return best
