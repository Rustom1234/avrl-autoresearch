def _init(n, rng):
    return np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])


def _penalty(v, n, iu, mu=10.0):
    """Sum of radii minus a squared pairwise-overlap penalty, with analytic gradient."""
    c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
    diff = c[iu[0]] - c[iu[1]]
    d = np.sqrt((diff ** 2).sum(1)) + 1e-12
    ov = np.maximum(0, r[iu[0]] + r[iu[1]] - d)
    f = -r.sum() + mu * (ov ** 2).sum()
    gr = -np.ones(n)
    np.add.at(gr, iu[0], 2 * mu * ov)
    np.add.at(gr, iu[1], 2 * mu * ov)
    w = (2 * mu * ov / d)[:, None] * diff
    gc = np.zeros((n, 2))
    np.add.at(gc, iu[0], -w)
    np.add.at(gc, iu[1], w)
    return f, np.concatenate([gc.ravel(), gr])

    iu = np.triu_indices(n, 1)
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    ests = []
