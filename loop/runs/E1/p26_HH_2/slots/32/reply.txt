def solve(n, seed, time_limit):
    rng, t0, best, longest = np.random.default_rng(seed), time.process_time(), None, 0.0
    while True:   # random restarts until the time is used, keeping a margin for the longest restart seen
        elapsed = time.process_time() - t0
        if elapsed >= TIME_FRACTION * time_limit or elapsed + longest > SAFE_FRACTION * time_limit:
            break
        ts = time.process_time()
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        longest = max(longest, time.process_time() - ts)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    return best
