def _symmetries(c):
    x, y = c[:, 0], c[:, 1]
    maps = [(x, y), (y, 1 - x), (1 - x, 1 - y), (1 - y, x), (1 - x, y), (x, 1 - y), (y, x), (1 - y, 1 - x)]
    return [np.stack(m, 1) for m in maps]


def solve(n, seed, time_limit):
    rng, t0 = np.random.default_rng(seed), time.process_time()
    budget = TIME_FRACTION * time_limit
    c, r = _solve_once(n, rng)                 # first random restart
    c = np.clip(c, 0, 1)
    best = (c, _repair(c, r))
    base_c, base_r = best
    for g in _symmetries(base_c):              # symmetry branch: jittered copies of the best packing
        if time.process_time() - t0 > SYM_FRACTION * budget:
            break
        x0 = np.concatenate([(g + rng.normal(0, SYM_JITTER, g.shape)).ravel(), base_r])
        c, r = _solve_once(n, rng, x0)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if r.sum() > best[1].sum():
            best = (c, r)
    while time.process_time() - t0 < budget:   # random restarts for the rest of the time
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if r.sum() > best[1].sum():
            best = (c, r)
    return best
