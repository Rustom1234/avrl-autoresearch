def _ring_start(n, rng):
    """Boundary-first start: about 4*sqrt(n) centres on the perimeter with a random offset, rest in the inner square."""
    m = min(n, int(round(4 * np.sqrt(n))))
    t = (np.arange(m) + rng.uniform()) / m * 4
    side, f = np.floor(t).astype(int), t % 1
    z, o = np.zeros(m), np.ones(m)
    x = np.choose(side, [f, o, 1 - f, z])
    y = np.choose(side, [z, f, o, 1 - f])
    inner = rng.uniform(0.25, 0.75, size=(n - m, 2))
    c = np.vstack([np.column_stack([x, y]), inner])
    return np.concatenate([c.ravel(), np.full(n, 0.02)])
