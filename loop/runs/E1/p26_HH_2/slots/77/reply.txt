def _voronoi_caps(c):
    """Per-circle radius caps: 0.9 x half the farthest distance from each centre to its Voronoi cell (grid estimate)."""
    g = (np.arange(200) + 0.5) / 200
    X, Y = np.meshgrid(g, g)
    P = np.stack([X.ravel(), Y.ravel()], 1)
    d = ((P[:, None, :] - c[None]) ** 2).sum(-1)
    owner = d.argmin(1)
    dist = np.sqrt(d[np.arange(len(P)), owner])
    far = np.zeros(len(c))
    np.maximum.at(far, owner, dist)
    return 0.9 * 0.5 * far
