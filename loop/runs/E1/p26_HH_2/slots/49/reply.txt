def _pool(n, rng):
    """Draw POOL random centre sets, rank them by squared pair overlap at radius POOL_R, keep the KEEP best."""
    c = rng.uniform(INIT_LOW, 1 - INIT_LOW, (POOL, n, 2))
    d = np.sqrt(((c[:, :, None] - c[:, None]) ** 2).sum(-1))
    ov = np.maximum(2 * POOL_R - d, 0)
    ov[:, np.arange(n), np.arange(n)] = 0
    e = (ov ** 2).sum((1, 2))
    return c[np.argsort(e)[:KEEP]]


def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    pool, k = _pool(n, rng), 0
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # restarts from the pool until the time is used
        c, r = _solve_once(n, pool[k % KEEP])
        k += 1
