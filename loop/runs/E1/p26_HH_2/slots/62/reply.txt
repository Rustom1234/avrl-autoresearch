def _rotate_cluster(c, r, rng):
    """Rotate the centres inside a random disc about a random point, then repair the radii."""
    c0 = rng.uniform(0, 1, 2)
    R = rng.uniform(0.2, 0.45)
    th = np.deg2rad(rng.uniform(-30, 30))
    rot = np.array([[np.cos(th), -np.sin(th)], [np.sin(th), np.cos(th)]])
    m = np.sqrt(((c - c0) ** 2).sum(1)) < R
    c = c.copy()
    c[m] = (c[m] - c0) @ rot.T + c0
    c = np.clip(c, 0, 1)
    return c, _repair(c, r.copy())


def solve(n, seed, time_limit):
    rng, t0, best, k = np.random.default_rng(seed), time.process_time(), None, 0
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        if best is not None and k % 2 == 1:
            c, r = _rotate_cluster(best[0], best[1], rng)
            c, r = _solve_once(n, rng, np.concatenate([c.ravel(), r]), [0.3, 1.0])
        else:
            c, r = _solve_once(n, rng)
        k += 1
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    return best
