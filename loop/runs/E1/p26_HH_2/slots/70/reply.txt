def _ring_centres(n, rng):
    k = int(rng.choice([2, 3, 4]))
    cuts = np.sort(rng.choice(np.arange(1, n), k - 1, replace=False))
    counts = np.diff(np.concatenate([[0], cuts, [n]]))
    scale = rng.uniform(0.25, 0.4)
    pts = []
    for j, m in enumerate(counts, 1):
        ang = rng.uniform(0, 2 * np.pi) + np.arange(m) * 2 * np.pi / m
        rad = scale * j / k
        pts.append(np.stack([0.5 + rad * np.cos(ang), 0.5 + rad * np.sin(ang)], 1))
    c = np.concatenate(pts) + rng.normal(0, 0.01, (n, 2))
    return np.clip(c, INIT_LOW, 1 - INIT_LOW)
