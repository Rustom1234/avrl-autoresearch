def _grid_start(rng):
    if rng.random() < 0.5:
        g = np.array([(i, j) for i in range(5) for j in range(5)], float)
        g = np.vstack([g, rng.uniform(0, 4, (1, 2))]) / 4
    else:
        g = np.array([(i, j) for i in range(6) for j in range(5)], float)
        g = np.delete(g, rng.choice(len(g), 4, replace=False), axis=0) / np.array([5, 4])
    c = np.clip(0.1 + 0.8 * g + rng.uniform(-0.05, 0.05, g.shape), 0, 1)
    return np.concatenate([c.ravel(), np.full(26, INIT_RADIUS)])
