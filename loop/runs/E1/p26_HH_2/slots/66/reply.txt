def solve(n, seed, time_limit):
    rng, t0, scouts = np.random.default_rng(seed), time.process_time(), []
    total = TIME_FRACTION * time_limit
    while time.process_time() - t0 < 0.25 * total:   # scout phase: short restarts
        c, r = _solve_once(n, rng, max_iter=60)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        scouts.append((r.sum(), c, r))
    scouts.sort(key=lambda s: -s[0])
    best = (scouts[0][1], scouts[0][2])
    for _, c0, r0 in scouts[:3]:   # deep phase: full continuation of the top 3 scout packings
        if time.process_time() - t0 >= total:
            break
        c, r = _solve_once(n, rng, x0=np.concatenate([c0.ravel(), r0]))
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if r.sum() > best[1].sum():
            best = (c, r)
    return best
