def _slsqp(n, x, caps):
    iu = np.triu_indices(n, 1)

    def cons(v, cap):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r, cap - r])

    for cap in caps:
        res = minimize(lambda v: -v[2 * n:].sum(), x, method="SLSQP",
                       constraints={"type": "ineq", "fun": lambda v, cap=cap: cons(v, cap)},
                       options={"maxiter": MAX_ITER})
        x = res.x
    return x


def _solve_once(n, rng):
    caps = list(np.arange(0.05, 0.4 + 1e-9, 0.02)) + [1.0]
    m = n + 1   # solve with one extra circle, then drop the smallest
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * m), np.full(m, INIT_RADIUS)])
    x = _slsqp(m, x0, caps)
    c, r = x[:2 * m].reshape(m, 2), x[2 * m:]
    keep = np.argsort(r)[1:]
    x = np.concatenate([c[keep].ravel(), r[keep]])
    x = _slsqp(n, x, [1.0])   # short polish with n circles
    return x[:2 * n].reshape(n, 2), x[2 * n:]
