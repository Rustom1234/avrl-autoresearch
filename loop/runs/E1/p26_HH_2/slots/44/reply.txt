def _greedy_start(n, rng):
    """Place circles one by one at the candidate with the largest feasible radius."""
    cand = rng.uniform(0, 1, (2000, 2))
    wall = np.minimum(np.minimum(cand[:, 0], cand[:, 1]), np.minimum(1 - cand[:, 0], 1 - cand[:, 1]))
    c, r, placed = np.full((n, 2), 0.5), np.zeros(n), []
    for i in rng.permutation(n):
        feas = wall.copy()
        if placed:
            idx = np.array(placed)
            d = np.sqrt(((cand[:, None, :] - c[idx][None]) ** 2).sum(-1)) - r[idx][None]
            feas = np.minimum(feas, d.min(1))
        k = int(np.argmax(feas))
        c[i], r[i] = cand[k], 0.9 * max(feas[k], 0.0)
        placed.append(i)
    return c, r
