def _slacks(v, n, iu):
    c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
    d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
    return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])


def _barrier(c, r):
    """Log-barrier refinement: keeps iterates strictly feasible, so the result stays valid."""
    n = len(r)
    iu = np.triu_indices(n, 1)
    x = np.concatenate([c.ravel(), r])
    if np.any(_slacks(x, n, iu) <= 0):
        return c, r

    def f(v, mu):
        s = _slacks(v, n, iu)
        if np.any(s <= 0):
            return np.inf
        return -v[2 * n:].sum() - mu * np.log(s).sum()
