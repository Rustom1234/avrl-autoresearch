def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    budget = TIME_FRACTION * time_limit
    while time.process_time() - t0 < budget:   # successive halving batches until the time is used
        scored = []
        for _ in range(N_STARTS):
            if time.process_time() - t0 >= budget:
                break
            x0 = _start(n, rng)
            c, r = _finish(n, x0, SHORT_ITER)
            scored.append((r.sum(), x0))
        scored.sort(key=lambda s: -s[0])
        for _, x0 in scored[:KEEP]:
            if best is not None and time.process_time() - t0 >= budget:
                break
            c, r = _finish(n, x0, MAX_ITER)
            if best is None or r.sum() > best[1].sum():
                best = (c, r)
    return best
