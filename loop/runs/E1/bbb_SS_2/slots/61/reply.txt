def knn_count_predict(train_smiles, y, test_smiles, k=15):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=2, fpSize=2048)
    def fp(smiles):
        return np.array([gen.GetCountFingerprintAsNumPy(Chem.MolFromSmiles(s)) for s in smiles], dtype=np.float64)
    A, B = fp(test_smiles), fp(train_smiles)
    l1 = cdist(A, B, "cityblock")
    s = A.sum(1)[:, None] + B.sum(1)[None, :]
    sim = (s - l1) / np.maximum(s + l1, 1e-9)
    idx = np.argpartition(-sim, k - 1, axis=1)[:, :k]
    w = np.take_along_axis(sim, idx, axis=1) ** 3 + 1e-9
    return (w * y[idx]).sum(1) / w.sum(1)
