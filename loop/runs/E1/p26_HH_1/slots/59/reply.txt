def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    walk, hist = None, []   # current walk's incumbent and its sum-of-radii history
    while time.process_time() - t0 < TIME_FRACTION * time_limit:
        if walk is None or (len(hist) >= 15 and hist[-1] - hist[-15] < 1e-6):
            c, r = _solve_once(n, rng)   # first walk or stalled walk: fresh random restart
            walk, hist = None, []
        else:              # local kick: re-randomise the centres in a disc, then re-optimise everything
            c, r = walk[0].copy(), walk[1].copy()
            p, R = rng.uniform(0, 1, 2), rng.uniform(0.15, 0.35)
            m = np.hypot(*(c - p).T) < R
            k = m.sum()
            ang, rad = rng.uniform(0, 2 * np.pi, k), R * np.sqrt(rng.uniform(0, 1, k))
            c[m] = np.clip(p + np.c_[rad * np.cos(ang), rad * np.sin(ang)], 0, 1)
            r[m] = 0
            c, r = _solve_once(n, rng, np.concatenate([c.ravel(), r]))
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if walk is None or r.sum() > walk[1].sum():
            walk = (c, r)
        hist.append(walk[1].sum())
        if best is None or walk[1].sum() > best[1].sum():
            best = walk
    return best
