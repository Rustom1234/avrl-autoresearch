def _polish(c, r):
    """Final trust-constr polish of one packing, with the same objective and stacked constraints as the SLSQP phase."""
    n = len(r)
    iu = np.triu_indices(n, 1)
    m = len(iu[0])

    def cons(v):
        c_, r_ = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c_[iu[0]] - c_[iu[1]]) ** 2).sum(1)) - r_[iu[0]] - r_[iu[1]]
        return np.concatenate([d, c_[:, 0] - r_, c_[:, 1] - r_, 1 - c_[:, 0] - r_, 1 - c_[:, 1] - r_, r_])

    def cons_jac(v):
        c_, r_ = v[:2 * n].reshape(n, 2), v[2 * n:]
        diff = c_[iu[0]] - c_[iu[1]]
        d = np.maximum(np.sqrt((diff ** 2).sum(1)), 1e-12)
        u = diff / d[:, None]
        J = np.zeros((m + 5 * n, 3 * n))
        k, i, j, a = np.arange(m), iu[0], iu[1], np.arange(n)
        J[k, 2 * i], J[k, 2 * i + 1] = u[:, 0], u[:, 1]
        J[k, 2 * j], J[k, 2 * j + 1] = -u[:, 0], -u[:, 1]
        J[k, 2 * n + i] = -1
        J[k, 2 * n + j] = -1
        o = m
        J[o + a, 2 * a] = 1
        J[o + a, 2 * n + a] = -1
        J[o + n + a, 2 * a + 1] = 1
        J[o + n + a, 2 * n + a] = -1
        J[o + 2 * n + a, 2 * a] = -1
        J[o + 2 * n + a, 2 * n + a] = -1
        J[o + 3 * n + a, 2 * a + 1] = -1
        J[o + 3 * n + a, 2 * n + a] = -1
        J[o + 4 * n + a, 2 * n + a] = 1
        return J

    grad = np.concatenate([np.zeros(2 * n), -np.ones(n)])
    v0 = np.concatenate([c.ravel(), r])
    res = minimize(lambda v: -v[2 * n:].sum(), v0, jac=lambda v: grad, method="trust-constr",
                   constraints=[NonlinearConstraint(cons, 0, np.inf, jac=cons_jac)],
                   options={"gtol": 1e-10, "xtol": 1e-12, "maxiter": 200})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
