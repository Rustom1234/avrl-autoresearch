def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    budget = TIME_FRACTION * time_limit

    def run(x0=None):
        c, r = _solve_once(n, rng, x0)
        c = np.clip(c, 0, 1)
        return c, _repair(c, r)

    # initial random restarts
    while best is None or time.process_time() - t0 < 0.3 * budget:
        c, r = run()
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    # monotone basin hopping from the incumbent
    while time.process_time() - t0 < budget:
        c, r = best
        sigma = rng.uniform(0.02, 0.05)
        c2 = c + rng.normal(0, sigma, c.shape)
        r2 = r.copy()
        idx = rng.choice(n, size=rng.integers(0, 3), replace=False)
        r2[idx] = INIT_RADIUS
        x0 = np.concatenate([np.clip(c2, 0, 1).ravel(), r2])
        c3, r3 = run(x0)
        if r3.sum() > best[1].sum():
            best = (c3, r3)
    return best
