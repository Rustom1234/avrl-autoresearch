def _penalty_descent(n, v):
    """Cheap L-BFGS-B descent on -sum(r) plus a quadratic overlap and wall penalty (weight 1e3)."""
    iu = np.triu_indices(n, 1)

    def f(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        walls = np.concatenate([c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r])
        pen = np.minimum(np.concatenate([d, walls]), 0)
        return -r.sum() + 1e3 * (pen ** 2).sum()

    return minimize(f, v, method="L-BFGS-B", options={"maxiter": 300}).x
