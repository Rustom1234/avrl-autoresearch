def _polish(n, x0):

def _solve_sym(n, rng):
    """Mirror-symmetric stage about x = 0.5: solve on reduced variables, then unconstrained polish."""
    m, odd = n // 2, n % 2
    R0 = 2 * m + odd                      # reduced number of centre coordinates
    M, b = np.zeros((3 * n, R0 + n)), np.zeros(3 * n)
    for k in range(m):
        M[2 * k, 2 * k] = M[2 * k + 1, 2 * k + 1] = 1
        M[2 * (m + k), 2 * k] = -1        # mirror image: x -> 1 - x
        b[2 * (m + k)] = 1
        M[2 * (m + k) + 1, 2 * k + 1] = 1
    if odd:                               # centre on the axis x = 0.5
        b[2 * n - 2] = 0.5
        M[2 * n - 1, 2 * m] = 1
    M[2 * n:, R0:] = np.eye(n)
    cons = _cons(n)
    u0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, R0), np.full(n, INIT_RADIUS)])
    res = minimize(lambda u: -u[R0:].sum(), u0, method="SLSQP",
                   constraints={"type": "ineq", "fun": lambda u: cons(M @ u + b)}, options={"maxiter": MAX_ITER})
    return _polish(n, M @ res.x + b)
