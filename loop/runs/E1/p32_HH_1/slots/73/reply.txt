def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    pulls, gains = np.zeros(2), np.zeros(2)   # arm 0: fresh random start, arm 1: guided from incumbent
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        T = pulls.sum()
        if best is None:
            arm = 0
        elif pulls.min() < 2:
            arm = int(np.argmin(pulls))
        else:
            arm = int(np.argmax(gains / pulls + np.sqrt(2 * np.log(T) / pulls)))
        c, r = _solve_once(n, rng, best[0] if arm == 1 else None)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        pulls[arm] += 1
        if best is None:
            best = (c, r)
        elif r.sum() > best[1].sum():
            gains[arm] += r.sum() - best[1].sum()
            best = (c, r)
    return best
