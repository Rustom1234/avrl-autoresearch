def _spectral(c, r, rng):
    """Displace centres along random low-frequency eigenvectors of the contact-graph Laplacian."""
    n = len(r)
    d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
    gap = d - r[:, None] - r[None]
    A = ((gap < 1e-3) & ~np.eye(n, dtype=bool)).astype(float)
    _, V = np.linalg.eigh(np.diag(A.sum(1)) - A)
    out = np.empty_like(c)
    for a in range(2):
        k = rng.integers(1, min(6, n))
        out[:, a] = c[:, a] + 0.05 * rng.choice([-1.0, 1.0]) * V[:, k]
    return np.clip(out, 0, 1)
