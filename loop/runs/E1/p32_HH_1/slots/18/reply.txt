def _halton_pool(rng, m):
    """m scrambled Halton points (bases 2 and 3) over [0.05, 0.95]^2; digit permutations and start index from rng."""
    idx = rng.integers(1, 10000) + np.arange(m)
    cols = []
    for b in (2, 3):
        perm = rng.permutation(b)
        f, r, i = 1.0, np.zeros(m), idx.copy()
        while i.any():
            f /= b
            r += f * perm[i % b]
            i //= b
        cols.append(r)
    return 0.05 + 0.9 * np.column_stack(cols)
