def solve(n, seed, time_limit):
    rng, t0 = np.random.default_rng(seed), time.process_time()
    budget = TIME_FRACTION * time_limit
    pop, sigma = [], 0.03   # pop: list of (radius sum, centres, radii), best first
    while time.process_time() - t0 < budget:
        if time.process_time() - t0 < 0.4 * budget or len(pop) < 8:   # random restarts, then population phase
            c, r = _solve_once(n, rng)
            c = np.clip(c, 0, 1)
            r = _repair(c, r)
        else:
            parent = pop[rng.integers(len(pop))]
            c, r = _solve_once(n, rng, parent[1], sigma)
            c = np.clip(c, 0, 1)
            r = _repair(c, r)
            ok = r.sum() > parent[0]
            sigma = float(np.clip(sigma * (1.22 if ok else 1.22 ** -0.25), 1e-3, 0.2))   # 1/5th-style rule
        pop.append((r.sum(), c, r))
        pop.sort(key=lambda p: -p[0])
        pop = pop[:8]
    return pop[0][1], pop[0][2]
