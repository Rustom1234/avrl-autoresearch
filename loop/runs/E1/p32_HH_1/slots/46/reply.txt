def _relocate(n, c, r, rng):
    """Move the smallest circle into the largest hole (sampled), then re-solve once."""
    k = int(np.argmin(r))
    keep = np.arange(n) != k
    pts = rng.uniform(0, 1, (2000, 2))
    clear = np.sqrt(((pts[:, None] - c[keep][None]) ** 2).sum(-1)) - r[keep][None]
    gap = np.minimum(clear.min(1), np.minimum(pts, 1 - pts).min(1))
    c2, r2 = c.copy(), r.copy()
    c2[k], r2[k] = pts[np.argmax(gap)], 0.0
    return _slsqp(n, np.concatenate([c2.ravel(), r2]))
