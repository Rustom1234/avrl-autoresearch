def _class_stage(n, rng, base):
    """Size-class homotopy stage: circles in the same class share one radius; k is 2 or 3."""
    k = int(rng.choice([2, 3]))
    cuts = np.sort(rng.choice(np.arange(1, n), k - 1, replace=False))
    sizes = np.diff(np.concatenate([[0], cuts, [n]]))
    lab = rng.permutation(np.repeat(np.arange(k), sizes))
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, R = v[:2 * n].reshape(n, 2), v[2 * n:]
        r = R[lab]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, R])

    x0 = np.concatenate([base.ravel(), np.full(k, INIT_RADIUS)])
    res = minimize(lambda v: -(sizes * v[2 * n:]).sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                   options={"maxiter": MAX_ITER})
    return res.x[:2 * n].reshape(n, 2), res.x[2 * n:][lab]
