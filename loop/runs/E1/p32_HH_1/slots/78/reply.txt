def _mc_start(n, rng, sweeps=800, sigma=0.01):
    """Hard-disk NPT Monte Carlo compression; returns unit-square centres and half-gap radii."""
    L, c = 1.0, rng.uniform(0.1, 0.9, (n, 2))
    for k in range(sweeps):
        pressure = 10.0 ** (1 + 5 * k / (sweeps - 1))
        for _ in range(n):
            i = rng.integers(n)
            new = c[i] + rng.uniform(-0.5 * sigma, 0.5 * sigma, 2)
            if new.min() < sigma / 2 or new.max() > L - sigma / 2:
                continue
            d2 = ((c - new) ** 2).sum(1)
            d2[i] = np.inf
            if d2.min() < sigma * sigma:
                continue
            c[i] = new
        eps = rng.uniform(-0.02, 0.02)
        L2, c2 = L * np.exp(eps), c * np.exp(eps)
        if c2.min() < sigma / 2 or c2.max() > L2 - sigma / 2:
            continue
        d2 = ((c2[:, None] - c2[None]) ** 2).sum(-1)
        np.fill_diagonal(d2, np.inf)
        if d2.min() < sigma * sigma:
            continue
        if rng.random() < np.exp(-pressure * (L2 * L2 - L * L) + 2 * n * eps):
            c, L = c2, L2
    u = c / L
    D = np.sqrt(((u[:, None] - u[None]) ** 2).sum(-1))
    np.fill_diagonal(D, np.inf)
    wall = np.minimum(u, 1 - u).min(1)
    return u, np.minimum(0.5 * D.min(1), wall)
