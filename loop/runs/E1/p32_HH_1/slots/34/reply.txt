def _best_candidate(n, rng, k=20):
    """Mitchell's best-candidate rule: each new centre is the candidate farthest from the centres placed so far."""
    pts = np.empty((n, 2))
    pts[0] = rng.uniform(INIT_LOW, 1 - INIT_LOW, 2)
    for i in range(1, n):
        cand = rng.uniform(INIT_LOW, 1 - INIT_LOW, (k, 2))
        d = np.sqrt(((cand[:, None] - pts[None, :i]) ** 2).sum(-1)).min(1)
        pts[i] = cand[np.argmax(d)]
    return pts
