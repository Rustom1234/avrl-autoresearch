def _quadrant_init(n, rng):
    """Split the circles into four random groups, pack each group in its own quadrant, and merge the centres."""
    labels = rng.permutation(n) % 4
    centers = np.zeros((n, 2))
    for q in range(4):
        idx = np.where(labels == q)[0]
        if len(idx) == 0:
            continue
        c, _ = _solve_once(len(idx), rng)
        centers[idx] = 0.5 * np.clip(c, 0, 1) + np.array([(q % 2) * 0.5, (q // 2) * 0.5])
    return centers
