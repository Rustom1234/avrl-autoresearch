def _langevin(n, rng, c, r, steps=400, dt=1e-3, t_hi=0.05, t_lo=1e-5):
    """Overlap-energy Langevin dynamics with geometrically decaying temperature; returns the relaxed start."""
    eye = np.eye(n)
    temps = t_hi * (t_lo / t_hi) ** np.linspace(0, 1, steps)
    for T in temps:
        d = c[:, None] - c[None]
        dist = np.sqrt((d ** 2).sum(-1) + eye)
        ov = np.maximum(r[:, None] + r[None] - dist, 0) * (1 - eye)
        gc = -2 * ((ov / dist)[:, :, None] * d).sum(1)
        gr = 2 * ov.sum(1)
        lx, rx = np.maximum(r - c[:, 0], 0), np.maximum(c[:, 0] + r - 1, 0)
        ly, ry = np.maximum(r - c[:, 1], 0), np.maximum(c[:, 1] + r - 1, 0)
        gc[:, 0] += -2 * lx + 2 * rx
        gc[:, 1] += -2 * ly + 2 * ry
        gr += 2 * (lx + rx + ly + ry) - 1
        noise = rng.normal(size=(n, 3)) * np.sqrt(2 * T * dt)
        c = c - dt * gc + noise[:, :2]
        r = r - dt * gr + noise[:, 2]
    return np.clip(c, 0, 1), np.maximum(r, 0)
