def _solve_once(n, rng, init=None, init_r=None):
    rad = np.full(n, INIT_RADIUS)
    if init is None:
        base = rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
    else:
        base = init.copy()
        k = rng.integers(1, 4)
        w = 1.0 / (init_r + 1e-3)
        idx = rng.choice(n, size=k, replace=False, p=w / w.sum())
        for i in idx:
            if rng.random() < 0.5:
                base[i] = rng.uniform(0, 1, 2)
            else:
                step = 0.5 * (1 - rng.random()) ** (-1 / 1.5)
                ang = rng.uniform(0, 2 * np.pi)
                base[i] = base[i] + step * np.array([np.cos(ang), np.sin(ang)])
            base[i] = np.clip(base[i], 0, 1)
        rad[idx] = init_r.min()
    x0 = np.concatenate([base.ravel(), rad])
