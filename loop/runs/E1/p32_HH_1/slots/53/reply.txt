def _solve_shell(n, rng, k=20):
    """Solve a k-circle boundary shell first, then insert the remaining circles in the interior."""
    m = rng.uniform(0.05, 0.15)
    s, t = 1 - 2 * m, rng.uniform(0, 4, k)
    e, f, z = np.floor(t).astype(int), t - np.floor(t), np.zeros(k)
    edges = np.array([[f, z], [z + 1, f], [1 - f, z + 1], [z, 1 - f]])  # shape (4, 2, k)
    p = edges[e, :, np.arange(k)]  # shape (k, 2), random points on the rectangle boundary
    cs, rs = _slsqp(np.clip(m + s * p, 0, 1), np.full(k, INIT_RADIUS))
    ci = rng.uniform(0.2, 0.8, (n - k, 2))
    return _slsqp(np.clip(np.vstack([cs, ci]), 0, 1), np.concatenate([rs, np.full(n - k, INIT_RADIUS)]))
