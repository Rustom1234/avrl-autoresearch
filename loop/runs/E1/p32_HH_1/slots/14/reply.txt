def _destroy(c, r, rng):
    """Re-place the circles whose centres lie in a random disc uniformly in that disc, with radius 0.01."""
    p, R = rng.uniform(0, 1, 2), rng.uniform(0.2, 0.4)
    c, r = c.copy(), r.copy()
    idx = np.where(np.hypot(*(c - p).T) < R)[0]
    ang, rad = rng.uniform(0, 2 * np.pi, len(idx)), R * np.sqrt(rng.uniform(0, 1, len(idx)))
    c[idx] = np.clip(p + np.stack([rad * np.cos(ang), rad * np.sin(ang)], 1), 0, 1)
    r[idx] = 0.01
    return c, r
