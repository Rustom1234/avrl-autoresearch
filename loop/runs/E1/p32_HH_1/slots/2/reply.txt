def _hex_init(n, rng):
    k = int(np.ceil(np.sqrt(n))) + 2
    i, j = np.meshgrid(np.arange(-k, k + 1), np.arange(-k, k + 1))
    pts = np.stack([i + 0.5 * j, j * np.sqrt(3) / 2], -1).reshape(-1, 2)
    t = rng.uniform(0, np.pi / 3)
    pts = pts @ np.array([[np.cos(t), -np.sin(t)], [np.sin(t), np.cos(t)]]).T
    pts = pts[np.argsort((pts ** 2).sum(1))[:n]]
    s = rng.uniform(0.9, 1.1) * np.sqrt(2 / (np.sqrt(3) * n))
    return np.clip(pts * s + 0.5, 0, 1)
