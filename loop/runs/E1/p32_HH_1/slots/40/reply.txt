def _riesz_init(n, rng):
    """Centres minimising a Riesz s-energy with random exponent s; the box bounds act as the wall term."""
    s = rng.uniform(1, 8)

    def energy(x):
        c = x.reshape(n, 2)
        diff = c[:, None] - c[None]
        d2 = (diff ** 2).sum(-1) + np.eye(n)
        e = d2 ** (-s / 2)
        w = e / d2
        np.fill_diagonal(e, 0)
        np.fill_diagonal(w, 0)
        grad = -2 * s * (w[:, :, None] * diff).sum(1)
        return e.sum(), grad.ravel()

    res = minimize(energy, rng.uniform(0.05, 0.95, 2 * n), jac=True, method="L-BFGS-B",
                   bounds=[(0.05, 0.95)] * (2 * n), options={"maxiter": 100})
    return res.x.reshape(n, 2)
