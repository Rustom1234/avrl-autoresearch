def _lp_repair(c, r):
    """Keep centres fixed and re-solve the radii as an LP that maximises their sum. Returns None on failure."""
    n = len(r)
    wall = np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1]))
    iu = np.triu_indices(n, 1)
    d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1))
    m = len(d)
    A = np.zeros((m, n))
    A[np.arange(m), iu[0]] = 1
    A[np.arange(m), iu[1]] = 1   # all pairs kept, so the LP stays feasible for any radii it returns
    res = linprog(-np.ones(n), A_ub=A, b_ub=d, bounds=list(zip(np.zeros(n), np.maximum(wall, 0))), method="highs")
    if res.status != 0:
        return None
    return np.clip(res.x - 1e-12, 0, np.maximum(wall, 0))
