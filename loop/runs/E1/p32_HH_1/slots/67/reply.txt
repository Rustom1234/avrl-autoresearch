def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    pool = []   # (repaired sum, centres, radii) of the stored packings, best first
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        guided = best is not None and time.process_time() - t0 > 0.5 * TIME_FRACTION * time_limit and rng.random() < 0.5
        parent = None
        if guided:
            sums = np.array([p[0] for p in pool])
            w = np.exp((sums - best[1].sum()) / WEIGHT_T)
            parent = pool[rng.choice(len(pool), p=w / w.sum())][1]
        c, r = _solve_once(n, rng, parent)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        pool.append((r.sum(), c, r))
        pool.sort(key=lambda p: -p[0])
        del pool[POOL_SIZE:]
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    return best
