def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    tt_gen = rdFingerprintGenerator.GetTopologicalTorsionGenerator(fpSize=1024)
    rows = []
    for s in smiles:
        mol = Chem.MolFromSmiles(s)
        rows.append(np.concatenate([gen.GetFingerprintAsNumPy(mol),
                                    tt_gen.GetCountFingerprintAsNumPy(mol),
                                    np.array(descriptors(mol), dtype=np.float32)]))
    return np.array(rows, dtype=np.float32)
