def scaffold_keys(smiles_list):
    murcko, generic = [], []
    for s in smiles_list:
        scaf = MurckoScaffold.GetScaffoldForMol(Chem.MolFromSmiles(s))
        murcko.append(Chem.MolToSmiles(scaf))
        generic.append(Chem.MolToSmiles(MurckoScaffold.MakeScaffoldGeneric(scaf)))
    return murcko, generic


def target_encode(fit_keys, fit_y, apply_keys, global_mean, smooth=5.0):
    sums, counts = {}, {}
    for k, t in zip(fit_keys, fit_y):
        sums[k] = sums.get(k, 0.0) + float(t)
        counts[k] = counts.get(k, 0) + 1
    enc = np.zeros(len(apply_keys))
    cnt = np.zeros(len(apply_keys))
    for i, k in enumerate(apply_keys):
        n = counts.get(k, 0)
        enc[i] = (sums.get(k, 0.0) + smooth * global_mean) / (n + smooth)
        cnt[i] = n
    return enc, cnt


def oof_scaffold_encode(generic_keys, groups, y, n_splits=5):
    keys = np.asarray(generic_keys, dtype=object)
    enc = np.zeros(len(y))
    cnt = np.zeros(len(y))
    for tr, va in GroupKFold(n_splits=n_splits).split(keys, y, np.asarray(groups)):
        e, c = target_encode(keys[tr], y[tr], keys[va], y[tr].mean())
        enc[va] = e
        cnt[va] = c
    return enc, cnt
