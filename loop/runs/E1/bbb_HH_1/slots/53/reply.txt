def cross_features(Ct, Ce, y):
    r = np.array([np.corrcoef(Ct[:, j], y)[0, 1] if Ct[:, j].std() > 0 else 0.0 for j in range(Ct.shape[1])])
    r = np.nan_to_num(r)
    top = np.argsort(-np.abs(r))[:12]
    mu = Ct[:, top].mean(axis=0)
    sd = Ct[:, top].std(axis=0) + 1e-9
    Zt = (Ct[:, top] - mu) / sd
    Ze = (Ce[:, top] - mu) / sd

    def cross(Z):
        pi, pj = np.triu_indices(12, 1)
        qi, qj = np.triu_indices(4, 1)
        return np.hstack([Z[:, pi] * Z[:, pj], Z[:, qi] - Z[:, qj]])

    return cross(Zt), cross(Ze)
