def density_weights(D):
    # Anti-clustering weights: mean distance to 20 nearest training neighbours (standardized descriptors).
    mu = D.mean(axis=0)
    sd = D.std(axis=0)
    sd[sd == 0] = 1.0
    Z = (D - mu) / sd
    nn = NearestNeighbors(n_neighbors=21).fit(Z)
    dist, _ = nn.kneighbors(Z)
    dens = dist[:, 1:].mean(axis=1)
    w = dens / dens.mean()
    w = np.clip(w, 0.2, 5.0)
    return w / w.mean()
