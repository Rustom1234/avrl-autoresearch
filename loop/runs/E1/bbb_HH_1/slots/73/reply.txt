    # Fisher discriminant direction with Ledoit-Wolf shrinkage on standardized descriptors
    sc = StandardScaler().fit(D_fit)
    lda = LinearDiscriminantAnalysis(solver='lsqr', shrinkage='auto')
    lda.fit(sc.transform(D_fit), y_fit)
    return lda.decision_function(sc.transform(D_eval))
