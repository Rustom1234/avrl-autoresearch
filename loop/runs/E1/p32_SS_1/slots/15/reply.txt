def _mixed_start(n, rng):
    r = np.sort(np.exp(rng.uniform(np.log(0.02), np.log(0.12), n)))[::-1]
    c = np.zeros((n, 2))
    for i in range(n):
        best, bo = None, None
        for _ in range(20):
            p = rng.uniform(r[i], 1 - r[i], 2)
            o = 0.0 if i == 0 else np.max(r[:i] + r[i] - np.sqrt(((c[:i] - p) ** 2).sum(1)))
            if best is None or o < bo:
                best, bo = p, o
            if o <= 0:
                break
        c[i] = best
    return np.concatenate([c.ravel(), r])
