def _relax(c0, n, iu, R=0.09):
    def f(v):
        c = v.reshape(n, 2)
        diff = c[iu[0]] - c[iu[1]]
        d = np.sqrt((diff ** 2).sum(1)) + 1e-12
        ov = np.maximum(2 * R - d, 0)
        lo, hi = np.maximum(R - c, 0), np.maximum(c - (1 - R), 0)
        e = (ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()
        g = np.zeros((n, 2))
        gp = (-2 * ov / d)[:, None] * diff
        np.add.at(g, iu[0], gp)
        np.add.at(g, iu[1], -gp)
        g += -2 * lo + 2 * hi
        return e, g.ravel()

    res = minimize(f, c0.ravel(), jac=True, method="L-BFGS-B", bounds=[(0, 1)] * (2 * n),
                   options={"maxiter": 30})
    return res.x, res.fun
