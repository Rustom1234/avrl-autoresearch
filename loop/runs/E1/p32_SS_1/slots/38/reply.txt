def _relax(c0, n, r0=0.085):
    def f(v):
        c = v.reshape(n, 2)
        diff = c[:, None] - c[None]
        d = np.sqrt((diff ** 2).sum(-1) + 1e-12)
        ov = np.maximum(2 * r0 - d, 0)
        np.fill_diagonal(ov, 0)
        lo, hi = np.maximum(r0 - c, 0), np.maximum(c - (1 - r0), 0)
        e = 0.5 * (ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()
        g = -2 * ((ov / d)[:, :, None] * diff).sum(1) - 2 * lo + 2 * hi
        return e, g.ravel()

    res = minimize(f, c0.ravel(), jac=True, method="L-BFGS-B", options={"maxiter": 200})
    return np.clip(res.x.reshape(n, 2), INIT_LOW / 2, 1 - INIT_LOW / 2)


def _fresh(n, rng):
    if rng.random() < 0.5:
        rows = int(rng.integers(5, 8))
        c = []
        for k, idx in enumerate(np.array_split(np.arange(n), rows)):
            m = len(idx)
            xs = (np.arange(m) + 0.25 + 0.5 * (k % 2)) / m
            c.extend((x, (k + 0.5) / rows) for x in xs)
        c = np.clip(np.array(c) + rng.normal(0, 0.03, (n, 2)), 0.05, 0.95)
    else:
        c = rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
    return np.concatenate([_relax(c, n).ravel(), np.full(n, INIT_RADIUS)])
