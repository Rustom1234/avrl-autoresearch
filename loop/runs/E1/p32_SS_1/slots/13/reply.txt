def _relax(x, n, steps=40):
    """Gradient descent on -sum(r) + mu * squared violations, mu rising 10 -> 1000."""
    c, r = x[:2 * n].reshape(n, 2).copy(), x[2 * n:].copy()
    eye = np.eye(n)
    for mu in np.geomspace(10, 1000, steps):
        diff = c[:, None] - c[None]
        d = np.sqrt((diff ** 2).sum(-1)) + eye
        over = np.maximum(r[:, None] + r[None] - d, 0) * (1 - eye)
        gr = -1 + 2 * mu * over.sum(1)
        gc = -2 * mu * ((over / d)[:, :, None] * diff).sum(1)
        for k in range(2):
            lo = np.maximum(r - c[:, k], 0)
            hi = np.maximum(r + c[:, k] - 1, 0)
            gr += 2 * mu * (lo + hi)
            gc[:, k] += 2 * mu * (hi - lo)
        lr = 0.1 / mu
        c -= lr * gc
        r = np.maximum(r - lr * gr, 0)
    return np.concatenate([c.ravel(), r])
