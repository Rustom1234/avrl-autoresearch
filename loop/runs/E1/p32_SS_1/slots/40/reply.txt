    """Spread random centres by minimising a quadratic overlap/wall penalty with radii fixed at R."""
    def f(x):
        c = x.reshape(n, 2)
        diff = c[:, None] - c[None]
        d = np.sqrt((diff ** 2).sum(-1)) + np.eye(n)
        ov = np.maximum(2 * R - d, 0)
