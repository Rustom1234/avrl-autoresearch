def _lp(c):
    from scipy.optimize import linprog
    n = len(c)
    iu = np.triu_indices(n, 1)
    d = np.maximum(np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)), 0)
    A = np.zeros((len(d), n))
    A[np.arange(len(d)), iu[0]] = 1
    A[np.arange(len(d)), iu[1]] = 1
    w = np.maximum(np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1])), 0)
    res = linprog(-np.ones(n), A_ub=A, b_ub=d, bounds=list(zip(np.zeros(n), w)), method="highs")
    return res.x if res.success else None
