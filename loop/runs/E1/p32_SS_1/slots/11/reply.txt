def _prerelax(n, rng):
    """Push random centres apart with a smooth overlap penalty while the common radius grows."""
    x = rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n)
    for r in (0.05, 0.075, 0.09):
        def f(x, r=r):
            c = x.reshape(n, 2)
            diff = c[:, None] - c[None]
            d = np.sqrt((diff ** 2).sum(-1)) + np.eye(n)
            ov = np.maximum(2 * r - d, 0)
            np.fill_diagonal(ov, 0)
            lo, hi = np.maximum(r - c, 0), np.maximum(c + r - 1, 0)
            e = 0.5 * (ov ** 2).sum() + 0.5 * (lo ** 2 + hi ** 2).sum()
            g = -(ov / d)[:, :, None] * diff
            return e, (g.sum(1) - lo + hi).ravel()
        x = minimize(f, x, jac=True, method="L-BFGS-B", options={"maxiter": 100}).x
    return x
