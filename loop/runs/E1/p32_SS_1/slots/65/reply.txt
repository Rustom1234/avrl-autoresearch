def _relax(c, rad=0.09):
    """Relax centres by L-BFGS-B on a quadratic overlap/wall penalty with equal radii."""
    n = len(c)
    iu = np.triu_indices(n, 1)

    def f(v):
        p = v.reshape(n, 2)
        diff = p[iu[0]] - p[iu[1]]
        d = np.sqrt((diff ** 2).sum(1)) + 1e-12
        o = np.maximum(2 * rad - d, 0)
        lo, hi = np.maximum(rad - p, 0), np.maximum(p - (1 - rad), 0)
        e = (o ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()
        g = -2 * lo + 2 * hi
        w = (-2 * o / d)[:, None] * diff
        np.add.at(g, iu[0], w)
        np.add.at(g, iu[1], -w)
        return e, g.ravel()

    res = minimize(f, c.ravel(), jac=True, method="L-BFGS-B", options={"maxiter": 100})
    return res.x.reshape(n, 2)


def _solve_once(n, rng, x0=None):
    if x0 is None:
        c0 = _relax(rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2)))
