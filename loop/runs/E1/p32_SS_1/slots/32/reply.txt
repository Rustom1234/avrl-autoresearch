def _relax(c0, n, r0=0.09):
    iu = np.triu_indices(n, 1)

    def f(x):
        c = x.reshape(n, 2)
        dv = c[iu[0]] - c[iu[1]]
        d = np.sqrt((dv ** 2).sum(1)) + 1e-12
        o = np.maximum(2 * r0 - d, 0)
        lo = np.maximum(r0 - c, 0)
        hi = np.maximum(c - (1 - r0), 0)
        e = (o ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()
        g = -2 * lo + 2 * hi
        w = (-2 * o / d)[:, None] * dv
        np.add.at(g, iu[0], w)
        np.add.at(g, iu[1], -w)
        return e, g.ravel()

    res = minimize(f, c0.ravel(), jac=True, method="L-BFGS-B", bounds=[(0, 1)] * (2 * n),
                   options={"maxiter": 100})
    return res.x.reshape(n, 2)


def _lattice(n, rng):
    k = int(rng.integers(5, 8))
    m = -(-n // k)
    pts = np.array([[(j + 0.5 + 0.5 * (i % 2)) / (m + 0.5), (i + 0.5) / k] for i in range(k) for j in range(m)])
    pts = pts[rng.choice(len(pts), n, replace=False)]
    return np.clip(pts + rng.normal(0, 0.02, pts.shape), INIT_LOW / 2, 1 - INIT_LOW / 2)
