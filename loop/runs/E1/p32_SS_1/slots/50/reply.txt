def _fresh(n, rng):
    R = rng.uniform(0.05, 0.13, n)
    R *= np.sqrt(0.8 / (np.pi * (R ** 2).sum()))
    RR = R[:, None] + R[None]

    def energy(x):
        c = x.reshape(n, 2)
        diff = c[:, None] - c[None]
        d = np.sqrt((diff ** 2).sum(-1) + 1e-12)
        ov = np.maximum(RR - d, 0)
        np.fill_diagonal(ov, 0)
        lo, hi = np.maximum(R[:, None] - c, 0), np.maximum(c + R[:, None] - 1, 0)
        e = 0.5 * (ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()
        g = -2 * (ov / d)[:, :, None] * diff
        return e, g.sum(1).ravel() - 2 * lo.ravel() + 2 * hi.ravel()

    c0 = rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n)
    res = minimize(energy, c0, jac=True, method="L-BFGS-B", options={"maxiter": 200})
    return np.concatenate([np.clip(res.x, 0, 1), 0.8 * R])
