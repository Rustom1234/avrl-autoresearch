def _lloyd(n, rng, iters=10, g=100):
    p = rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
    s = (np.arange(g) + 0.5) / g
    grid = np.stack(np.meshgrid(s, s), -1).reshape(-1, 2)
    for _ in range(iters):
        lab = ((grid[:, None] - p[None]) ** 2).sum(-1).argmin(1)
        for k in range(n):
            m = lab == k
            if m.any():
                p[k] = grid[m].mean(0)
    return p
