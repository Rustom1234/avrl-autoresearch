def _relax(n, rng, rad=0.09):
    iu = np.triu_indices(n, 1)

    def energy(p):
        c = p.reshape(n, 2)
        diff = c[iu[0]] - c[iu[1]]
        d = np.sqrt((diff ** 2).sum(1)) + 1e-12
        ov = np.maximum(2 * rad - d, 0)
        g = np.zeros_like(c)
        w = (-2 * ov / d)[:, None] * diff
        np.add.at(g, iu[0], w)
        np.add.at(g, iu[1], -w)
        lo, hi = np.maximum(rad - c, 0), np.maximum(c - (1 - rad), 0)
        e = (ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()
        g += -2 * lo + 2 * hi
        return e, g.ravel()
