def _relax(c0, t, iters=60):
    n = len(t)
    iu = np.triu_indices(n, 1)
    s = t[iu[0]] + t[iu[1]]

    def f(v):
        c = v.reshape(n, 2)
        dv = c[iu[0]] - c[iu[1]]
        d = np.sqrt((dv ** 2).sum(1)) + 1e-12
        o = np.maximum(s - d, 0)
        lo = np.maximum(t[:, None] - c, 0)
        hi = np.maximum(c + t[:, None] - 1, 0)
        e = (o ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()
        w = (-2 * o / d)[:, None] * dv
        g = -2 * lo + 2 * hi
        np.add.at(g, iu[0], w)
        np.add.at(g, iu[1], -w)
        return e, g.ravel()

    res = minimize(f, c0.ravel(), jac=True, method="L-BFGS-B", options={"maxiter": iters})
    return np.clip(res.x.reshape(n, 2), 0.01, 0.99)
