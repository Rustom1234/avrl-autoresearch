def _reinsert(pool, n, rng):
    c, r = pool[0]
    keep = np.argsort(r)[rng.integers(2, 4):]
    c, r = c[keep], r[keep]
    while len(r) < n:
        cand = rng.uniform(0, 1, (200, 2))
        d = np.sqrt(((cand[:, None] - c[None]) ** 2).sum(-1)) - r[None]
        score = np.minimum(d.min(1), np.minimum(cand.min(1), (1 - cand).min(1)))
        c = np.vstack([c, cand[np.argmax(score)]])
        r = np.append(r, 0.02)
    return np.concatenate([c.ravel(), r])
