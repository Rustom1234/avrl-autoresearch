def _fresh(n, rng, R=0.09):
    iu = np.triu_indices(n, 1)

    def f(z):
        c = z.reshape(n, 2)
        diff = c[iu[0]] - c[iu[1]]
        d = np.sqrt((diff ** 2).sum(1) + 1e-12)
        ov = np.maximum(2 * R - d, 0)
        lo, hi = np.maximum(R - c, 0), np.maximum(c - (1 - R), 0)
        g = np.zeros_like(c)
        gp = (-2 * ov / d)[:, None] * diff
        np.add.at(g, iu[0], gp)
        np.add.at(g, iu[1], -gp)
        g += -2 * lo + 2 * hi
        return (ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum(), g.ravel()
