def _relax(n, rng, R=0.09):
    """Relax random centres with L-BFGS-B on a quadratic overlap/wall penalty (all radii R)."""
    def f(v):
        c = v.reshape(n, 2)
        D = c[:, None] - c[None]
        d = np.sqrt((D ** 2).sum(-1)) + np.eye(n)
        o = np.clip(2 * R - d, 0, None)
        np.fill_diagonal(o, 0)
        lo, hi = np.clip(R - c, 0, None), np.clip(c + R - 1, 0, None)
        e = 0.5 * (o ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()
        g = (-2 * o[:, :, None] * D / d[:, :, None]).sum(1) - 2 * lo + 2 * hi
        return e, g.ravel()

    res = minimize(f, rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), jac=True, method="L-BFGS-B",
                   options={"maxiter": 100})
    return np.concatenate([np.clip(res.x, 0.02, 0.98), np.full(n, INIT_RADIUS)])
