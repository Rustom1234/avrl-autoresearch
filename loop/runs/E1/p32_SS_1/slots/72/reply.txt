def _relax(c0, n, R=0.09):
    iu = np.triu_indices(n, 1)

    def f(v):
        c = v.reshape(n, 2)
        diff = c[iu[0]] - c[iu[1]]
        d = np.sqrt((diff ** 2).sum(1)) + 1e-12
        ov = np.maximum(0, 2 * R - d)
        g = np.zeros((n, 2))
        coef = (-2 * ov / d)[:, None] * diff
        np.add.at(g, iu[0], coef)
        np.add.at(g, iu[1], -coef)
        lo, hi = np.maximum(0, R - c), np.maximum(0, c + R - 1)
        e = (ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()
        g += -2 * lo + 2 * hi
        return e, g.ravel()

    res = minimize(f, c0.ravel(), jac=True, method="L-BFGS-B", options={"maxiter": 60})
    return res.x.reshape(n, 2)
