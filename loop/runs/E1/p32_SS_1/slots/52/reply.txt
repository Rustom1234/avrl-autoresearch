def _cross(pool, n, rng):
    """Relocation: move one circle (chosen with prob ~ 1/radius) to a random place with a small radius."""
    c, r = pool[rng.integers(len(pool))]
    c, r = c.copy(), r.copy()
    p = 1.0 / np.maximum(r, 1e-3)
    k = rng.choice(n, p=p / p.sum())
    c[k] = rng.uniform(INIT_LOW, 1 - INIT_LOW, 2)
    r[k] = 0.02
    return np.concatenate([c.ravel(), r])

        d = np.sqrt((dv ** 2).sum(1)) + 1e-12
