_TCACHE = {}


def atypes_one(smi):
    # Proxy for Crippen atom types (RDKit does not expose the type labels directly).
    if smi not in _TCACHE:
        m = Chem.MolFromSmiles(smi)
        _TCACHE[smi] = Counter(
            (a.GetSymbol(), a.GetIsAromatic(), a.GetTotalNumHs(), a.GetDegree()) for a in m.GetAtoms())
    return _TCACHE[smi]
