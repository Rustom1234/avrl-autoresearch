    X_tr, X_te = X_tr[:, keep], X_te[:, keep]
    qt = QuantileTransformer(n_quantiles=len(X_tr), output_distribution='normal', random_state=seed)
    X_tr = qt.fit_transform(X_tr)
    X_te = qt.transform(X_te)
