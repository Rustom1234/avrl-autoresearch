def _darts(n, rng):
    """Sequential dart throwing, largest first, of a bimodal set of target discs."""
    k = round(0.375 * n)
    rt = np.array([0.11] * k + [0.06] * (n - k))
    c, i, fails = np.zeros((n, 2)), 0, 0
    while i < n:
        p = rng.uniform(rt[i], 1 - rt[i], 2)
        if i == 0 or (np.hypot(*(c[:i] - p).T) >= rt[:i] + rt[i]).all():
            c[i], i, fails = p, i + 1, 0
        else:
            fails += 1
            if fails >= 200:
                rt, fails = rt * 0.9, 0
    return c, rt
