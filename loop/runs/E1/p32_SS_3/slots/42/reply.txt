def _compress(n, rng, steps=150):
    p = rng.uniform(0.1, 0.9, (n, 2))
    for k in range(steps):
        rad = 0.09 * (k + 1) / steps
        dv = p[:, None] - p[None]
        d = np.sqrt((dv ** 2).sum(-1)) + np.eye(n)
        f = np.where(d < 4 * rad, 1 / d ** 3, 0.0)
        np.fill_diagonal(f, 0)
        F = (f[..., None] * dv / d[..., None]).sum(1)
        p = np.clip(p + np.clip(1e-5 * F, -0.01, 0.01), rad, 1 - rad)
    return p.ravel()
