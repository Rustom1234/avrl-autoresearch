def _local(n, rng, c, r):
    """SLSQP on a random cluster (a circle and its 8 nearest neighbours); other circles frozen."""
    i = rng.integers(n)
    k = np.argsort(((c - c[i]) ** 2).sum(1))[:9]
    f = np.setdiff1d(np.arange(n), k)
    m = len(k)
    ii, jj = np.triu_indices(m, 1)
    cf, rf = c[f], r[f]
    x0 = np.concatenate([np.clip(c[k] + rng.normal(0, 0.03, (m, 2)), 0.01, 0.99).ravel(), r[k]])

    def cons(v):
        ck, rk = v[:2 * m].reshape(m, 2), v[2 * m:]
        d1 = np.sqrt(((ck[ii] - ck[jj]) ** 2).sum(1)) - rk[ii] - rk[jj]
        d2 = np.sqrt(((ck[:, None] - cf[None]) ** 2).sum(-1)) - rk[:, None] - rf[None]
        return np.concatenate([d1, d2.ravel(), ck[:, 0] - rk, ck[:, 1] - rk, 1 - ck[:, 0] - rk,
                               1 - ck[:, 1] - rk, rk])

    res = minimize(lambda v: -v[2 * m:].sum(), x0, method="SLSQP",
                   constraints={"type": "ineq", "fun": cons}, options={"maxiter": 100})
    c2, r2 = c.copy(), r.copy()
    c2[k] = res.x[:2 * m].reshape(m, 2)
    r2[k] = res.x[2 * m:]
    return c2, r2
