def _polish(best, n, t0, time_limit):
    from scipy.optimize import NonlinearConstraint, LinearConstraint
    iu = np.triu_indices(n, 1)
    m = len(iu[0])
    rows = np.arange(m)

    def f(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        return np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]

    def jac(v):
        c = v[:2 * n].reshape(n, 2)
        diff = c[iu[0]] - c[iu[1]]
        u = diff / np.maximum(np.sqrt((diff ** 2).sum(1)), 1e-12)[:, None]
        J = np.zeros((m, 3 * n))
        for k in range(2):
            J[rows, 2 * iu[0] + k] = u[:, k]
            J[rows, 2 * iu[1] + k] = -u[:, k]
        J[rows, 2 * n + iu[0]] = -1
        J[rows, 2 * n + iu[1]] = -1
        return J

    A = np.zeros((4 * n, 3 * n))
    for i in range(n):
        A[i, 2 * i], A[i, 2 * n + i] = 1, -1
        A[n + i, 2 * i + 1], A[n + i, 2 * n + i] = 1, -1
        A[2 * n + i, 2 * i], A[2 * n + i, 2 * n + i] = -1, -1
        A[3 * n + i, 2 * i + 1], A[3 * n + i, 2 * n + i] = -1, -1
    lb = np.concatenate([np.zeros(2 * n), -np.ones(2 * n)])
    x0 = np.concatenate([best[0].ravel(), best[1]])
    deadline = t0 + 0.95 * time_limit
    res = minimize(lambda v: -v[2 * n:].sum(), x0, jac=lambda v: np.concatenate([np.zeros(2 * n), -np.ones(n)]),
                   method="trust-constr",
                   constraints=[NonlinearConstraint(f, 0, np.inf, jac=jac), LinearConstraint(A, lb, np.inf)],
                   callback=lambda xk, st: time.process_time() > deadline,
                   options={"maxiter": 300})
    c = np.clip(res.x[:2 * n].reshape(n, 2), 0, 1)
    r = _repair(c, res.x[2 * n:].copy())
    return (c, r) if r.sum() > best[1].sum() else best
