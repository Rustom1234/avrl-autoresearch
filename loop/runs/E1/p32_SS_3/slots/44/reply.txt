def _reinsert(best, rng):
    c, r = best[0].copy(), best[1].copy()
    idx = np.argsort(r)[:rng.integers(1, 4)]
    active = np.ones(len(r), bool)
    active[idx] = False
    for i in idx:
        cand = rng.uniform(0, 1, (2000, 2))
        d = np.sqrt(((cand[:, None] - c[active][None]) ** 2).sum(-1)) - r[active][None]
        clear = np.minimum(d.min(1), np.minimum(cand, 1 - cand).min(1))
        k = np.argmax(clear)
        c[i], r[i] = cand[k], max(clear[k], 0.01)
        active[i] = True
    return c, r
