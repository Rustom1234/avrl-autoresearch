from xgboost import XGBRegressor
from sklearn.kernel_ridge import KernelRidge

KRR_ALPHA = 0.3


def tanimoto(A, B):
    A = (A > 0).astype(np.float64)
    B = (B > 0).astype(np.float64)
    inter = A @ B.T
    denom = A.sum(1)[:, None] + B.sum(1)[None, :] - inter
    return inter / np.maximum(denom, 1e-12)
