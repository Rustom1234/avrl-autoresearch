def fit_predict(train_smiles, train_y, test_smiles, seed):
    names = [n for n, _ in Descriptors.descList]
    mono = [0] * (N_BITS + len(names))
    for name in ('TPSA', 'NumHDonors'):
        mono[N_BITS + names.index(name)] = -1
    model = lgb.LGBMRegressor(objective='l1', learning_rate=0.03, n_estimators=600, max_depth=6,
                              subsample=0.8, subsample_freq=1, colsample_bytree=0.3, min_child_samples=3,
                              monotone_constraints=mono, random_state=seed, n_jobs=8, verbose=-1)
    model.fit(featurize(train_smiles), np.asarray(train_y, dtype=np.float64))
    return model.predict(featurize(test_smiles))
