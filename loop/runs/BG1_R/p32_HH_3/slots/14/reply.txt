def _greedy_start(n, rng):
    """Place circles one at a time in decreasing target radius, each at the grid point with the largest clearance."""
    g = np.linspace(0, 1, 200)
    P = np.stack(np.meshgrid(g, g), -1).reshape(-1, 2)
    wall = np.minimum(np.minimum(P[:, 0], P[:, 1]), np.minimum(1 - P[:, 0], 1 - P[:, 1]))
    targets = np.sort(rng.uniform(0.05, 0.25, n))[::-1]
    c, r = np.zeros((0, 2)), np.zeros(0)
    for t in targets:
        clear = wall.copy()
        if len(r):
            clear = np.minimum(clear, (np.sqrt(((P[:, None] - c[None]) ** 2).sum(-1)) - r[None]).min(1))
        k = np.argmax(clear)
        c = np.vstack([c, P[k]])
        r = np.append(r, max(min(clear[k], t), 0))
    return np.concatenate([c.ravel(), r])
