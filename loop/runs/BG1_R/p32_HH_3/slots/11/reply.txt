def _shrink_start(n, rng):
    """Solve n+1 circles, drop the smallest one, and return the remaining n circles as a start vector."""
    c, r = _solve_once(n + 1, rng)
    k = np.argmin(r)
    c, r = np.delete(c, k, axis=0), np.delete(r, k)
    return np.concatenate([np.clip(c, 0, 1).ravel(), np.maximum(r, 0)])
