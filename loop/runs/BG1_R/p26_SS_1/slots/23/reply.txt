def _hex_centres(n, rng):
    rows = rng.permutation([5, 5, 6, 5, 5]) if n == 26 else None
    if rows is None:
        return rng.uniform(INIT_LOW, 1 - INIT_LOW, (n, 2))
    pts = []
    for i, k in enumerate(rows):
        y = (i + 0.5) / len(rows)
        xs = (np.arange(k) + (0.5 if k == 6 else 0.5 + 0.5 * (i % 2 - 0.5) / k)) / k
        pts += [(x, y) for x in xs]
    pts = np.array(pts) + rng.normal(0, 0.03, (n, 2))
    return np.clip(pts, 0.03, 0.97)
