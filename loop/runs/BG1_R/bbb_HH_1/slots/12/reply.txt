def knn_score(X_train, y, X_test, k=10):
    A = X_train[:, :N_BITS]
    B = X_test[:, :N_BITS]
    inter = B @ A.T
    union = B.sum(1)[:, None] + A.sum(1)[None, :] - inter
    sim = inter / np.maximum(union, 1e-9)
    idx = np.argsort(-sim, axis=1)[:, :k]
    w = np.take_along_axis(sim, idx, axis=1) ** 3
    return (w * y[idx]).sum(1) / np.maximum(w.sum(1), 1e-12)
