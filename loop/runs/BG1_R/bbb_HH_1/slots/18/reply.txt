def physchem_stack(Xb_train, T_train, Xb_test, seed, alpha=10.0):
    # Out-of-fold Ridge predictions of physchem targets from Morgan bits (training rows),
    # and full-fold Ridge predictions for the test rows.
    oof = np.zeros((len(Xb_train), T_train.shape[1]))
    test = np.zeros((len(Xb_test), T_train.shape[1]))
    kf = KFold(n_splits=5, shuffle=True, random_state=seed)
    for j in range(T_train.shape[1]):
        t = T_train[:, j]
        for tr, va in kf.split(Xb_train):
            oof[va, j] = Ridge(alpha=alpha).fit(Xb_train[tr], t[tr]).predict(Xb_train[va])
        test[:, j] = Ridge(alpha=alpha).fit(Xb_train, t).predict(Xb_test)
    return oof, test
