    labels = []

        labels.append(_h((atom.GetSymbol(), atom.GetDegree(), atom.GetFormalCharge(),
                          atom.GetIsAromatic(), crip_bin)))
    all_labels = list(labels)
    for _ in range(N_WL_ROUNDS):
        labels = [_h((labels[a.GetIdx()], sorted(labels[n.GetIdx()] for n in a.GetNeighbors())))
                  for a in mol.GetAtoms()]
        all_labels.extend(labels)
    counts = np.zeros(N_WL_BINS, dtype=np.float32)
    for lab in all_labels:
        counts[lab % N_WL_BINS] += 1
    return counts
