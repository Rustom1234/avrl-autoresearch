_SHAPE_CACHE = {}


def shape_features(smiles):
    try:
        mol = Chem.AddHs(Chem.MolFromSmiles(smiles))
        ps = AllChem.ETKDGv3()
        ps.randomSeed = 42
        if AllChem.EmbedMolecule(mol, ps) != 0:
            return [0.0] * 8
        AllChem.MMFFOptimizeMolecule(mol, maxIters=200)
        return [rdMolDescriptors.CalcPMI1(mol), rdMolDescriptors.CalcPMI2(mol), rdMolDescriptors.CalcPMI3(mol),
                rdMolDescriptors.CalcNPR1(mol), rdMolDescriptors.CalcNPR2(mol), rdMolDescriptors.CalcPBF(mol),
                rdMolDescriptors.CalcRadiusOfGyration(mol), rdMolDescriptors.CalcAsphericity(mol)]
    except Exception:
        return [0.0] * 8

    for s in smiles_list:

    arr = np.array([_SHAPE_CACHE[s] for s in smiles_list], dtype=np.float64)
