def _relax(c0, r):
    n = len(r)
    iu = np.triu_indices(n, 1)

    def f(x):
        c = x.reshape(n, 2)
        diff = c[iu[0]] - c[iu[1]]
        d = np.sqrt((diff ** 2).sum(1)) + 1e-12
        ov = np.maximum(0, r[iu[0]] + r[iu[1]] - d)
        lo = np.maximum(0, r[:, None] - c)
        hi = np.maximum(0, c + r[:, None] - 1)
        e = (ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()
        coef = (-2 * ov / d)[:, None] * diff
        g = -2 * lo + 2 * hi
        np.add.at(g, iu[0], coef)
        np.add.at(g, iu[1], -coef)
        return e, g.ravel()

    res = minimize(f, c0.ravel(), jac=True, method="L-BFGS-B", options={"maxiter": 200})
    return res.x.reshape(n, 2)
