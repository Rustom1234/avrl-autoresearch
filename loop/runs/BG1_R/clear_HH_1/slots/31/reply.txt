def laplacian(A, B, gamma, chunk=64):
    out = np.empty((A.shape[0], B.shape[0]))
    for i in range(0, A.shape[0], chunk):
        d = np.abs(A[i:i + chunk, None, :] - B[None, :, :]).sum(axis=2)
        out[i:i + chunk] = np.exp(-gamma * d)
    return out
