def _child(n, rng, pool):
    a, b = rng.choice(len(pool), 2, replace=False)
    (ca, ra), (cb, rb) = pool[a], pool[b]
    t = rng.uniform(0.3, 0.7)
    ma, mb = ca[:, 0] < t, cb[:, 0] >= t
    c = np.concatenate([ca[ma], cb[mb]])
    r = np.concatenate([ra[ma], rb[mb]]) * 0.8
    if len(c) > n:
        k = rng.permutation(len(c))[:n]
        c, r = c[k], r[k]
    elif len(c) < n:
        extra = n - len(c)
        c = np.concatenate([c, rng.uniform(0.05, 0.95, (extra, 2))])
        r = np.concatenate([r, np.full(extra, 0.03)])
    return c, r
