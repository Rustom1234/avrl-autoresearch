def _child(n, rng, A, B):
    t = rng.uniform(0.35, 0.65)
    ma, mb = A[1][:, 0] < t, B[1][:, 0] >= t
    c = np.vstack([A[1][ma], B[1][mb]])
    r = np.concatenate([A[2][ma], B[2][mb]])
    if len(r) > n:
        keep = np.argsort(-r)[:n]
        c, r = c[keep], r[keep]
    elif len(r) < n:
        k = n - len(r)
        c = np.vstack([c, rng.uniform(0.05, 0.95, (k, 2))])
        r = np.concatenate([r, np.full(k, 0.02)])
    return c, 0.9 * r


def _insert(elite, c, r):
    s = r.sum()
    if any(abs(s - e[0]) < 1e-7 for e in elite):
        return
    elite.append((s, c, r))
    elite.sort(key=lambda e: -e[0])
    del elite[5:]
