def _drop_start(n, rng):
    """Packs n+1 circles, drops the smallest-radius circle, and returns the remaining n centres (flat)."""
    _, p = _relax(rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * (n + 1)), n + 1)
    c, r = _solve_once(n + 1, p)
    c = np.clip(c, 0, 1)
    r = _repair(c, r)
    return c[np.argsort(r)[1:]].ravel()
