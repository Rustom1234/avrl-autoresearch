def _halton(m, rng):
    """m scrambled Halton points in 2D (bases 2 and 3), mapped into [INIT_LOW, 1 - INIT_LOW]."""
    offset = rng.integers(1, 10000)
    pts = np.zeros((m, 2))
    for dim, base in enumerate((2, 3)):
        perm = np.concatenate([[0], 1 + rng.permutation(base - 1)])
        for i in range(m):
            k, f, x = i + offset, 1.0, 0.0
            while k > 0:
                f /= base
                x += f * perm[k % base]
                k //= base
            pts[i, dim] = x
    return INIT_LOW + (1 - 2 * INIT_LOW) * pts
