def _equal_radius(n, p):
    """Phase one: bisect for the largest common radius at which the centres can be made feasible; returns (r, centres)."""
    iu = np.triu_indices(n, 1)

    def energy(q, r):
        c = q.reshape(n, 2)
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1) + 1e-12)
        ov = np.maximum(0, 2 * r - d)
        lo = np.maximum(0, r - c)
        hi = np.maximum(0, c - (1 - r))
        return (ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()

    lo_r, hi_r, q = 0.0, 0.5 / np.sqrt(n) * 2, p
    for _ in range(20):
        mid = 0.5 * (lo_r + hi_r)
        res = minimize(energy, q, args=(mid,), method="L-BFGS-B", options={"maxiter": 200})
        if res.fun < 1e-12:
            lo_r, q = mid, res.x
        else:
            hi_r = mid
    return lo_r, q
