def solve(n, seed, time_limit):
    rng, t0, pool = np.random.default_rng(seed), time.process_time(), []
    total = TIME_FRACTION * time_limit
    while not pool or time.process_time() - t0 < TIER1_FRACTION * total:   # tier 1: many cheap restarts
        c, r = _solve_once(n, rng, maxiter=CHEAP_ITER)
        c = np.clip(c, 0, 1)
        rr = _repair(c, r)
        pool.append((rr.sum(), c, rr, r))
    pool.sort(key=lambda p: -p[0])
    best = (pool[0][1], pool[0][2])
    for _, c, rr, r in pool[:TOP_K]:   # tier 2: re-optimise the top few with the full cap
        if time.process_time() - t0 > total:
            break
        c2, r2 = _solve_once(n, rng, x0=np.concatenate([c.ravel(), r]), maxiter=MAX_ITER)
        c2 = np.clip(c2, 0, 1)
        r2 = _repair(c2, r2)
        if r2.sum() > best[1].sum():

    return best
