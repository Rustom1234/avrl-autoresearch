def _penalty(x, r, iu):
    """Quadratic overlap and wall-violation penalty at fixed radii, with analytic gradient."""
    n = len(r)
    c = x.reshape(n, 2)
    diff = c[iu[0]] - c[iu[1]]
    d = np.sqrt((diff ** 2).sum(1)) + 1e-12
    ov = np.maximum(0, r[iu[0]] + r[iu[1]] - d)
    lo = np.maximum(0, r[:, None] - c)
    hi = np.maximum(0, c + r[:, None] - 1)
    f = (ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum()
    coef = -2 * ov[:, None] * diff / d[:, None]
    g = np.zeros((n, 2))
    np.add.at(g, iu[0], coef)
    np.add.at(g, iu[1], -coef)
    g += -2 * lo + 2 * hi
    return PENALTY_WEIGHT * f, PENALTY_WEIGHT * g.ravel()
