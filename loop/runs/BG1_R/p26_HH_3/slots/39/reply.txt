def _penalise(n, x0):
    """Penalty continuation: L-BFGS-B on -sum(r) + mu * squared violations, for increasing mu."""
    iu = np.triu_indices(n, 1)

    def pen(v, mu):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        ov = np.maximum(0, r[iu[0]] + r[iu[1]] - np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)))
        bd = np.concatenate([np.maximum(0, r - c[:, 0]), np.maximum(0, r - c[:, 1]),
                             np.maximum(0, c[:, 0] + r - 1), np.maximum(0, c[:, 1] + r - 1), np.maximum(0, -r)])
        return -r.sum() + mu * ((ov ** 2).sum() + (bd ** 2).sum())

    v = x0
    for mu in (10, 100, 1000, 1e4):
        v = minimize(pen, v, args=(mu,), method="L-BFGS-B", options={"maxiter": 200}).x
    return v
