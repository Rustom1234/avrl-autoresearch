def _spread_start(n, rng, k=2000, steps=30):
    """Pick the most spread-out of k random candidates after a cheap repulsion relaxation."""
    c = rng.uniform(0.05, 0.95, (k, n, 2))
    eye = np.eye(n)
    mask = eye.astype(bool)
    for _ in range(steps):
        diff = c[:, :, None] - c[:, None, :]
        d2 = (diff ** 2).sum(-1) + eye
        c = np.clip(c + 1e-4 * (diff / d2[..., None] ** 1.5).sum(2), 0.05, 0.95)
    diff = c[:, :, None] - c[:, None, :]
    d = np.sqrt((diff ** 2).sum(-1))
    d[:, mask] = np.inf
    pair = d.min(axis=(1, 2))
    wall = np.minimum(c, 1 - c).min(axis=(1, 2))
    best = np.argmax(np.minimum(pair / 2, wall))
    return c[best], np.full(n, min(pair[best] / 2, 0.1))
