def _solve_once(n, rng, extra=3):
    m = n + extra
    x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * m), np.full(m, INIT_RADIUS)])
    c, r = _optimise(m, x0)
    c = np.clip(c, 0, 1)
    r = _repair(c, r)
    while len(r) > n:   # drop the smallest circle and re-optimise
        i = np.argmin(r)
        c, r = np.delete(c, i, 0), np.delete(r, i)
        c, r = _optimise(len(r), np.concatenate([c.ravel(), r]))
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
    return c, r
