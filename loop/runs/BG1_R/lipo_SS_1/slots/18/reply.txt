def minmax_kernel(A, B, tmax=10):
    A, B = np.minimum(A, tmax), np.minimum(B, tmax)
    inter = np.zeros((len(A), len(B)), dtype=np.float32)
    for t in range(1, tmax + 1):
        inter += (A >= t).astype(np.float32) @ (B >= t).astype(np.float32).T
    union = A.sum(1)[:, None] + B.sum(1)[None, :] - inter
    return inter / np.maximum(union, 1e-9)
