def solve(n, seed, time_limit):
    rng, t0, best, scores = np.random.default_rng(seed), time.process_time(), None, []
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        v = _solve_once(n, rng, maxiter=PROBE_ITER)   # short probe run
        scores.append(v[2 * n:].sum())
        if scores[-1] >= np.quantile(scores, 0.75):   # continue only promising basins
            v = _solve_once(n, rng, x0=v, maxiter=MAX_ITER)
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        c = np.clip(c, 0, 1)
