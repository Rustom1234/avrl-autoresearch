def _crossover(pa, pb, rng):
    """Child start: circles left of a random cut from packing a, the rest from packing b; count fixed to n."""
    ca, cb = pa[0], pb[0]
    n = len(ca)
    axis, cut = rng.integers(2), rng.uniform(0.3, 0.7)
    child = np.vstack([ca[ca[:, axis] < cut], cb[cb[:, axis] >= cut]])
    if len(child) > n:
        child = child[rng.permutation(len(child))[:n]]
    elif len(child) < n:
        child = np.vstack([child, rng.uniform(0.1, 0.9, (n - len(child), 2))])
    return child
