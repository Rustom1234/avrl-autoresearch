def solve(n, seed, time_limit):
    rng, t0, pop = np.random.default_rng(seed), time.process_time(), []
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # population search until the time is used
        full = len(pop) >= POP_SIZE
        c, r = _solve_once(n, rng, pop[rng.integers(len(pop))][0] if full else None)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if not full:
            pop.append((c, r))
        else:
            w = min(range(len(pop)), key=lambda i: pop[i][1].sum())
            if r.sum() > pop[w][1].sum():
                pop[w] = (c, r)
    return max(pop, key=lambda p: p[1].sum())
