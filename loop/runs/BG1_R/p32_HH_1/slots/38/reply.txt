def _greedy_start(n, rng):
    """Place circles one at a time at the grid point with the largest clearance (walls and placed centres)."""
    g = np.linspace(0, 1, 60)
    G = np.stack(np.meshgrid(g, g), -1).reshape(-1, 2)
    wall = np.minimum(np.minimum(G[:, 0], G[:, 1]), np.minimum(1 - G[:, 0], 1 - G[:, 1]))
    tie = rng.uniform(0, 1e-9, len(G))
    dmin = np.full(len(G), np.inf)
    c, r = np.empty((n, 2)), np.empty(n)
    for k in range(n):
        i = np.argmax(np.minimum(wall, dmin) + tie)
        r[k] = 0.9 * min(wall[i], dmin[i])
        c[k] = G[i]
        dmin = np.minimum(dmin, np.sqrt(((G - G[i]) ** 2).sum(1)))
    return c, r
