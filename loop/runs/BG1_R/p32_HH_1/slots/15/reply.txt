def _best_packing(n, rng, seconds):
    t0, best = time.process_time(), None
    while time.process_time() - t0 < seconds:
        c, r = _solve_once(n, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    return best


def _grow(c, r):
    """Add one circle at the grid point with the largest clearance from circles and walls."""
    g = (np.arange(200) + 0.5) / 200
    p = np.stack(np.meshgrid(g, g), -1).reshape(-1, 2)
    gap = np.sqrt(((p[:, None] - c[None]) ** 2).sum(-1)) - r[None]
    gap = np.minimum(gap.min(1), np.minimum(p, 1 - p).min(1))
    return np.vstack([c, p[np.argmax(gap)]])

    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    prev = _best_packing(n - 1, rng, 0.15 * time_limit) if n > 2 else None   # best packing for n-1 circles
