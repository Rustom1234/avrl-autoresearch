def minmax_kernel(A, B):
    sa = A.sum(1)[:, None]
    sb = B.sum(1)[None, :]
    l1 = cdist(A, B, metric="cityblock")
    return (sa + sb - l1) / np.maximum(sa + sb + l1, 1e-9)
