def tanimoto_k(A, B):
    inter = A @ B.T
    return inter / np.maximum(A.sum(1)[:, None] + B.sum(1)[None, :] - inter, 1e-9)

    A = morgan_bits(train_smiles)

    m = SVR(kernel='precomputed', C=5, epsilon=0.1).fit(tanimoto_k(A, A), y - mu)
