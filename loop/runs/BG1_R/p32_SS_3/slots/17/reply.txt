def _repair(c, r):
    """Optimal radii for fixed centres via LP, then shrink radii until the packing is valid."""
    n = len(r)
    try:
        from scipy.optimize import linprog
        iu = np.triu_indices(n, 1)
        m = len(iu[0])
        A = np.zeros((m, n))
        A[np.arange(m), iu[0]] = 1
        A[np.arange(m), iu[1]] = 1
        b = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1))
        w = np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1]))
        w = np.maximum(w, 0)
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=list(zip(np.zeros(n), w)), method="highs")
        if res.status == 0:
            r = np.maximum(res.x - 2e-7, 0)
    except Exception:
        pass
