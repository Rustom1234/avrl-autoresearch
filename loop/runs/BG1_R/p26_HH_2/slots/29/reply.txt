def _halton_centres(n, rng):
    # scrambled Halton points in bases 2 and 3, with a random digit permutation and offset per restart
    idx = np.arange(1, n + 1) + rng.integers(0, 1000)
    cols = []
    for base in (2, 3):
        perm = rng.permutation(base)
        vals, f, k = np.zeros(n), 1.0, idx.copy()
        while (k > 0).any():
            f /= base
            vals += f * perm[k % base] * (k > 0)
            k //= base
        cols.append(vals)
    u = np.stack(cols, 1)
    return INIT_LOW + (1 - 2 * INIT_LOW) * u + rng.uniform(-0.02, 0.02, (n, 2))
