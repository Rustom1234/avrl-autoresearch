def _insert_point(c, r, rng, samples=2000):
    """Point with the largest clearance to existing circles and walls, and a small radius for it."""
    p = rng.uniform(0, 1, (samples, 2))
    d = np.sqrt(((p[:, None] - c[None]) ** 2).sum(-1)) - r[None]
    walls = np.minimum(np.minimum(p[:, 0], p[:, 1]), np.minimum(1 - p[:, 0], 1 - p[:, 1]))
    clear = np.minimum(d.min(1), walls)
    return p[np.argmax(clear)], 0.5 * max(clear.max(), 0)


def solve(n, seed, time_limit):
    rng, t0, best, base = np.random.default_rng(seed), time.process_time(), None, None
    budget = TIME_FRACTION * time_limit
    while time.process_time() - t0 < budget / 3:   # random restarts for n-1 circles
        c, r = _solve_once(n - 1, rng)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if base is None or r.sum() > base[1].sum():
            base = (c, r)
    while time.process_time() - t0 < budget:       # insert one circle and optimise all n
        p, rn = _insert_point(base[0], base[1], rng)
        x0 = np.concatenate([np.vstack([base[0], p]).ravel(), np.append(base[1], rn)])
        c, r = _solve_once(n, rng, x0)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    return best
