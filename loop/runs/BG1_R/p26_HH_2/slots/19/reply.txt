def _maxmin_init(n, rng):
    c = np.empty((n, 2))
    c[0] = rng.uniform(INIT_LOW, 1 - INIT_LOW, 2)
    for k in range(1, n):
        s = rng.uniform(0, 1, (500, 2))
        wall = np.minimum(s, 1 - s).min(1)
        d = np.sqrt(((s[:, None] - c[None, :k]) ** 2).sum(-1)).min(1)
        c[k] = s[np.argmax(np.minimum(d, wall))]
    return c
