def _row_start(n, rng):
    rows = int(rng.choice([4, 5, 6]))
    while True:
        counts = rng.integers(3, 8, rows)
        if counts.sum() == n:
            break
    pts = []
    for k, y in zip(counts, np.linspace(0.1, 0.9, rows)):
        sp = 0.8 / (k - 1)
        xs = np.linspace(0.1, 0.9, k) + rng.uniform(-sp / 2, sp / 2)
        pts.append(np.stack([xs, np.full(k, y)], 1))
    c = np.clip(np.concatenate(pts) + rng.uniform(-0.02, 0.02, (n, 2)), 0, 1)
    if rng.random() < 0.5:
        c = c[:, ::-1].copy()
    return c
