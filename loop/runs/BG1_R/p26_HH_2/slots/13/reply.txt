def _reinsert(c, r, rng):
    """Move the smallest circle to the most open of 2000 random points and reset all radii."""
    n = len(r)
    i = np.argmin(r)
    others = np.delete(np.arange(n), i)
    S = rng.uniform(0, 1, (2000, 2))
    d = np.sqrt(((S[:, None] - c[None, others]) ** 2).sum(-1)).min(1)
    b = np.minimum(np.minimum(S[:, 0], S[:, 1]), np.minimum(1 - S[:, 0], 1 - S[:, 1]))
    c = c.copy()
    c[i] = S[np.argmax(np.minimum(d, b))]
    return np.concatenate([c.ravel(), np.full(n, INIT_RADIUS)])
