def _insert(c, r):
    """Add one circle at the grid point with the largest clearance (distance to nearest circle edge or wall)."""
    g = (np.arange(200) + 0.5) / 200
    X, Y = np.meshgrid(g, g)
    P = np.stack([X.ravel(), Y.ravel()], 1)
    gap = np.sqrt(((P[:, None] - c[None]) ** 2).sum(-1)) - r[None]
    wall = np.minimum(np.minimum(P[:, 0], P[:, 1]), np.minimum(1 - P[:, 0], 1 - P[:, 1]))
    clear = np.minimum(gap.min(1), wall)
    k = np.argmax(clear)
    return np.vstack([c, P[k]]), np.append(r, max(clear[k], 0.0))

def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    prev = None
    if n > 1:   # best-effort packing of n-1 circles to seed the insertion starts
        pc, pr = _solve_once(n - 1, rng)
        pc = np.clip(pc, 0, 1)
        prev = (pc, _repair(pc, pr))
    k = 0
    while best is None or time.process_time() - t0 < TIME_FRACTION * time_limit:   # restarts until the time is used
        if prev is not None and k % 2 == 0:
            c, r = _solve_once(n, rng, _insert(*prev))
