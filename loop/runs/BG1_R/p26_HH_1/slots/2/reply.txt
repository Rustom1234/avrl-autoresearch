def _dropout(c, r, rng):
    """Remove ~20% of circles, re-solve the rest, reinsert the removed ones at random and re-solve all."""
    n = len(r)
    k = max(1, int(round(0.2 * n)))
    idx = rng.choice(n, k, replace=False)
    keep = np.setdiff1d(np.arange(n), idx)
    ck, rk = _slsqp(np.concatenate([c[keep].ravel(), r[keep]]), len(keep))
    c_new, r_new = c.copy(), r.copy()
    c_new[keep], r_new[keep] = ck, rk
    c_new[idx] = rng.uniform(INIT_LOW, 1 - INIT_LOW, (k, 2))
    r_new[idx] = INIT_RADIUS
    return _slsqp(np.concatenate([c_new.ravel(), r_new]), n)
