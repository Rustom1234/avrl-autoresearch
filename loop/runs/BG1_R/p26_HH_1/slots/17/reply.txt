def _lattice_centers(n, rng):
    """Square lattice with at least n sites, random extras dropped, small jitter added."""
    m = int(np.ceil(np.sqrt(n)))
    g = (np.arange(m) + 0.5) / m
    pts = np.array([(x, y) for x in g for y in g])
    pts = pts[rng.permutation(len(pts))[:n]]
    return np.clip(pts + rng.normal(0, 0.01, pts.shape), 0.01, 0.99)
