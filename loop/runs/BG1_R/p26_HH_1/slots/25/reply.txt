def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    budget = TIME_FRACTION * time_limit
    pool, it = [], MIN_ITER
    while best is None or time.process_time() - t0 < budget:
        if not pool:   # refill with fresh starts once the pool has been halved to nothing
            pool, it = [_start(n, rng) for _ in range(POOL)], MIN_ITER
        scored = []
        for x0 in pool:
            c, r = _solve_once(n, x0, it)
            c = np.clip(c, 0, 1)
            r = _repair(c, r)
            scored.append((r.sum(), c, r, x0))
            if time.process_time() - t0 >= budget:
                break
        scored.sort(key=lambda s: -s[0])
        if best is None or scored[0][0] > best[1].sum():
            best = (scored[0][1], scored[0][2])
        pool = [s[3] for s in scored[:len(scored) // 2]]   # keep the top half
        it = min(2 * it, MAX_ITER)
    return best
