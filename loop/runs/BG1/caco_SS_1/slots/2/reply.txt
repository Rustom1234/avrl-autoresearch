def minmax_kernel(A, B):
    K = np.zeros((len(A), len(B)))
    for j, b in enumerate(B):
        K[:, j] = np.minimum(A, b).sum(1) / np.maximum(np.maximum(A, b).sum(1), 1e-9)
    return K
