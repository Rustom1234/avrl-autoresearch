def _reinsert(best, rng):
    c, r = best[0].copy(), best[1] * 0.7
    idx = np.argsort(best[1])[:rng.integers(2, 4)]
    keep = [i for i in range(len(r)) if i not in idx]
    for i in idx:
        p = rng.uniform(0, 1, (2000, 2))
        d = np.sqrt(((p[:, None] - c[keep][None]) ** 2).sum(-1)) - r[keep]
        s = np.minimum(d.min(1), np.minimum(p.min(1), (1 - p).min(1)))
        c[i], r[i] = p[np.argmax(s)], 0.02
        keep.append(i)
    return np.concatenate([c.ravel(), r])
