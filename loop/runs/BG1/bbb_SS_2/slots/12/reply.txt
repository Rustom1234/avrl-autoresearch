def clean_mol(s):
    mol = Chem.MolFromSmiles(s)
    try:
        frags = Chem.rdmolops.GetMolFrags(mol, asMols=True)
        mol = max(frags, key=lambda m: m.GetNumHeavyAtoms())
        mol = rdMolStandardize.Uncharger().uncharge(mol)
        Chem.SanitizeMol(mol)
    except Exception:
        mol = Chem.MolFromSmiles(s)
    return mol
