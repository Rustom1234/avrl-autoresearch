def _reinsert(c, r, rng):
    """Move the smallest circle to the sampled point with the largest clearance to all other circles and walls."""
    k = np.argmin(r)
    others = np.delete(np.arange(len(r)), k)
    p = rng.uniform(0, 1, (2000, 2))
    d = np.sqrt(((p[:, None] - c[others][None]) ** 2).sum(-1))
    clear = np.minimum((d - r[others]).min(1), np.minimum(p, 1 - p).min(1))
    c, r = c.copy(), r.copy()
    c[k], r[k] = p[np.argmax(clear)], max(clear.max(), 0)
    return c, r
