def solve(n, seed, time_limit):
    rng, t0 = np.random.default_rng(seed), time.process_time()
    budget = TIME_FRACTION * time_limit
    K, SWAP_EVERY = 4, 10
    temps = 1e-4 * (1e-2 / 1e-4) ** (np.arange(K) / (K - 1))
    chains = [None] * K
    best = None
    rounds = 0
    while time.process_time() - t0 < budget:
        for k in range(K):
            if time.process_time() - t0 >= budget:
                break
            cur = chains[k]
            if cur is None:
                c, r = _solve_once(n, rng)
            else:
                c0 = np.clip(cur[0] + rng.normal(0, 0.05, (n, 2)), 0, 1)
                c, r = _solve_once(n, rng, np.concatenate([c0.ravel(), cur[1]]))
            c = np.clip(c, 0, 1)
            r = _repair(c, r)
            if cur is None or r.sum() >= cur[1].sum() or np.exp((r.sum() - cur[1].sum()) / temps[k]) > rng.uniform():
                chains[k] = (c, r)
            if best is None or r.sum() > best[1].sum():
                best = (c, r)
        rounds += 1
        if rounds % SWAP_EVERY == 0:
            for k in range(K - 1):
                if chains[k] is None or chains[k + 1] is None:
                    continue
                d = (chains[k + 1][1].sum() - chains[k][1].sum()) * (1 / temps[k] - 1 / temps[k + 1])
                if np.exp(min(d, 0)) > rng.uniform():
                    chains[k], chains[k + 1] = chains[k + 1], chains[k]
    return best
