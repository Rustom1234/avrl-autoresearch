def _inflate(n, rng, steps=200):
    """Grow radii to the largest value that fits the current centres, pushing overlapping centres apart."""
    c = rng.uniform(0.1, 0.9, (n, 2))
    for _ in range(steps):
        diff = c[:, None] - c[None]
        d = np.sqrt((diff ** 2).sum(-1)) + np.eye(n) * 1e9
        walls = np.minimum(c, 1 - c).min(1)
        r = np.maximum(np.minimum(walls, (d - 0 * d.min(1)).min(1) * 0 + (d - np.zeros(n)[None]).min(1)), 0)
        over = np.maximum(r[:, None] + r[None] - d, 0)
        np.fill_diagonal(over, 0)
        u = diff / d[..., None]
        c = np.clip(c + 0.5 * (over[..., None] * u).sum(1), 1e-3, 1 - 1e-3)
    return c, r
