def _solve_once(n, rng, x0=None):
    if x0 is None:
        x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * n), np.full(n, INIT_RADIUS)])
    all_i, all_j = np.triu_indices(n, 1)
    c0, r0 = x0[:2 * n].reshape(n, 2), x0[2 * n:]
    d0 = np.sqrt(((c0[all_i] - c0[all_j]) ** 2).sum(1))
    sel = d0 < 2 * r0.max() + 0.15
    for _ in range(3):
        iu = (all_i[sel], all_j[sel])

        def cons(v):
            c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
            d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
            return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

        res = minimize(lambda v: -v[2 * n:].sum(), x0, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                       options={"maxiter": MAX_ITER})
        c, r = res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
        d = np.sqrt(((c[all_i] - c[all_j]) ** 2).sum(1))
        over = r[all_i] + r[all_j] - d > 0
        if not np.any(over & ~sel):
            break
        sel = sel | over
        x0 = res.x
    return c, r
