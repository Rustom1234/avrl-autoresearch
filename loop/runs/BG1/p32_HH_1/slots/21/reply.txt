def _grow(c, r, rng):
    """Greedily grow each radius to the largest value its walls and neighbours allow. Only increases radii."""
    wall = np.minimum(c, 1 - c).min(1)
    for _ in range(2):
        for i in rng.permutation(len(r)):
            d = np.sqrt(((c - c[i]) ** 2).sum(1))
            d[i] = np.inf
            lim = min(wall[i], (d - r).min())
            if lim > r[i]:
                r[i] = lim
    return r
