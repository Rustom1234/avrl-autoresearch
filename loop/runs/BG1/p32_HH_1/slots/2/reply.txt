def _rattlers(c, r):
    """Circles with fewer than two near-contacts (gap below 1e-4) to other circles or walls."""
    d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
    gap = d - r[:, None] - r[None]
    np.fill_diagonal(gap, np.inf)
    wall = np.stack([c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r], 1)
    contacts = (gap < 1e-4).sum(1) + (wall < 1e-4).sum(1)
    return contacts < 2
