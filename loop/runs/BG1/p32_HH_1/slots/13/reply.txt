def _penalty(v, n, iu, w):
    c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
    diff = c[iu[0]] - c[iu[1]]
    d = np.sqrt((diff ** 2).sum(1)) + 1e-12
    ov = np.maximum(0, r[iu[0]] + r[iu[1]] - d)
    lo = np.maximum(0, r[:, None] - c)
    hi = np.maximum(0, c + r[:, None] - 1)
    f = -r.sum() + w * ((ov ** 2).sum() + (lo ** 2).sum() + (hi ** 2).sum())
    gc, gr = np.zeros((n, 2)), -np.ones(n)
    pair = (2 * w * ov / d)[:, None] * diff
    np.add.at(gc, iu[0], -pair)
    np.add.at(gc, iu[1], pair)
    np.add.at(gr, iu[0], 2 * w * ov)
    np.add.at(gr, iu[1], 2 * w * ov)
    gc += -2 * w * lo + 2 * w * hi
    gr += 2 * w * lo.sum(1) + 2 * w * hi.sum(1)
    return f, np.concatenate([gc.ravel(), gr])
