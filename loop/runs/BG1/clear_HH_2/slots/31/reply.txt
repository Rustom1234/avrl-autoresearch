def leaf_onehot(leaves, width=31):
    # one-hot encode (tree, leaf) pairs; width = num_leaves of the LightGBM model
    n, T = leaves.shape
    X = np.zeros((n, width * T))
    X[np.arange(n)[:, None], leaves + width * np.arange(T)[None, :]] = 1.0
    return X
