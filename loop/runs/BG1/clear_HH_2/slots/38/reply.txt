def choose_sigma2(K, y, smiles):
    scaffolds = [MurckoScaffold.MurckoScaffoldSmiles(smiles=s) for s in smiles]
    groups = np.unique(scaffolds, return_inverse=True)[1]
    best_score, best_p, best_s2 = -np.inf, 1, 0.3
    for p in (1, 2, 3):
        Kp = K ** p
        for s2 in (0.1, 0.3, 1.0):
            pred = np.zeros(len(y))
            for tr, va in GroupKFold(n_splits=5).split(Kp, y, groups):
                pred[va] = gp_mean(Kp[np.ix_(tr, tr)], y[tr], Kp[np.ix_(va, tr)], s2)
            score = spearmanr(pred, y).correlation
            if score > best_score:
                best_score, best_p, best_s2 = score, p, s2
    return best_p, best_s2
