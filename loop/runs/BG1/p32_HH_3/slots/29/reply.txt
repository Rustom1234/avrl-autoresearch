def _quadrant_start(n, rng):
    """Hierarchical start: solve each quadrant separately, map back, and return the concatenated (unpolished) vector."""
    counts = 1 + rng.multinomial(n - 4, [0.25] * 4)
    cs, rs = [], []
    for k, (ox, oy) in zip(counts, [(0, 0), (0.5, 0), (0, 0.5), (0.5, 0.5)]):
        c, r = _solve_once(k, rng)
        c = np.clip(c, 0, 1)
        r = np.maximum(np.minimum(r, np.minimum(c, 1 - c).min(1)), 0)
        cs.append(np.array([ox, oy]) + 0.5 * c)
        rs.append(0.5 * r)
    return np.concatenate([np.vstack(cs).ravel(), np.concatenate(rs)])
