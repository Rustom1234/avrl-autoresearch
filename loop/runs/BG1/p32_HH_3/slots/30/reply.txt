def _sym_centres(n, rng):
    m = int(np.ceil(np.sqrt(n)))
    g = (np.arange(m) + 0.5) / m
    p = np.stack(np.meshgrid(g, g), -1).reshape(-1, 2)
    p = p[np.argsort(((p - 0.5) ** 2).sum(1), kind="stable")[:n]]
    return np.clip(p + 0.01 * rng.standard_normal(p.shape), 0, 1)
