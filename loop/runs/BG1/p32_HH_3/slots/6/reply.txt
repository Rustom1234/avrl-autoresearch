def _hole_start(c, r):
    """Remove the smallest circle and re-insert it at the largest empty spot (200x200 grid), tiny radius."""
    k = np.argmin(r)
    keep = np.arange(len(r)) != k
    g = np.linspace(0, 1, 200)
    gx, gy = np.meshgrid(g, g)
    pts = np.stack([gx.ravel(), gy.ravel()], 1)
    d = np.sqrt(((pts[:, None] - c[keep][None]) ** 2).sum(-1)) - r[keep][None]
    wall = np.minimum(np.minimum(pts[:, 0], 1 - pts[:, 0]), np.minimum(pts[:, 1], 1 - pts[:, 1]))
    clear = np.minimum(d.min(1), wall)
    c2, r2 = c.copy(), r.copy()
    c2[k], r2[k] = pts[np.argmax(clear)], 1e-3
    return np.concatenate([c2.ravel(), r2])


def solve(n, seed, time_limit):
    rng, t0, best = np.random.default_rng(seed), time.process_time(), None
    while time.process_time() - t0 < TIME_FRACTION * time_limit:   # random restarts until the time is used
        if best is not None and rng.random() < 0.5:   # refine the incumbent: move its weakest circle into the largest hole
            c, r = _solve_once(n, rng, _hole_start(*best))
        else:
            c, r = _solve_once(n, rng)
