def _grow(c, r, rng):
    """Greedily grow each radius to the largest value its neighbours and the walls allow."""
    d = np.sqrt(((c[:, None] - c[None]) ** 2).sum(-1))
    np.fill_diagonal(d, np.inf)
    wall = np.minimum(np.minimum(c[:, 0], c[:, 1]), np.minimum(1 - c[:, 0], 1 - c[:, 1]))
    for _ in range(1000):
        change = 0.0
        for i in rng.permutation(len(r)):
            m = min(wall[i], (d[i] - r).min())
            if m > r[i]:
                change = max(change, m - r[i])
                r[i] = m
        if change <= 1e-12:
            break
    return r
