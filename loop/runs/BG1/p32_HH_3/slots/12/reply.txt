def _lloyd_centres(n, rng, iters=30, grid=60):
    """Spread-out initial centres: Lloyd relaxation of n random points against a fixed sample grid."""
    s = (np.arange(grid) + 0.5) / grid
    samples = np.stack(np.meshgrid(s, s), -1).reshape(-1, 2)
    c = rng.uniform(0, 1, (20 * n, 2))[:n]
    for _ in range(iters):
        lab = np.argmin(((samples[:, None] - c[None]) ** 2).sum(-1), 1)
        cnt = np.bincount(lab, minlength=n)
        for k in range(2):
            sums = np.bincount(lab, samples[:, k], minlength=n)
            c[:, k] = np.where(cnt > 0, sums / np.maximum(cnt, 1), c[:, k])
    return c
