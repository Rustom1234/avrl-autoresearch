def solve(n, seed, time_limit):
    rng, t0 = np.random.default_rng(seed), time.process_time()
    total = TIME_FRACTION * time_limit
    pool = []
    while time.process_time() - t0 < 0.4 * total:   # coarse screening: many restarts with few SLSQP iterations
        c, r = _solve_once(n, rng, maxiter=60)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        pool.append((r.sum(), c, r))
    pool.sort(key=lambda p: -p[0])
    top = [(c, r) for _, c, r in pool[:3]]
    k = 0
    while time.process_time() - t0 < total:   # fine refinement of the top packings in turn
        c, r = top[k % len(top)]
        c, r = _solve_once(n, rng, np.concatenate([c.ravel(), r]), maxiter=400)
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if r.sum() > top[k % len(top)][1].sum():
            top[k % len(top)] = (c, r)
        k += 1
    return max(top, key=lambda p: p[1].sum())
