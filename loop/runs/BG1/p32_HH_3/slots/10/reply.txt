def _solve_once(n, rng, start=None):
    if start is None:
        # solve n-1 circles, then insert one circle into the largest free hole
        m = n - 1
        x0 = np.concatenate([rng.uniform(INIT_LOW, 1 - INIT_LOW, 2 * m), np.full(m, INIT_RADIUS)])
        c, r = _slsqp(m, x0, MAX_ITER // 2)
        dist = np.sqrt(((GRID[:, None] - c[None]) ** 2).sum(-1)) - r[None]
        wall = np.minimum(np.minimum(GRID[:, 0], GRID[:, 1]), np.minimum(1 - GRID[:, 0], 1 - GRID[:, 1]))
        free = np.minimum(dist.min(1), wall)
        k = int(np.argmax(free + 1e-12 * rng.uniform(size=len(free))))
        start = np.concatenate([np.vstack([c, GRID[k]]).ravel(), np.append(r, 0.01)])
    return _slsqp(n, start, MAX_ITER)
