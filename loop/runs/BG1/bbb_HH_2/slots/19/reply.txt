def _shape3d_one(s, seed):
    m = Chem.AddHs(Chem.MolFromSmiles(s))
    params = rdDistGeom.ETKDGv3()
    params.randomSeed = seed
    params.maxIterations = 10
    if rdDistGeom.EmbedMolecule(m, params) < 0:
        return [0.0] * 5
    try:
        rdForceFieldHelpers.MMFFOptimizeMolecule(m)
    except Exception:
        pass
    return [
        rdMolDescriptors.CalcNPR1(m),
        rdMolDescriptors.CalcNPR2(m),
        rdMolDescriptors.CalcAsphericity(m),
        rdMolDescriptors.CalcEccentricity(m),
        rdMolDescriptors.CalcRadiusOfGyration(m),
    ]


def shape3d(smiles, seed):
    rows = []
    for s in smiles:
        key = (s, seed)
        if key not in _SHAPE_CACHE:
            _SHAPE_CACHE[key] = _shape3d_one(s, seed)
        rows.append(_SHAPE_CACHE[key])
    return np.array(rows, dtype=np.float32)
