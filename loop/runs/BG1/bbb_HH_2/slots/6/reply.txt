def fit_predict(train_smiles, train_y, test_smiles, seed):
    y = np.asarray(train_y, dtype=np.int64)
    fp_tr, desc_tr = fingerprints(train_smiles)
    fp_te, desc_te = fingerprints(test_smiles)
    desc_tr = np.nan_to_num(desc_tr, nan=0.0, posinf=0.0, neginf=0.0)
    desc_te = np.nan_to_num(desc_te, nan=0.0, posinf=0.0, neginf=0.0)
    lo, hi = np.percentile(desc_tr, [1, 99], axis=0)
    desc_tr = np.clip(desc_tr, lo, hi)
    desc_te = np.clip(desc_te, lo, hi)
    x_tr = np.hstack([fp_tr, desc_tr]).astype(np.float32)
