def fit_predict(train_smiles, train_y, test_smiles, seed):
    y = np.asarray(train_y, dtype=np.int64)
    train_sets = fingerprints(train_smiles)
    test_sets = fingerprints(test_smiles)
    # Pass 1: score test molecules with the base ensemble
    score = ensemble_scores(train_sets, y, test_sets, seed)
    k = int(round(0.15 * len(test_smiles)))
    order = np.argsort(score)
    neg_idx, pos_idx = order[:k], order[len(order) - k:]
    # Pass 2: append confident test pseudo-labels and refit the same ensemble
    aug_sets = [np.vstack([tr, te[pos_idx], te[neg_idx]]) for tr, te in zip(train_sets, test_sets)]
    aug_y = np.concatenate([y, np.ones(k, dtype=np.int64), np.zeros(k, dtype=np.int64)])
    return ensemble_scores(aug_sets, aug_y, test_sets, seed)
