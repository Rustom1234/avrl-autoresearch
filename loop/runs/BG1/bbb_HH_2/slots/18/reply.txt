def fit_predict(train_smiles, train_y, test_smiles, seed):
    y = np.asarray(train_y, dtype=np.int64)
    train_sets = fingerprints(train_smiles)
    test_sets = fingerprints(test_smiles)
    scaffolds = np.array([MurckoScaffold.MurckoScaffoldSmiles(mol=Chem.MolFromSmiles(s)) for s in train_smiles])
    unique_scaf = np.unique(scaffolds)
    rng = np.random.default_rng(seed)
    total = np.zeros(len(test_smiles))
    for bag in range(N_BAGS):
        drawn = rng.choice(unique_scaf, size=len(unique_scaf), replace=True)
        idx = np.flatnonzero(np.isin(scaffolds, drawn))
        for x_tr, x_te in zip(train_sets, test_sets):
            model = XGBClassifier(n_estimators=250, max_depth=5, learning_rate=0.05, subsample=0.8, colsample_bytree=0.3,
                                  min_child_weight=2, eval_metric='logloss', random_state=seed + bag, n_jobs=8)
            model.fit(x_tr[idx], y[idx])
            total += rankdata(model.predict_proba(x_te)[:, 1])
    total = (total - total.min()) / (total.max() - total.min() + 1e-12)
    return total
