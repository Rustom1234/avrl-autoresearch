def fit_predict(train_smiles, train_y, test_smiles, seed):
    X_train = featurize(train_smiles)
    X_test = featurize(test_smiles)
    y = np.asarray(train_y, dtype=np.int64)
    preds = []
    for i in range(5):
        model = LGBMClassifier(objective="binary", n_estimators=400, learning_rate=0.05, num_leaves=31,
                               colsample_bytree=0.3, subsample=0.7, subsample_freq=1, min_child_samples=10,
                               n_jobs=-1, random_state=seed * 10 + i)
        model.fit(X_train, y)
        preds.append(model.predict_proba(X_test)[:, 1])
    return np.mean(preds, axis=0)
