def featurize(smiles):
    mols = [Chem.MolFromSmiles(s) for s in smiles]
    fps = []
    for r in (1, 2, 3):
        gen = rdFingerprintGenerator.GetMorganGenerator(radius=r, fpSize=1024)
        fps.append(np.array([gen.GetCountFingerprintAsNumPy(m) for m in mols], dtype=np.float32))
