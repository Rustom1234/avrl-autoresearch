def standardize(smi):
    mol = Chem.MolFromSmiles(smi)
    if mol is None:
        return mol
    try:
        frags = Chem.GetMolFrags(mol, asMols=True)
        big = max(frags, key=lambda f: f.GetNumHeavyAtoms())
        return rdMolStandardize.Uncharger().uncharge(big)
    except Exception:
        return mol
