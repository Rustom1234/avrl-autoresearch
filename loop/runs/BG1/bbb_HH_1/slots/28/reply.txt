def featurize(mols, counts, vocab):
    fp = np.zeros((len(mols), len(vocab)), dtype=np.float32)
    for i, c in enumerate(counts):
        for k, v in c.items():
            j = vocab.get(k)
            if j is not None:
                fp[i, j] = v
