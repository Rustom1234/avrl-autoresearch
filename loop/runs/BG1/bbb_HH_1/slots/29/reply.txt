    systems = []
    for r in rings:
        new = set(r)
        keep = []
        for s in systems:
            if s & new:
                new |= s
            else:
                keep.append(s)
        systems = keep + [new]
    ring_atoms = set().union(*rings) if rings else set()

        max((len(s) for s in systems), default=0),

        len(rings) - n_arom,
        rdMolDescriptors.CalcFractionCSP3(mol),
