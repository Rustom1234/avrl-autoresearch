_UNCHARGER = rdMolStandardize.Uncharger()


def standardize(smiles):
    mol = Chem.MolFromSmiles(smiles)
    try:
        frag = rdMolStandardize.LargestFragmentChooser().choose(mol)
        frag = _UNCHARGER.uncharge(frag)
        if frag is not None:
            return frag
    except Exception:
        pass
    return mol
