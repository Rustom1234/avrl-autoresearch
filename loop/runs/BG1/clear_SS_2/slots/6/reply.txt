def minmax_kernel(A, B):
    A = A.astype(np.float64)
    B = B.astype(np.float64)
    K = np.empty((len(A), len(B)))
    sb = B.sum(1)
    for i, a in enumerate(A):
        mn = np.minimum(a, B).sum(1)
        K[i] = mn / np.maximum(a.sum() + sb - mn, 1e-9)
    return K
