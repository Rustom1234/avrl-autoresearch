    """Move the smallest circle to the grid point with the largest clearance from the other circles."""
    k = np.argmin(r)
    others = np.delete(np.arange(len(r)), k)
    g = np.linspace(0, 1, 40)
    P = np.stack(np.meshgrid(g, g), -1).reshape(-1, 2)
    clear = (np.sqrt(((P[:, None] - c[others][None]) ** 2).sum(-1)) - r[others][None]).min(1)
    c = c.copy()
    c[k] = P[np.argmax(clear)]
    return c
