def _kick(best, rng):
    """Move 1 to 3 small circles (chosen with probability proportional to 1/r) to random points with radius 0.01."""
    c, r = best[0].copy(), best[1].copy()
    k = rng.integers(1, 4)
    p = 1 / (r + 1e-9)
    idx = rng.choice(len(r), size=k, replace=False, p=p / p.sum())
    c[idx] = rng.uniform(0, 1, (k, 2))
    r[idx] = 0.01
    return c, r
