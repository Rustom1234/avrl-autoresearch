def _walls(c, r):
    """Number of circles touching a wall (centre within r + 1e-9 of an edge)."""
    t = r[:, None] + 1e-9
    return int(((c <= t) | (1 - c <= t)).any(1).sum())


def _better(c, r, best):
    """Radius sum decides, except that near-equal sums (within 1e-4) are broken by wall contacts."""
    s, bs = r.sum(), best[1].sum()
    if abs(s - bs) <= 1e-4 and _walls(c, r) != _walls(*best):
        return _walls(c, r) > _walls(*best)
    return s > bs
