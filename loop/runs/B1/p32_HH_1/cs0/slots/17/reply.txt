def _rows_start(n, rng):
    k = int(rng.integers(5, 7))
    counts = np.full(k, n // k)
    counts[rng.choice(k, n - counts.sum(), replace=False)] += 1
    pts = [((j + 0.25 + 0.5 * (i % 2)) / (m + 0.5), (i + 0.5) / k) for i, m in enumerate(counts) for j in range(m)]
    pts = np.array(pts) + rng.normal(0, 0.02, (n, 2))
    pts = np.clip(pts, 0.02, 0.98)
    return pts[:, ::-1] if rng.random() < 0.5 else pts
