def _row_init(rng):
    rows = ROWS[rng.integers(len(ROWS))]
    ys = np.linspace(0.1, 0.9, len(rows))
    pts = np.array([((i + 0.5) / k, y) for k, y in zip(rows, ys) for i in range(k)])
    dx, dy = 1 / max(rows), 0.8 / (len(rows) - 1)
    pts = np.clip(pts + rng.normal(0, 0.02, pts.shape), 0.02, 0.98)
    return rng.permutation(pts), 0.5 * min(dx, dy) * 0.9
