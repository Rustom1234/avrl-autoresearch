def _reinsert(c, r):
    """Drop the smallest circle and put it back where the free radius on a 40x40 grid is largest."""
    k = np.argmin(r)
    cc, rr = np.delete(c, k, 0), np.delete(r, k)
    g = np.linspace(0, 1, 40)
    P = np.stack(np.meshgrid(g, g), -1).reshape(-1, 2)
    free = (np.sqrt(((P[:, None] - cc[None]) ** 2).sum(-1)) - rr[None]).min(1)
    free = np.minimum(free, np.minimum(P.min(1), (1 - P).min(1)))
    j = np.argmax(free)
    return np.vstack([cc, P[j]]), np.append(rr, max(free[j], 1e-3))
