def _swap_kick(c, r, rng):
    """Swap the centres of two circles whose radii differ by more than a factor 1.3."""
    n = len(r)
    for _ in range(100):
        i, j = rng.choice(n, 2, replace=False)
        if max(r[i], r[j]) > 1.3 * min(r[i], r[j]):
            c = c.copy()
            c[[i, j]] = c[[j, i]]
            break
    return c
