    n = len(r)

    r = np.maximum(r, 0)
    I, J = np.triu_indices(n, 1)
    for _ in range(50):
        changed = False
        for k in rng.permutation(len(I)):
            i, j = I[k], J[k]
            d = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
            if r[i] + r[j] > d:
                s = d / (r[i] + r[j])
                r[i], r[j] = r[i] * s, r[j] * s
                changed = True
        if not changed:
            break
    return np.maximum(r - SHRINK, 0)
