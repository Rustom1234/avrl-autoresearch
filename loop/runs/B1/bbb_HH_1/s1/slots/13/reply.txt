def largest_fragment(smiles_list):
    out = []
    for s in smiles_list:
        mol = Chem.MolFromSmiles(s)
        frags = Chem.GetMolFrags(mol, asMols=True)
        out.append(Chem.MolToSmiles(max(frags, key=lambda m: m.GetNumHeavyAtoms())))
    return out
