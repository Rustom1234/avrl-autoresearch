def scaffold_groups(smiles_list):
    keys = {}
    groups = []
    for s in smiles_list:
        scaf = MurckoScaffold.MurckoScaffoldSmiles(mol=Chem.MolFromSmiles(s))
        groups.append(keys.setdefault(scaf, len(keys)))
    return np.array(groups)


def choose_weight(X, y, groups, seed):
    scores = {w: [] for w in W_GRID}

        p_gb = scaled_rank(gb.predict_proba(X[te])[:, 1])
        for w in W_GRID:
