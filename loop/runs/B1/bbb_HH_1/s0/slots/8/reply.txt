def choose_weight(X, y, smiles, seed):
    scaf = [MurckoScaffold.MurckoScaffoldSmiles(smiles=s) or s for s in smiles]
    ids = {}
    groups = np.array([ids.setdefault(g, len(ids)) for g in scaf])
    oof_rf = np.zeros(len(y))
    oof_gb = np.zeros(len(y))
    for tr, va in GroupKFold(n_splits=3).split(X, y, groups):
        p_rf, p_gb = member_probs(X[tr], y[tr], X[va], seed)
        oof_rf[va] = scaled_rank(p_rf)
        oof_gb[va] = scaled_rank(p_gb)
    # Grid ordered so that ties resolve toward 0.5
    grid = sorted(np.round(np.arange(0.0, 1.01, 0.1), 2), key=lambda w: abs(w - 0.5))
    aucs = [roc_auc_score(y, w * oof_rf + (1 - w) * oof_gb) for w in grid]
    return float(grid[int(np.argmax(aucs))])
