def ratio_features(smiles):
    rows = []
    for s in smiles:
        mol = Chem.MolFromSmiles(s)
        n = max(mol.GetNumHeavyAtoms(), 1)
        mw = max(Descriptors.MolWt(mol), 1.0)
        tpsa = rdMolDescriptors.CalcTPSA(mol)
        hbd = rdMolDescriptors.CalcNumHBD(mol)
        hba = rdMolDescriptors.CalcNumHBA(mol)
        logp = Crippen.MolLogP(mol)
        rot = rdMolDescriptors.CalcNumRotatableBonds(mol)
        atoms = mol.GetAtoms()
        hetero = sum(1 for a in atoms if a.GetSymbol() in ("N", "O", "S", "F", "Cl", "Br", "I"))
        arom = sum(1 for a in atoms if a.GetIsAromatic())
        nitro = sum(1 for a in atoms if a.GetSymbol() == "N")
        rows.append([tpsa / mw, tpsa / n, (hbd + hba) / n, logp / n, hetero / n, arom / n, nitro / n, rot / n])
    return np.array(rows, dtype=np.float64)
