def minmax(A, B):
    # MinMax (generalized Tanimoto) kernel on counts capped at 8: sum(min)/sum(max), with sum(max)=sum(a)+sum(b)-sum(min).
    # sum(min) = sum_t (A>=t)@(B>=t).T over thresholds t=1..8.
    Ac, Bc = np.minimum(A, 8), np.minimum(B, 8)
    smin = np.zeros((len(A), len(B)))
    for t in range(1, 9):
        smin += (Ac >= t).astype(np.float64) @ (Bc >= t).astype(np.float64).T
    sa = Ac.sum(axis=1)[:, None]
    sb = Bc.sum(axis=1)[None, :]
    return smin / np.maximum(sa + sb - smin, 1e-9)
