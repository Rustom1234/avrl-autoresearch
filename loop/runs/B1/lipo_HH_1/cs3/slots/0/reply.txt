def minmax(A, B):
    # sum_i min(a_i,b_i) = sum_t [a>=t].[b>=t] for integer counts capped at 8 (exact, via matmuls).
    dot = 0.0
    for t in range(1, 9):
        dot = dot + (A >= t).astype(np.float32) @ (B >= t).astype(np.float32).T
    dot = dot.astype(np.float64)
    sa = A.sum(axis=1).astype(np.float64)[:, None]
    sb = B.sum(axis=1).astype(np.float64)[None, :]
    return dot / np.maximum(sa + sb - dot, 1e-9)
