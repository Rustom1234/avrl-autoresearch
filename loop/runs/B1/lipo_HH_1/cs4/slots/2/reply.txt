def scaffold_te(gen_a, ga, ya, gen_b, m=5.0):
    # Smoothed generic-scaffold target mean; training rows exclude their own Murcko group.
    mu = ya.mean()
    S, N, S2, N2 = {}, {}, {}, {}
    for g, k, v in zip(gen_a, ga, ya):
        S[g] = S.get(g, 0.0) + v
        N[g] = N.get(g, 0) + 1
        S2[(g, k)] = S2.get((g, k), 0.0) + v
        N2[(g, k)] = N2.get((g, k), 0) + 1
    tr = np.array([(S[g] - S2[(g, k)] + m * mu) / (N[g] - N2[(g, k)] + m) for g, k in zip(gen_a, ga)])
    te = np.array([(S.get(g, 0.0) + m * mu) / (N.get(g, 0) + m) for g in gen_b])
    return tr, te
