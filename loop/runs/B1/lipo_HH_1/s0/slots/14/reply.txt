def tanimoto(A, B):
    dot = A @ B.T
    na = (A * A).sum(axis=1)[:, None]
    nb = (B * B).sum(axis=1)[None, :]
    return dot / np.maximum(na + nb - dot, 1e-9)

def minmax(A, B):
    # MinMax kernel on clipped counts: sum_j min(a_j, b_j) / sum_j max(a_j, b_j).
    # For integer counts in 0..4, sum of mins = sum over t=1..4 of co-occurrences of (count >= t).
    smin = np.zeros((len(A), len(B)))
    for t in range(1, 5):
        smin += (A >= t).astype(np.float64) @ (B >= t).astype(np.float64).T
    na = A.sum(axis=1)[:, None]
    nb = B.sum(axis=1)[None, :]
    return smin / np.maximum(na + nb - smin, 1e-9)
