def krr_grid_search(Xa, Fa, ya, ga):
    # Picks the kernel-ridge (lam, wr) once per fit_predict call, by scaffold-grouped 4-fold CV on the full training set.
    Za = Xa[:, 2 * N_BITS:].astype(np.float64)
    Za = (Za - Za.mean(axis=0)) / (Za.std(axis=0) + 1e-9)
    gamma = 1.0 / (2 * Za.shape[1])
    sq = (Za * Za).sum(axis=1)
    Ra = np.exp(-gamma * np.maximum(sq[:, None] + sq[None, :] - 2 * Za @ Za.T, 0))
    Ta = tanimoto(Fa, Fa)
