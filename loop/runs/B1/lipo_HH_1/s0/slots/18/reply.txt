def count_features(smiles):
    # Radius-3 Morgan count fingerprint, clipped at N_LEVELS.
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=3, fpSize=N_BITS)
    return np.array([np.minimum(gen.GetCountFingerprintAsNumPy(Chem.MolFromSmiles(s)), N_LEVELS) for s in smiles],
                    dtype=np.float32)


def minmax_kernel(A, B):
    # Generalised Tanimoto sum(min)/sum(max) on clipped counts; sum(min) = sum_t [a>=t][b>=t] exactly.
    smin = np.zeros((len(A), len(B)), dtype=np.float32)
    for t in range(1, N_LEVELS + 1):
        smin += (A >= t).astype(np.float32) @ (B >= t).astype(np.float32).T
    sa = A.sum(axis=1)[:, None]
    sb = B.sum(axis=1)[None, :]
    return smin / np.maximum(sa + sb - smin, 1e-9)
