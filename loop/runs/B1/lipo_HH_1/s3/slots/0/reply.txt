def generic_scaffold(smiles):
    scaf = MurckoScaffold.GetScaffoldForMol(Chem.MolFromSmiles(smiles))
    return Chem.MolToSmiles(MurckoScaffold.MakeScaffoldGeneric(scaf))


def target_encode(gen_a, ya, ga, gen_b, m=5):
    # Smoothed mean of y per generic scaffold, shrunk toward the global mean with weight m.
    # Training rows leave out their own Murcko group (leave-one-group-out), so no self-leakage.
    mu = ya.mean()
    sums, cnts, pair_s, pair_c = {}, {}, {}, {}
    for g, k, y in zip(gen_a, ga, ya):
        sums[g] = sums.get(g, 0.0) + y
        cnts[g] = cnts.get(g, 0) + 1
        pair_s[(g, k)] = pair_s.get((g, k), 0.0) + y
        pair_c[(g, k)] = pair_c.get((g, k), 0) + 1
    ea = np.array([(sums[g] - pair_s[(g, k)] + m * mu) / (cnts[g] - pair_c[(g, k)] + m) for g, k in zip(gen_a, ga)])
    eb = np.array([(sums.get(g, 0.0) + m * mu) / (cnts.get(g, 0) + m) for g in gen_b])
    return ea, eb
