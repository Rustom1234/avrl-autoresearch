def l1_meta(P, y):
    # Non-negative weights and free intercept minimising sum |y - (b + P w)| via LP; None on failure.
    n, k = P.shape
    X = sp.csr_matrix(np.hstack([P, np.ones((n, 1))]))
    I = sp.identity(n, format="csr")
    A_ub = sp.vstack([sp.hstack([-X, -I]), sp.hstack([X, -I])]).tocsr()
    b_ub = np.concatenate([-y, y])
    c = np.concatenate([np.zeros(k + 1), np.ones(n)])
    bounds = [(0, None)] * k + [(None, None)] + [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
    return res.x if res.status == 0 else None
