def minmax_feats(smiles):
    # Count Morgan (r=3); stacking [count>=t] for t=1..4 makes binary Tanimoto equal to MinMax on counts capped at 4.
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=3, fpSize=N_BITS)
    rows = []
    for s in smiles:
        c = gen.GetCountFingerprintAsNumPy(Chem.MolFromSmiles(s))
        rows.append(np.concatenate([(c >= t) for t in (1, 2, 3, 4)]))
    return np.array(rows, dtype=np.float64)
