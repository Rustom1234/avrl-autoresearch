def _polish(n, c, r, deadline):
    """Re-run free-radius SLSQP from its own output until the radius sum gains less than 1e-10."""
    iu = np.triu_indices(n, 1)

    def cons(v):
        c, r = v[:2 * n].reshape(n, 2), v[2 * n:]
        d = np.sqrt(((c[iu[0]] - c[iu[1]]) ** 2).sum(1)) - r[iu[0]] - r[iu[1]]
        return np.concatenate([d, c[:, 0] - r, c[:, 1] - r, 1 - c[:, 0] - r, 1 - c[:, 1] - r, r])

    x, s = np.concatenate([c.ravel(), r]), r.sum()
    while time.process_time() < deadline:
        res = minimize(lambda v: -v[2 * n:].sum(), x, method="SLSQP", constraints={"type": "ineq", "fun": cons},
                       options={"maxiter": 500, "ftol": 1e-14})
        ns = res.x[2 * n:].sum()
        if ns <= s:
            break
        gain, x, s = ns - s, res.x, ns
        if gain < 1e-10:
            break
    return x[:2 * n].reshape(n, 2), x[2 * n:]
