def solve(n, seed, time_limit):
    rng, t0 = np.random.default_rng(seed), time.process_time()
    budget = TIME_FRACTION * time_limit
    best, cur, stall = None, None, 0
    while time.process_time() - t0 < budget:
        restart = cur is None or (stall >= 60 and budget - (time.process_time() - t0) >= 0.25 * budget)
        if restart:        # first start, or a stagnant chain: a fresh random restart
            c, r = _solve_once(n, rng)
        else:              # local kick of the chain's current state
            c, r = cur[0].copy(), cur[1].copy()
            p, R = rng.uniform(0, 1, 2), rng.uniform(0.15, 0.35)
            m = np.hypot(*(c - p).T) < R
            k = m.sum()
            ang, rad = rng.uniform(0, 2 * np.pi, k), R * np.sqrt(rng.uniform(0, 1, k))
            c[m] = np.clip(p + np.c_[rad * np.cos(ang), rad * np.sin(ang)], 0, 1)
            r[m] = 0
            c, r = _solve_once(n, rng, np.concatenate([c.ravel(), r]))
        c = np.clip(c, 0, 1)
        r = _repair(c, r)
        if restart or r.sum() > cur[1].sum():
            cur, stall = (c, r), 0
        else:
            stall += 1
        if best is None or r.sum() > best[1].sum():
            best = (c, r)
    return best
