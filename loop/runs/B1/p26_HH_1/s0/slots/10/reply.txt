def _smallest_kick(c, r, rng):
    """Move 1-3 of the 6 smallest circles to the emptiest of 200 random points (clearance to the other circles and walls)."""
    n = len(r)
    move = rng.choice(np.argsort(r)[:6], rng.integers(1, 4), replace=False)
    keep = np.ones(n, bool)
    keep[move] = False
    for i in move:
        cand = rng.uniform(0, 1, (200, 2))
        d = np.sqrt(((cand[:, None] - c[keep][None]) ** 2).sum(-1)) - r[keep]
        score = np.minimum(d.min(1), np.minimum(cand, 1 - cand).min(1))
        c[i] = cand[np.argmax(score)]
        r[i] = 0
    return c, r
