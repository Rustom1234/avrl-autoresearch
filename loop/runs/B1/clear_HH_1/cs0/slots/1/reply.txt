def get_mol(s):
    m = Chem.MolFromSmiles(s)
    frags = Chem.rdmolops.GetMolFrags(m, asMols=True)
    if len(frags) > 1:
        m = max(frags, key=lambda f: f.GetNumHeavyAtoms())
    return m
