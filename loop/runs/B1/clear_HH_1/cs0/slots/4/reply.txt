def physchem(smiles):

def gobbi(smiles, n=4096):
    out = np.zeros((len(smiles), n))
    for i, s in enumerate(smiles):
        fp = Generate.Gen2DFingerprint(Chem.MolFromSmiles(s), Gobbi_Pharm2D.factory)
        for b in fp.GetOnBits():
            out[i, b % n] = 1.0
    return out
