def gasteiger(smiles):
    rows = []
    for s in smiles:
        m = Chem.MolFromSmiles(s)
        rdPartialCharges.ComputeGasteigerCharges(m)
        q = np.array([a.GetDoubleProp("_GasteigerCharge") for a in m.GetAtoms()])
        q = np.nan_to_num(q, nan=0.0, posinf=0.0, neginf=0.0)
        rows.append([np.sum(q < -0.25), np.sum((q >= -0.25) & (q < -0.05)),
                     np.sum((q >= -0.05) & (q < 0.05)), np.sum((q >= 0.05) & (q < 0.25)),
                     np.sum(q >= 0.25), np.abs(q).max() if q.size else 0.0])
    return np.array(rows, dtype=np.float64)
