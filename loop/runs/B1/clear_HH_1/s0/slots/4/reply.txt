def pharm(smiles):
    rows = np.zeros((len(smiles), 4096), dtype=np.float64)
    for i, s in enumerate(smiles):
        fp = Generate.Gen2DFingerprint(Chem.MolFromSmiles(s), Gobbi_Pharm2D.factory)
        for b in fp.GetOnBits():
            rows[i, b % 4096] = 1.0
    return rows


def physchem(smiles):
    rows = []
    for s in smiles:
        m = Chem.MolFromSmiles(s)
        rows.append([Descriptors.MolWt(m), Descriptors.MolLogP(m), rdMolDescriptors.CalcTPSA(m),
                     rdMolDescriptors.CalcNumHBD(m), rdMolDescriptors.CalcNumHBA(m),
                     rdMolDescriptors.CalcNumRotatableBonds(m), rdMolDescriptors.CalcNumRings(m)])
    return np.array(rows, dtype=np.float64)
