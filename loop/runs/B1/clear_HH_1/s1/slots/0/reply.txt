FUNCS = dict(Descriptors.descList)


def descriptors(smiles):
    rows = []
    for s in smiles:
        m = Chem.MolFromSmiles(s)
        row = [FUNCS[n](m) for n in CURATED]
        row.append(QED.qed(m))
        atoms = list(m.GetAtoms())
        syms = [a.GetSymbol() for a in atoms]
        row.append(sum(x in ("F", "Cl", "Br", "I") for x in syms))
        row.append(syms.count("N"))
        row.append(syms.count("S"))
