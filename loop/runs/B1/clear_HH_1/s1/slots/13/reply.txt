    model.fit(K_train, (y - mu) / sd)

def fit_predict(train_smiles, train_y, test_smiles, seed):
    X = featurize(train_smiles)
    Xt = featurize(test_smiles)
    D = descriptors(train_smiles)
    Dt = descriptors(test_smiles)
    keep = D.std(axis=0) > 1e-9
    F = np.hstack([X, D[:, keep]])
    Ft = np.hstack([Xt, Dt[:, keep]])
    y = np.asarray(train_y, dtype=np.float64)
    y = (rankdata(y) - 0.5) / len(y)
    model = lgb.LGBMRegressor(
        objective="huber", alpha=0.2, min_child_samples=20, reg_lambda=5.0,
        colsample_bytree=0.3, n_estimators=500, learning_rate=0.03,
        random_state=seed, n_jobs=8, verbose=-1,
    )
    model.fit(F, y)
    return model.predict(Ft)
