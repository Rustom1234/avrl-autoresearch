def minmax(A, B):
    # MinMax (generalized Jaccard) kernel for count vectors.
    # sum_i min(a_i, b_i) = sum_k sum_i [a_i >= k][b_i >= k], k = 1..max count.
    # sum_i max(a_i, b_i) = sum(a) + sum(b) - sum_i min(a_i, b_i).
    inter = np.zeros((A.shape[0], B.shape[0]))
    for k in range(1, int(max(A.max(), B.max())) + 1):
        inter += (A >= k).astype(np.float64) @ (B >= k).astype(np.float64).T
    na = A.sum(axis=1)[:, None]
    nb = B.sum(axis=1)[None, :]
    return inter / np.maximum(na + nb - inter, 1e-9)
