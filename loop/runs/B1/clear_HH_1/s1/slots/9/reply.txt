def featurize(smiles):
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=RADIUS, fpSize=N_BITS)
    ap = rdFingerprintGenerator.GetAtomPairGenerator(fpSize=1024)
    rows = []
    for s in smiles:
        if s not in _FP_CACHE:
            m = Chem.MolFromSmiles(s)
            _FP_CACHE[s] = np.concatenate([
                gen.GetFingerprintAsNumPy(m).astype(np.float32),
                np.array(list(MACCSkeys.GenMACCSKeys(m)), dtype=np.float32),
                ap.GetCountFingerprintAsNumPy(m).astype(np.float32),
            ])
        rows.append(_FP_CACHE[s])
    return np.array(rows, dtype=np.float32)
