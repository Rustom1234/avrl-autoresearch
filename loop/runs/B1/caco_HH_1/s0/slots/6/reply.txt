def tanimoto_count(A, B):
    K = np.empty((A.shape[0], B.shape[0]))
    for i in range(A.shape[0]):
        mn = np.minimum(A[i], B).sum(axis=1)
        mx = np.maximum(A[i], B).sum(axis=1)
        K[i] = mn / np.maximum(mx, 1e-12)
    return K
