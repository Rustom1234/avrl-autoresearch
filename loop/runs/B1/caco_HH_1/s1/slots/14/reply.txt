def knn_feature(Btr, ytr, Bq, exclude_self):
    inter = Bq @ Btr.T
    sim = inter / np.maximum(Bq.sum(1)[:, None] + Btr.sum(1)[None, :] - inter, 1e-9)
    if exclude_self:
        np.fill_diagonal(sim, -1.0)
    idx = np.argsort(-sim, axis=1)[:, :5]
    return ytr[idx].mean(1)
